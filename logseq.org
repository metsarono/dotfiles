* Launcher

The ~logseq-db~ wrapper ensures ttyd is running before handing off to the AppImage.
The AppImage release conf lives in ~package-appimage.org~ (managed by ~apprelease~).

** Wrapper script

#+name: logseq-db-launcher
#+BEGIN_SRC bash :tangle all/.local/bin/logseq-db :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755) :comments no
# Ensure ttyd terminal server is running on port 7681 before opening Logseq
load_private_env() {
  local line key
  [ -f "$HOME/.env" ] || return 0
  while IFS= read -r line || [ -n "$line" ]; do
    line="${line%%#*}"
    case "$line" in
      export\ *) line="${line#export }" ;;
    esac
    case "$line" in *=*) ;; *) continue ;; esac
    key="${line%%=*}"
    key="${key#"${key%%[![:space:]]*}"}"
    case "$key" in *[!A-Za-z0-9_]*) continue ;; esac
    [ -n "$key" ] && eval "export $line"
  done < "$HOME/.env"
}

load_private_env

if ! ss -tlnp 2>/dev/null | grep -q ':7681'; then
  TTYD_FONT_SIZE="${LOGSEQ_TTYD_FONT_SIZE:-16}"
  TTYD_FONT_FAMILY="${LOGSEQ_TTYD_FONT_FAMILY:-SFMono Nerd Font, monospace}"
  # logseq-ttyd-index owns the patched index (Shift+Enter + websocket tracking).
  TTYD_INDEX="$("$HOME/.local/bin/logseq-ttyd-index")" || exit 1
  # Loopback only (-i lo): the tailnet face is the Caddy basic_auth gate
  # (services-web-terminal.org), never ttyd itself.
  setsid /home/metsatron/.local/bin/ttyd -W -i lo -p 7681 -I "$TTYD_INDEX" -t "fontSize=$TTYD_FONT_SIZE" -t "fontFamily=\"$TTYD_FONT_FAMILY\"" zsh </dev/null &>/dev/null &
  disown
fi
exec "$HOME/Applications/Logseq-linux-x86_64-latest.AppImage" --no-sandbox "$@"
#+END_SRC

** ttyd index builder

~logseq-ttyd-index~ builds and refreshes the patched ttyd index cache and prints
its path. It is the single owner of the index patch (Shift+Enter handling and
websocket tracking): both the =logseq-db= launcher and the Caddy web terminal
(=services-web-terminal.org=) call it. Never fork this logic into a second copy.

#+name: logseq-ttyd-index-script
#+BEGIN_SRC bash :tangle all/.local/bin/logseq-ttyd-index :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755) :comments no
# Builds the patched ttyd index (see ttyd Terminal Plugin) and prints its path.
set -euo pipefail

ensure_logseq_ttyd_index() {
  local ttyd_bin cache_dir index stamp build_stamp build_port log pid
  ttyd_bin="${LOGSEQ_TTYD_BIN:-/home/metsatron/.local/bin/ttyd}"
  cache_dir="${XDG_CACHE_HOME:-$HOME/.cache}/logseq-ttyd"
  index="$cache_dir/index.html"
  stamp="$cache_dir/ttyd.stamp"
  build_port="${LOGSEQ_TTYD_INDEX_PORT:-17681}"
  log="$cache_dir/index-build.log"

  mkdir -p "$cache_dir"
  build_stamp="$(stat -c '%Y:%s' "$ttyd_bin" 2>/dev/null || printf unknown)"
  if [ -s "$index" ] && [ -f "$stamp" ] && [ "$(cat "$stamp" 2>/dev/null)" = "$build_stamp" ]; then
    printf '%s\n' "$index"
    return 0
  fi

  "$ttyd_bin" -p "$build_port" zsh >"$log" 2>&1 &
  pid="$!"
  for _ in $(seq 1 40); do
    if curl -fsS "http://127.0.0.1:$build_port/" -o "$index.raw" 2>>"$log"; then
      break
    fi
    sleep 0.1
  done
  kill "$pid" >/dev/null 2>&1 || true
  wait "$pid" >/dev/null 2>&1 || true

  if [ ! -s "$index.raw" ]; then
    printf 'logseq-db: failed to build ttyd index; see %s\n' "$log" >&2
    return 1
  fi

  python3 - "$index.raw" "$index.tmp" <<'PY'
import pathlib
import sys

source = pathlib.Path(sys.argv[1])
target = pathlib.Path(sys.argv[2])
html = source.read_text(encoding="utf-8")
patch = r'''<script>
(() => {
  const NativeWebSocket = window.WebSocket;
  const sockets = new Set();
  const state = {
    shiftEnterCount: 0,
    lastShiftEnterAt: 0,
    lastOpenSocketCount: 0,
  };

  function track(socket) {
    sockets.add(socket);
    socket.addEventListener("close", () => sockets.delete(socket));
    socket.addEventListener("error", () => sockets.delete(socket));
    return socket;
  }

  const nativeSend = NativeWebSocket.prototype.send;
  NativeWebSocket.prototype.send = function(data) {
    sockets.add(this);
    return nativeSend.call(this, data);
  };

  window.WebSocket = new Proxy(NativeWebSocket, {
    construct(target, args) {
      return track(new target(...args));
    },
    apply(target, thisArg, args) {
      return track(target.apply(thisArg, args));
    },
  });

  function openSockets() {
    return Array.from(sockets).filter((socket) => socket.readyState === NativeWebSocket.OPEN);
  }

  function sendShiftEnter() {
    const open = openSockets();
    state.lastOpenSocketCount = open.length;
    if (!open.length) return false;
    state.shiftEnterCount += 1;
    state.lastShiftEnterAt = Date.now();
    for (const socket of open) socket.send("0\n");
    return true;
  }

  function isShiftEnter(event) {
    return event.shiftKey
      && !event.ctrlKey
      && !event.altKey
      && !event.metaKey
      && (event.key === "Enter" || event.code === "Enter" || event.code === "NumpadEnter");
  }

  document.addEventListener("keydown", (event) => {
    if (!isShiftEnter(event)) return;
    if (!sendShiftEnter()) return;
    event.preventDefault();
    event.stopImmediatePropagation();
  }, true);

  window.__logseqTtydShiftEnter = { sendShiftEnter, state, openSockets };
})();
</script>'''

needle = "</head><body><script"
if needle not in html:
    raise SystemExit("ttyd index shape changed: missing script insertion point")
html = html.replace(needle, patch + needle, 1)
target.write_text(html, encoding="utf-8")
PY

  mv "$index.tmp" "$index"
  printf '%s\n' "$build_stamp" >"$stamp"
  rm -f "$index.raw"
  printf '%s\n' "$index"
}

ensure_logseq_ttyd_index
#+END_SRC

* App Config

Logseq DB writes its settings to =~/.config/Logseq/configs.edn= via the UI.
That full config directory is private mutable app state and is owned by
=~/HelmCortex/NEXUS/stow/linux/.config/Logseq= via =helmstow=.

DotCortex must not tangle, stow, template, or migrate =~/.config/Logseq=.
Only the public launcher, HTTP API helper, and custom plugin source live here.
The Claude Code token lives outside DotCortex in =~/.env= as =LOGSEQ_API_TOKEN=
and in =~/.mcp.json=.

* HTTP API

=logseq-api= is a thin curl wrapper for the Logseq HTTP API (Plugin SDK method proxy).
The server runs on port 12315 when =:server/mcp-enabled?= is true in configs.edn.
Token auth reads from =$LOGSEQ_API_TOKEN= in the environment (sourced from =~/.env=).

The MCP endpoint at =http://127.0.0.1:12315/mcp= is wired into Claude Code via
=~/.mcp.json= (outside DotCortex).

#+BEGIN_SRC bash :tangle all/.local/bin/logseq-api :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755) :comments no

TOKEN="${LOGSEQ_API_TOKEN:-}"
TOKEN_NAME="${LOGSEQ_API_NAME:-claude-code}"

usage() {
  cat >&2 <<'USAGE'
Usage: logseq-api check
       logseq-api METHOD [ARGS_JSON]

Examples:
  logseq-api check
  logseq-api logseq.Editor.getAllPages
  logseq-api logseq.Editor.getPage '["MyPage"]'
  logseq-api logseq.Editor.getPageBlocksTree '["MyPage"]'

* ttyd Terminal Plugin

Custom Logseq plugin embedding ttyd (web PTY server) as a pinned right-side panel.
Tangled directly to =~/.logseq/plugins/logseq-ttyd/= — not via stow, since Logseq
manages =~/.logseq/= and the plugin directory must remain writable by the app.

=lsplugin.user.js= is the Logseq Plugin SDK runtime (~90KB, distributed with the
plugin template). It is not authored — keep the copy at
=~/.logseq/plugins/logseq-ttyd/lsplugin.user.js= in place and do not tangle it.

The ttyd server is started by the =logseq-db= launcher before Logseq loads.

To iterate on the plugin: edit this file, then =loom logseq:tangle= or
=tangle-one logseq.org=. Changes land in =~/.logseq/plugins/logseq-ttyd/=
immediately; reload the plugin in Logseq (Settings -> Plugins -> disable/enable).

** index.html

Main plugin entry point. Injects a pinned right-edge iframe via the Plugin SDK.

CSS strategy: =.lsp-ui-float-container= injects =.resizable-handle= and =.draggable-handle=
children that sit above the iframe and eat pointer events. Using
=#logseq-ttyd--ttyd-panel.lsp-ui-float-container .resizable-handle= (specificity 1,2,0)
beats Logseq's internal rules; =display: none= removes the elements from hit-testing
entirely as a belt-and-suspenders.

#+name: logseq-ttyd-index
#+BEGIN_SRC html :tangle ~/.logseq/plugins/logseq-ttyd/index.html :comments no
<!DOCTYPE html>
<html>
<head><meta charset="UTF-8"></head>
<body>
<script src="./lsplugin.user.js"></script>
<script>
const TTYD_URL = 'http://localhost:7681';

const settingsSchema = [
  {
    key: 'terminalUrl',
    type: 'string',
    title: 'Terminal URL',
    description: 'URL loaded in the terminal iframe.',
    default: TTYD_URL,
  },
  {
    key: 'panelWidth',
    type: 'number',
    title: 'Panel width',
    description: 'Pinned terminal panel width in pixels.',
    default: 768,
  },
  {
    key: 'toolbarHeight',
    type: 'number',
    title: 'Toolbar offset',
    description: 'Top offset in pixels for the pinned terminal panel.',
    default: 40,
  },
  {
    key: 'startVisible',
    type: 'boolean',
    title: 'Show terminal on startup',
    description: 'Open the terminal panel when Logseq loads.',
    default: true,
  },
  {
    key: 'hideIframeScrollbar',
    type: 'boolean',
    title: 'Hide iframe scrollbar',
    description: 'Hide scrollbars around the embedded iframe wrapper.',
    default: true,
  },
  {
    key: 'fontSize',
    type: 'number',
    title: 'Terminal font size',
    description: 'Font size used by ttyd when launched from logseq-db.',
    default: 16,
  },
  {
    key: 'fontFamily',
    type: 'string',
    title: 'Terminal font family',
    description: 'Font family used by ttyd when launched from logseq-db.',
    default: 'SFMono Nerd Font, monospace',
  },
  {
    key: 'customKeyBindings',
    type: 'string',
    title: 'Custom key bindings',
    description: 'JSON object for terminal keybinding intent. Shift+Enter defaults to LF.',
    default: '{"Shift+Enter":"\\n"}',
  },
];

logseq.useSettingsSchema(settingsSchema);

const TTYD_IFRAME_VERSION = 'shift-enter-v3';
let terminalVisible = true;
let styleVersion = 0;

function setting(key, fallback) {
  const value = logseq.settings && logseq.settings[key];
  return value === undefined || value === null || value === '' ? fallback : value;
}

function px(value, fallback) {
  const parsed = Number(value);
  return Number.isFinite(parsed) && parsed > 0 ? `${parsed}px` : `${fallback}px`;
}

function terminalSrc(url) {
  try {
    const parsed = new URL(url);
    parsed.searchParams.set('logseqTtyd', TTYD_IFRAME_VERSION);
    return parsed.toString();
  } catch (_error) {
    return url;
  }
}

async function renderTerminalPanel() {
  const panelWidth = px(setting('panelWidth', 768), 768);
  const toolbarHeight = px(setting('toolbarHeight', 40), 40);
  const terminalUrl = terminalSrc(String(setting('terminalUrl', TTYD_URL)));
  const hideScrollbar = setting('hideIframeScrollbar', true) !== false;
  const overflowMode = hideScrollbar ? 'hidden' : 'auto';
  const displayMode = terminalVisible ? 'block' : 'none';
  const version = ++styleVersion;

  await logseq.provideStyle(`
    /* logseq-ttyd dynamic style version ${version} */
    div#logseq-ttyd_lsp_main,
    iframe#logseq-ttyd_iframe {
      pointer-events: none !important;
    }

    #logseq-ttyd--ttyd-panel.lsp-ui-float-container .resizable-handle,
    #logseq-ttyd--ttyd-panel.lsp-ui-float-container .draggable-handle {
      display: none !important;
      pointer-events: none !important;
    }

    #logseq-ttyd--ttyd-panel.lsp-ui-float-container {
      display: ${displayMode} !important;
      position: fixed !important;
      right: 0 !important;
      left: auto !important;
      top: ${toolbarHeight} !important;
      width: ${panelWidth} !important;
      height: calc(100vh - ${toolbarHeight}) !important;
      z-index: 100 !important;
      margin: 0 !important;
      padding: 0 !important;
      transform: none !important;
      border-radius: 0 !important;
      box-shadow: none !important;
      overflow: ${overflowMode} !important;
      pointer-events: all !important;
      scrollbar-width: none !important;
    }

    #logseq-ttyd--ttyd-panel.lsp-ui-float-container::-webkit-scrollbar,
    #logseq-ttyd--ttyd-panel.lsp-ui-float-container *::-webkit-scrollbar {
      display: none !important;
    }

    #logseq-ttyd--ttyd-panel.lsp-ui-float-container > * {
      width: 100% !important;
      height: 100% !important;
      overflow: ${overflowMode} !important;
      pointer-events: all !important;
      scrollbar-width: none !important;
    }
  `);

  await logseq.provideUI({
    key: 'ttyd-panel',
    template: `<iframe src="${terminalUrl}" scrolling="${hideScrollbar ? 'no' : 'auto'}" style="width:100%;height:100%;border:none;display:block;pointer-events:all;overflow:${overflowMode};"></iframe>`,
    style: {
      width:    '100%',
      height:   '100%',
      overflow: overflowMode,
    }
  });
}

logseq.ready(async () => {
  terminalVisible = setting('startVisible', true) !== false;

  logseq.provideModel({
    async toggleTerminal() {
      terminalVisible = !terminalVisible;
      await renderTerminalPanel();
    },
    async showTerminalSettings() {
      logseq.showSettingsUI();
    },
  });

  logseq.App.registerUIItem('toolbar', {
    key: 'logseq-ttyd-toggle',
    template: `
      <a class="button" data-on-click="toggleTerminal" title="Terminal">
        <svg xmlns="http://www.w3.org/2000/svg" width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
          <polyline points="4 17 10 11 4 5"></polyline>
          <line x1="12" y1="19" x2="20" y2="19"></line>
        </svg>
      </a>
    `,
  });

  logseq.onSettingsChanged(async () => {
    await renderTerminalPanel();
  });

  await renderTerminalPanel();
}).catch(console.error);
</script>
</body>
</html>
