* Codex live config apply
Codex keeps credentials, state, marketplace cache paths, and machine-local
project trust entries under =~/.codex=. DotCortex should not own that whole
tree. This helper patches only the declarative TUI status-line list, the two
local HelmCortex MCP startup timeouts, the single DotCortex stow-target hook,
the workspace profile's network permission required by local async MCP stdio,
and the minimal =codex-warm= lifecycle hooks in the live private config. The
hook source itself is public and tangled from =agents-hooks.org=; only the
private =~/.codex/hooks.json= registration and the narrow trust state for that
exact audited idle hook are applied here. Herdr's installed
Codex session hook is deliberately removed: identity and PTY-derived state
belong to =codex-warm=, while these hooks only notify its internal compaction
state machine.

Codex's =PreToolUse= hook sees Bash and unified =exec_command= calls. It runs
the same target guard as Claude Code and =safe-stow=, so a mismatched
=loom stow:<machine>= is denied before the shell starts. Codex requires changed
command hooks to be reviewed/trusted in =/hooks= before they execute.

#+BEGIN_SRC bash :tangle all/.local/bin/codex-config-apply :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

config="${CODEX_CONFIG:-$HOME/.codex/config.toml}"
desired='status_line = ["model-with-reasoning", "current-dir", "context-used", "five-hour-limit", "weekly-limit", "context-window-size"]'

apply_codex_stow_hook() {
  local hooks_config guard_command hooks_input hooks_new hooks_backup
  hooks_config="${CODEX_HOOKS:-$HOME/.codex/hooks.json}"
  guard_command="${DOTCORTEX_STOW_GUARD:-$HOME/.local/bin/dotcortex-stow-target-guard}"
  [ -x "$guard_command" ] || guard_command="$HOME/DotCortex/all/.local/bin/dotcortex-stow-target-guard"
  [ -x "$guard_command" ] || {
    echo "codex-config-apply: missing dotcortex-stow-target-guard" >&2
    return 1
  }
  command -v jq >/dev/null 2>&1 || {
    echo "codex-config-apply: jq is required to update $hooks_config" >&2
    return 1
  }

  mkdir -p "$(dirname "$hooks_config")"
  hooks_input="$(mktemp)"
  hooks_new="$(mktemp)"
  if [ -f "$hooks_config" ]; then
    cp -f "$hooks_config" "$hooks_input"
  else
    printf '{"hooks":{}}\n' >"$hooks_input"
  fi

  local idle_hook="${CODEX_IDLE_HOOK:-$HOME/.local/bin/codex-hook-idle-event}"
  [ -x "$idle_hook" ] || idle_hook="$HOME/DotCortex/all/.local/bin/codex-hook-idle-event"
  local recall_hook="${CODEX_RECALL_HOOK:-$HOME/.local/bin/codex-hook-recall-context}"
  [ -x "$recall_hook" ] || recall_hook="$HOME/DotCortex/all/.local/bin/codex-hook-recall-context"

  jq --arg command "$guard_command" --arg idle_hook "$idle_hook" --arg recall_hook "$recall_hook" '
    .hooks = (.hooks // {}) |
    .hooks |= with_entries(
      .value = [
        .value[]?
        | .hooks = [(.hooks[]? | select((.command // "") != $recall_hook))]
        | select((.hooks | length) > 0)
      ]
    ) |
    # Remove the separate Herdr Codex lifecycle/session registration.  The
    # supervisor registers identity and derives visible state from PTY output;
    # this private hook must not compete with that ownership boundary.
    .hooks |= with_entries(
      .value = [
        .value[]?
        | .hooks = [
            (.hooks[]? | select(
              ((.type // "") != "command") or
              (((.command // "") | contains("herdr-agent-state.sh")) | not)
            ))
          ]
        | select((.hooks | length) > 0)
      ]
    ) |
    .hooks.PreToolUse = (.hooks.PreToolUse // []) |
    if any(.hooks.PreToolUse[]?;
      ((.matcher // "") == "^Bash$" or (.matcher // "") == "Bash") and
      any(.hooks[]?; .type == "command" and .command == ($command + " --codex-json")))
    then .
    else .hooks.PreToolUse += [{
      "matcher": "^Bash$",
      "hooks": [{
        "type": "command",
        "command": ($command + " --codex-json"),
        "timeout": 30,
        "statusMessage": "Checking DotCortex stow target"
      }]
    }]
    end |

    # Minimal lifecycle hooks for codex-warm internal idle-compaction state
    def ensure_idle_hook($event; $idle_hook; $timeout):
      .hooks[$event] = (.hooks[$event] // []) |
      if any(.hooks[$event][]?; any(.hooks[]?; .type == "command" and .command == $idle_hook))
      then .hooks[$event] |= map(
        .hooks |= map(
          if .type == "command" and .command == $idle_hook
          then .timeout = $timeout
          else .
          end
        )
      )
      else .hooks[$event] += [{"hooks": [{"type": "command", "command": $idle_hook, "timeout": $timeout}]}]
      end;

    ensure_idle_hook("Stop"; $idle_hook; 5) |
    ensure_idle_hook("PostCompact"; $idle_hook; 5) |
    ensure_idle_hook("UserPromptSubmit"; $idle_hook; 5) |
    ensure_idle_hook("PreCompact"; $idle_hook; 5) |
    ensure_idle_hook("SessionEnd"; $idle_hook; 3) |
    ensure_idle_hook("SessionStart"; $idle_hook; 5)
  ' "$hooks_input" >"$hooks_new"

  if cmp -s "$hooks_new" "$hooks_config" 2>/dev/null; then
    rm -f "$hooks_input" "$hooks_new"
    echo "codex-config-apply: stow hook already current"
    return 0
  fi
  if [ -f "$hooks_config" ]; then
    hooks_backup="${hooks_config}.backup-$(date +%Y%m%d-%H%M%S)"
    cp -a "$hooks_config" "$hooks_backup"
    echo "codex-config-apply: hook backup $hooks_backup"
  fi
  cat "$hooks_new" >"$hooks_config"
  chmod 600 "$hooks_config"
  rm -f "$hooks_input" "$hooks_new"
  echo "codex-config-apply: installed minimal codex-warm hooks in $hooks_config"
}

apply_codex_stow_hook

enable_codex_warm_trust() {
  local hooks_config idle_hook
  hooks_config="${CODEX_HOOKS:-$HOME/.codex/hooks.json}"
  idle_hook="${CODEX_IDLE_HOOK:-$HOME/.local/bin/codex-hook-idle-event}"
  [ -x "$idle_hook" ] || idle_hook="$HOME/DotCortex/all/.local/bin/codex-hook-idle-event"
  [ -f "$hooks_config" ] || return 0
  command -v python3 >/dev/null 2>&1 || {
    echo "codex-config-apply: python3 is required to reconcile Codex hook trust" >&2
    return 1
  }

  python3 - "$hooks_config" "$config" "$idle_hook" <<'PY'
import json
import os
import re
import shutil
import sys
import tempfile
import time
from pathlib import Path

hooks_path = Path(sys.argv[1]).expanduser()
config_path = Path(sys.argv[2]).expanduser()
idle_hook = os.path.abspath(os.path.expanduser(sys.argv[3]))
hooks_key_path = str(hooks_path if hooks_path.is_absolute() else hooks_path.resolve())

try:
    hooks = json.loads(hooks_path.read_text())
except (OSError, ValueError) as exc:
    print(f"codex-config-apply: cannot read {hooks_path}: {exc}", file=sys.stderr)
    raise SystemExit(1)

def event_key(name):
    return re.sub(r"(?<!^)(?=[A-Z])", "_", name).lower()

idle_keys = set()
for event, groups in (hooks.get("hooks") or {}).items():
    for group_index, group in enumerate(groups or []):
        for hook_index, hook in enumerate(group.get("hooks") or []):
            if hook.get("type") == "command" and os.path.abspath(os.path.expanduser(hook.get("command", ""))) == idle_hook:
                idle_keys.add(f"{hooks_key_path}:{event_key(event)}:{group_index}:{hook_index}")

if not idle_keys or not config_path.exists():
    print("codex-config-apply: no existing disabled codex-warm trust state to reconcile")
    raise SystemExit(0)

text = config_path.read_text()
header = re.compile(r'^\[hooks\.state\."([^"]+)"\]\s*$', re.MULTILINE)
enabled_false = re.compile(r'^(\s*enabled\s*=\s*)false(\s*(?:#.*)?)$', re.MULTILINE)
lines = text.splitlines(keepends=True)
current = None
changed = 0
out = []
for line in lines:
    match = header.match(line.rstrip("\r\n"))
    if match:
        current = match.group(1)
    if current in idle_keys and enabled_false.match(line.rstrip("\r\n")):
        newline = "\r\n" if line.endswith("\r\n") else "\n" if line.endswith("\n") else ""
        body = line[:-len(newline)] if newline else line
        line = enabled_false.sub(r"\1true\2", body) + newline
        changed += 1
    out.append(line)

if not changed:
    print("codex-config-apply: codex-warm trust already enabled (or not yet recorded)")
    raise SystemExit(0)

backup = config_path.with_name(f"{config_path.name}.backup-{time.strftime('%Y%m%d-%H%M%S')}")
if backup.exists():
    backup = config_path.with_name(f"{backup.name}.{os.getpid()}")
shutil.copy2(config_path, backup)
fd, temp_name = tempfile.mkstemp(prefix=f".{config_path.name}.", dir=config_path.parent)
try:
    with os.fdopen(fd, "w") as stream:
        stream.write("".join(out))
    os.chmod(temp_name, config_path.stat().st_mode & 0o777)
    os.replace(temp_name, config_path)
finally:
    try:
        os.unlink(temp_name)
    except FileNotFoundError:
        pass
print(f"codex-config-apply: enabled {changed} audited codex-warm trust entr{'y' if changed == 1 else 'ies'}")
print(f"codex-config-apply: trust backup {backup}")
PY
}

enable_codex_warm_trust

mkdir -p "$(dirname "$config")"

# Codex permission profiles supersede the legacy sandbox_workspace_write table.
# Extend :workspace so .git and .agents retain their built-in protection, then
# reopen only DotCortex's project-local .codex tree.  That narrow exception lets
# Codex tangle its own harness-specific skills without granting broad filesystem
# access.  Existing HelmCortex roots remain additional workspaces when present.
[ -f "$config" ] || : >"$config"
profile_name=dotcortex-workspace
dotcortex_root="${DOTCORTEX_ROOT:-$HOME/DotCortex}"
if resolved_root="$(timeout 2 realpath "$dotcortex_root" 2>/dev/null)"; then
  dotcortex_root="$resolved_root"
fi

profile_roots=()
for candidate in "$HOME/HelmCortex" "$HOME/mnt/t1700/HelmCortex"; do
  timeout 2 test -d "$candidate" 2>/dev/null || continue
  resolved_root="$(timeout 2 realpath "$candidate" 2>/dev/null)" || continue
  duplicate=0
  for existing_root in "${profile_roots[@]}"; do
    if [ "$existing_root" = "$resolved_root" ]; then
      duplicate=1
      break
    fi
  done
  [ "$duplicate" -eq 1 ] || profile_roots+=("$resolved_root")
done
unset candidate duplicate existing_root resolved_root

toml_quote() {
  local value=$1
  value=${value//\\/\\\\}
  value=${value//\"/\\\"}
  printf '"%s"' "$value"
}

body="$(mktemp)"
tmp="$(mktemp)"
trap 'rm -f "$body" "$tmp"' EXIT

awk -v desired="$desired" -v profile="$profile_name" \
  -v mcp_timeout="startup_timeout_sec = 60" \
  -v recall_approval="default_tools_approval_mode = \"approve\"" '
  BEGIN {
    in_tui = 0
    seen_tui = 0
    wrote_status = 0
    in_slow_mcp = 0
    wrote_mcp_timeout = 0
    in_recall_mcp = 0
    wrote_recall_approval = 0
    section = ""
    skip_section = 0
    pending_blanks = 0
  }

  function is_profile_section(line, prefix) {
    prefix = "[permissions." profile
    return index(line, prefix "]") == 1 || index(line, prefix ".") == 1
  }

  function flush_blanks(i) {
    for (i = 0; i < pending_blanks; i++) {
      print ""
    }
    pending_blanks = 0
  }

  /^\[[^]]+\]$/ {
    if (in_tui && !wrote_status) {
      flush_blanks()
      print desired
      wrote_status = 1
    }
    if (in_slow_mcp && !wrote_mcp_timeout) {
      print mcp_timeout
      wrote_mcp_timeout = 1
    }
    if (in_recall_mcp && !wrote_recall_approval) {
      print recall_approval
      wrote_recall_approval = 1
    }

    in_tui = 0
    in_slow_mcp = 0
    in_recall_mcp = 0
    if ($0 == "[sandbox_workspace_write]" || is_profile_section($0)) {
      section = $0
      skip_section = 1
      pending_blanks = 0
      next
    }

    skip_section = 0
    section = $0
    in_tui = ($0 == "[tui]")
    in_slow_mcp = ($0 == "[mcp_servers.recall]" || $0 == "[mcp_servers.timeark]")
    in_recall_mcp = ($0 == "[mcp_servers.recall]")
    wrote_mcp_timeout = 0
    wrote_recall_approval = 0
    if (in_tui) {
      seen_tui = 1
    }
    flush_blanks()
    print
    next
  }

  skip_section { next }

  /^[[:space:]]*$/ {
    pending_blanks++
    next
  }

  section == "" && /^[[:space:]]*(default_permissions|sandbox_mode)[[:space:]]*=/ {
    next
  }

  in_tui && /^[[:space:]]*status_line[[:space:]]*=/ {
    flush_blanks()
    if (!wrote_status) {
      print desired
      wrote_status = 1
    }
    next
  }

  in_slow_mcp && /^[[:space:]]*startup_timeout_sec[[:space:]]*=/ {
    flush_blanks()
    if (!wrote_mcp_timeout) {
      print mcp_timeout
      wrote_mcp_timeout = 1
    }
    next
  }

  in_recall_mcp && /^[[:space:]]*default_tools_approval_mode[[:space:]]*=/ {
    flush_blanks()
    if (!wrote_recall_approval) {
      print recall_approval
      wrote_recall_approval = 1
    }
    next
  }

  {
    flush_blanks()
    print
  }

  END {
    if (!seen_tui) {
      flush_blanks()
      print ""
      print "[tui]"
      print desired
    } else if (in_tui && !wrote_status) {
      flush_blanks()
      print desired
    }
    if (in_slow_mcp && !wrote_mcp_timeout) {
      print mcp_timeout
    }
    if (in_recall_mcp && !wrote_recall_approval) {
      print recall_approval
    }
  }
' "$config" >"$body"

{
  printf 'default_permissions = "%s"\n' "$profile_name"
  cat "$body"
  printf '\n[permissions.%s]\n' "$profile_name"
  printf 'description = "Workspace editing with DotCortex Codex skill authoring."\n'
  printf 'extends = ":workspace"\n'

  if [ "${#profile_roots[@]}" -gt 0 ]; then
    printf '\n[permissions.%s.workspace_roots]\n' "$profile_name"
    for existing_root in "${profile_roots[@]}"; do
      printf '%s = true\n' "$(toml_quote "$existing_root")"
    done
  fi

  printf '\n[permissions.%s.network]\n' "$profile_name"
  printf 'enabled = true\n'

  printf '\n[permissions.%s.filesystem]\n' "$profile_name"
  printf '%s = "write"\n' "$(toml_quote "$dotcortex_root/.codex")"
  # MailCortex outbound lane: the Codex adapters (HelmAstra, HelmDuctor) deposit
  # mail into recipient maildirs under ~/Mail/consorts (atomic tmp -> new), so
  # grant that one subtree write -- never ~/Mail itself, never $HOME.  Sealed
  # 2026-10-20 (Admiral): a coordinator that can build but cannot write its own
  # MailCortex receipt is a broken membrane.
  printf '%s = "write"\n' "$(toml_quote "$HOME/Mail/consorts")"
} >"$tmp"

if cmp -s "$tmp" "$config"; then
  rm -f "$body" "$tmp"
  trap - EXIT
  echo "codex-config-apply: already current"
  exit 0
fi

backup="${config}.backup-$(date +%Y%m%d-%H%M%S)"
if [ -e "$backup" ]; then
  backup="${backup}.$$"
fi
cp -a "$config" "$backup"
cat "$tmp" >"$config"
rm -f "$body" "$tmp"
trap - EXIT
chmod 600 "$config"
echo "codex-config-apply: updated $config"
echo "codex-config-apply: backup $backup"
#+END_SRC
