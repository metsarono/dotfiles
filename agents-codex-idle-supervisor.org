* Design notes

This launcher is the owner for ordinary interactive Codex CLI processes. It uses Python's =pty.fork= so the Codex child receives a real controlling terminal while the parent owns the PTY master for the entire process lifetime. The parent proxies bytes in both directions, propagates terminal geometry and signals, and is the only component permitted to write the automatic =/compact= submission. Codex hook helpers communicate with the per-session Unix socket; they never write to a terminal.

The default values match the Emacs governor heuristic: 27 minutes idle before compaction, a 30-minute warm-cache TTL, and an absolute 70,000 current input-context threshold. Automatic compaction is fail-closed when =current_tokens= is unknown: a timer alone is not evidence that the context is large enough to compact. The environment variables documented in =--help= are the supervisor's configurable surface.

This is a standalone adaptation of =claude-warm='s architecture. It reuses the patterns (PTY supervisor, lifecycle socket, idle timer, compaction gate, Herdr integration) but is its own file. The two supervisors diverge in: channel prefix (=codex-= vs =claude-=), environment namespace (=CODEX_IDLE_*= vs =CLAUDE_IDLE_*=), Herdr agent label, absence of transcript inotify, absence of cache profiles, and absence of channel guard.

Herdr identity is registered from the supervisor, but Codex lifecycle hooks do not
own Herdr state. The supervisor's PTY proxy copies every child byte unchanged to
the outer pane and a small screen-stream classifier mirrors the installed Codex
detector's visible spinner, prompt, and blocker evidence. That adapter is only a
fallback for Herdr's process detector: it is driven by PTY output, never by the
Codex lifecycle socket or hook events. This matters because Herdr sees the outer
Python supervisor as the foreground process while the nested Codex child owns the
inner PTY.


Resolver output is untrusted: only exact endpoint/model identity, supported
confidence and positive ordered integer windows select measured timing.
Malformed output retains this consumer's fixed fallback. Automatic actions
refresh the cache window at their final gate; a later deadline preserves the
idle window. Launch arguments and disk configuration are declarations, not
observations of the active session. The owning lifecycle hook exposes neither
current model nor endpoint, so the runtime identity remains unknown and uses
fixed fallback timing. Extra socket fields cannot supply that missing evidence.
Measured runtime timing requires a separately verified owning observation surface;
this change does not add one. Custom endpoint identity is never inferred.
Observed evidence requires the explicit =CODEX_CACHE_POLICY_ALLOW_OBSERVED= opt-in.

def configured_cache_identity(argv: list[str]) -> tuple[str, str]:
    """Read declared selection; unrecognized/custom endpoints stay unsupported."""
    config_root = pathlib.Path(os.environ.get("CODEX_HOME", str(pathlib.Path.home() / ".codex"))).expanduser()
    try:
        config = tomllib.loads((config_root / "config.toml").read_text(encoding="utf-8"))
    except FileNotFoundError:
        config = {}
    except (OSError, ValueError):
        return "unknown", "unknown"
    selected = dict(config)
    overrides = []
    launch_model = None
    endpoint_known = not os.environ.get("OPENAI_BASE_URL")
    for index, value in enumerate(argv):
        if value.startswith("--model="):
            launch_model = value.split("=", 1)[1]
        elif value in {"--model", "-m"} and index + 1 < len(argv):
            launch_model = argv[index + 1]
        elif value in {"--profile", "-p"} or value.startswith("--profile="):
            profile = value.split("=", 1)[1] if "=" in value else argv[index + 1] if index + 1 < len(argv) else ""
            profiles = config.get("profiles", {})
            row = profiles.get(profile) if isinstance(profiles, dict) else None
            if not isinstance(row, dict):
                return "unknown", "unknown"
            selected.update(row)
        elif value in {"--config", "-c"} and index + 1 < len(argv):
            overrides.append(argv[index + 1])
        elif value.startswith("--config="):
            overrides.append(value.split("=", 1)[1])
        elif value in {"--oss", "--local-provider"} or value.startswith("--local-provider="):
            endpoint_known = False
    for override in overrides:
        try:
            row = tomllib.loads(override)
        except ValueError:
            return "unknown", "unknown"
        if any(key not in {"model", "model_provider", "model_reasoning_effort"} for key in row):
            endpoint_known = False
        selected.update(row)
    provider = selected.get("model_provider", "openai")
    providers = config.get("model_providers", {})
    if provider != "openai" or not isinstance(providers, dict) or "openai" in providers:
        endpoint_known = False
    model = launch_model or os.environ.get("CODEX_IDLE_MODEL") or selected.get("model")
    if not isinstance(model, str) or not model.strip() or model.strip() != model:
        model = "unknown"
    return model, "openai-codex-oauth" if endpoint_known else "unknown"


def configured_model(argv: list[str]) -> str:

def validated_cache_decision(decision: object, provider: str, endpoint: str,
                             model: str, allow_observed: bool = False) -> dict:
    """Accept only the resolver's exact identity and ordered integer window."""
    if not isinstance(decision, dict):
        raise ValueError("cache-policy-not-object")
    mode, confidence = decision.get("mode"), decision.get("confidence")
    if mode == "fallback" and confidence == "fallback":
        # The caller owns fallback timing; stdout cannot redefine its profile.
        return decision
    if not (mode == "measured" and confidence == "supported" or
            allow_observed and mode == "observed" and confidence == "observed"):
        raise ValueError("cache-policy-confidence-invalid")
    def canonical(value: str) -> str:
        return value.strip().lower().replace("_", "-")
    expected_provider = canonical(provider)
    expected_provider = {"openai-codex": "openai",
                         "openai-codex-oauth": "openai"}.get(expected_provider, expected_provider)
    expected_endpoint = canonical(endpoint)
    if expected_provider == "openai" and expected_endpoint in {"openai", "openai-codex", "openai-codex-oauth"}:
        expected_endpoint = "openai-codex-oauth"
    expected_model = canonical(model)
    if expected_provider == "anthropic":
        expected_model = re.sub(r"-+", "-", re.sub(r"[ .]+", "-", expected_model)).strip("-")
        if not expected_model.startswith("claude-"):
            expected_model = "claude-" + expected_model
    for key, expected in (("provider", expected_provider), ("endpoint_id", expected_endpoint), ("model", expected_model)):
        if decision.get(key) != expected:
            raise ValueError("cache-policy-identity-mismatch")
    compact, expiry = decision.get("compact_before_seconds"), decision.get("cache_expires_seconds")
    if (type(compact) is not int or type(expiry) is not int or
            not 0 < compact < expiry or expiry > 2**31 - 1):
        raise ValueError("cache-policy-window-invalid")
    return decision


def measured_window(model: str, endpoint: str = "openai-codex-oauth") -> tuple[int, int, dict]:
    if model == "unknown" or endpoint == "unknown":
        return IDLE_SECONDS, TTL_SECONDS, {"mode": "fallback", "reason": "identity-unknown"}
    helper = pathlib.Path(os.environ.get(
        "FLEET_CACHE_POLICY_HELPER", str(pathlib.Path.home() / "HelmCortex/FORGE/bin/cache-policy")
    )).expanduser()
    policy = pathlib.Path(os.environ.get(
        "FLEET_CACHE_POLICY_PATH",
        str(pathlib.Path.home() / "HelmCortex/ROOTS/DotRoot/agent-session/centre/data/cache-policy.json"),
    )).expanduser()
    if not helper.is_file() or not policy.is_file():
        return IDLE_SECONDS, TTL_SECONDS, {
            "mode": "fallback",
            "reason": "policy-unavailable",
        }
    try:
        command = [str(helper), "resolve", "--policy", str(policy), "--harness", "codex",
                   "--provider", "openai", "--endpoint", endpoint,
                   "--model", model, "--fallback-compact", str(IDLE_SECONDS),
                   "--fallback-cache", str(TTL_SECONDS)]
        if ALLOW_OBSERVED_POLICY:
            command.append("--allow-observed")
        completed = subprocess.run(
            command,
            capture_output=True, text=True, timeout=5, check=True,
        )
        decision = validated_cache_decision(json.loads(completed.stdout), "openai", endpoint, model, ALLOW_OBSERVED_POLICY)
        if decision["mode"] == "fallback":
            return IDLE_SECONDS, TTL_SECONDS, decision
        return decision["compact_before_seconds"], decision["cache_expires_seconds"], decision
    except (OSError, TypeError, ValueError, KeyError, OverflowError, subprocess.SubprocessError):
        return IDLE_SECONDS, TTL_SECONDS, {"mode": "fallback", "reason": "resolver-error"}

def atomic_json(path: pathlib.Path, payload: dict) -> None:
    temporary = path.with_name(f".{path.name}.tmp.{os.getpid()}")
    data = json.dumps(payload, sort_keys=True, separators=(",", ":")) + "\n"
    fd = os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o600)
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as stream:
            stream.write(data)
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temporary, path)
    finally:
        try:
            temporary.unlink()
        except FileNotFoundError:
            pass

def resolve_codex() -> str:
    explicit = os.environ.get("CODEX_IDLE_REAL_EXECUTABLE")
    candidates = [explicit] if explicit else []
    found = shutil.which("codex")
    if found:
        candidates.append(found)
    candidates.extend(
        [
            str(pathlib.Path.home() / ".local/bin/codex"),
            str(pathlib.Path.home() / ".npm-global/bin/codex"),
            str(pathlib.Path.home() / ".cargo/bin/codex"),
        ]
    )
    wrapper = pathlib.Path(__file__).resolve()
    for candidate in candidates:
        if not candidate:
            continue
        path = pathlib.Path(candidate).expanduser()
        try:
            resolved = path.resolve()
        except OSError:
            continue
        if resolved == wrapper or not resolved.is_file() or not os.access(resolved, os.X_OK):
            continue
        if resolved.name in {"codex-warm", "codex-warmctl"}:
            continue
        return str(resolved)
    raise RuntimeError("real Codex executable not found; set CODEX_IDLE_REAL_EXECUTABLE")

def write_all(fd: int, data: bytes) -> None:
    offset = 0
    while offset < len(data):
        try:
            written = os.write(fd, data[offset:])
        except InterruptedError:
            continue
        except BlockingIOError:
            time.sleep(0.002)
            continue
        if written <= 0:
            raise OSError(errno.EIO, "PTY write returned zero bytes")
        offset += written


class TerminalInputClassifier:
    """Classify raw outer-terminal input without suppressing any forwarded byte."""

    PASTE_START = b"\x1b[200~"
    PASTE_END = b"\x1b[201~"

    def __init__(self, escape_timeout: float = INPUT_ESCAPE_TIMEOUT_SECONDS):
        self.escape_timeout = escape_timeout
        self.buffer = bytearray()
        self.in_bracketed_paste = False
        self.pending_since: float | None = None

    @staticmethod
    def _event(kind: str, length: int, human: bool) -> tuple[str, int, bool]:
        return kind, length, human

    def feed(self, data: bytes) -> list[tuple[str, int, bool]]:
        if not data:
            return []
        self.buffer.extend(data)
        events: list[tuple[str, int, bool]] = []
        if self.in_bracketed_paste:
            end = bytes(self.buffer).find(self.PASTE_END)
            if end >= 0:
                if end:
                    events.append(self._event("bracketed-paste-payload", end, True))
                events.append(self._event("bracketed-paste-end", len(self.PASTE_END), False))
                del self.buffer[: end + len(self.PASTE_END)]
                self.in_bracketed_paste = False
            else:
                events.append(self._event("bracketed-paste-payload", len(self.buffer), True))
                self.buffer.clear()
            self.pending_since = None if not self.buffer else monotonic()
            return events
        raw = bytes(self.buffer)
        paste_start = raw.find(self.PASTE_START)
        if paste_start >= 0:
            if paste_start:
                events.append(self._event("keyboard-input", paste_start, True))
            events.append(self._event("bracketed-paste-start", len(self.PASTE_START), False))
            del self.buffer[: paste_start + len(self.PASTE_START)]
            self.in_bracketed_paste = True
            self.pending_since = None if not self.buffer else monotonic()
            return events
        esc_pos = raw.find(0x1B)
        if esc_pos < 0:
            length = len(self.buffer)
            self.buffer.clear()
            self.pending_since = None
            events.append(self._event("keyboard-input", length, True))
            return events
        if esc_pos > 0:
            events.append(self._event("keyboard-input", esc_pos, True))
            del self.buffer[:esc_pos]
        if len(self.buffer) >= 2:
            length = len(self.buffer)
            self.buffer.clear()
            self.pending_since = None
            events.append(self._event("escape-sequence", length, False))
        else:
            if self.pending_since is None:
                self.pending_since = monotonic()
        return events

    def flush_expired(self) -> list[tuple[str, int, bool]]:
        if not self.buffer or self.pending_since is None:
            return []
        if monotonic() - self.pending_since < self.escape_timeout:
            return []
        length = len(self.buffer)
        self.buffer.clear()
        self.pending_since = None
        self.in_bracketed_paste = False
        return [self._event("unknown-input", length, True)]


class CodexScreenClassifier:
    """Classify visible Codex PTY output without consuming or rewriting it."""

    SPINNER_RE = re.compile(r"(?:^| )[⠋⠙⠹⠸⠼⠴⠦⠧⠇⠏](?: |$)")
    OSC_TITLE_RE = re.compile(
        r"\x1b\](?:0|1|2);(.*?)(?:\x07|\x1b\\)", re.DOTALL
    )
    ANSI_RE = re.compile(
        r"\x1b(?:\[[0-?]*[ -/]*[@-~]|\][^\x07\x1b]*(?:\x07|\x1b\\)|[@-_])"
    )
    STRONG_BLOCKERS = (
        "press enter to confirm or esc to cancel",
        "enter to submit answer",
        "enter to submit all",
        "allow command?",
    )

    def __init__(self) -> None:
        self.raw = bytearray()
        self.osc_title = ""
        self.state: str | None = None

    def reset(self) -> None:
        self.raw.clear()
        self.osc_title = ""
        self.state = None

    def _visible_text(self) -> str:
        text = self.ANSI_RE.sub("", bytes(self.raw).decode("utf-8", "replace"))
        return "".join(character for character in text if character in "\n\r\t" or ord(character) >= 0x20)

    def _classify(self, visible: str) -> str:
        title = self.osc_title
        if "Action Required" in title:
            return "blocked"
        if self.SPINNER_RE.search(title):
            return "working"

        recent = visible[-HERDR_SCREEN_BYTES:]
        lowered = recent.lower()
        if any(marker in lowered for marker in self.STRONG_BLOCKERS):
            return "blocked"
        if "[y/n]" in lowered or "yes (y)" in lowered:
            return "blocked"
        if (
            "do you want to" in lowered
            and ("yes" in lowered or "❯" in recent)
        ) or (
            "would you like to" in lowered
            and ("yes" in lowered or "❯" in recent)
        ):
            return "blocked"

        lines = [line.strip() for line in recent.splitlines() if line.strip()]
        bottom = "\n".join(lines[-3:])
        if (
            re.search(r"^[•◦]\s+Working \([^)]*esc to interrupt\)(?: · .*)?$", bottom, re.MULTILINE)
            and "■ Conversation interrupted" not in bottom
        ):
            return "working"
        if title.strip():
            return "idle"
        return "unknown"

    def feed(self, data: bytes) -> str | None:
        if not data:
            return None
        self.raw.extend(data)
        if len(self.raw) > HERDR_SCREEN_BYTES:
            del self.raw[:-HERDR_SCREEN_BYTES]
        titles = list(self.OSC_TITLE_RE.finditer(bytes(self.raw).decode("utf-8", "replace")))
        if titles:
            self.osc_title = titles[-1].group(1)
        state = self._classify(self._visible_text())
        if state == self.state:
            return None
        self.state = state
        return state


class HerdrReporter:
    """Register identity and optionally report PTY-derived state through Herdr."""

    def __init__(self) -> None:
        socket_path = os.environ.get("HERDR_SOCKET_PATH", "")
        pane_id = os.environ.get("HERDR_PANE_ID", "")
        agent = os.environ.get("CODEX_WARM_HERDR_AGENT", "") or "Codex"
        self.socket_path = pathlib.Path(socket_path) if socket_path else None
        self.pane_id = pane_id if HERDR_PANE_RE.fullmatch(pane_id) else None
        self.agent = agent if HERDR_AGENT_RE.fullmatch(agent) else None
        self.enabled = (
            os.environ.get("HERDR_ENV") == "1"
            and self.socket_path is not None
            and self.socket_path.is_absolute()
            and len(str(self.socket_path)) < 100
            and self.pane_id is not None
            and self.agent is not None
        )
        self.source = HERDR_SOURCE
        self.sequence = 0
        self.registered = False
        self.last_report_at: float | None = None
        self.last_report_state: str | None = None
        self.last_error: str | None = None
        self.next_heartbeat = 0.0
        self.child_pid: int | None = None
        self.agent_session_id: str | None = None

    def state(self) -> dict[str, object]:
        return {
            "herdr_enabled": self.enabled,
            "herdr_pane_id": self.pane_id,
            "herdr_agent": self.agent,
            "herdr_source": self.source,
            "herdr_registered": self.registered,
            "herdr_last_report_at": self.last_report_at,
            "herdr_last_report_state": self.last_report_state,
            "herdr_last_error": self.last_error,
            "herdr_child_pid": self.child_pid,
            "herdr_agent_session_id": self.agent_session_id,
        }

    @staticmethod
    def _valid_session_id(value: object) -> str | None:
        if isinstance(value, str) and SESSION_RE.fullmatch(value):
            return value
        return None

    def _send(self, method: str, params: dict[str, object]) -> bool:
        if not self.enabled or self.socket_path is None:
            return False
        self.sequence += 1
        request = {
            "id": f"{self.source}:{self.sequence}",
            "method": method,
            "params": params,
        }
        try:
            with socket.socket(socket.AF_UNIX, socket.SOCK_STREAM) as connection:
                connection.settimeout(0.5)
                connection.connect(str(self.socket_path))
                connection.sendall((json.dumps(request, separators=(",", ":")) + "\n").encode())
                response = connection.recv(4096)
            if response:
                decoded = json.loads(response.decode("utf-8"))
                if isinstance(decoded, dict) and decoded.get("error"):
                    error = decoded["error"]
                    code = error.get("code") if isinstance(error, dict) else "request-failed"
                    self.last_error = str(code)[:96]
                    return False
            return True
        except (OSError, ValueError, UnicodeError) as error:
            self.last_error = type(error).__name__
            return False

    def _base(self, session_id: object = None) -> dict[str, object]:
        base = {
            "pane_id": self.pane_id,
            "source": self.source,
            "agent": self.agent,
        }
        safe_session_id = self._valid_session_id(session_id)
        if safe_session_id:
            base["agent_session_id"] = safe_session_id
        return base

    def _metadata(self) -> dict[str, object]:
        tokens = {
            "supervisor_pid": str(os.getpid()),
            "pty_owner": "codex-warm",
        }
        if isinstance(self.child_pid, int) and self.child_pid > 0:
            tokens["codex_pid"] = str(self.child_pid)
        return {
            "pane_id": self.pane_id,
            "source": self.source,
            "agent": self.agent,
            "display_agent": self.agent,
            "tokens": tokens,
            "ttl_ms": HERDR_REGISTRATION_TTL_MS,
        }

    def register(self, session_id: object = None) -> None:
        """Refresh identity/metadata; this path never sends a state update."""
        if not self.enabled:
            return
        safe_session_id = self._valid_session_id(session_id)
        base = self._base(safe_session_id)
        if safe_session_id and safe_session_id != self.agent_session_id:
            self._send("pane.report_agent_session", dict(base))
            self.agent_session_id = safe_session_id
        success = self._send("pane.report_metadata", self._metadata())
        self.registered = success or self.registered
        self.last_report_at = now()
        self.next_heartbeat = monotonic() + HERDR_HEARTBEAT_SECONDS

    def report_state(self, state: str, session_id: object = None) -> None:
        """Report only a state derived from PTY output, never from a hook event."""
        if not self.enabled or state not in {"idle", "working", "blocked", "unknown"}:
            return
        self.register(session_id)
        report = self._base(session_id)
        report["state"] = state
        self._send("pane.report_agent", report)
        self.last_report_state = state
        self.last_report_at = now()
        self.next_heartbeat = monotonic() + HERDR_HEARTBEAT_SECONDS

    def heartbeat(self, state: str | None = None, session_id: object = None) -> None:
        if self.enabled and monotonic() >= self.next_heartbeat:
            if state in {"idle", "working", "blocked", "unknown"}:
                self.report_state(state, session_id)
            else:
                self.register(session_id)

    def release(self) -> None:
        if not self.enabled or self.socket_path is None:
            return
        params = {
            "pane_id": self.pane_id,
            "source": self.source,
            "agent": self.agent,
        }
        if self._send("pane.release_agent", params):
            self.registered = False
        self.last_report_state = "released"
        self.last_report_at = now()


class Supervisor:
    """Own one Codex PTY, lifecycle socket, state file, and idle timer."""

    def __init__(self, argv: list[str], executable: str):

        self.executable = executable
        self.channel = "codex-" + uuid.uuid4().hex
        self.root = runtime_root()

        self.socket_path = self.session_dir / "control.sock"
        self.state_path = self.session_dir / "state.json"
        self.socket = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
        self.socket.bind(str(self.socket_path))
        os.chmod(self.socket_path, 0o600)
        self.socket.listen(8)
        self.socket.setblocking(False)
        self.selector = selectors.DefaultSelector()
        self.selector.register(self.socket, selectors.EVENT_READ, "server")
        self.pid: int | None = None
        self.master_fd: int | None = None
        self.child_tty: str | None = None
        self.child_status: int | None = None
        self.stdin_eof = False
        self.terminal_attributes = None
        self.stdin_flags: int | None = None
        self.session_id: str | None = None
        self.status = "starting"
        self.enabled = True
        self.dirty = False
        self.compacting = False
        self.compaction_requested_at: float | None = None
        self.compaction_ignore_next_stop = False
        self.current_tokens: int | None = None
        self._refresh_current_cache_window()
        self.last_normal_stop: float | None = None
        self.last_local_input: float | None = None
        self.last_local_input_classification: str | None = None
        self.last_lifecycle_activity: float | None = None
        self.input_classifier = TerminalInputClassifier()
        self.timer_deadline: float | None = None
        self.timer_kind: str | None = None
        self.timer_generation = 0
        self.last_cancellation_reason: str | None = None
        self.last_gate_rejection_reason: str | None = None
        self.last_injection_result: str | None = None
        self.last_state_transition = "supervisor-started"
        self.shutdown_requested = False
        self.child_exited_at: float | None = None
        self.herdr = HerdrReporter()
        self.screen_classifier = CodexScreenClassifier()
        self.screen_state: str | None = None
        self._write_state()

    def state(self) -> dict:
        return {
            "channel": self.channel,
            "pid": self.pid,
            "supervisor_pid": os.getpid(),
            "pty_master_owned": self.master_fd is not None,
            "pty_child_terminal": self.child_tty,
            "control_socket": str(self.socket_path),
            "session_id": self.session_id,
            "status": self.status,
            "current_input_context_tokens": self.current_tokens,
            "model": self.model,
            "endpoint_id": self.endpoint,
            "cache_policy": self.cache_policy,
            "dirty": self.dirty,
            "compacting": self.compacting,
            "compaction_ignore_next_stop": self.compaction_ignore_next_stop,
            "last_normal_stop": self.last_normal_stop,
            "last_local_input": self.last_local_input,
            "last_local_input_classification": self.last_local_input_classification,
            "last_lifecycle_activity": self.last_lifecycle_activity,
            "timer_deadline": self.timer_deadline,
            "timer_kind": self.timer_kind,
            "timer_generation": self.timer_generation,
            "last_cancellation_reason": self.last_cancellation_reason,
            "last_gate_rejection_reason": self.last_gate_rejection_reason,
            "last_injection_result": self.last_injection_result,
            "last_state_transition": self.last_state_transition,
            "debug_log": str(self.debug_log_path),
            "herdr_screen_state": self.screen_state,
            **self.herdr.state(),
        }

    def _debug(self, message: str) -> None:
        append_debug(self.debug_log_path, message)

    def _write_state(self) -> None:
        try:
            atomic_json(self.state_path, self.state())
        except OSError as error:
            self._debug(f"state-write-failed channel={short_id(self.channel)} error={error}")

    def _register_herdr(self) -> None:
        self.herdr.register(self.session_id)
        self._write_state()

    def _report_screen_state(self, state: str) -> None:
        self.screen_state = state
        self.herdr.report_state(state, self.session_id)
        self._write_state()

    def _bound_debug_log(self) -> None:
        try:
            size = self.debug_log_path.stat().st_size
            if size <= DEBUG_LOG_MAX_BYTES:
                return
            with self.debug_log_path.open("rb") as stream:
                stream.seek(-DEBUG_LOG_MAX_BYTES, os.SEEK_END)
                tail = stream.read()
            with self.debug_log_path.open("wb") as stream:
                stream.write(tail)
        except OSError:
            pass

    def cancel(self, reason: str) -> None:
        self.timer_deadline = None
        self.timer_kind = None
        self.timer_generation += 1
        self.last_cancellation_reason = reason
        self.last_state_transition = f"cancelled:{reason}"
        self._debug(f"cancelled:{reason} channel={short_id(self.channel)}")
        self._write_state()

    def arm(self, preserve_transition: bool = False) -> None:
        if not (
            self.enabled
            and self.dirty
            and self.status == "idle"
            and not self.compacting
            and self.last_normal_stop
            and child_alive(self.pid)
            and self.timer_deadline is None
        ):
            return
        self._refresh_current_cache_window()
        self.timer_generation += 1
        self.timer_deadline = self.last_normal_stop + self.compact_after_seconds
        self.timer_kind = "compact"
        if not preserve_transition:
            self.last_state_transition = "armed"
        self._debug(
            f"armed channel={short_id(self.channel)} "
            f"delay={self.compact_after_seconds} policy={self.cache_policy.get('mode')}"
        )
        self._write_state()

    def _refresh_current_cache_window(self) -> None:
        # The installed hook forwards lifecycle facts only. No authenticated
        # model/endpoint observation exists here; declarations must not guess it.
        self.model, self.endpoint = "unknown", "unknown"
        self.compact_after_seconds, self.cache_expires_seconds, self.cache_policy = measured_window(self.model, self.endpoint)

    def _refresh_cache_policy_for_action(self) -> bool:
        self._refresh_current_cache_window()
        deadline = self.last_normal_stop + self.compact_after_seconds if self.last_normal_stop else None
        if deadline is not None and deadline > now():
            self.timer_generation += 1
            self.timer_deadline = deadline
            self.timer_kind = "compact"
            self.last_state_transition = "policy-refreshed-rearmed"
            self._write_state()
            return False
        self._write_state()
        return True

    def _compaction_failed(self, reason: str) -> None:
        if not self.compacting:
            return
        self.compacting = False
        self.compaction_requested_at = None
        self.compaction_ignore_next_stop = True
        self.dirty = True
        self.last_cancellation_reason = "compaction-failed"
        self.last_injection_result = "failed"
        self.last_state_transition = "compaction-failed"
        self._debug(
            f"compaction-failed channel={short_id(self.channel)} reason={reason}"
        )
        self._write_state()

    def _recover_compaction(self, reason: str) -> None:
        """Clear a missing PostCompact and re-arm from a safe idle point."""
        if not self.compacting:
            return
        self._compaction_failed(reason)
        current = now()
        self.status = "idle"
        self.last_normal_stop = current
        self.last_lifecycle_activity = current
        self.last_local_input = None
        # The failure is already being handled at the Stop/timeout boundary;
        # do not consume the next genuine turn's Stop hook as a late failure.
        self.compaction_ignore_next_stop = False
        self.last_state_transition = "compaction-recovered"
        self.cancel("compaction-recovered")
        self.last_state_transition = "compaction-recovered"
        self.arm(preserve_transition=True)
        self._debug(
            f"compaction-recovered channel={short_id(self.channel)} reason={reason}"
        )
        self._write_state()

    def _check_compaction_timeout(self) -> None:
        if not self.compacting or not self.compaction_requested_at:
            return
        if now() - self.compaction_requested_at < COMPACTION_TIMEOUT_SECONDS:
            return
        # A visible spinner means Codex may still be compacting. Keep the
        # request pending and let its lifecycle hooks finish rather than
        # injecting a second command into an active TUI turn.
        if self.screen_state == "working":
            self.compaction_requested_at = now()
            self.last_state_transition = "compaction-timeout-working"
            self._debug(
                f"compaction-timeout-working channel={short_id(self.channel)}"
            )
            self._write_state()
            return
        self._recover_compaction("timeout")

    def _install_terminal(self) -> None:
        fd = sys.stdin.fileno()
        is_terminal = os.isatty(fd)
        self.stdin_flags = fcntl.fcntl(fd, fcntl.F_GETFL)
        if is_terminal:
            self.terminal_attributes = termios.tcgetattr(fd)
            tty.setraw(fd)
        os.set_blocking(fd, False)
        try:
            self.selector.register(sys.stdin, selectors.EVENT_READ, "stdin")
        except OSError:
            if is_terminal:
                raise
            self._debug("stdin-not-pollable")
        if is_terminal:
            self._resize()

    def _restore_terminal(self) -> None:
        if self.terminal_attributes is not None:
            try:
                termios.tcsetattr(sys.stdin.fileno(), termios.TCSADRAIN, self.terminal_attributes)
            except OSError:
                pass
            self.terminal_attributes = None
        if self.stdin_flags is not None:
            try:
                fcntl.fcntl(sys.stdin.fileno(), fcntl.F_SETFL, self.stdin_flags)
            except OSError:
                pass
            self.stdin_flags = None

    def _cleanup_terminal_modes(self) -> None:
        if not os.isatty(sys.stdout.fileno()):
            return
        try:
            write_all(sys.stdout.fileno(), TERMINAL_CLEANUP_BYTES)
            sys.stdout.flush()
        except OSError:
            pass

    def _resize(self) -> None:
        if self.master_fd is None or not os.isatty(sys.stdout.fileno()):
            return
        try:
            size = shutil.get_terminal_size(fallback=(80, 24))
            winsize = struct.pack("HHHH", size.lines, size.columns, 0, 0)
            fcntl.ioctl(self.master_fd, termios.TIOCSWINSZ, winsize)
            if self.pid and child_alive(self.pid):
                os.kill(self.pid, signal.SIGWINCH)
        except OSError:
            pass

    def _forward_signal(self, signum: int, _frame: object) -> None:
        if self.pid and child_alive(self.pid):
            try:
                os.killpg(self.pid, signum)
            except OSError:
                try:
                    os.kill(self.pid, signum)
                except OSError:
                    pass
        if signum in {signal.SIGTERM, signal.SIGHUP, signal.SIGQUIT}:
            self.shutdown_requested = True

    def _install_signals(self) -> None:
        for signum in (signal.SIGINT, signal.SIGTERM, signal.SIGHUP, signal.SIGQUIT):
            signal.signal(signum, self._forward_signal)
        signal.signal(signal.SIGWINCH, lambda _signum, _frame: self._resize())

    def launch(self) -> None:
        environment = os.environ.copy()
        environment.update(
            {
                "CODEX_IDLE_CHANNEL": self.channel,
                "CODEX_IDLE_OWNER": "pty-supervisor",
                "CODEX_IDLE_CONTROL_SOCKET": str(self.socket_path),
                "CODEX_IDLE_RUNTIME_DIR": str(self.session_dir),
                "CODEX_IDLE_SUPERVISOR_PID": str(os.getpid()),
            }
        )
        pid, master_fd = pty.fork()
        if pid == 0:
            try:
                os.execvpe(self.executable, [self.executable, *self.argv], environment)
            except BaseException as error:
                print(f"codex-warm: failed to exec Codex: {error}", file=sys.stderr)
                os._exit(127)
        self.pid = pid
        self.master_fd = master_fd
        os.set_blocking(master_fd, False)
        self.child_tty = self._read_child_tty()
        self.selector.register(master_fd, selectors.EVENT_READ, "master")
        self._install_signals()
        self._install_terminal()
        self.herdr.child_pid = pid
        self._register_herdr()
        self._write_state()

    def _read_child_tty(self) -> str | None:
        if not self.pid:
            return None
        try:
            candidate = os.readlink(f"/proc/{self.pid}/fd/0")
            return candidate if candidate.startswith("/dev/") else None
        except OSError:
            return None

    def _pty_still_owned(self) -> bool:
        if not self.pid or self.master_fd is None or not child_alive(self.pid):
            return False
        if self.child_tty is None:
            self.child_tty = self._read_child_tty()
        try:
            return (
                os.getpgid(self.pid) == self.pid
                and self.child_tty is not None
                and os.readlink(f"/proc/{self.pid}/fd/0") == self.child_tty
            )
        except OSError:
            return False

    def _gate_reason(self, purpose: str = "compact") -> str | None:
        if not self.enabled:
            return "session-disabled"
        if self.status != "idle":
            return "session-active"
        if self.compacting:
            return "already-compacting"
        if not self._pty_still_owned():
            return "pty-or-process-gone"
        if not self.session_id or not SESSION_RE.fullmatch(self.session_id):
            return "session-unbound"
        if not self.last_normal_stop:
            return "no-normal-stop"
        if self.last_lifecycle_activity and self.last_lifecycle_activity > self.last_normal_stop:
            return "lifecycle-activity"
        if self.last_local_input and self.last_local_input > self.last_normal_stop:
            return "user-input"
        if purpose == "compact":
            if not self.dirty:
                return "clean"
            if MIN_TOKENS > 0 and self.current_tokens is None:
                return "token-count-unknown"
            if self.current_tokens is not None and self.current_tokens < MIN_TOKENS:
                return "below-token-threshold"
        elif purpose != "manual":
            return "unknown-injection-purpose"
        return None

    def attempt(
        self,
        expected_generation: int | None = None,
        command: str = "/compact",
        purpose: str = "compact",
    ) -> bool:
        if expected_generation is not None and expected_generation != self.timer_generation:
            self.last_gate_rejection_reason = "stale-timer"
            self.last_state_transition = "gate-refused:stale-timer"
            self._write_state()
            return False
        if (purpose == "compact" and expected_generation is not None
                and not self._refresh_cache_policy_for_action()):
            return False
        reason = self._gate_reason(purpose)
        if reason:
            self.last_cancellation_reason = reason
            self.last_gate_rejection_reason = reason
            self.last_state_transition = f"gate-refused:{reason}"
            self._debug(f"cancelled:{reason} channel={short_id(self.channel)}")
            self._write_state()
            return False
        if command not in {"/compact"}:
            self.last_gate_rejection_reason = "inject-command-refused"
            self._write_state()
            return False
        self.status = "active"
        self.last_gate_rejection_reason = None
        self.last_injection_result = "pending"
        self.last_state_transition = "injection-attempted"
        self.timer_deadline = None
        self.timer_kind = None
        self.timer_generation += 1
        if purpose == "compact":
            self.compacting = True
            self.compaction_requested_at = now()
            self.compaction_ignore_next_stop = False
            self.last_cancellation_reason = "compaction-requested"
        else:
            self.last_cancellation_reason = "manual-injection-requested"
        self._write_state()
        try:
            if self.master_fd is None:
                raise OSError(errno.EIO, "PTY master unavailable")
            write_all(self.master_fd, (command + "\r").encode("utf-8"))
        except OSError as error:
            if purpose == "compact":
                self.compacting = False
                self.compaction_requested_at = None
                self.dirty = True
                self.last_cancellation_reason = "compaction-injection-failed"
                self.last_injection_result = "failed"
                self.last_state_transition = "injection-failed"
                self._debug(f"compaction-failed channel={short_id(self.channel)} error={error}")
            else:
                self.status = "idle"
                self.last_cancellation_reason = "manual-injection-failed"
                self.last_injection_result = "failed"
                self.last_state_transition = "injection-failed"
            self._write_state()
            return False
        if purpose == "compact":
            self.last_injection_result = "requested"
            self.last_state_transition = "compaction-requested"
            self._debug(f"compaction-requested channel={short_id(self.channel)}")
        self._write_state()
        return True

    def _handle_event(self, message: dict) -> dict:
        if message.get("kind") != "event" or message.get("channel") != self.channel:
            return {"ok": False, "error": "invalid-event-owner"}
        event = message.get("event")
        session_id = message.get("session_id")
        if event not in EVENTS or not isinstance(session_id, str) or not SESSION_RE.fullmatch(session_id):
            return {"ok": False, "error": "invalid-event"}
        if event == "session-start":
            new_session = self.session_id != session_id
            if self.session_id and new_session:
                self.cancel("session-replaced")
                self.dirty = False
                self.compaction_requested_at = None
                self.compaction_ignore_next_stop = False
            self.session_id = session_id
            self.status = "starting"
            self.screen_classifier.reset()
            self.screen_state = None
            self.last_lifecycle_activity = now()
            self._register_herdr()
            self._write_state()
            return {"ok": True}
        if self.session_id != session_id:
            return {"ok": False, "error": "session-mismatch"}
        current = now()
        if event == "stop":
            if message.get("stop_hook_active") is not False:
                return {"ok": True, "ignored": "stop-hook-active"}
            if self.compaction_ignore_next_stop:
                self.compaction_ignore_next_stop = False
                self.status = "idle"
                self.dirty = True
                self.last_normal_stop = current
                self.last_lifecycle_activity = current
                self.last_local_input = None
                self.cancel("compaction-failed-stop")
                self.arm()
                return {"ok": True, "ignored": "compaction-failed-stop"}
            if self.compacting:
                self._recover_compaction("stop")
                return {"ok": True, "ignored": "compaction-stop"}
            if (
                self.last_normal_stop
                and self.status == "idle"
                and self.dirty
                and current - self.last_normal_stop < 0.5
            ):
                return {"ok": True, "ignored": "duplicate-stop"}
            self.status = "idle"
            self.dirty = True
            self.last_normal_stop = current
            self.last_lifecycle_activity = current
            self.last_local_input = None
            self.cancel("new-normal-turn")
            self.arm()
            return {"ok": True}
        if event == "user-prompt-submit":
            self.status = "active"
            self.last_lifecycle_activity = current
            self.compaction_ignore_next_stop = False
            self.cancel("prompt-submit")
        elif event == "post-compact":
            self.status = "idle"
            self.dirty = False
            self.compacting = False
            self.compaction_requested_at = None
            self.compaction_ignore_next_stop = False
            self.current_tokens = None
            self.last_injection_result = "succeeded"
            self.last_lifecycle_activity = current
            self.cancel("compaction-completed")
            self.last_state_transition = "compaction-completed"
            self._debug(f"compaction-completed channel={short_id(self.channel)}")
        elif event == "session-end":
            self.status = "ended"
            self.last_lifecycle_activity = current
            self.cancel("session-end")
            self.herdr.release()
        elif event == "stop-failure":
            if self.compacting:
                self._compaction_failed("stop-failure")
                self.last_injection_result = "failed"
                self.last_state_transition = "stop-failure"
            self.status = "error"
            self.last_lifecycle_activity = current
            self.cancel("stop-failure")
        else:
            self.status = "active"
            self.last_lifecycle_activity = current
            self.cancel("session-activity")
        self._write_state()
        return {"ok": True}

    def _handle_control(self, message: dict) -> dict:
        if message.get("kind") != "control":
            return {"ok": False, "error": "invalid-control"}
        if message.get("channel") != self.channel:
            return {"ok": False, "error": "invalid-control-owner"}
        command = message.get("command")
        if command not in CONTROL_COMMANDS:
            return {"ok": False, "error": "unknown-control"}
        if command == "state":
            return {"ok": True, "state": self.state()}
        if command == "enable":
            self.enabled = True
            self.arm()
        elif command == "disable":
            self.enabled = False
            self.cancel("session-disabled")
        elif command == "cancel":
            self.cancel("manual-cancel")
        elif command == "trigger":
            success = self.attempt()
            self._write_state()
            return {"ok": success, "state": self.state()}
        elif command == "inject":
            text = message.get("text")
            if text != "/compact":
                return {"ok": False, "error": "inject-command-refused", "state": self.state()}
            success = self.attempt(command=text, purpose="compact")
            self._write_state()
            return {"ok": success, "state": self.state()}
        self._write_state()
        return {"ok": True, "state": self.state()}

    def _serve_one(self, connection: socket.socket) -> None:
        try:
            connection.settimeout(1.0)
            data = connection.recv(65536)
            if not data:
                return
            message = json.loads(data.split(b"\n", 1)[0].decode("utf-8"))
            if not isinstance(message, dict):
                response = {"ok": False, "error": "invalid-json"}
            elif message.get("kind") == "event":
                response = self._handle_event(message)
            else:
                response = self._handle_control(message)
            connection.sendall((json.dumps(response, separators=(",", ":")) + "\n").encode())
        except (OSError, ValueError, UnicodeError) as error:
            self._debug(f"ipc-error channel={short_id(self.channel)} error={error}")
        finally:
            connection.close()

    def _record_input_observations(
        self,
        observations: list[tuple[str, int, bool]],
    ) -> None:
        if not observations:
            return
        human = any(is_human for _, _, is_human in observations)
        if not human:
            return
        self.last_local_input = now()
        self.last_local_input_classification = ",".join(
            sorted({kind for kind, _, is_human in observations if is_human})
        )
        if self.timer_deadline is not None and self.last_normal_stop and self.last_local_input > self.last_normal_stop:
            self.cancel("user-input")
        else:
            self._write_state()

    def _flush_input_classifier(self) -> None:
        self._record_input_observations(self.input_classifier.flush_expired())

    def _read_stdin(self) -> None:
        try:
            data = os.read(sys.stdin.fileno(), 65536)
        except (OSError, BlockingIOError):
            return
        if not data:
            self.stdin_eof = True
            if self.master_fd is not None:
                try:
                    write_all(self.master_fd, b"\x04")
                except OSError:
                    pass
            try:
                self.selector.unregister(sys.stdin)
            except Exception:
                pass
            return
        if self.master_fd is not None:
            try:
                write_all(self.master_fd, data)
            except OSError:
                self.shutdown_requested = True
        try:
            observations = self.input_classifier.feed(data)
        except Exception:
            self._debug("terminal-input-classifier-error")
            observations = [("unknown-input", len(data), True)]
        self._record_input_observations(observations)

    def _read_master(self) -> bool:
        if self.master_fd is None:
            return False
        try:
            data = os.read(self.master_fd, 65536)
        except OSError as error:
            if error.errno in {errno.EIO, errno.EBADF}:
                return False
            return True
        if not data:
            return False
        state = self.screen_classifier.feed(data)
        try:
            write_all(sys.stdout.fileno(), data)
            sys.stdout.flush()
        except OSError:
            self.shutdown_requested = True
        if state is not None:
            self._report_screen_state(state)
        return True

    def _child_has_exited(self) -> bool:
        if self.child_status is not None or not self.pid:
            return self.child_status is not None
        try:
            waited_pid, status = os.waitpid(self.pid, os.WNOHANG)
        except ChildProcessError:
            self.child_status = 1 << 8
            return True
        if waited_pid == self.pid:
            self.child_status = status
            self.child_exited_at = now()
            self.status = "exited"
            self.cancel("process-exit")
            return True
        return False

    def run(self) -> int:
        self.launch()
        master_open = True
        next_debug_bound = monotonic() + 5
        try:
            while master_open:
                child_exited = self._child_has_exited()
                self.herdr.heartbeat(self.screen_state, self.session_id)
                self._flush_input_classifier()
                self._check_compaction_timeout()
                if self.timer_deadline is not None and now() >= self.timer_deadline:
                    generation = self.timer_generation
                    kind = self.timer_kind
                    self.timer_deadline = None
                    self.timer_kind = None
                    if kind == "compact":
                        self.attempt(generation)
                if monotonic() >= next_debug_bound:
                    next_debug_bound = monotonic() + 5
                    self._bound_debug_log()
                if child_exited:
                    master_open = self._read_master()
                    if not master_open:
                        break
                timeout = 1.0
                if self.timer_deadline is not None:
                    timeout = max(0.0, min(timeout, self.timer_deadline - now()))
                events = self.selector.select(timeout)
                for key, _mask in events:
                    if key.data == "server":
                        try:
                            connection, _ = self.socket.accept()
                            self._serve_one(connection)
                        except BlockingIOError:
                            pass
                    elif key.data == "master":
                        master_open = self._read_master()
                    elif key.data == "stdin":
                        self._read_stdin()
                if self.shutdown_requested and self.pid and child_alive(self.pid):
                    self._forward_signal(signal.SIGHUP, None)
            self._child_has_exited()
            if self.child_status is None and self.pid:
                try:
                    _, self.child_status = os.waitpid(self.pid, 0)
                except ChildProcessError:
                    self.child_status = 1 << 8
            return os.waitstatus_to_exitcode(self.child_status or 0)
        finally:
            self._cleanup_terminal_modes()
            self._restore_terminal()
            self.cleanup()

    def cleanup(self) -> None:
        self.herdr.release()
        try:
            self.selector.close()
        except Exception:
            pass
        if self.master_fd is not None:
            try:
                os.close(self.master_fd)
            except OSError:
                pass
            self.master_fd = None
        try:
            self.socket.close()
        except OSError:
            pass
        try:
            shutil.rmtree(self.session_dir)
        except OSError:
            pass


def control_socket(channel: str) -> pathlib.Path:

def send_control(channel: str, command: str, **fields: object) -> dict:
    path = control_socket(channel)
    message = {"kind": "control", "channel": channel, "command": command, **fields}
    connection = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
    try:
        connection.settimeout(2.0)
        connection.connect(str(path))
        connection.sendall((json.dumps(message, separators=(",", ":")) + "\n").encode())
        data = connection.recv(65536)
        return json.loads(data.decode("utf-8"))
    finally:
        connection.close()

* Multi-account boot owner on OpenRC hosts

On a headless OpenRC host (Honey) warm consorts start from =/etc/init.d=, not XDG autostart. Until 2026-09-26 that meant one hand-written script per kind, =claude-warm= and =codex-warm=, both running as the operator (=gille=) and sharing one runtime dir =/run/warm=. The agent-user wall (=agents-users.org= "Agent User Bootstrap") wants consorts to run as the account that owns their auth, and the host is to be replaced (processes move to Milk), so the boot owner must be per account and host-agnostic.

One body serves both kinds and every account, in OpenRC's own multi-instance idiom (the =net.lo= / =net.eth0= pattern): the body is installed as =/etc/init.d/claude-warm= and =/etc/init.d/codex-warm=, and each instance is a symlink =<kind>-warm.<account>= to it. =RC_SVCNAME= carries both halves:

| Derived from =RC_SVCNAME= | =claude-warm.agent-claude= | bare =claude-warm= (compat) |
|---|---|---|
| kind (binary =<home>/.local/bin/<kind>-warm=, else the system checkout's =/usr/local/share/dotcortex/DotCortex/all/.local/bin/<kind>-warm= (=warm_checkout=), else the account's own =~/DotCortex/all/.local/bin/<kind>-warm=) | =claude= | =claude= |
| account (=su -l=, supervisor owner) | =agent-claude= | =warm_user= from conf.d, required |
| tmux socket (=tmux -L=) and session | =claude-warm= / =claude-warm= | default socket / conf.d =warm_session= |
| =XDG_RUNTIME_DIR= (0700, account-owned) | =/run/claude-warm.agent-claude= | conf.d =xdg= |
| log (0640, account-owned) | =/var/log/claude-warm.agent-claude.log= | =/var/log/claude-warm.log= |
| pid file (written by the pane just before its =exec=) | =/run/claude-warm.agent-claude/claude-warm.agent-claude.pid= | =<xdg>/claude-warm.pid= |
| Herdr agent name (fleet-unique; exported for =claude= only) | =<host>-claude-agent-claude= | conf.d =warm_agent= |

Per-instance values (working directory, model, extra arguments, overrides) live in =/etc/conf.d/<kind>-warm.<account>=, which OpenRC sources automatically (falling back to =/etc/conf.d/<kind>-warm=). A bare =claude-warm= / =codex-warm= refuses to start unless its conf.d names a =warm_user=: that is the *compat path*. Today's two services keep their names, runlevel links, tmux sessions (=warm=, =codex=), shared =/run/warm= and Herdr name (=honey-opus=) by carrying those exact values in =/etc/conf.d/claude-warm= and =/etc/conf.d/codex-warm=, so replacing the hand-written scripts with the body is a no-op until an instance is switched. The body never runs as root and refuses uid 0. The compat conf.d also pins =warm_path= to the scripts' exact =PATH= (they replace it; an instance appends =~/.local/bin= to the login =PATH=).

*Revised 2026-09-26 after the Honey tanuki bots landed* (=agents-bots-honey.org=): three =claude-warm= supervisors now run under one account there (=agent-claude=), which is exactly the case where "the account's process running this binary" stops naming one supervisor. The body takes their lesson: the pane writes =<xdg>/<svc>.pid= and =exec='s (same pid), and stop/status trust that file, checked against the process command line. Only a supervisor started by a pre-body script (no pid file yet) is found by owner and binary path, which is what the compat swap needs. An agent account is not stowed, so the binary comes from the host's root-owned system checkout (=warm_checkout=, default =/usr/local/share/dotcortex/DotCortex=), as the bots' launchers do (=agents-bots-honey.org= "Service lane"), and only then from the account's own DotCortex checkout. The bots' own services (=honey-opus=, =honey-sonnet=, =honey-haiku=) are Telegram bodies, not warm consorts, and stay as they are.

Which instances a host runs is declared in =fleet.ssv='s =warm= column (=agents-users.org=). *Next commit, specified here but not yet written:* =warm-openrc-apply= reads that row: =--dry-run= (default) prints the plan (body, symlinks, conf.d stubs it would create, runlevel links), =--check= compares the installed body, =--apply= needs =--host= equal to the live hostname and an OpenRC service manager, installs the body and symlinks, writes only *missing* conf.d files, adds runlevel links and verifies them in =/etc/runlevels/default=. It never starts, stops or restarts a consort (bounce via Nurse Joy) and never removes a link the row no longer lists (it prints the =rc-update del= for the operator). It supersedes the single-instance renderer proposed in MR !7 (=claude-warm-openrc-apply=) for hosts that need more than one account.

esac

# Instance defaults, derived from the service name; conf.d wins.

# The supervisor's pid: the pid file the pane writes just before its exec
# (same pid), checked against the command line; else, for a supervisor started
# by a pre-body script, the account's process running this exact binary.
# Several supervisors of one kind can share an account (agents-bots-honey.org),
# so the pid file is the source of truth once the body has started it.
warm_pid() {
    local pid bin
    bin="$(warm_find_bin)"
    if [ -r "${warm_pidfile}" ]; then
        pid="$(cat "${warm_pidfile}" 2>/dev/null)"
        if [ -n "${pid}" ] && tr '\0' ' ' < "/proc/${pid}/cmdline" 2>/dev/null | grep -qF "${bin}"; then
            printf '%s' "${pid}"; return 0
        fi
        return 1
    fi
    pgrep -u "${warm_user}" -f "${bin}" 2>/dev/null | head -n 1
}

    warm_valid || return 1

    # carries its own HERDR_SOURCE.
    [ "${warm_kind}" = claude ] && herdr="CLAUDE_WARM_HERDR_AGENT='${warm_agent}'"
    # exec makes the pane's process the supervisor; the pid file names it.
    su -l "${warm_user}" -c "export XDG_RUNTIME_DIR='${xdg}' PATH=${env_path} ${herdr}; \
        $(warm_tmux) new-session -d -s '${warm_session}' -c '${warm_cwd}' \
        -e XDG_RUNTIME_DIR='${xdg}' -e PATH=\"\${PATH}\" ${herdr:+-e ${herdr}} \
        'echo \$\$ > ${warm_pidfile}; exec ${warm_bin} ${warm_args} >> ${warm_log} 2>&1'"
    eend $? "Failed to start ${RC_SVCNAME}"
}

stop() {
    local pid i=0
    ebegin "Stopping ${description}"
    [ -n "${warm_user}" ] || { eend 1 "no account"; return 1; }
    # The supervisor ignores SIGHUP and survives its pane: SIGTERM it by pid,
    # then tear down this instance's tmux session only.
    pid="$(warm_pid)"
    if [ -n "${pid}" ]; then
        kill -TERM "${pid}" 2>/dev/null
        while kill -0 "${pid}" 2>/dev/null && [ $i -lt 10 ]; do sleep 1; i=$((i+1)); done
        kill -0 "${pid}" 2>/dev/null && kill -KILL "${pid}" 2>/dev/null
    fi
    rm -f "${warm_pidfile}"
    su -l "${warm_user}" -c "$(warm_tmux) kill-session -t '${warm_session}'" 2>/dev/null
    eend 0
}

status() {
    local pid
    pid="$( [ -n "${warm_user}" ] && warm_pid )"
    if [ -n "${pid}" ]; then
        einfo "${RC_SVCNAME} supervisor is running as ${warm_user} (pid ${pid})"
        return 0
    fi
    einfo "${RC_SVCNAME} is not running"
    return 3
}

#+BEGIN_SRC python :tangle linux/.local/share/dotcortex/tests/test-warm-consort.py :shebang "#!/usr/bin/env python3" :comments no :noweb no
from __future__ import annotations

import subprocess
import tempfile
import time
import unittest
from pathlib import Path


ROOT = Path(__file__).resolve().parents[5]
BODY = ROOT / "linux/.local/share/dotcortex/openrc/warm-consort"
CONF = ROOT / "linux/.local/share/dotcortex/openrc/conf.d"

STUBS = r"""
ebegin() { :; }; eend() { return "${1:-0}"; }
einfo() { echo "INFO $*"; }; eerror() { echo "ERR $*"; }
checkpath() { echo "CHECKPATH $*"; }
hostname() { echo testhost; }
getent() {
    case "$2" in
        gille) echo "gille:x:1000:1000::$FAKE_HOME:/bin/sh" ;;
        agent-claude) echo "agent-claude:x:2002:2002::$FAKE_HOME:/bin/bash" ;;
        rooty) echo "rooty:x:0:0::$FAKE_HOME:/bin/sh" ;;
        *) return 2 ;;
    esac
}
su() { printf 'SU %s\n' "$*"; }
"""


class WarmConsortTests(unittest.TestCase):
    def setUp(self) -> None:
        self.tmp = tempfile.TemporaryDirectory()
        self.home = Path(self.tmp.name) / "home"
        (self.home / "metsatron-peer").mkdir(parents=True)
        self.run_dir = Path(self.tmp.name) / "run"
        self.run_dir.mkdir()

    def tearDown(self) -> None:
        self.tmp.cleanup()

    def put_bin(self, rel: str) -> Path:
        path = self.home / rel
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text("#!/bin/sh\nexec sleep 30\n")
        path.chmod(0o755)
        return path

    def body(self, svc: str, script: str, conf: str = "") -> subprocess.CompletedProcess[str]:
        prog = f"{STUBS}\nFAKE_HOME='{self.home}'\nRC_SVCNAME='{svc}'\n{conf}\n. '{BODY}'\n{script}\n"
        return subprocess.run(["bash", "-c", prog], capture_output=True, text=True, timeout=30)

    def test_instance_derives_from_the_service_name(self) -> None:
        self.put_bin(".local/bin/claude-warm")
        out = self.body("claude-warm.agent-claude",
                        'warm_valid && echo "$warm_user|$warm_socket|$warm_session|$xdg|$warm_pidfile|$warm_log|$warm_log_mode|$warm_agent|$warm_bin"')
        self.assertEqual(out.returncode, 0, out.stdout + out.stderr)
        self.assertIn(
            f"agent-claude|claude-warm|claude-warm|/run/claude-warm.agent-claude|"
            f"/run/claude-warm.agent-claude/claude-warm.agent-claude.pid|/var/log/claude-warm.agent-claude.log|0640|"
            f"testhost-claude-agent-claude|{self.home}/.local/bin/claude-warm", out.stdout)

    def test_unstowed_account_falls_back_to_its_checkout(self) -> None:
        bin_ = self.put_bin("DotCortex/all/.local/bin/codex-warm")
        out = self.body("codex-warm.agent-claude", 'warm_valid && echo "BIN $warm_bin"',
                        f"warm_checkout='{self.tmp.name}/none'")
        self.assertIn(f"BIN {bin_}", out.stdout)

    def test_unstowed_account_prefers_the_system_checkout(self) -> None:
        self.put_bin("DotCortex/all/.local/bin/claude-warm")
        lane = Path(self.tmp.name) / "system"
        sys_bin = lane / "all/.local/bin/claude-warm"
        sys_bin.parent.mkdir(parents=True)
        sys_bin.write_text("#!/bin/sh\nexec sleep 30\n")
        sys_bin.chmod(0o755)
        out = self.body("claude-warm.agent-claude", 'warm_valid && echo "BIN $warm_bin"', f"warm_checkout='{lane}'")
        self.assertIn(f"BIN {sys_bin}", out.stdout)

    def test_agent_reference_conf_keeps_the_herdr_name_and_needs_its_lane(self) -> None:
        conf = (CONF / "claude-warm.agent-claude").read_text()
        out = self.body("claude-warm.agent-claude", 'echo "$warm_user|$warm_agent|$warm_cwd"', conf)
        self.assertIn("agent-claude|honey-opus|/home/agent-claude/metsatron-peer", out.stdout)
        missing = f"{self.tmp.name}/no-lane"
        out = self.body("claude-warm.agent-claude", 'warm_valid; echo "RC=$?"', conf + f"\nwarm_cwd='{missing}'")
        self.assertIn("RC=1", out.stdout)
        self.assertIn(f"working dir {missing} missing", out.stdout)

    def test_bare_service_needs_conf(self) -> None:
        self.put_bin(".local/bin/claude-warm")
        out = self.body("claude-warm", "warm_valid; echo rc=$?")
        self.assertIn("rc=1", out.stdout)
        self.assertIn("no account", out.stdout)

    def test_compat_conf_reproduces_the_hand_written_scripts(self) -> None:
        self.put_bin(".local/bin/claude-warm")
        conf = (CONF / "claude-warm").read_text().replace("/home/gille", str(self.home))
        out = self.body("claude-warm", "start", conf)
        self.assertEqual(out.returncode, 0, out.stdout + out.stderr)
        su = next(line for line in out.stdout.splitlines() if line.startswith("SU "))
        self.assertIn("-l gille", su)
        self.assertIn("XDG_RUNTIME_DIR='/run/warm'", su)
        self.assertIn(f"PATH='{self.home}/.local/bin:/usr/local/bin:/usr/bin:/bin'", su)
        self.assertIn("CLAUDE_WARM_HERDR_AGENT='honey-opus'", su)
        self.assertIn("tmux new-session -d -s 'warm'", su, "default socket, session warm")
        self.assertNotIn("-L", su)
        self.assertIn("echo $$ > /run/warm/claude-warm.pid; exec ", su)
        self.assertIn("--model claude-opus-4-8 >> /var/log/claude-warm.log", su)

    def test_codex_gets_no_claude_herdr_name(self) -> None:
        self.put_bin(".local/bin/codex-warm")
        conf = (CONF / "codex-warm").read_text().replace("/home/gille", str(self.home))
        out = self.body("codex-warm", "start", conf)
        su = next(line for line in out.stdout.splitlines() if line.startswith("SU "))
        self.assertNotIn("CLAUDE_WARM_HERDR_AGENT", su)
        self.assertIn("-s 'codex'", su)
        self.assertIn("echo $$ > /run/warm/codex-warm.pid", su)

    def test_instance_appends_to_the_login_path(self) -> None:
        self.put_bin(".local/bin/claude-warm")
        out = self.body("claude-warm.agent-claude", "start")
        su = next(line for line in out.stdout.splitlines() if line.startswith("SU "))
        self.assertIn('PATH="${PATH}:${HOME}/.local/bin"', su)
        self.assertIn("tmux -L 'claude-warm' new-session", su)

    def test_root_is_refused(self) -> None:
        self.put_bin(".local/bin/claude-warm")
        out = self.body("claude-warm.rooty", "warm_valid; echo rc=$?")
        self.assertIn("rc=1", out.stdout)
        self.assertIn("root", out.stdout)

    def test_unknown_account_and_kind_are_refused(self) -> None:
        self.assertIn("rc=1", self.body("claude-warm.nobody-here", "warm_valid; echo rc=$?").stdout)
        self.assertIn("rc=1", self.body("gemini-warm.gille", "warm_valid; echo rc=$?").stdout)

    def test_pid_file_names_one_supervisor_among_several(self) -> None:
        bin_ = self.put_bin(".local/bin/claude-warm")
        procs = [subprocess.Popen(["bash", "-c", f"exec -a '{bin_}' sleep 30"]) for _ in range(2)]
        try:
            time.sleep(0.2)
            pidfile = self.run_dir / "svc.pid"
            pidfile.write_text(f"{procs[1].pid}\n")
            conf = f"warm_user=gille\nwarm_pidfile='{pidfile}'"
            out = self.body("claude-warm", "warm_pid", conf)
            self.assertEqual(out.stdout.strip(), str(procs[1].pid))
            pidfile.write_text("999999\n")
            out = self.body("claude-warm", 'warm_pid; echo "rc=$?"', conf)
            self.assertEqual(out.stdout.strip(), "rc=1", "a stale pid file is not a running supervisor")
        finally:
            for p in procs:
                p.kill()
                p.wait()


if __name__ == "__main__":
    unittest.main()
