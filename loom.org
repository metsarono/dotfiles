;; Run one shell string via bash -lc <cmd> with inherited TTY/stdout/stderr,
;; so long-running tasks stream live output and color-capable tools keep ANSI.
(define (exit-status status)
  (cond
   ((not status) 127)
   ((status:exit-val status) => identity)
   ((status:term-sig status) => (lambda (sig) (+ 128 sig)))
   (else 1)))

(define (sh cmd)
  (exit-status (system* "bash" "-lc" cmd)))

(define (run-sequential . cmds)
  (let loop ((cmds cmds)
             (first-failure 0))
    (if (null? cmds)
        first-failure
        (let ((code (sh (car cmds))))
          (loop (cdr cmds)
                (if (and (zero? first-failure) (not (zero? code)))
                    code
                    first-failure))))))

;; Generic target runner. The root Makefile includes every all/.mk/*.mk fragment,

(define (git-submodules-latest-cmd)
  (string-append
   "git submodule foreach --recursive '"
   "branch=$(git -C \"$toplevel/$name\" remote show origin | sed -n \"s#.*HEAD branch: ##p\"); "
   "if [ -z \"$branch\" ]; then branch=$(git config -f \"$toplevel/.gitmodules\" --get submodule.$name.branch || echo master); fi; "
   "git fetch origin \"$branch\" && git checkout -q FETCH_HEAD'"))

(define (git-submodules-sync-cmd)
  "git submodule sync --recursive && git submodule update --init --recursive")

(define (git-submodules-status-cmd)
  "git submodule status --recursive")

;; After stow, reload tmux config in the running server (if any).
;; Ensures that changed options like @resurrect-save-script-path take effect
;; without requiring a full server restart.
;; Bash login shells spawned by Guile don't inherit Guix PATH, so we prepend it.
(define (stow-then-reload-tmux stow-cmd)
  (let ((result (sh (string-append "cd \"" HOME "/DotCortex\" && " stow-cmd))))
    (sh (string-append
         "PATH=\"" HOME "/.guix-extra-profiles/core/core/bin:$PATH\" "
         "tmux source-file ~/.config/tmux/tmux.conf >/dev/null 2>&1 "
         "&& printf '[loom] tmux config reloaded\\n' "
         "|| printf '[loom] tmux: not running, skipped reload\\n'"))
    result))

;; --- Pretty printing for help ---

(define (pad-right s width)
  (let ((n (string-length s)))
    (if (>= n width)
        s
        (string-append s (make-string (- width n) #\space)))))

(define (task-name-str t)
  (symbol->string (task-name t)))

(define (print-group title pred)
  (let* ((ts (filter pred tasks))
         (ts (sort ts (lambda (a b)
                        (string<? (task-name-str a)
                                  (task-name-str b))))))
    (when (pair? ts)
      (format #t "~a~%" title)
      (for-each
       (lambda (t)
         (let ((nm (pad-right (task-name-str t) 22)))
           (format #t "  ~a ~a~%" nm (task-desc t))))
       ts)
      (newline))))

;; Strip a literal "--" and everything before it in ARGV
(define (drop-dashdash xs)
  (cond ((and (pair? xs) (string=? (car xs) "--"))
         (drop-dashdash (cdr xs)))
        (else xs)))

(define (main args)
  (let* ((rest (drop-dashdash (cdr args))))
    (match rest
      (()            (begin (usage) 0))
      (("-h")        (begin (usage) 0))
      (("--help")    (begin (usage) 0))
      (("-V")        (begin (print-version) 0))
      (("--version") (begin (print-version) 0))
      (("help")      (begin (usage) 0))
      (("list")      (run 'list))
      ((cmd . _more) (run (string->symbol cmd)))
      (_             (begin (usage) 1)))))

;; ---- census: does org parse every block the text declares? ------------------
;; An unbalanced block, or an unescaped column-0 headline inside a block, makes
;; org-element parse FEWER src blocks than the raw text declares - org-babel
;; then tangles fewer outputs, silently. The census compares both counts per
;; file and fails loudly on any divergence. (Sealed 2026-07-16 after two such
;; drops shipped: claude.mk orphaned for four days, redstone gtk.css dropped.)
(defun dotfiles--textual-src-count (file)
  "Count the src blocks FILE's raw text declares, as a reader would.
A simple state machine: a BEGIN_SRC line opens a src block; BEGIN_EXAMPLE
and BEGIN_EXPORT open verbatim blocks whose content is not counted; the
matching END_ closes.  Nested or unbalanced markers are the linter's job."
  (with-temp-buffer
    (insert-file-contents file)
    (goto-char (point-min))
    (let ((case-fold-search t) (count 0) (inblk nil))
      (while (not (eobp))
        (cond
         ((and (not inblk) (looking-at "[ \t]*#\\+begin_src\\b"))
          (setq count (1+ count) inblk t))
         ((and (not inblk) (looking-at "[ \t]*#\\+begin_\\(example\\|export\\)\\b"))
          (setq inblk t))
         ((looking-at "[ \t]*#\\+end_\\(src\\|example\\|export\\)\\b")
          (setq inblk nil)))
        (forward-line 1))
      count)))

(defun dotfiles-batch-census (root)
  "Fail loudly if org-element parses fewer src blocks than ROOT/*.org declare."
  (interactive "DRoot: ")
  (setq root (file-name-as-directory (expand-file-name root)))
  (let ((bad '()))
    (dolist (f (dotfiles-root-org-files root))
      (let* ((textual (dotfiles--textual-src-count f))
             (parsed (with-current-buffer (find-file-noselect f)
                       (org-mode)
                       (prog1
                           (length (org-element-map (org-element-parse-buffer)
                                       'src-block #'identity))
                         (kill-buffer)))))
        (if (= textual parsed)
            (message "census ok   %-34s %3d blocks" (file-name-nondirectory f) parsed)
          (push (format "%s: text declares %d src blocks, org parses %d - blocks are being DROPPED"
                        (file-name-nondirectory f) textual parsed)
                bad))))
    (if bad
        (progn
          (message "== CENSUS FAILURES ==")
          (dolist (b (nreverse bad)) (message "!! %s" b))
          (kill-emacs 1))
      (message "census: every file parses exactly as declared"))))

#!/bin/env bash
set -euo pipefail

usage() {
  echo "Usage: org-style-lint"
}

case "${1:-}" in
  -h|--help|help) usage; exit 0 ;;
  "") ;;
  *) echo "org-style-lint: unknown argument: $1" >&2; usage >&2; exit 2 ;;
esac

# Behavior:
# - FATAL on structural breakage in root *.org sources: unbalanced
#   #+BEGIN_/#+END_ blocks, or a column-0 headline inside a block. Both make
#   org-element parse fewer blocks than the text declares, so org-babel
#   silently drops tangled outputs (sealed 2026-07-16: claude.mk orphaned
#   for four days by a lost #+END_SRC; redstone gtk.css dropped by a bare
#   CSS universal selector at column 0). A broken parse must never tangle.
# - Warn on *lowercase* #+begin_src / #+end_src (style).
# - Warn when a root source exceeds the 100 KB module budget (~25k tokens):
#   every module must stay loadable by a small-model context window.
# - Set STRICT_LINT=1 if you want the warnings to be fatal too.

shopt -s nullglob globstar
STATUS=0
FATAL=0

echo "== org-style-lint =="

# --- Conflict markers (FATAL). Tangling a conflicted source emits markers
# --- into stowed outputs, breaking every affected script (2026-08-05). -------
for f in *.org; do
  if grep -qE '^<<<<<<< |^=======$|^>>>>>>> ' "$f"; then
    echo "!! $f: unresolved merge conflict markers — tangling would break stowed scripts"
    grep -nE '^<<<<<<< |^=======$|^>>>>>>> ' "$f" | head -5 | sed 's/^/   /'
    FATAL=1
  fi
done

# --- Structural integrity (FATAL). Root sources only: matches tangle scope. ---
for f in *.org; do
  out="$(awk '
    { low = tolower($0); sub(/^[ \t]+/, "", low) }
    low ~ /^#\+begin_(src|example|export)/ {
      if (inblk) printf "%s:%d: #+BEGIN inside the block opened at line %d - close it or escape as ,#+\n", FILENAME, FNR, inblk
      else inblk = FNR
      next
    }
    low ~ /^#\+end_(src|example|export)/ {
      if (!inblk) printf "%s:%d: #+END without a matching #+BEGIN (stray or duplicated)\n", FILENAME, FNR
      inblk = 0
      next
    }
    inblk && /^\*+ / {
      printf "%s:%d: column-0 headline inside a block breaks the block - escape it with a leading comma (,*)\n", FILENAME, FNR
    }
    END { if (inblk) printf "%s: block opened at line %d never closed before EOF\n", FILENAME, inblk }
  ' "$f")"
  if [ -n "$out" ]; then
    printf '%s\n' "$out" | sed 's/^/!! /'
    FATAL=1
  fi
done

# --- Module size budget (warn) ----------------------------------------------
BUDGET_KB=100
for f in *.org; do
  kb=$(( $(stat -c %s "$f") / 1024 ))
  if [ "$kb" -gt "$BUDGET_KB" ]; then
    echo "?? $f is ${kb} KB (budget ${BUDGET_KB} KB) - split it; see the module map in README.org"
    STATUS=1
  fi
done

# A root module either carries a family prefix documented in README.org's
# "Module namespaces" table, or is one of the standalone modules below that
# predate this check. The prefixes are read from that table, so documenting a
# family IS what makes its prefix legal: a new family is added to the table in
# the same commit, never invented in a filename. Names are lowercase-kebab.
# Never grow NAMING_STANDALONE to silence a new module (added 2026-09-23 after
# agents kept inventing prefixes; see agents.org Key Files).
# layers is pre-allowlisted for MR !1 (layers.org, standalone stow-layer
# registry) so nothing warns whichever order !1 and this lint merge in.
NAMING_STANDALONE=" app backup browser CODE_OF_SOVEREIGNTY css defaults dotcortex fonts gitreleases guests helmcortex-shims helmcortex-skills-mirror icons layers logseq loom lutris menu music mux nfs odin qemu README root shell shortcuts storage-maintenance style submodules term win x11 "
NAMING_CASE_EXEMPT=" README.org CODE_OF_SOVEREIGNTY.org "
naming_prefixes="$(grep -oE '^\| =[a-z][a-z0-9]*-\*=' README.org 2>/dev/null | sed -E 's/^\| =([a-z0-9]+)-\*=/\1/' | tr '\n' ' ' || true)"
[ -n "$naming_prefixes" ] || naming_prefixes="package desktop emacs agents services sanctuary tools"
for f in *.org; do
  base="${f%.org}"
  case "$NAMING_CASE_EXEMPT" in
    *" $f "*) ;;
    *)
      if ! printf '%s\n' "$f" | grep -qE '^[a-z0-9]+(-[a-z0-9]+)*\.org$'; then
        echo "?? naming: $f is not lowercase-kebab (a-z, 0-9, single hyphens)"
        STATUS=1
      fi
      ;;
  esac
  case "$NAMING_STANDALONE" in *" $base "*) continue ;; esac
  naming_ok=0
  for prefix in $naming_prefixes; do
    case "$base" in "$prefix"|"$prefix"-*) naming_ok=1; break ;; esac
  done
  if [ "$naming_ok" -eq 0 ]; then
    echo "?? naming: $f matches no documented namespace (README.org \"Module namespaces\"): use an existing prefix, or add the family to that table in the same commit"
    STATUS=1
  fi
done

# --- Style: lowercase begin/end (warn) ---------------------------------------
LOWER_BEGIN_RE='^[[:space:]]*#\+begin_src\b'
LOWER_END_RE='^[[:space:]]*#\+end_src\b'
while IFS= read -r -d '' f; do
  if grep -nE "$LOWER_BEGIN_RE|$LOWER_END_RE" "$f" >/dev/null 2>&1; then
    echo "!! Lowercase #+begin_src/#+end_src found (prefer UPPERCASE) → $f"
    grep -nE "$LOWER_BEGIN_RE|$LOWER_END_RE" "$f" || true
    STATUS=1
  fi
done < <(printf '%s\0' **/*.org)

if [ "$FATAL" -ne 0 ]; then
  echo "== org-style-lint: STRUCTURAL ERRORS - a broken parse silently drops outputs. Refusing to pass. =="
  exit 1
fi
if [ "${STRICT_LINT:-0}" = "1" ] && [ "$STATUS" -ne 0 ]; then
  exit 1
fi
exit 0

# Safe stow with timestamped backups of real files
# LESSON: stow 2.4+ changed conflict message format from:
#   "existing target is neither a link nor a directory: FILE"
# to:
#   "cannot stow PKG/FILE over existing target FILE since neither a link nor a directory"
# We match both patterns plus:
#   "existing target is not owned by stow: FILE"
# If preview reports "existing target is stowed to a different package",
# abort before backing up/removing anything: that target is already repo-owned.
#
# LESSON (2026-07-27, the .codex migration): repair_mutable_agent_dir converts a
# whole-directory agent symlink (~/.codex -> NEXUS/stow/userspace/.codex) back
# into a real local directory and moves the runtime state home. Across a mount
# boundary `mv` is copy-then-unlink, so a LIVE process writing into the source
# makes the unlink fail AFTER the copy has already succeeded. Under `set -e`
# that aborted the entire safe-stow: the migration was actually complete, but
# stow never ran and the failure read as "nothing worked".
# A failed move is now non-fatal and distinguishes the two cases — destination
# already arrived (partial move, only the source needs clearing) versus nothing
# copied (fall back to cp -a). Leftovers are reported loudly and stow continues.
# Never re-tighten this into a bare `mv` under set -e.
#
# Host stow lanes (package-host-excludes.org, "Stow lanes"): a host's `stow`
# rows become --ignore arguments for every package and its `stow-hook` rows
# skip post-stow helpers. `stow -D` honours ignores too, so links made before a
# path was excluded are pruned first (links into this checkout only, never real
# files). A host without stow rows gets no arguments and no prune.
safe-stow:
| set -euo pipefail; \
| cd $(HOME)/DotCortex; \
| STOW_GUARD="$(HOME)/.local/bin/dotcortex-stow-target-guard"; \
| [ -x "$$STOW_GUARD" ] || STOW_GUARD="$(HOME)/DotCortex/all/.local/bin/dotcortex-stow-target-guard"; \
| [ -x "$$STOW_GUARD" ] || { echo "safe-stow: missing dotcortex-stow-target-guard — refusing to continue" >&2; exit 1; }; \
| "$$STOW_GUARD" --packages "$(STOW_PKGS)"; \
| if [ "$(STOW_CONFLICTS)" = adopt ]; then \
|   ADOPT="$(HOME)/.local/bin/dotcortex-stow-adopt"; \
|   [ -x "$$ADOPT" ] || ADOPT="$(HOME)/DotCortex/all/.local/bin/dotcortex-stow-adopt"; \
|   echo ">> adopt pre-existing files (STOW_CONFLICTS=adopt)"; \
|   "$$ADOPT" --apply --dir "$(HOME)/DotCortex" --target "$(STOW_TARGET)" --packages "$(STOW_PKGS)"; \
| fi; \
| repair_mutable_agent_dir() { \
|   local rel="$$1" target="$(HOME)/$$1" resolved entry name rel_entry ts backup; \
|   [ -L "$$target" ] || return 0; \
|   resolved=$$(readlink -f "$$target" 2>/dev/null || true); \
|   echo ">> repair mutable agent dir $$target"; \
|   rm -f "$$target"; \
|   mkdir -p "$$target"; \
|   [ -n "$$resolved" ] && [ -d "$$resolved" ] || return 0; \
|   for entry in "$$resolved"/* "$$resolved"/.[!.]* "$$resolved"/..?*; do \
|     [ -e "$$entry" ] || [ -L "$$entry" ] || continue; \
|     name=$$(basename "$$entry"); \
|     rel_entry=$${entry#$(HOME)/DotCortex/}; \
|     if git ls-files --error-unmatch "$$rel_entry" >/dev/null 2>&1 || git ls-files "$$rel_entry/" | grep -q .; then \
|       continue; \
|     fi; \
|     if [ -e "$$target/$$name" ] || [ -L "$$target/$$name" ]; then \
|       ts=$$(date +%Y%m%d-%H%M%S); \
|       backup="$$target/$$name.bak.$$ts"; \
|       echo "   backup existing $$target/$$name -> $$backup"; \
|       mv "$$target/$$name" "$$backup"; \
|     fi; \
|     echo "   move runtime $$entry -> $$target/$$name"; \
|     if mv "$$entry" "$$target/$$name" 2>/dev/null; then \
|       continue; \
|     fi; \
|     if [ -e "$$target/$$name" ] || [ -L "$$target/$$name" ]; then \
|       echo "   !! partial move: $$target/$$name arrived, source not released"; \
|     else \
|       cp -a "$$entry" "$$target/$$name" 2>/dev/null || { \
|         echo "   !! FAILED to migrate $$entry — left in place, stow continuing"; \
|         continue; \
|       }; \
|     fi; \
|     rm -rf "$$entry" 2>/dev/null || \
|       echo "   !! source still held: $$entry — stop the app writing it, then re-run"; \
|   done; \
| }; \
| repair_mutable_agent_dir .claude; \
| repair_mutable_agent_dir .agents; \
| repair_mutable_agent_dir .opencode; \
| repair_mutable_agent_dir .codex; \
| EXCL="$(HOME)/.local/bin/dotcortex-stow-excludes"; \
| [ -x "$$EXCL" ] || EXCL="$(HOME)/DotCortex/all/.local/bin/dotcortex-stow-excludes"; \
| HOST_STOW_IGNORES=""; \
| if [ -x "$$EXCL" ]; then \
|   HOST_STOW_IGNORES=$$("$$EXCL" ignore-args | tr '\n' ' '); \
|   if [ -n "$$HOST_STOW_IGNORES" ]; then \
|     echo ">> host stow excludes (package-host-excludes.org): $$("$$EXCL" paths | tr '\n' ' ')"; \
|     "$$EXCL" prune --apply --target "$(STOW_TARGET)" --repo "$(HOME)/DotCortex"; \
|   fi; \
| fi; \
| for pkg in $(STOW_PKGS); do \
|   preview_file=$$(mktemp); \
|   echo ">> preview $$pkg"; \
|   stow $(STOW_FLAGS) -n --ignore='\.bak\.' $$HOST_STOW_IGNORES $$pkg >"$$preview_file" 2>&1 || true; \
|   if grep -q 'existing target is stowed to a different package:' "$$preview_file"; then \
|     cat "$$preview_file"; \
|     rm -f "$$preview_file"; \
|     exit 1; \
|   fi; \
|   sed -n \
|       -e 's/.*existing target is neither a link nor a directory: \(.*\)$$/\1/p' \
|       -e 's/.*over existing target \(.*\) since neither.*/\1/p' \
|       -e 's/.*existing target is not owned by stow: \(.*\)$$/\1/p' \
|       "$$preview_file" \
|     | { grep -v '^HelmCortex$$' || true; } \
|     | while read -r t; do \
|         case "$$t" in "$(STOW_TARGET)"/*) abs="$$t" ;; /*) abs="$$t" ;; *) abs="$(STOW_TARGET)/$$t" ;; esac; \
|         if [ -e "$$abs" ] || [ -L "$$abs" ]; then \
|           if [ "$$t" = ".config/opencode/opencode.json" ]; then \
|             HELPER="$(STOW_TARGET)/.local/bin/dotcortex-opencode-config-merge-guard"; \
|             [ -x "$$HELPER" ] || HELPER="$(STOW_TARGET)/DotCortex/all/.local/bin/dotcortex-opencode-config-merge-guard"; \
|             if [ -x "$$HELPER" ]; then \
|               "$$HELPER" apply; \
|               continue; \
|             fi; \
|           fi; \
|           ts=$$(date +%Y%m%d-%H%M%S); \
|           echo "   backup $$abs -> $$abs.bak.$$ts"; \
|           cp -a "$$abs" "$$abs.bak.$$ts"; \
|           echo "   remove  $$abs"; \
|           rm -rf "$$abs"; \
|         fi; \
|       done; \
|   rm -f "$$preview_file"; \
|   echo ">> stow $$pkg"; \
|   if stow $(STOW_FLAGS) --ignore='\.bak\.' $$HOST_STOW_IGNORES $$pkg 2>&1; then \
|     true; \
|   else \
|     echo ">> stow $$pkg conflict — retrying with --ignore=HelmCortex"; \
|     stow $(STOW_FLAGS) --ignore='HelmCortex' --ignore='\.bak\.' $$HOST_STOW_IGNORES $$pkg; \
|   fi; \
| done; \
| case " $(STOW_PKGS) " in \
|   *" all "*|*" linux "*) \
|     HELPER="$(HOME)/.local/bin/icons-home-sync"; \
|     [ -x "$$HELPER" ] || HELPER="$(HOME)/DotCortex/all/.local/bin/icons-home-sync"; \
|     if [ -x "$$EXCL" ] && "$$EXCL" skip-hook icons-home-sync; then \
|       echo ">> icons-sync skipped (host stow-hook exclude)"; \
|     elif [ -x "$$HELPER" ]; then \
|       echo ">> icons-sync"; \
|       "$$HELPER"; \
|     fi; \
|     ;; \
| esac
| case " $(STOW_PKGS) " in \
|   *" t480s "*) \
|     HELPER="$(HOME)/.local/bin/t480s-plank-apply"; \
|     [ -x "$$HELPER" ] || HELPER="$(HOME)/DotCortex/linux/.local/bin/t480s-plank-apply"; \
|     [ -x "$$HELPER" ] || { echo "safe-stow: missing T480s Plank apply helper: $$HELPER" >&2; exit 1; }; \
|     echo ">> t480s-plank-apply"; \
|     "$$HELPER"; \
|     ;; \
| esac
