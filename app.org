* App manifest
Columns: NAME METHOD URL ROOT BIN_DIR LINK_MODE NOTE R1
- METHOD: how to install (curl_sh, curl_bash, tar_gz, curl_bin)
  - curl_bin: download URL straight to ~/.local/bin/NAME and chmod +x. The
    token =\{arch\}= in the URL expands to =uname -m= (x86_64 / aarch64) so one
    manifest line serves every architecture in the fleet, including the phone.
- ROOT: absolute or ~ path to app root, default ~/.local/NAME.app
- BIN_DIR: relative directory under ROOT containing executables, default bin
- LINK_MODE:
  - all   = symlink all executables from BIN_DIR into ~/.local/bin
  - main:NAME = symlink only that executable name from BIN_DIR
  - none  = do not create symlinks (for odd cases)
- NOTE/R1: method-specific extras; =tar_gz= and =curl_bin= accept =sha256=<digest>= or =sha512=<digest>= (verified on install; always pin one); prefix a field with =env:= to pass an environment assignment to curl-based installers

* herdr-web tailnet launcher
Serves the herdr-web browser bridge bound to this machine's Tailscale IPv4 only. The bridge has no authentication of its own, so the tailnet is the auth boundary — it must never listen on 0.0.0.0 or a LAN interface. The IP is derived at runtime from the tailscale CLI, keeping addresses out of DotCortex source (rule 21). Requires a running herdr daemon at protocol 16+ (herdr >= 0.7.2).

#+BEGIN_SRC bash :tangle all/.local/bin/herdr-web-tailnet :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/herdr-web-tailnet
set -euo pipefail

# Herdr sessions inherit the server process environment. NO_COLOR is a
# client-side display preference that may be injected by an agent harness or an
# interactive launcher; carrying it into the daemon makes Flyline suppress all
# Bash prompt colour in every Herdr session. Keep the preference outside the
# server boundary without altering the user's shell or desktop environment.
unset NO_COLOR
# XDG autostart and agent launchers run without a real terminal; the server
# inherits TERM=linux and passes it to every pane, suppressing Flyline colours.
# Pin a colour-capable TERM at the daemon boundary.
export TERM=xterm-256color

port="${HERDR_WEB_PORT:-4000}"
tailscale_bin="${HERDR_WEB_TAILSCALE_BIN:-tailscale}"
herdr_bin="${HERDR_WEB_HERDR_BIN:-herdr}"
retry_seconds="${HERDR_WEB_RETRY_SECONDS:-5}"

# --autostart: XDG-autostart mode — wait out the tailscaled login race and
# respawn the bridge if it dies, logging to ~/.local/state/herdr-web/.
autostart=0
if [ "${1:-}" = "--autostart" ]; then
  autostart=1
  shift
fi

resolve_ip() {
  # A missing/not-yet-ready tailscaled is expected during desktop login.
  # `set -e -o pipefail` would otherwise exit on the first failed probe,
  # before --autostart reaches its retry loop.
  "$tailscale_bin" ip -4 2>/dev/null | head -n1 || true
}

log_dir=""
if [ "$autostart" -eq 1 ]; then
  log_dir="$HOME/.local/state/herdr-web"
  mkdir -p "$log_dir"
  exec 9>"$log_dir/supervisor.lock"
  if ! flock -n 9; then
    # The boot watchdog and XDG fallback may meet.  Exactly one supervisor
    # owns both the daemon and bridge; every later invocation exits cleanly.
    exit 0
  fi
  printf '%s herdr-web-tailnet: autostart supervisor starting\n' \
    "$(date -Is)" >> "$log_dir/autostart.log"
fi

ip="$(resolve_ip)"
if [ "$autostart" -eq 1 ]; then
  tries=0
  while [ -z "$ip" ]; do
    if [ "$tries" -eq 0 ] || [ $((tries % 12)) -eq 0 ]; then
      printf '%s herdr-web-tailnet: waiting for Tailscale IPv4\n' \
        "$(date -Is)" >> "$log_dir/autostart.log"
    fi
    sleep "$retry_seconds"
    ip="$(resolve_ip)"
    tries=$((tries + 1))
  done
fi
if [ -z "$ip" ]; then
  msg="herdr-web-tailnet: no Tailscale IPv4 address after startup wait (is tailscaled running?)"
  echo "$msg" >&2
  [ -n "$log_dir" ] && printf '%s %s\n' "$(date -Is)" "$msg" >> "$log_dir/autostart.log"
  exit 1
fi

echo "herdr-web-tailnet: serving http://$ip:$port (tailnet only)"
# Invoke the app-root script, not the ~/.local/bin symlink: upstream's bin/herdr-web
# locates its static web assets via ${BASH_SOURCE[0]} without resolving symlinks, so
# a symlinked invocation serves 404s for the entire UI.
# --allow-origin http://localhost: the bundled Android app serves its UI from a
# localhost origin and connects out to the bridge; without this the origin guard
# rejects it ("failed to fetch"). Browsers connect same-origin and are unaffected.
app="${HERDR_WEB_APP:-$HOME/.local/herdr-web.app/bin/herdr-web}"

if [ "$autostart" -eq 1 ]; then
  while :; do
    # The bridge refuses to start without a live herdr daemon (protocol 16+),
    # and nothing else raises one at login — so the autostart loop owns it.
    # Grep the status text: exit codes unverified for the down case
    # (observed output is "status: running" / "status: not running").
    if ! "$herdr_bin" status server 2>/dev/null | grep -q "^status: running"; then
      echo "herdr-web-tailnet: no herdr daemon, starting one" >> "$log_dir/autostart.log"
      nohup "$herdr_bin" server >> "$log_dir/herdr-server.log" 2>&1 &
      sleep 3
    fi
    "$app" --host "$ip" --port "$port" --allow-origin http://localhost "$@" >> "$log_dir/autostart.log" 2>&1 || true
    sleep 5
    ip="$(resolve_ip)"
    while [ -z "$ip" ]; do sleep "$retry_seconds"; ip="$(resolve_ip)"; done
  done
fi

exec "$app" --host "$ip" --port "$port" --allow-origin http://localhost "$@"
#+END_SRC

* herdr stack update gate

Deterministic update gate for the herdr + herdr-web pair: BRIDGE LEADS, HERDR FOLLOWS. herdr-web (kcosr/herdr-web, third-party) pins a maximum herdr daemon protocol; a herdr release that outruns it silently kills the phone/web access path (herdr 0.7.5/protocol 17 vs bridge ceiling 16, discovered 2026-07-27 when Hermes killed a live server chasing exactly this). The compatibility oracle is a real handshake in a fully isolated sandbox (own HOME/XDG, own HERDR_SOCKET_PATH via env -i, random loopback port) — never a static grep, because the bridge's ceiling only exists in its runtime refusal text. herdr itself never self-updates unattended (disassembly-verified: the background updater thread is check-only; the install path is reachable only from the =herdr update= CLI), so this script is the sole mover of both versions. Default is =--dry-run=; =--selftest= validates the classifier against known-good and known-bad pairs before any verdict is trusted. Validated live on the X230 2026-07-27/28: bridge v0.3.2 -> v0.3.3 promoted under a running daemon, herdr correctly held at protocol 16.

#+BEGIN_SRC bash :tangle all/.local/bin/herdr-stack-update :shebang "#!/usr/bin/env bash" :mkdirp yes
# herdr-stack-update — deterministic update gate for the herdr + herdr-web stack.
#
# Policy: BRIDGE LEADS, HERDR FOLLOWS.
# herdr-web (kcosr/herdr-web) is a third-party companion that pins a maximum
# Herdr daemon protocol. A herdr release whose protocol exceeds the bridge's
# ceiling silently decapitates the phone/web access path. So no herdr version is
# ever promoted until a real bridge has been observed to serve against it.
#
# The compatibility oracle is a live handshake, never a static grep: the bridge's
# protocol ceiling is not present as a number in its binary — only its runtime
# refusal text reveals it.
#
# Every handshake runs against a fully isolated sandbox daemon (own HOME, own
# XDG_*, own HERDR_SOCKET_PATH) on a random 127.0.0.1 port. The live daemon and
# the live bridge are never contacted, signalled, or reconfigured.
#
# Default mode is --dry-run: nothing is installed, every decision is printed.

set -uo pipefail

# ---------------------------------------------------------------- configuration

HERDR_REPO="herdrdev/herdr"  # upstream moved orgs 2026-08 (was ogulcancelik/herdr)
BRIDGE_REPO="kcosr/herdr-web"

LAB="${HERDR_GATE_LAB:-$HOME/tmp/herdr-gate-lab}"
STAGE="$LAB/stage"
RUN="$LAB/run"
LOGS="$LAB/logs"

INSTALLED_HERDR="${HERDR_GATE_INSTALLED_HERDR:-$HOME/.local/bin/herdr}"
INSTALLED_BRIDGE_ROOT="${HERDR_GATE_INSTALLED_BRIDGE:-$HOME/.local/herdr-web.app}"

DRY_RUN=1
HANDOFF=0
KEEP=0
SELFTEST=0
PLATFORM_HERDR_ASSET="herdr-linux-x86_64"
PLATFORM_BRIDGE_ASSET_GLOB="linux-x86_64.tar.gz"

# Sandbox PIDs started by this script. Torn down by exact PID only — never by
# name pattern, because a name pattern would match the live production daemon.
SANDBOX_PIDS=()

# ---------------------------------------------------------------------- logging

log()  { printf '%s\n' "$*"; }
info() { printf '[info]  %s\n' "$*"; }
warn() { printf '[warn]  %s\n' "$*" >&2; }
err()  { printf '[error] %s\n' "$*" >&2; }
step() { printf '\n=== %s ===\n' "$*"; }
# Every skip/drop is announced. No silent decisions.
skip() { printf '[skip]  %s\n' "$*"; }

usage() {
  cat <<'USAGE'
usage: herdr-stack-update [--dry-run|--apply] [--handoff] [--keep] [--help]

  --dry-run   (default) resolve, stage, handshake-test, and print verdicts only.
  --apply     perform promotion. Refuses while a live `herdr server` is running
              unless --handoff is also given and the installed herdr supports
              live handoff.
  --handoff   acknowledge that promotion may use herdr's live-handoff path.
  --keep      keep sandbox homes/sockets after the run (they are kept anyway
              inside the lab dir; this also keeps the extracted candidates).
  --selftest  validate the classifier itself against known ground truth: a
              known-good pair must classify PASS and a known-bad pair must
              classify FAIL-newer. Installs nothing. Run this before trusting
              any verdict — a detector that cannot fail is not a detector.
USAGE
}

while [ $# -gt 0 ]; do
  case "$1" in
    --dry-run) DRY_RUN=1 ;;
    --apply)   DRY_RUN=0 ;;
    --handoff) HANDOFF=1 ;;
    --keep)    KEEP=1 ;;
    --selftest) SELFTEST=1 ;;
    -h|--help) usage; exit 0 ;;
    *) err "unknown argument: $1"; usage; exit 2 ;;
  esac
  shift
done

# --------------------------------------------------------------------- teardown

teardown_sandbox() {
  local pid
  for pid in "${SANDBOX_PIDS[@]:-}"; do
    [ -n "${pid:-}" ] || continue
    if kill -0 "$pid" 2>/dev/null; then
      kill "$pid" 2>/dev/null
    fi
  done
  sleep 1
  for pid in "${SANDBOX_PIDS[@]:-}"; do
    [ -n "${pid:-}" ] || continue
    kill -0 "$pid" 2>/dev/null && kill -9 "$pid" 2>/dev/null
  done
  SANDBOX_PIDS=()
}
trap teardown_sandbox EXIT INT TERM

# ------------------------------------------------------------------- discovery

# latest_tag <owner/repo> -> vX.Y.Z on stdout
# Primary: GitHub API. Fallback (rate limit / API outage): the Location header of
# the /releases/latest redirect, which needs no token and no quota.
latest_tag() {
  local repo="$1" tag=""
  tag=$(curl -sf --max-time 20 "https://api.github.com/repos/$repo/releases/latest" 2>/dev/null \
        | sed -n 's/.*"tag_name": *"\([^"]*\)".*/\1/p' | head -1)
  if [ -n "$tag" ]; then
    printf '%s' "$tag"; return 0
  fi
  warn "GitHub API gave no tag for $repo (rate limit?); falling back to redirect header"
  tag=$(curl -sI --max-time 20 "https://github.com/$repo/releases/latest" 2>/dev/null \
        | tr -d '\r' | sed -n 's|^[Ll]ocation:.*/releases/tag/||p' | head -1)
  [ -n "$tag" ] || return 1
  printf '%s' "$tag"
}

asset_url() {
  local repo="$1" pattern="$2"
  curl -sf --max-time 20 "https://api.github.com/repos/$repo/releases/latest" 2>/dev/null \
    | sed -n 's/.*"browser_download_url": *"\([^"]*\)".*/\1/p' \
    | grep -- "$pattern" | grep -v '\.sha256$' | head -1
}

ver_num() { printf '%s' "${1#v}"; }

# ver_gt A B -> true if A > B (dotted numeric)
ver_gt() {
  local a b
  a=$(ver_num "$1"); b=$(ver_num "$2")
  [ "$a" = "$b" ] && return 1
  [ "$(printf '%s\n%s\n' "$a" "$b" | sort -V | tail -1)" = "$a" ]
}

protocol_of_binary() {
  strings "$1" 2>/dev/null | grep -oE '"protocol": *[0-9]+' | grep -oE '[0-9]+' | sort -u | head -1
}

# The bridge exposes no --version and embeds no version string, so identity is
# established by content hash. A candidate whose binary hashes equal to the
# installed one is the same build, whatever the tag says.
bridge_fingerprint() {
  [ -x "$1/bin/herdr-web-bridge" ] || { printf 'absent'; return; }
  sha256sum "$1/bin/herdr-web-bridge" | cut -d' ' -f1
}

free_port() {
  python3 - <<'PY'
import socket
s = socket.socket(); s.bind(("127.0.0.1", 0))
print(s.getsockname()[1]); s.close()
PY
}

# ---------------------------------------------------------------- the handshake

# handshake <herdr-binary> <bridge-root> <tag>
# Prints one classification line to stdout:
#   PASS <http-code>
#   FAIL-newer <daemon-protocol> <bridge-ceiling>
#   FAIL-older <daemon-protocol> <bridge-floor>
#   FAIL-other <reason>
# Diagnostics go to stderr and to $LOGS/<tag>-{daemon,bridge}.log.
handshake() {
  local herdr_bin="$1" bridge_root="$2" tag="$3"
  local sh="$RUN/$tag-home" sock="$RUN/$tag.sock" csock="$RUN/$tag-client.sock"
  local dlog="$LOGS/$tag-daemon.log" blog="$LOGS/$tag-bridge.log"
  local port dpid bpid i code

  if [ ! -x "$herdr_bin" ]; then printf 'FAIL-other missing-herdr-binary\n'; return; fi
  if [ ! -x "$bridge_root/bin/herdr-web-bridge" ]; then printf 'FAIL-other missing-bridge-binary\n'; return; fi

  rm -rf "$sh"; mkdir -p "$sh/.config" "$sh/.local/state" "$sh/.local/share" "$sh/.cache"
  rm -f "$sock" "$csock"
  port=$(free_port)

  # env -i: the sandbox inherits nothing. Any stray HERDR_SOCKET_PATH in the
  # operator's shell would otherwise aim the sandbox at the live daemon.
  env -i PATH=/usr/bin:/bin TERM=xterm-256color HOME="$sh" \
      XDG_CONFIG_HOME="$sh/.config" XDG_STATE_HOME="$sh/.local/state" \
      XDG_DATA_HOME="$sh/.local/share" XDG_CACHE_HOME="$sh/.cache" \
      XDG_RUNTIME_DIR="$RUN" \
      HERDR_SOCKET_PATH="$sock" HERDR_CLIENT_SOCKET_PATH="$csock" \
      "$herdr_bin" server >"$dlog" 2>&1 &
  dpid=$!
  SANDBOX_PIDS+=("$dpid")

  for i in $(seq 1 60); do [ -S "$sock" ] && break; sleep 0.25; done
  if [ ! -S "$sock" ]; then
    printf 'FAIL-other sandbox-daemon-no-socket\n'
    kill "$dpid" 2>/dev/null
    return
  fi
  # Isolation assertion: the sandbox socket must live under the lab tree.
  case "$sock" in
    "$LAB"/*) : ;;
    *) err "REFUSING: sandbox socket escaped the lab dir: $sock"; kill "$dpid" 2>/dev/null; printf 'FAIL-other socket-escaped-lab\n'; return ;;
  esac

  env -i PATH=/usr/bin:/bin HOME="$sh" \
      XDG_CONFIG_HOME="$sh/.config" XDG_STATE_HOME="$sh/.local/state" \
      XDG_DATA_HOME="$sh/.local/share" XDG_CACHE_HOME="$sh/.cache" \
      XDG_RUNTIME_DIR="$RUN" HERDR_SOCKET_PATH="$sock" \
      "$bridge_root/bin/herdr-web-bridge" \
        --host 127.0.0.1 --port "$port" \
        --static-dir "$bridge_root/share/herdr-web/web" >"$blog" 2>&1 &
  bpid=$!
  SANDBOX_PIDS+=("$bpid")

  # Give the bridge time either to bind or to refuse.
  code=""
  for i in $(seq 1 20); do
    sleep 0.5
    kill -0 "$bpid" 2>/dev/null || break
    code=$(curl -s -o /dev/null -w '%{http_code}' --max-time 5 "http://127.0.0.1:$port/" 2>/dev/null)
    case "$code" in 200|30[0-9]) break ;; esac
  done

  local verdict
  if kill -0 "$bpid" 2>/dev/null && case "$code" in 200|30[0-9]) true ;; *) false ;; esac; then
    verdict="PASS $code"
  else
    local msg dproto ceiling floor
    msg=$(tr -d '\r' < "$blog")
    dproto=$(printf '%s' "$msg" | sed -n 's/.*Herdr daemon protocol \([0-9]\+\).*/\1/p' | head -1)
    if printf '%s' "$msg" | grep -q 'is newer than this herdr-web bridge supports'; then
      ceiling=$(printf '%s' "$msg" | sed -n 's/.*need protocol \([0-9]\+\) or older.*/\1/p' | head -1)
      verdict="FAIL-newer ${dproto:-?} ${ceiling:-?}"
    elif printf '%s' "$msg" | grep -q 'is too old for herdr-web'; then
      floor=$(printf '%s' "$msg" | sed -n 's/.*need protocol \([0-9]\+\).*/\1/p' | head -1)
      verdict="FAIL-older ${dproto:-?} ${floor:-?}"
    else
      verdict="FAIL-other $(printf '%s' "$msg" | head -1 | tr ' ' '_' | cut -c1-160)"
      [ -n "$(printf '%s' "$msg")" ] || verdict="FAIL-other bridge-died-without-message(http=${code:-none})"
    fi
  fi

  kill "$bpid" 2>/dev/null; kill "$dpid" 2>/dev/null
  sleep 1
  kill -0 "$bpid" 2>/dev/null && kill -9 "$bpid" 2>/dev/null
  kill -0 "$dpid" 2>/dev/null && kill -9 "$dpid" 2>/dev/null
  printf '%s\n' "$verdict"
}

  step "live services after selftest"
  LIVE_AFTER=$(pgrep -fa 'herdr server|herdr-web-bridge' | grep -v 'herdr-stack-update' || true)

    || { err "LIVE PROCESS SET CHANGED"; rc=4; }
  exit $rc
fi

step "installed versions"
INST_HERDR_VER=""
if [ -x "$INSTALLED_HERDR" ]; then
  INST_HERDR_VER=$("$INSTALLED_HERDR" --version 2>/dev/null | awk '{print $2}')
  info "herdr        $INST_HERDR_VER  ($INSTALLED_HERDR, protocol $(protocol_of_binary "$INSTALLED_HERDR"))"
else

INST_BRIDGE_FP=$(bridge_fingerprint "$INSTALLED_BRIDGE_ROOT")

BRIDGE_TAG=$(latest_tag "$BRIDGE_REPO") || { err "cannot resolve latest $BRIDGE_REPO tag"; exit 1; }

  info "downloading herdr $HERDR_TAG"

CAND_BRIDGE_ROOT="$STAGE/bridge-$(ver_num "$BRIDGE_TAG")"
if [ -x "$CAND_BRIDGE_ROOT/bin/herdr-web-bridge" ]; then
  skip "candidate bridge already staged: $CAND_BRIDGE_ROOT"
else

step "verdicts"
PROMOTE_HERDR=0
PROMOTE_BRIDGE=0
HERDR_REASON=""
BRIDGE_REASON=""

if [ "${T1%% *}" = "PASS" ]; then
  if [ -n "$INST_HERDR_VER" ] && ! ver_gt "$HERDR_TAG" "$INST_HERDR_VER"; then
    HERDR_REASON="already at or ahead of $HERDR_TAG"
  else
    PROMOTE_HERDR=1
    HERDR_REASON="pair handshake PASS with $BEST_BRIDGE_LABEL bridge"
  fi
  if [ "$BRIDGE_IS_NEW" = 1 ]; then
    PROMOTE_BRIDGE=1
    BRIDGE_REASON="pair handshake PASS; promoted together with herdr"
  else
    BRIDGE_REASON="candidate is byte-identical to installed bridge"
  fi
else

    FAIL-newer*)

      HERDR_REASON="HELD: daemon protocol ${2} exceeds bridge ceiling ${3} ($BEST_BRIDGE_LABEL bridge refuses to start)"
      ;;
    FAIL-older*)
      set -- $T1
      HERDR_REASON="HELD: daemon protocol ${2} is below bridge floor ${3}"
      ;;
    *)
      HERDR_REASON="HELD: handshake inconclusive ($T1) — see $LOGS/t1-bridge.log"
      ;;
  esac
  # Bridge may still advance on its own if it works with the daemon we run today.
  if [ "$BRIDGE_IS_NEW" = 1 ] && [ "${T2%% *}" = "PASS" ]; then
    PROMOTE_BRIDGE=1
    BRIDGE_REASON="passes against the CURRENTLY INSTALLED herdr ${INST_HERDR_VER}; safe to advance alone"
  elif [ "$BRIDGE_IS_NEW" = 0 ]; then
    BRIDGE_REASON="no new bridge build available"
  else
    BRIDGE_REASON="HELD: candidate bridge did not pass against installed herdr ${INST_HERDR_VER} ($T2)"
  fi
fi

printf 'VERDICT herdr-web %-8s %s\n'  "$BRIDGE_TAG" "$([ "$PROMOTE_BRIDGE" = 1 ] && echo "PROMOTE — $BRIDGE_REASON" || echo "HOLD — $BRIDGE_REASON")"

step "promotion"
if [ "$DRY_RUN" = 1 ]; then
  info "DRY-RUN: nothing will be installed."
  [ "$PROMOTE_HERDR" = 1 ]  && log "WOULD install $CAND_HERDR -> $INSTALLED_HERDR"        || skip "would NOT touch $INSTALLED_HERDR"
  [ "$PROMOTE_BRIDGE" = 1 ] && log "WOULD install $CAND_BRIDGE_ROOT -> $INSTALLED_BRIDGE_ROOT" || skip "would NOT touch $INSTALLED_BRIDGE_ROOT"
else
  # Bridge promotion is safe with a live daemon AND a live bridge: the install is
  # a directory swap via mv (rename), so the running bridge keeps its old inode
  # and the new build takes effect on its next (re)start. Only herdr promotion
  # needs the live-server guard.
  if [ "$PROMOTE_BRIDGE" = 1 ]; then
    log "installing bridge $BRIDGE_TAG (directory swap; running bridge keeps old inode until restarted)"
    STAGED_NEW="$INSTALLED_BRIDGE_ROOT.new.$$"
    rm -rf "$STAGED_NEW"
    cp -a "$CAND_BRIDGE_ROOT" "$STAGED_NEW" || { err "staging copy failed"; exit 5; }
    rm -rf "$INSTALLED_BRIDGE_ROOT.prev"
    mv "$INSTALLED_BRIDGE_ROOT" "$INSTALLED_BRIDGE_ROOT.prev" || { err "backup move failed"; rm -rf "$STAGED_NEW"; exit 5; }
    if mv "$STAGED_NEW" "$INSTALLED_BRIDGE_ROOT"; then
      info "bridge installed; previous kept at $INSTALLED_BRIDGE_ROOT.prev"
    else
      err "swap failed; restoring previous bridge"
      mv "$INSTALLED_BRIDGE_ROOT.prev" "$INSTALLED_BRIDGE_ROOT"
      rm -rf "$STAGED_NEW"
      exit 5
    fi
  else
    skip "bridge not promoted: $BRIDGE_REASON"
  fi
  if [ "$PROMOTE_HERDR" = 1 ]; then
    if [ -n "${LIVE_HERDR_PIDS// /}" ] && [ "$HANDOFF" != 1 ]; then
      err "REFUSING herdr promotion: live herdr server running (pids: $LIVE_HERDR_PIDS) and --handoff not given."
      err "Re-run with --handoff, or promote while no server is running."
      exit 3
    fi
    # Install the STAGED, GATE-TESTED binary — never `herdr update`, which pulls
    # whatever is latest at that moment and can leap past the gate.
    log "installing herdr $HERDR_TAG (staged binary, mv swap)"
    cp "$CAND_HERDR" "$INSTALLED_HERDR.new.$$" && chmod +x "$INSTALLED_HERDR.new.$$" || { err "herdr staging copy failed"; exit 5; }
    cp -p "$INSTALLED_HERDR" "$INSTALLED_HERDR.prev" 2>/dev/null || true
    mv "$INSTALLED_HERDR.new.$$" "$INSTALLED_HERDR" || { err "herdr swap failed"; rm -f "$INSTALLED_HERDR.new.$$"; exit 5; }
    info "herdr installed; previous kept at $INSTALLED_HERDR.prev"
    if [ -n "${LIVE_HERDR_PIDS// /}" ] && [ "$HANDOFF" = 1 ]; then
      if "$INSTALLED_HERDR" server --help 2>&1 | grep -q 'live-handoff'; then
        log "attempting live handoff of the running server"
        "$INSTALLED_HERDR" server live-handoff || warn "live handoff failed; old server still running old binary until restarted"
      else
        warn "installed herdr has no live-handoff; running server keeps old binary until restarted"
      fi
    fi
  else
    skip "herdr not promoted: $HERDR_REASON"
  fi
fi

step "live services after run"
teardown_sandbox
LIVE_AFTER=$(pgrep -fa 'herdr server|herdr-web-bridge' | grep -v 'herdr-stack-update' || true)
printf '%s\n' "${LIVE_AFTER:-<none>}"
if [ "$LIVE_BEFORE" = "$LIVE_AFTER" ]; then
  info "live herdr/bridge processes unchanged"
elif [ "$DRY_RUN" = 1 ]; then
  # A dry run must never change the live process set — that is a hard failure.
  err "LIVE PROCESS SET CHANGED DURING A DRY RUN — investigate immediately"
  exit 4
else
  # An --apply run with --handoff may legitimately replace the server PID.
  warn "live process set changed (expected only if a handoff/restart was part of promotion) — review above"
fi

[ "$KEEP" = 1 ] || skip "sandbox homes left in $RUN for inspection (nothing is auto-deleted in lab mode)"
exit 0

* obsidian-web tailnet gate (caddy login screen)

Serves obsidian-web to the tailnet through a caddy basic_auth login screen. obsidian-web itself has NO authentication and its =POST /api/vaults/open= accepts arbitrary filesystem paths, so it must ONLY ever bind 127.0.0.1 — the gate is the sole tailnet-facing listener. The tailscale IP is derived at runtime (rule 21). The Caddyfile is machine-local at =~/.config/obsidian-web/Caddyfile= and is NOT in DotCortex: it holds the bcrypt password hash, and DotCortex is public. Create it on the serving machine from this template (hash from =caddy hash-password=, password lives in KeePass):

#+BEGIN_EXAMPLE
{
	auto_https off
}

http://{$OBSIDIAN_GATE_IP}:{$OBSIDIAN_GATE_PORT} {
	basic_auth {
		metsatron <bcrypt-hash>
	}
	reverse_proxy 127.0.0.1:38231
}
#+END_EXAMPLE

#+BEGIN_SRC bash :tangle all/.local/bin/obsidian-web-gate :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/obsidian-web-gate
set -euo pipefail

config="${OBSIDIAN_GATE_CONFIG:-$HOME/.config/obsidian-web/Caddyfile}"
if [ ! -r "$config" ]; then
  echo "obsidian-web-gate: missing $config (machine-local basic_auth Caddyfile — template in DotCortex app.org)" >&2
  exit 1
fi

ip="$(tailscale ip -4 2>/dev/null | head -n1)"
if [ -z "$ip" ]; then
  echo "obsidian-web-gate: no Tailscale IPv4 address (is tailscaled running?)" >&2
  exit 1
fi

export OBSIDIAN_GATE_IP="$ip"
export OBSIDIAN_GATE_PORT="${OBSIDIAN_GATE_PORT:-8443}"

echo "obsidian-web-gate: serving http://$ip:$OBSIDIAN_GATE_PORT -> 127.0.0.1:38231 (basic auth)"
exec caddy run --config "$config" --adapter caddyfile
#+END_SRC

chmod +x ~/DotCortex/all/.local/bin/obsidian-web-gate

* FileBrowser Quantum tailnet file manager (kikin-kushi)

Serves FileBrowser Quantum (gtsteffaniak fork, v1.5.x) from kikin-kushi so Mètsàtron can browse and download fleet files from any tailnet device. The binary comes from the app manifest (=curl_bin=, sha256-pinned). The service binds loopback only; the existing kikin Caddy TLS front (port 1996, =kikin-kushi.sphinx-draco.ts.net=) reverse-proxies to it, so FileBrowser itself never listens on a tailnet or LAN interface.

** Why loopback + Caddy rather than tailnet-direct

Unlike herdr-web (no auth at all), FileBrowser Quantum has its own password auth (=auth.methods.password=). Loopback-bind + Caddy TLS still wins: one TLS front already lives on the machine, the hostname gets a real certificate, and the service stays invisible off-tailnet. Same law as obsidian-web: the gate is the sole listener with a tailnet/lAN presence.

** Config (machine-local, NEVER in DotCortex)

FileBrowser Quantum config is YAML at =$HOME/.config/filebrowser/config.yaml=. It holds no secrets (passwords are created in-app and stored in its own DB), but it IS machine-specific — kikin's source list has no business on other hosts, and the admin password lives in KeePass (provision on first run, stop-point for Fable). DotCortex is PUBLIC: the template below uses =$HOME= placeholders only; the provisioning step renders real paths into the machine-local copy.

#+BEGIN_EXAMPLE
server:
  # Loopback only; Caddy TLS front (caddy-serve :9090) is the tailnet listener.
  port: 9091
  listen: 127.0.0.1
  # Machine-local SQLite DB (users, settings). Provisioned beside the config;
  # keep it out of any DotCortex-synced path.
  database: $HOME/.local/share/filebrowser/database.db
  # Absolute cache path: the default "tmp" is RELATIVE to the cwd — running the
  # binary from a worktree litters it with cache dirs (observed 2026-10-09).
  cacheDir: $HOME/.local/state/filebrowser/cache
  sources:
    # Hard drive root: read-only whole-filesystem view. The Admiral's
    # 'hard drive' icon opens here. readOnly is a real per-source flag
    # (v1.5.6 config.generated.yaml) — UI, WebDAV, and API writes are all
    # disabled for this source.
    - name: root
      path: /
      config:
        defaultEnabled: true
        readOnly: true
        private: true   # no shares from this source
        rules:
          - folderPath: "/NEXUS"                    # KeePass vaults + repo mirrors
          - folderPath: "/home/RENDERED_USER/.ssh"   # rendered at provisioning
          - folderPath: "/root/.ssh"
          - folderName: ".ssh"
          - folderPath: "/etc/caddy"
          - folderPath: "/etc/ssl/private"
          - folderPath: "/etc/letsencrypt/live"
          - fileName: ".env"
          - fileEndsWith: ".env"
          - folderName: "node_modules"
          - fileEndsWith: ".kdbx"
          - fileEndsWith: ".kdbx.key"
    # Home: the writable working source.
    - name: home
      path: $HOME
      config:
        defaultEnabled: true
        private: true   # no shares — tailnet-only by policy (sovereignty rule)
        rules:
          - folderPath: "NEXUS"
          - folderPath: ".ssh"
          - folderPath: ".config/filebrowser"
          - folderPath: ".local/share/dotcortex/tests"
          - fileName: ".env"
          - fileEndsWith: ".env"
          - fileEndsWith: ".kdbx"
          - fileEndsWith: ".kdbx.key"
    # Mounted fleet shares.
    - name: mnt
      path: $HOME/mnt
      config:
        defaultEnabled: true
        private: true
    # Pictures gallery.
    - name: pictures
      path: $HOME/Pictures
      config:
        defaultEnabled: true
        private: true
    # Sanctuary Cortex guest home (deep-linked from the Centre desktop icons).
    - name: sanctuary-cortex
      path: $HOME/.local/share/dotcortex/guests/sanctuary-cortex/home
      config:
        defaultEnabled: true
        private: true
auth:
  methods:
    password:
      enabled: true
      signup: false
#+END_EXAMPLE

Notes on rules: Quantum rules are OR-based; each entry is an exclude filter (verified against the v1.5.6-stable =config.generated.yaml=, fetched from the pinned tag 2026-10-09). Key corrections from that reference, Fable's review round: =listen:= is the bind-address key (=host:= does not exist); there is no =frontend.disableShares= — shares are gated per-source with =private: true=; per-source =readOnly: true= exists and covers UI, WebDAV, and API writes. On the root source, ignoreHidden is deliberately NOT set: hidden dot-directories under / are system entries Mètsàtron expects to see in a hard-drive view; the explicit deny rules above carry the security weight. ** Supervisor: filebrowser-supervise

Mirrors herdr-web-tailnet's autostart loop (the correct daemon-owning shape): flock singleton supervisor that respawns the foreground launcher forever. NOT the agentify bounded-attempts shape — that one probes an idempotent owner; a daemon needs a persistent owner loop or the service dies with every supervisor exit. Nothing in it is machine-specific except the host guard — the launcher owns config/port, the cron-apply owns the contract.

#+BEGIN_SRC bash :tangle linux/.local/bin/filebrowser-supervise :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

TAG="filebrowser-supervise"
EXPECTED_HOST=""
DELAY=15
RETRY_SECONDS=5
LAUNCHER="${FILEBROWSER_SUPERVISE_LAUNCHER:-$HOME/.local/bin/filebrowser-serve}"
HOSTNAME_BIN="${FILEBROWSER_SUPERVISE_HOSTNAME_BIN:-hostname}"

while [ "$#" -gt 0 ]; do
  case "$1" in
    --expected-host) [ "$#" -ge 2 ] || { echo "$TAG: --expected-host requires a value" >&2; exit 2; }; EXPECTED_HOST="$2"; shift ;;
    --delay) [ "$#" -ge 2 ] || { echo "$TAG: --delay requires a value" >&2; exit 2; }; DELAY="$2"; shift ;;
    --retry-seconds) [ "$#" -ge 2 ] || { echo "$TAG: --retry-seconds requires a value" >&2; exit 2; }; RETRY_SECONDS="$2"; shift ;;
    -h|--help|help) echo "Usage: $TAG --expected-host HOST [--delay N] [--retry-seconds N]"; exit 0 ;;
    *) echo "$TAG: unknown argument: $1" >&2; exit 2 ;;
  esac
  shift
done

[ -n "$EXPECTED_HOST" ] || { echo "$TAG: --expected-host is required" >&2; exit 2; }
CURRENT_HOST="$($HOSTNAME_BIN -s)"
[ "$CURRENT_HOST" = "$EXPECTED_HOST" ] || { echo "$TAG: wrong host ($CURRENT_HOST != $EXPECTED_HOST), nothing to do" >&2; exit 1; }
[ -x "$LAUNCHER" ] || { echo "$TAG: launcher is not executable: $LAUNCHER" >&2; exit 69; }

LOG_DIR="$HOME/.local/state/filebrowser"
mkdir -p "$LOG_DIR"
exec 9>"$LOG_DIR/supervisor.lock"
if ! flock -n 9; then
  # Boot watchdog and liveness poll may meet; only one supervisor owns the service.
  exit 0
fi

log() { printf '%s %s %s\n' "$(date -Is)" "$TAG" "$*" >> "$LOG_DIR/supervisor.log"; }

log "supervisor starting (host=$CURRENT_HOST, launcher=$LAUNCHER)"

# Boot path: give the machine a moment, then own the daemon forever,
# respawning on death — herdr-web-tailnet --autostart shape.
sleep "$DELAY"
while :; do
  "$LAUNCHER" >> "$LOG_DIR/service.log" 2>&1 || true
  log "service exited, respawning in ${RETRY_SECONDS}s"
  sleep "$RETRY_SECONDS"
done
#+END_SRC

chmod +x ~/DotCortex/linux/.local/bin/filebrowser-supervise
#+END_SRC

_emit_norm() {
  awk -v FS='[ \t]+' '
    function dequote(s){
      if (s == "\"\"") return ""
      if (substr(s,1,1) == "\"" && substr(s,length(s),1) == "\"")
        return substr(s, 2, length(s)-2)
      return s
    }
    $0 ~ /^[[:space:]]*#/ || NF==0 { next }
    {
      n1=$1; n2=$2; n3=$3; n4=$4; n5=$5; n6=$6; extra="";
      # Dequote each extra field individually and drop empties. Joining the raw
      # fields and dequoting the whole string once mangles "launch=n" "" into
      # launch=n" " (only the outer quote pair is stripped, inner quotes survive)
      # and leaks a trailing "" into note. Per-field dequoting keeps note clean
      # for the curl_sh -> collect_installer_fields path.
      for (i=7; i<=NF; i++) {
        f = dequote($i)
        if (f != "") extra = extra (extra ? " " : "") f
      }
      printf "%s|%s|%s|%s|%s|%s|%s|%s\n",
        dequote(n1),dequote(n2),dequote(n3),dequote(n4),
        dequote(n5),dequote(n6),extra,""
    }
  ' "$MAN"
}

expand_root() {
  local name="$1" root="$2"

  if [ -z "$root" ]; then
    printf '%s\n' "$HOME/.local/${name}.app"
  elif [ "${root:0:1}" = "~" ]; then
    printf '%s\n' "${root/#\~/$HOME}"
  else
    printf '%s\n' "$root"
  fi
}

meta_value() {
  local key="$1"
  shift

  local field
  for field in "$@"; do
    [ -n "$field" ] || continue
    for field in $field; do
      case "$field" in
        "$key"=*)
          printf '%s\n' "${field#*=}"
          return 0
          ;;
      esac
    done
  done

  return 1
}

expand_env_assignment() {
  local assignment="$1" key value

  key="${assignment%%=*}"
  value="${assignment#*=}"

  case "$value" in
    '~')
      value="$HOME"
      ;;
    '~/'*)
      value="${value#\~}"
      value="$HOME$value"
      ;;
  esac

  printf '%s=%s\n' "$key" "$value"
}

collect_installer_fields() {
  INSTALLER_ENV=()
  INSTALLER_ARGS=()

  local field tok
  for field in "$@"; do
    [ -n "$field" ] || continue
    # Word-split note into tokens (mirrors meta_value) so multi-arg installers
    # like calibre (install_dir=... isolated=y) get separate argv elements
    # instead of one space-joined blob passed as a single argument.
    for tok in $field; do
      case "$tok" in
        env:*=*)
          INSTALLER_ENV+=("$(expand_env_assignment "${tok#env:}")")
          ;;
        *)
          INSTALLER_ARGS+=("$tok")
          ;;
      esac
    done
  done
}

sha256_file() {
  local path="$1"

  if command -v sha256sum >/dev/null 2>&1; then
    sha256sum "$path" | awk '{print $1}'
  else
    shasum -a 256 "$path" | awk '{print $1}'
  fi
}

sha512_file() {
  local path="$1"

  if command -v sha512sum >/dev/null 2>&1; then
    sha512sum "$path" | awk '{print $1}'
  else
    shasum -a 512 "$path" | awk '{print $1}'
  fi
}

install_tar_gz() {
  local name="$1" url="$2" root="$3" note="$4" r1="$5"
  local tmpdir archive expected_sha actual_sha hash_alg strip_components copy_spec copy_src copy_dst

  root="$(expand_root "$name" "$root")"
  tmpdir="$(mktemp -d)"
  archive="$tmpdir/archive.tar.gz"

  echo ">> [$name] install via tar.gz: $url"
  curl -fL "$url" -o "$archive"

  expected_sha="$(meta_value sha512 "$note" "$r1" || true)"
  hash_alg="sha512"
  if [ -z "$expected_sha" ]; then
    expected_sha="$(meta_value sha256 "$note" "$r1" || true)"
    hash_alg="sha256"
  fi
  if [ -n "$expected_sha" ]; then
    if [ "$hash_alg" = "sha512" ]; then
      actual_sha="$(sha512_file "$archive")"
    else
      actual_sha="$(sha256_file "$archive")"
    fi
    if [ "$actual_sha" != "$expected_sha" ]; then
      echo "!! [$name] $hash_alg mismatch: expected $expected_sha, got $actual_sha" >&2

  local symlink_real bin_real

  root="$(expand_root "$name" "$root")"

  [ -z "$bindir" ] && bindir="bin"
  bin_path="${root%/}/$bindir"

  [ -d "$bin_path" ] || {
    echo ">> [$name] no bin dir at $bin_path, skipping link sync"
    return 0
  }

  mkdir -p "$symlink_dir"
  symlink_real="$(readlink -f "$symlink_dir" 2>/dev/null || printf '%s\n' "$symlink_dir")"
  bin_real="$(readlink -f "$bin_path" 2>/dev/null || printf '%s\n' "$bin_path")"

  if [ "$bin_real" = "$symlink_real" ]; then
    echo ">> [$name] bin dir already on PATH at $bin_real, skipping link sync"
    return 0
  fi

  echo ">> [$name] syncing symlinks from $bin_path into $symlink_dir (mode=$linkmode)"

  # Remove any existing symlinks in ~/.local/bin that point into this app's bin dir
  find "$symlink_dir" -maxdepth 1 -type l | while read -r link; do
    target="$(readlink "$link" 2>/dev/null || true)"
    case "$target" in
      "$bin_path"/*)
        rm -f "$link"
        ;;
    esac
  done

  case "${linkmode:-all}" in
    all|"")
      # Link all executable entries in bin_path, including symlinks
      while IFS= read -r exe; do
        local base
        [ -x "$exe" ] || continue
        base="$(basename "$exe")"
        ln -sf "$exe" "$symlink_dir/$base"
      done < <(find "$bin_path" -maxdepth 1 \( -type f -o -type l \))
      ;;

    main:*)
      local main
      main="${linkmode#main:}"
      if [ -x "$bin_path/$main" ]; then
        ln -sf "$bin_path/$main" "$symlink_dir/$main"
      else
        echo ">> [$name] main '$main' not found in $bin_path"
      fi
      ;;

    none)
      # Intentionally no links
      ;;

    *)
      echo ">> [$name] unknown LINK_MODE '$linkmode', skipping linking"
      ;;
  esac
}

# Link XDG share files (desktop entries, icons) from ROOT/share into ~/.local/share.
# Desktop entries are written as real files (not symlinks) with Exec/TryExec
# rewritten to the absolute binary path so they work when the DE's PATH omits
# ~/.local/bin (common on Devuan/sysv-init display managers).
sync_share_for_app() {
  local name="$1" root="$2" bindir="${3:-bin}"
  local share_src share_dst="$HOME/.local/share"
  local preserve_exec=0

  root="$(expand_root "$name" "$root")"
  share_src="${root%/}/share"

  case "$name" in
    antigravity-hub)
      preserve_exec=1
      ;;
  esac

  [ -d "$share_src" ] || return 0

  # Resolve the primary binary — used to rewrite Exec/TryExec to full path
  local bin_exe=""
  local bin_root="${root%/}/${bindir}"
  [ "$bindir" = "." ] && bin_root="${root%/}"
  if [ -n "$bindir" ] && [ "$bindir" != "none" ] && [ -x "$bin_root/$name" ]; then
    bin_exe="$bin_root/$name"
  fi

  local apps_src="$share_src/applications"
  if [ -d "$apps_src" ]; then
    mkdir -p "$share_dst/applications"
    while IFS= read -r f; do
      local base dst
      base="$(basename "$f")"
      dst="$share_dst/applications/$base"
      if [ "$preserve_exec" -eq 0 ] && [ -n "$bin_exe" ]; then
        # Write a corrected copy with absolute Exec/TryExec so the launcher
        # appears even when the DE session PATH does not include ~/.local/bin.
        # Write to a tempfile first — $dst may be a symlink pointing back to
        # $f, and `sed $f > $dst` would truncate $f before reading it.
        local _tmp
        _tmp="$(mktemp)"
        sed -e "s|^TryExec=.*|TryExec=$bin_exe|" \
            -e "s|^Exec=\([^ 	]*\)\(.*\)|Exec=$bin_exe\2|" \
            "$f" > "$_tmp"
        rm -f "$dst"
        mv "$_tmp" "$dst"
      else
        ln -sf "$f" "$dst"
      fi
    done < <(find "$apps_src" -maxdepth 1 -name '*.desktop' -type f)
    { update-desktop-database "$share_dst/applications" 2>/dev/null || true; }
  fi

  local icons_src="$share_src/icons"
  if [ -d "$icons_src" ]; then
    while IFS= read -r f; do
      local rel dst
      rel="${f#${icons_src}/}"
      dst="$share_dst/icons/$rel"
      mkdir -p "$(dirname "$dst")"
      ln -sf "$f" "$dst"
    done < <(find "$icons_src" \( -type f -o -type l \) \( -name '*.png' -o -name '*.svg' \))
    echo ">> [$name] wrote XDG share entries from $share_src"
  fi
}

echo "== app apply =="

while IFS='|' read -r name method url root bindir linkmode note r1
do
  [ -n "$name" ] || continue
  [ -n "$APP_ONLY" ] && [ "$name" != "$APP_ONLY" ] && continue

  method="${method:-curl_sh}"

  case "$method" in
    curl_sh)
      if [ -z "$url" ]; then
        echo "!! [$name] curl_sh with empty URL, skipping"
      else
        echo ">> [$name] install via curl script: $url"
        collect_installer_fields "$note" "$r1"
        # Build the sh command with optional extra args from NOTE and R1
        cmd=(sh /dev/stdin)
        [ "${#INSTALLER_ARGS[@]}" -gt 0 ] && cmd+=("${INSTALLER_ARGS[@]}")
        if ! curl -L "$url" | env "${INSTALLER_ENV[@]}" "${cmd[@]}"; then
          echo "!! [$name] install failed"
        fi
      fi
      ;;

    curl_bash)
      if [ -z "$url" ]; then
        echo "!! [$name] curl_bash with empty URL, skipping"
      else
        echo ">> [$name] install via curl+bash script: $url"
        collect_installer_fields "$note" "$r1"
        cmd=(bash)
        [ "${#INSTALLER_ARGS[@]}" -gt 0 ] && cmd+=("${INSTALLER_ARGS[@]}")
        if ! curl -L "$url" | env "${INSTALLER_ENV[@]}" "${cmd[@]}"; then
          echo "!! [$name] install failed"
        fi
      fi
      ;;

    tar_gz)
      if [ -z "$url" ]; then
        echo "!! [$name] tar_gz with empty URL, skipping"
      else
        if ! install_tar_gz "$name" "$url" "$root" "$note" "$r1"; then
          echo "!! [$name] install failed"
        fi
      fi
      ;;

    curl_bin)
      if [ -z "$url" ]; then
        echo "!! [$name] curl_bin with empty URL, skipping"
      else
        # {arch} expands to `uname -m` (x86_64 / aarch64) so one manifest line
        # serves every architecture in the fleet, including the aarch64 phone.
        resolved_url="${url//\{arch\}/$(uname -m)}"
        dest="$HOME/.local/bin/${name}"
        mkdir -p "$HOME/.local/bin"
        echo ">> [$name] install binary via curl: $resolved_url -> $dest"
        # Download to a temp file and rename: writing onto $dest in place
        # fails with "text file busy" whenever the binary is a running
        # daemon (herdr, 2026-07-11); rename is atomic and always lands.
        tmp_dest="$dest.tmp.$$"
        if curl -fsSL "$resolved_url" -o "$tmp_dest"; then
          # Checksum gate (same law as tar_gz): sha512= preferred, sha256= fallback.
          # A curl_bin line without a digest installs UNVERIFIED — always pin one.
          expected_sha="$(meta_value sha512 "$note" "$r1" || true)"
          hash_alg="sha512"
          if [ -z "$expected_sha" ]; then
            expected_sha="$(meta_value sha256 "$note" "$r1" || true)"
            hash_alg="sha256"
          fi
          if [ -n "$expected_sha" ]; then
            if [ "$hash_alg" = "sha512" ]; then
              actual_sha="$(sha512_file "$tmp_dest")"
            else
              actual_sha="$(sha256_file "$tmp_dest")"
            fi
            if [ "$actual_sha" != "$expected_sha" ]; then
              echo "!! [$name] $hash_alg mismatch: expected $expected_sha, got $actual_sha" >&2
              rm -f "$tmp_dest"
              echo "!! [$name] install failed"
            else
              chmod +x "$tmp_dest"
              mv -f "$tmp_dest" "$dest"
            fi
          else
            chmod +x "$tmp_dest"
            mv -f "$tmp_dest" "$dest"
          fi
        else
          rm -f "$tmp_dest"
          echo "!! [$name] install failed"
        fi
      fi
      ;;

    ""|none)
      echo ">> [$name] method empty, skipping install, still syncing links if root exists"
      ;;

    *)
      echo ">> [$name] unknown method '$method', skipping install, still syncing links if root exists"
      ;;
  esac

  # Always attempt to sync links and XDG share entries for this app
  sync_links_for_app "$name" "$root" "$bindir" "$linkmode"
  sync_share_for_app "$name" "$root" "$bindir"

done < <(_emit_norm)

echo "Done."
#+END_SRC

* Health
#+BEGIN_SRC bash :tangle all/.local/bin/app-health :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/app-health
set -euo pipefail

MAN="$HOME/DotCortex/all/.app/manifest/apps.ssv"
APP_ONLY="${APP_ONLY:-}"

if [ ! -f "$MAN" ]; then
  echo "No app manifest at $MAN"
  exit 0
fi

_emit_norm() {
  awk -v FS='[ \t]+' '
    function dequote(s){
      if (s == "\"\"") return ""
      if (substr(s,1,1) == "\"" && substr(s,length(s),1) == "\"")
        return substr(s, 2, length(s)-2)
      return s
    }
    $0 ~ /^[[:space:]]*#/ || NF==0 { next }
    {
      n1=$1; n2=$2; n3=$3; n4=$4; n5=$5; n6=$6; n7=$7; n8=$8;
      printf "%s|%s|%s|%s|%s|%s|%s|%s\n",
        dequote(n1),dequote(n2),dequote(n3),dequote(n4),
        dequote(n5),dequote(n6),dequote(n7),dequote(n8)
    }
  ' "$MAN"
}

echo "== app health =="

while IFS='|' read -r name method url root bindir linkmode note r1
do
  [ -n "$name" ] || continue
  [ -n "$APP_ONLY" ] && [ "$name" != "$APP_ONLY" ] && continue

  # Expand root, derive defaults to mirror app-apply
  if [ -z "$root" ]; then
    root="$HOME/.local/${name}.app"
  elif [ "${root:0:1}" = "~" ]; then
    root="${root/#\~/$HOME}"
  fi

  [ -z "$bindir" ] && bindir="bin"
  bin_path="${root%/}/$bindir"

  if command -v "$name" >/dev/null 2>&1; then
    echo "[$name] in PATH at $(command -v "$name")"
  elif [ -x "$bin_path/$name" ]; then
    echo "[$name] installed at $bin_path/$name but not in PATH"
  elif [ -d "$root" ]; then
    echo "[$name] app root present at $root but binary '$name' not found in $bin_path"
  else
    echo "[$name] not found (no PATH entry, no app root at $root)"
  fi

done < <(_emit_norm)
