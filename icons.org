* Home sync helper
Keep icon and theme payloads in DotCortex while making the live home roots real
directories. Icon themes become real directories with repo-backed contents so
only cache files live in =$HOME=. GTK/XFWM theme entries stay repo-backed as
top-level symlinks inside real home theme roots so Stow never tries to replace
the whole tree.

#+BEGIN_SRC python :tangle all/.local/bin/icons-home-sync :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
from __future__ import annotations

import argparse
import shutil
import subprocess
from pathlib import Path


ICON_SYNC_MAP = (
    ("all/.icons", ".icons"),
    ("linux/.icons", ".icons"),
    ("all/.local/share/icons", ".local/share/icons"),
    ("linux/.local/share/icons", ".local/share/icons"),
)

THEME_SYNC_MAP = (
    ("all/.themes", ".themes"),
    ("linux/.themes", ".themes"),
    ("all/.local/share/themes", ".local/share/themes"),
    ("linux/.local/share/themes", ".local/share/themes"),
)

CACHE_NAMES = {"icon-theme.cache", ".icon-theme.cache"}
ROOT_SKIP_NAMES = {".git"}
ICON_SKIP_NAMES = CACHE_NAMES | ROOT_SKIP_NAMES


def remove_path(path: Path) -> None:
    if path.is_dir() and not path.is_symlink():
        shutil.rmtree(path)
    elif path.exists() or path.is_symlink():
        path.unlink()


def ensure_real_dir(path: Path, dry_run: bool) -> None:
    if dry_run:
        if path.is_symlink():
            print(f"plan detach root {path}")
        elif not path.exists():
            print(f"plan mkdir {path}")
        return
    if path.is_symlink():
        path.unlink()
    elif path.exists() and not path.is_dir():
        path.unlink()
    path.mkdir(parents=True, exist_ok=True)


def sync_icon_dir(srcs: list[Path], dst: Path, dry_run: bool) -> None:
    if dry_run:
        if dst.is_symlink():
            print(f"plan detach theme {dst}")
        elif dst.exists() and not dst.is_dir():
            print(f"plan replace non-dir {dst}")
        elif not dst.exists():
            print(f"plan mkdir {dst}")
    else:
        if dst.is_symlink() or (dst.exists() and not dst.is_dir()):
            remove_path(dst)
        dst.mkdir(parents=True, exist_ok=True)

    desired: dict[str, Path] = {}
    for src in srcs:
        for child in sorted(src.iterdir()):
            if child.name in ICON_SKIP_NAMES:
                continue
            desired[child.name] = child

    if dst.exists() and dst.is_dir() and not dst.is_symlink():
        for existing in sorted(dst.iterdir()):
            if existing.name in CACHE_NAMES:
                continue
            if existing.name not in desired:
                if dry_run:
                    print(f"plan remove stale {existing}")
                else:
                    remove_path(existing)

    for name, child in sorted(desired.items()):
        target = dst / name
        child_resolved = child.resolve(strict=False)
        if target.is_symlink() and target.resolve(strict=False) == child_resolved:
            continue
        if dry_run:
            print(f"plan link {target} -> {child}")
            continue
        if target.exists() or target.is_symlink():
            remove_path(target)
        target.symlink_to(child)


def remove_repo_caches(repo_root: Path, dry_run: bool) -> int:
    removed = 0
    seen: set[Path] = set()
    for src_rel, _dst_rel in ICON_SYNC_MAP:
        src_root = repo_root / src_rel
        if not src_root.exists():
            continue
        for cache_name in CACHE_NAMES:
            for cache_path in src_root.rglob(cache_name):
                if cache_path in seen:
                    continue
                seen.add(cache_path)
                if dry_run:
                    print(f"plan remove repo cache {cache_path}")
                else:
                    remove_path(cache_path)
                removed += 1
    return removed


def sync_icon_entry(srcs: list[Path], dst_root: Path, dry_run: bool) -> bool:
    src = srcs[-1]
    dst = dst_root / src.name
    if src.is_dir():
        sync_icon_dir(srcs, dst, dry_run)
    else:
        src_resolved = src.resolve(strict=False)
        if dst.is_symlink() and dst.resolve(strict=False) == src_resolved:
            return False
        if dry_run:
            print(f"plan link {dst} -> {src}")
        else:
            dst.parent.mkdir(parents=True, exist_ok=True)
            if dst.exists() or dst.is_symlink():
                remove_path(dst)
            dst.symlink_to(src)
    return True


def is_under_any(path: Path, roots: list[Path]) -> bool:
    resolved = path.resolve(strict=False)
    for root in roots:
        root_resolved = root.resolve(strict=False)
        try:
            resolved.relative_to(root_resolved)
            return True
        except ValueError:
            continue
    return False


def sync_theme_entries(src_roots: list[Path], dst_root: Path, dry_run: bool) -> int:
    desired: dict[str, Path] = {}
    for src_root in src_roots:
        for src in sorted(src_root.iterdir()):
            if src.name in ROOT_SKIP_NAMES:
                continue
            desired[src.name] = src

    synced = 0
    if dst_root.exists() and dst_root.is_dir() and not dst_root.is_symlink():
        for existing in sorted(dst_root.iterdir()):
            if existing.name in desired:
                continue
            if existing.is_symlink() and is_under_any(existing, src_roots):
                if dry_run:
                    print(f"plan remove stale {existing}")
                else:
                    existing.unlink()

    for name, src in sorted(desired.items()):
        dst = dst_root / name
        src_resolved = src.resolve(strict=False)
        if dst.is_symlink() and dst.resolve(strict=False) == src_resolved:
            continue
        if dry_run:
            action = "relink" if dst.exists() or dst.is_symlink() else "link"
            print(f"plan {action} {dst} -> {src}")
        else:
            if dst.exists() or dst.is_symlink():
                remove_path(dst)
            dst.symlink_to(src)
        synced += 1
    return synced


def rebuild_caches(themes: set[Path], dry_run: bool) -> int:
    gtk_cache = shutil.which("gtk-update-icon-cache")
    if not gtk_cache:
        return 0
    count = 0
    for theme in sorted(themes):
        if not theme.exists() or not theme.is_dir() or theme.is_symlink():
            continue
        if not (theme / "index.theme").is_file():
            continue
        for stale in (theme / ".icon-theme.cache", theme / "icon-theme.cache"):
            if stale.exists() or stale.is_symlink():
                if dry_run:
                    print(f"plan remove stale cache {stale}")
                else:
                    stale.unlink()
        if dry_run:
            print(f"plan cache {theme}")
        else:
            subprocess.run(
                [gtk_cache, "-f", "-t", str(theme)],
                check=False,
                stdout=subprocess.DEVNULL,
                stderr=subprocess.DEVNULL,
            )
        count += 1
    return count


def main() -> int:
    parser = argparse.ArgumentParser(
        description="Keep repo-backed icons and themes while rebuilding only local icon caches in home roots."
    )
    parser.add_argument(
        "--repo-root",
        default=str(Path.home() / "DotCortex"),
        help="DotCortex repo root",
    )
    parser.add_argument(
        "--home-root",
        help="home directory to update (default: the invoking user's home)",
    )
    parser.add_argument("--dry-run", action="store_true", help="show planned actions only")
    args = parser.parse_args()

    repo_root = Path(args.repo_root).expanduser().resolve()
    home = (
        Path(args.home_root).expanduser().resolve()
        if args.home_root
        else Path.home()
    )
    repo_caches_removed = remove_repo_caches(repo_root, args.dry_run)
    synced = 0
    icon_themes_to_rebuild: set[Path] = set()

    sync_roots: dict[str, list[Path]] = {}
    for src_rel, dst_rel in ICON_SYNC_MAP:
        src_root = repo_root / src_rel
        if not src_root.exists():
            continue
        sync_roots.setdefault(dst_rel, []).append(src_root)

    for dst_rel, src_roots in sync_roots.items():
        dst_root = home / dst_rel
        ensure_real_dir(dst_root, args.dry_run)
        entries: dict[str, list[Path]] = {}
        for src_root in src_roots:
            for src in sorted(src_root.iterdir()):
                if src.name in ICON_SKIP_NAMES:
                    continue
                entries.setdefault(src.name, []).append(src)
        for name in sorted(entries):
            if sync_icon_entry(entries[name], dst_root, args.dry_run):
                synced += 1
                dst = dst_root / name
                if dst.exists() and dst.is_dir() and not dst.is_symlink():
                    icon_themes_to_rebuild.add(dst)

    theme_roots: dict[str, list[Path]] = {}
    for src_rel, dst_rel in THEME_SYNC_MAP:
        src_root = repo_root / src_rel
        if not src_root.exists():
            continue
        theme_roots.setdefault(dst_rel, []).append(src_root)

    for dst_rel, src_roots in theme_roots.items():
        dst_root = home / dst_rel
        ensure_real_dir(dst_root, args.dry_run)
        synced += sync_theme_entries(src_roots, dst_root, args.dry_run)

    cached = rebuild_caches(icon_themes_to_rebuild, args.dry_run)

    mode = "planned" if args.dry_run else "linked"
    print(
        f"icons-home-sync: {mode}_entries={synced} "
        f"repo_caches_removed={repo_caches_removed} cache_roots={cached}"
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

* KDE-Classic deterministic importer

KDE 1 Alpha uses the native KDE-Classic icon corpus from KDE Artwork 4.14.3,
not a recoloured modern theme.  The importer accepts only the pinned upstream
archive, extracts only =IconThemes/kdeclassic=, preserves every native PNG,
repairs the malformed =64x64apps= index entry, adds the Redstone fallback chain,
and fills missing 22/24px slots with nearest-neighbour scaling.  Repository-side
icon caches are forbidden.

#+BEGIN_SRC python :tangle all/.local/bin/kde-classic-import :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
from __future__ import annotations

import argparse
import hashlib
import os
import re
import shutil
import tarfile
import tempfile
from pathlib import Path, PurePosixPath

from PIL import Image


ARCHIVE_SHA256 = "93dce7cbde334b31f07d7ebfcc00b3431695f685f8ccd9c8ef8f94d883f80ef4"
ARCHIVE_PREFIX = PurePosixPath("kdeartwork-4.14.3/IconThemes/kdeclassic")
TARGET_SIZES = (22, 24)
CACHE_NAMES = {"icon-theme.cache", ".icon-theme.cache"}
CONTEXTS = {
    "actions": "Actions",
    "animations": "Animations",
    "apps": "Applications",
    "categories": "Categories",
    "devices": "Devices",
    "emblems": "Emblems",
    "filesystems": "FileSystems",
    "mimetypes": "MimeTypes",
    "places": "Places",
}


def digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def extract_theme(archive: Path, destination: Path) -> None:
    """Extract regular files below the one pinned theme prefix, without links."""
    with tarfile.open(archive, "r:*") as tf:
        selected = 0
        for member in tf.getmembers():
            member_path = PurePosixPath(member.name)
            try:
                relative = member_path.relative_to(ARCHIVE_PREFIX)
            except ValueError:
                continue
            if not relative.parts:
                continue
            if member.issym() or member.islnk() or member.isdev():
                raise SystemExit(f"unsafe archive member: {member.name}")
            if relative.is_absolute() or ".." in relative.parts:
                raise SystemExit(f"unsafe archive path: {member.name}")
            target = destination.joinpath(*relative.parts)
            if member.isdir():
                target.mkdir(parents=True, exist_ok=True)
                continue
            if not member.isfile():
                continue
            source = tf.extractfile(member)
            if source is None:
                raise SystemExit(f"cannot read archive member: {member.name}")
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(source.read())
            os.chmod(target, member.mode & 0o777)
            selected += 1
    if selected == 0:
        raise SystemExit(f"archive contains no files below {ARCHIVE_PREFIX}")


def native_pngs(theme: Path) -> tuple[list[tuple[int, Path, Path]], int]:
    entries = []
    nonstandard = 0
    for size_dir in sorted(theme.iterdir()):
        match = re.fullmatch(r"(\d+)x\1", size_dir.name)
        if not match or not size_dir.is_dir():
            continue
        size = int(match.group(1))
        for icon in sorted(size_dir.rglob("*.png")):
            relative = icon.relative_to(size_dir)
            with Image.open(icon) as image:
                if image.size != (size, size):
                    # KDE-Classic intentionally carries a few glyph-shaped
                    # native canvases (for example 16x16/actions/3downarrow.png
                    # is 12x18). Inventory them, but never normalize or rewrite
                    # upstream art merely because its logical directory is
                    # square.
                    nonstandard += 1
            entries.append((size, relative, icon))
    return entries, nonstandard


def fill_panel_sizes(theme: Path, native: list[tuple[int, Path, Path]]) -> int:
    by_relative: dict[Path, list[tuple[int, Path]]] = {}
    for size, relative, icon in native:
        by_relative.setdefault(relative, []).append((size, icon))

    generated = 0
    for target_size in TARGET_SIZES:
        for relative, sources in sorted(by_relative.items(), key=lambda item: str(item[0])):
            target = theme / f"{target_size}x{target_size}" / relative
            if target.exists():
                continue
            source_size, source = min(
                sources,
                key=lambda item: (abs(item[0] - target_size), -item[0]),
            )
            target.parent.mkdir(parents=True, exist_ok=True)
            with Image.open(source) as image:
                image.resize(
                    (target_size, target_size), Image.Resampling.NEAREST
                ).save(target, optimize=True)
            generated += 1
    return generated


def indexed_directories(theme: Path) -> list[str]:
    directories = []
    for size_dir in theme.iterdir():
        match = re.fullmatch(r"(\d+)x\1", size_dir.name)
        if not match or not size_dir.is_dir():
            continue
        for category in size_dir.iterdir():
            if category.is_dir() and any(category.glob("*.png")):
                directories.append(f"{size_dir.name}/{category.name}")
    return sorted(
        directories,
        key=lambda value: (int(value.split("x", 1)[0]), value.split("/", 1)[1]),
    )


def repair_index(theme: Path) -> None:
    index = theme / "index.theme"
    text = index.read_text(encoding="utf-8")
    text = text.replace("64x64apps", "64x64/apps")
    if not re.search(r"^Inherits=", text, re.MULTILINE):
        text = text.replace(
            "DisplayDepth=32\n",
            "DisplayDepth=32\nInherits=Chicago95,hicolor\n",
            1,
        )

    directories = indexed_directories(theme)
    text = re.sub(
        r"^Directories=.*$",
        "Directories=" + ",".join(directories),
        text,
        count=1,
        flags=re.MULTILINE,
    )
    sections = set(re.findall(r"^\[([0-9]+x[0-9]+/[^]]+)\]$", text, re.MULTILINE))
    additions = []
    for directory in directories:
        if directory in sections:
            continue
        size_text, category = directory.split("/", 1)
        size = int(size_text.split("x", 1)[0])
        context = CONTEXTS.get(category, category.title())
        additions.append(
            f"\n[{directory}]\nSize={size}\nContext={context}\nType=Threshold\n"
        )
    index.write_text(text.rstrip() + "\n" + "".join(additions), encoding="utf-8")


def tree_fingerprint(root: Path) -> list[tuple[str, str]]:
    return [
        (str(path.relative_to(root)), digest(path))
        for path in sorted(root.rglob("*"))
        if path.is_file()
    ]


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "archive",
        nargs="?",
        default=str(Path.home() / "Downloads/kdeartwork-4.14.3.tar.xz"),
    )
    parser.add_argument("--repo-root", default=str(Path.home() / "DotCortex"))
    args = parser.parse_args()

    archive = Path(args.archive).expanduser().resolve()
    repo = Path(args.repo_root).expanduser().resolve()
    if not archive.is_file():
        raise SystemExit(f"missing KDE Artwork archive: {archive}")
    actual_sha = digest(archive)
    if actual_sha != ARCHIVE_SHA256:
        raise SystemExit(
            f"KDE Artwork checksum mismatch: expected {ARCHIVE_SHA256}, got {actual_sha}"
        )

    output = repo / "linux/.local/share/icons/KDE-Classic"
    output.parent.mkdir(parents=True, exist_ok=True)
    with tempfile.TemporaryDirectory(prefix=".kde-classic-", dir=output.parent) as tmp:
        built = Path(tmp) / "KDE-Classic"
        built.mkdir()
        extract_theme(archive, built)
        for required in ("LICENSE", "AUTHORS", "index.theme"):
            if not (built / required).is_file():
                raise SystemExit(f"archive theme missing {required}")
        native, native_nonstandard = native_pngs(built)
        generated = fill_panel_sizes(built, native)
        repair_index(built)

        caches = [path for path in built.rglob("*") if path.name in CACHE_NAMES]
        if caches:
            raise SystemExit(f"archive/import produced forbidden icon cache: {caches[0]}")

        status = "generated"
        if output.is_dir() and tree_fingerprint(output) == tree_fingerprint(built):
            status = "unchanged"
        else:
            old = output.parent / f".{output.name}.old-{os.getpid()}"
            if output.exists():
                output.rename(old)
            built.rename(output)
            if old.exists():
                shutil.rmtree(old)

    total_png = sum(1 for _ in output.rglob("*.png"))
    print(f"kde-classic-import: archive_sha256={actual_sha}")
    print(
        f"kde-classic-import: native_png={len(native)} "
        f"native_nonstandard_dimensions={native_nonstandard} "
        f"generated_22_24={generated} total_png={total_png} status={status}"
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

* Original KDE 1 icon variants

The late =KDE-Classic= corpus above remains the broad, practical default.  For
period comparison, this second importer reconstructs four additive themes from
the official =KDE/kde1-kdebase= history.  Each theme is pinned to the earliest
commit where its distinct visual style appears: the initial 1997 corpus, the
warm pre-Beta 1 art, the silver Classic art, and the later HiColor replacement.
Palette corrections and releases which merely carry a style forward do not
create duplicate themes.

Git commit, tree, and deterministic archive SHA-256 values are all pinned.  The
importer keeps an exact PNG rendering of every usable native XPM/PNG below
=upstream-native/=, then constructs XDG-sized 16/22/24/32/48px views with
nearest-neighbour scaling.  Missing modern names inherit from
=KDE-Classic,Chicago95,hicolor=.  It never removes or replaces =KDE-Classic=,
and it refuses to replace any variant directory that lacks its own generated
marker.

def run_git(repo: Path, *args: str, stdout: int | None = None) -> subprocess.CompletedProcess:
    return subprocess.run(
        ["git", "-C", str(repo), *args],
        check=True,
        stdout=stdout if stdout is not None else subprocess.PIPE,
        stderr=subprocess.PIPE,
    )


def archive_for(repo: Path, snapshot: Snapshot) -> bytes:
    resolved = run_git(
        repo, "rev-parse", "--verify", f"{snapshot.commit}^{{commit}}"
    ).stdout.decode().strip()
    if resolved != snapshot.commit:
        raise SystemExit(
            f"{snapshot.key}: commit mismatch: expected {snapshot.commit}, got {resolved}"
        )
    tree = run_git(repo, "show", "-s", "--format=%T", snapshot.commit).stdout.decode().strip()
    if tree != snapshot.tree:
        raise SystemExit(
            f"{snapshot.key}: tree mismatch: expected {snapshot.tree}, got {tree}"
        )
    archive = run_git(repo, "archive", "--format=tar", snapshot.commit).stdout
    actual_sha = hashlib.sha256(archive).hexdigest()
    if actual_sha != snapshot.archive_sha256:
        raise SystemExit(
            f"{snapshot.key}: archive checksum mismatch: "
            f"expected {snapshot.archive_sha256}, got {actual_sha}"
        )
    return archive


def normalized_name(source: PurePosixPath) -> str:
    name = source.stem.lower().replace("_", "-").replace(" ", "-")
    name = re.sub(r"^(mini|large|tiny)-", "", name)
    name = re.sub(r"[^a-z0-9.+-]+", "-", name).strip("-")
    return name or "unnamed"


def category_for(source: PurePosixPath, name: str) -> str:
    words = set(re.split(r"[-.+]", name))
    source_text = str(source).lower()
    if words & PLACE_WORDS:
        return "places"
    if "mimetype" in source_text or words & MIME_WORDS:
        return "mimetypes"
    if "toolbar" in source_text or words & ACTION_WORDS:
        return "actions"
    return "apps"


def square_canvas(image: Image.Image) -> Image.Image:
    rgba = image.convert("RGBA")
    if rgba.width == rgba.height:
        return rgba
    side = max(rgba.size)
    square = Image.new("RGBA", (side, side), (0, 0, 0, 0))
    square.paste(rgba, ((side - rgba.width) // 2, (side - rgba.height) // 2))
    return square


def source_priority(icon: SourceIcon) -> tuple[int, int, str]:
    parts = icon.source.parts
    if len(parts) == 2 and parts[0] == "pics":
        rank = 0
    elif "pics" in parts:
        rank = 1
    elif "icons" in parts:
        rank = 2
    else:
        rank = 3
    return rank, len(parts), str(icon.source)


def xpm_color(value: str) -> tuple[int, int, int, int]:
    color = value.strip()
    if color.lower() == "none":
        return 0, 0, 0, 0
    if color.startswith("#") and len(color) in {4, 7, 10, 13}:
        digits = len(color) - 1
        width = digits // 3
        maximum = (16 ** width) - 1
        channels = [
            round(int(color[1 + offset : 1 + offset + width], 16) * 255 / maximum)
            for offset in range(0, digits, width)
        ]
        return channels[0], channels[1], channels[2], 255
    if color.lower().startswith("rgb:"):
        channels = color[4:].split("/")
        if len(channels) == 3 and all(re.fullmatch(r"[0-9a-fA-F]{1,4}", item) for item in channels):
            values = [
                round(int(item, 16) * 255 / ((16 ** len(item)) - 1))
                for item in channels
            ]
            return values[0], values[1], values[2], 255
    grey = re.fullmatch(r"gr[ae]y([0-9]{1,3})", color, re.IGNORECASE)
    if grey:
        value = round(min(100, int(grey.group(1))) * 255 / 100)
        return value, value, value, 255
    shade = re.fullmatch(r"([a-z][a-z ]*?)([1-4])", color, re.IGNORECASE)
    if shade:
        base_name = shade.group(1).strip().lower()
        x11_bases = {
            "green": (0, 255, 0),
            "maroon": (176, 48, 96),
            "purple": (160, 32, 240),
        }
        base = x11_bases.get(base_name, ImageColor.getrgb(base_name))
        factor = {1: 1.0, 2: 238 / 255, 3: 205 / 255, 4: 139 / 255}[
            int(shade.group(2))
        ]
        return tuple(round(channel * factor) for channel in base) + (255,)
    rgb = ImageColor.getrgb(color)
    return rgb[0], rgb[1], rgb[2], 255


def decode_xpm_fallback(payload: bytes) -> Image.Image:
    strings = []
    for line in payload.decode("latin-1").splitlines():
        start = line.find('"')
        end = line.rfind('"')
        if start < 0 or end <= start:
            continue
        try:
            strings.append(ast.literal_eval(line[start : end + 1]))
        except (SyntaxError, ValueError):
            continue

    header_index = -1
    header = None
    for index, value in enumerate(strings):
        fields = value.split()
        if len(fields) >= 4 and all(field.isdigit() for field in fields[:4]):
            header_index = index
            header = tuple(int(field) for field in fields[:4])
            break
    if header is None:
        raise ValueError("XPM header not found")

    width, height, color_count, chars_per_pixel = header
    color_lines = strings[header_index + 1 : header_index + 1 + color_count]
    pixel_lines = strings[
        header_index + 1 + color_count : header_index + 1 + color_count + height
    ]
    if len(color_lines) != color_count or len(pixel_lines) != height:
        raise ValueError("truncated XPM palette or pixel rows")

    palette = {}
    for line in color_lines:
        key = line[:chars_per_pixel]
        definition = line[chars_per_pixel:]
        match = re.search(
            r"(?:^|\s)c\s+(.+?)(?=\s+[msg]\s+|$)",
            definition,
        )
        if not match:
            if definition.strip().lower() == "none":
                palette[key] = (0, 0, 0, 0)
                continue
            raise ValueError(f"XPM color definition missing for {key!r}")
        palette[key] = xpm_color(match.group(1))

    image = Image.new("RGBA", (width, height))
    pixels = image.load()
    transparent_key = next(
        (key for key, rgba in palette.items() if rgba[3] == 0),
        next(iter(palette)),
    )
    for y, row in enumerate(pixel_lines):
        if len(row) < width * chars_per_pixel:
            missing = width - (len(row) // chars_per_pixel)
            row += transparent_key * missing
        for x in range(width):
            key = row[x * chars_per_pixel : (x + 1) * chars_per_pixel]
            if key not in palette:
                raise ValueError(f"unknown XPM palette key {key!r}")
            pixels[x, y] = palette[key]
    return image


def decode_image(payload: bytes, suffix: str) -> tuple[Image.Image, str]:
    try:
        with Image.open(io.BytesIO(payload)) as opened:
            return opened.convert("RGBA"), "pillow"
    except (OSError, ValueError, UnidentifiedImageError):
        if suffix.lower() != ".xpm":
            raise
    return decode_xpm_fallback(payload), "xpm-fallback"


def load_sources(archive: bytes) -> tuple[list[SourceIcon], dict[str, bytes], list[str]]:
    icons = []
    metadata = {}
    skipped = []
    with tarfile.open(fileobj=io.BytesIO(archive), mode="r:") as tf:
        for member in tf.getmembers():
            source = PurePosixPath(member.name)
            if not member.isfile() or source.is_absolute() or ".." in source.parts:
                continue
            payload_file = tf.extractfile(member)
            if payload_file is None:
                continue
            payload = payload_file.read()
            if str(source) in {"AUTHORS", "COPYING", "README"}:
                metadata[str(source)] = payload
            if source.suffix.lower() not in {".png", ".xpm"}:
                continue
            try:
                image, decoder = decode_image(payload, source.suffix)
            except (OSError, ValueError, UnidentifiedImageError) as exc:
                detail = str(exc).replace("\t", " ").replace("\n", " ")
                if detail == "XPM header not found":
                    reason = "not-xpm-image"
                elif source.suffix.lower() == ".xpm":
                    reason = "malformed-xpm"
                else:
                    reason = "decode-error"
                skipped.append(f"{source}\t{reason}\t{detail}")
                continue
            if min(image.size) < 8 or max(image.size) > 128:
                skipped.append(f"{source}\tnon-icon-dimensions\t{image.width}x{image.height}")
                continue
            if max(image.size) > min(image.size) * 2:
                skipped.append(f"{source}\tnon-icon-aspect\t{image.width}x{image.height}")
                continue
            name = normalized_name(source)
            icons.append(
                SourceIcon(
                    source=source,
                    name=name,
                    category=category_for(source, name),
                    image=image,
                    decoder=decoder,
                )
            )
    for required in ("AUTHORS", "COPYING", "README"):
        if required not in metadata:
            raise SystemExit(f"upstream snapshot missing {required}")
    if not icons:
        raise SystemExit("upstream snapshot yielded no usable icons")
    return icons, metadata, skipped


def choose_source(candidates: list[SourceIcon], target_size: int) -> SourceIcon:
    return min(
        candidates,
        key=lambda icon: (
            abs(icon.side - target_size),
            source_priority(icon),
            -icon.side,
        ),
    )


def write_index(theme: Path, snapshot: Snapshot) -> None:
    directories = [
        f"{size}x{size}/{category}"
        for size in TARGET_SIZES
        for category in CATEGORIES
    ]
    lines = [
        "[Icon Theme]",
        f"Name={snapshot.title}",
        f"Comment={snapshot.description}",
        "Inherits=KDE-Classic,Chicago95,hicolor",
        "Directories=" + ",".join(directories),
        "",
    ]
    for size in TARGET_SIZES:
        for category in CATEGORIES:
            lines.extend(
                [
                    f"[{size}x{size}/{category}]",
                    f"Size={size}",
                    f"Context={CONTEXTS[category]}",
                    "Type=Threshold",
                    "",
                ]
            )
    (theme / "index.theme").write_text("\n".join(lines), encoding="utf-8")


def file_fingerprint(root: Path) -> list[tuple[str, str]]:
    return [
        (
            str(item.relative_to(root)),
            hashlib.sha256(item.read_bytes()).hexdigest()
            if not item.is_symlink()
            else f"link:{os.readlink(item)}",
        )
        for item in sorted(root.rglob("*"))
        if item.is_file() or item.is_symlink()
    ]


def build_snapshot(archive: bytes, snapshot: Snapshot, destination: Path) -> dict[str, int]:
    destination.mkdir()
    icons, metadata, skipped = load_sources(archive)

    (destination / GENERATED_MARKER).write_text(snapshot.commit + "\n", encoding="ascii")
    (destination / ".gitattributes").write_text(
        "AUTHORS -text -diff\nCOPYING -text -diff\nREADME.upstream -text -diff\n",
        encoding="ascii",
    )
    (destination / "AUTHORS").write_bytes(metadata["AUTHORS"])
    (destination / "COPYING").write_bytes(metadata["COPYING"])
    (destination / "README.upstream").write_bytes(metadata["README"])
    (destination / "PROVENANCE").write_text(
        "\n".join(
            (
                f"Upstream: {UPSTREAM_URL}",
                f"Snapshot: {snapshot.key}",
                f"Date: {snapshot.date}",
                f"Description: {snapshot.description}",
                f"Commit: {snapshot.commit}",
                f"Tree: {snapshot.tree}",
                f"Git archive SHA-256: {snapshot.archive_sha256}",
                "Conversion: Pillow RGBA; transparent square padding; nearest-neighbour standard sizes.",
                "Native renderings: upstream-native/<original path>.png",
                "",
            )
        ),
        encoding="utf-8",
    )

    for icon in icons:
        native = destination / "upstream-native" / Path(str(icon.source)).with_suffix(".png")
        native.parent.mkdir(parents=True, exist_ok=True)
        icon.image.save(native, optimize=True)

    families: dict[str, list[SourceIcon]] = {}
    for icon in icons:
        families.setdefault(icon.name, []).append(icon)

    written: dict[tuple[int, str], Path] = {}
    mapping_lines = [
        "icon_name\ttarget_size\tcategory\tsource\tsource_dimensions\tdecoder"
    ]
    for name, candidates in sorted(families.items()):
        category = min(candidates, key=source_priority).category
        for size in TARGET_SIZES:
            source = choose_source(candidates, size)
            target = destination / f"{size}x{size}" / category / f"{name}.png"
            target.parent.mkdir(parents=True, exist_ok=True)
            image = square_canvas(source.image)
            if image.size != (size, size):
                image = image.resize((size, size), Image.Resampling.NEAREST)
            image.save(target, optimize=True)
            written[(size, name)] = target
            mapping_lines.append(
                f"{name}\t{size}\t{category}\t{source.source}\t"
                f"{source.image.width}x{source.image.height}\t{source.decoder}"
            )

    aliases_written = 0
    for alias, (category, donors) in sorted(ALIASES.items()):
        for size in TARGET_SIZES:
            donor = next((written.get((size, name)) for name in donors if (size, name) in written), None)
            if donor is None:
                continue
            target = destination / f"{size}x{size}" / category / f"{alias}.png"
            target.parent.mkdir(parents=True, exist_ok=True)
            if target.exists():
                continue
            target.write_bytes(donor.read_bytes())
            written[(size, alias)] = target
            aliases_written += 1

    (destination / "source-map.tsv").write_text(
        "\n".join(mapping_lines) + "\n", encoding="utf-8"
    )
    (destination / "skipped-images.tsv").write_text(
        "source\treason\tdetail\n" + "\n".join(skipped) + ("\n" if skipped else ""),
        encoding="utf-8",
    )
    write_index(destination, snapshot)

    caches = [item for item in destination.rglob("*") if item.name in CACHE_NAMES]
    if caches:
        raise SystemExit(f"forbidden icon cache generated: {caches[0]}")
    return {
        "source_images": len(icons),
        "xpm_fallback": sum(icon.decoder == "xpm-fallback" for icon in icons),
        "families": len(families),
        "standard_png": len(written),
        "aliases": aliases_written,
        "skipped": len(skipped),
    }


def install_built(built: Path, output: Path) -> str:
    if output.exists() and not (output / GENERATED_MARKER).is_file():
        raise SystemExit(f"refusing to replace unmarked icon theme: {output}")
    if output.is_dir() and file_fingerprint(output) == file_fingerprint(built):
        return "unchanged"

    old = output.parent / f".{output.name}.old-{os.getpid()}"
    if output.exists():
        output.rename(old)
    built.rename(output)
    if old.exists():
        shutil.rmtree(old)
    return "generated"

    parser = argparse.ArgumentParser(
        description="Build additive XDG icon themes from pinned official KDE 1 history."
    )
    parser.add_argument("source_repo", help="clone of https://github.com/KDE/kde1-kdebase.git")
    parser.add_argument("--repo-root", default=str(Path.home() / "DotCortex"))
    parser.add_argument(
        "--only",
        choices=tuple(snapshot.key for snapshot in SNAPSHOTS),
        help="build one snapshot instead of all four",
    )
    args = parser.parse_args()

    source_repo = Path(args.source_repo).expanduser().resolve()
    repo_root = Path(args.repo_root).expanduser().resolve()
    if not (source_repo / ".git").is_dir():
        raise SystemExit(f"not a Git worktree: {source_repo}")

    selected = [
        snapshot
        for snapshot in SNAPSHOTS
        if args.only is None or snapshot.key == args.only
    ]
    output_root = repo_root / "linux/.local/share/icons"
    output_root.mkdir(parents=True, exist_ok=True)

    for snapshot in selected:
        archive = archive_for(source_repo, snapshot)

        )
        print(
            "kde1-icon-variants-import: "
            + " ".join(f"{key}={value}" for key, value in counts.items())
            + f" status={status}"
        )
    return 0

* App icon injection helper
Scale a source image into one or more repo-managed icon themes, using the
theme's directory grammar. This is for app icon aliases such as =kitty=,
=firedragon=, or =org.garudalinux.firedragon=.

Pixel art defaults to nearest-neighbor scaling so crisp source pixels do not get
blurred. Use =--resample lanczos= for painterly or antialiased source art.
Use =--magic-pink= when an icon generator gives you a PNG with a fake
=#ff00ff= transparency matte instead of an alpha channel.
Use =--trim-transparent= after color-keying to crop away transparent borders
while preserving the icon's aspect ratio on a square transparent canvas.
The first =--names= entry is written as the real PNG. Additional aliases are
relative symlinks by default to avoid duplicate binary payloads.

def file_digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def art_exists_elsewhere(theme_root: Path, digest: str, exclude: Path) -> list[Path]:
    """Every other file in the theme carrying byte-identical art.

    This is the whole basis of the uniqueness guard: overwriting a name whose art
    is duplicated under other names destroys nothing, because the art survives.
    Overwriting the ONLY copy destroys it. On 2026-07-26 that distinction was the
    difference between a lossless edit to Win98SE/SE98 (art present under eight
    other names) and genuine destruction in Chicago95 (art present nowhere else).
    """
    matches = []
    exclude = exclude.resolve() if exclude.exists() else exclude
    for candidate in theme_root.rglob("*"):
        if not candidate.is_file() or candidate.is_symlink():
            continue
        try:
            if candidate.resolve() == exclude:
                continue
            if file_digest(candidate) == digest:
                matches.append(candidate)
        except OSError:
            continue
    return matches


def guard_destination(dest: Path, theme_root: Path, preserve_as: str | None,
                      allow_destroy: bool) -> None:
    """Refuse to destroy art that exists nowhere else in the theme.

    Also refuses to write THROUGH a symlink. cp -f onto a symlink follows it and
    overwrites the link's target — on 2026-07-26 that silently replaced
    Chicago95's generic places/{22,24}/folder.png while the intended destination
    was folder-documents.png. Symlinks are unlinked and replaced, never followed.
    """
    if dest.is_symlink():
        target = os.readlink(dest)
        print(f"  replacing symlink {dest.name} -> {target} (not writing through it)")
        dest.unlink()
        return
    if not dest.exists():
        return

    digest = file_digest(dest)
    duplicates = art_exists_elsewhere(theme_root, digest, dest)
    if duplicates:
        rels = ", ".join(sorted(p.name for p in duplicates)[:4])
        print(f"  replacing {dest.name}; art survives as {rels}")
        return

    if preserve_as:
        keep = dest.with_name(f"{preserve_as}{dest.suffix}")
        if keep.exists() or keep.is_symlink():
            raise SystemExit(f"--preserve-as target already exists: {keep}")
        dest.rename(keep)
        print(f"  preserved unique art: {dest.name} -> {keep.name}")
        return

    if allow_destroy:
        print(f"  DESTROYING unique art at {dest} (--force-destroy)")
        return

    raise SystemExit(
        f"refusing to destroy unique art: {dest}\n"
        f"  This is the only copy of this image in {theme_root.name}.\n"
        f"  Re-run with --preserve-as <name> to rename it first, "
        f"or --force-destroy to overwrite it anyway."
    )

        raise argparse.ArgumentTypeError("expected RGB hex color such as #ff00ff") from exc


def remove_color_key(image: Image.Image, color: tuple[int, int, int], tolerance: int) -> Image.Image:
    keyed = image.convert("RGBA")
    pixels = keyed.load()
    width, height = keyed.size
    cr, cg, cb = color
    for y in range(height):
        for x in range(width):
            r, g, b, a = pixels[x, y]
            if (
                abs(r - cr) <= tolerance
                and abs(g - cg) <= tolerance
                and abs(b - cb) <= tolerance
            ):
                pixels[x, y] = (r, g, b, 0)
            elif a == 0:
                pixels[x, y] = (r, g, b, 0)
    return keyed


def trim_transparent(image: Image.Image, padding: int) -> Image.Image:
    alpha = image.getchannel("A")
    bbox = alpha.getbbox()
    if bbox is None:
        return image

    left, top, right, bottom = bbox
    if padding > 0:
        left = max(0, left - padding)
        top = max(0, top - padding)
        right = min(image.width, right + padding)
        bottom = min(image.height, bottom + padding)

    cropped = image.crop((left, top, right, bottom))
    return pad_to_square(cropped)


def pad_to_square(image: Image.Image) -> Image.Image:
    side = max(image.width, image.height)
    if image.width == image.height:
        return image
    square = Image.new("RGBA", (side, side), (0, 0, 0, 0))
    square.paste(image, ((side - image.width) // 2, (side - image.height) // 2))
    return square

                print(f"{alias.relative_to(repo_root)} -> {primary.name} ({mode})")
                written += 1

    if skipped:
        print(f"icon-theme-inject: skipped {skipped} size(s) with no donor/directory")

    print(f"icon-theme-inject: wrote {written} icon files")
    return 0

    raise SystemExit(main())

* System-wide icon cache rebuild
=icons-home-sync='s own cache rebuild only covers themes it just synced into
=$HOME=. A missing =icon-theme.cache= on a theme outside that scope (a
system-installed pack under =/usr/share/icons=, root-owned, never touched by
DotCortex) silently forces GTK to fall back to a full directory stat/walk on
every icon lookup instead of one indexed read — found chasing a slow Rofi
=show-icons: true= launch back to =Catalina-Dark= having no cache at all
(2026-07-09). This sweeps every icon theme root, user and system alike, and
rebuilds whatever cache is missing (or all of them, with =--force=), batching
the root-owned ones behind a single =sudo= prompt instead of one per theme.

#+BEGIN_SRC python :tangle all/.local/bin/icon-cache-rebuild :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
from __future__ import annotations

import argparse
import os
import shutil
import subprocess
import sys
from pathlib import Path

CACHE_NAMES = ("icon-theme.cache", ".icon-theme.cache")

DEFAULT_ROOTS = (
    Path("/usr/share/icons"),
    Path("/usr/local/share/icons"),
    Path.home() / ".local/share/icons",
    Path.home() / ".icons",
)


def find_theme_dirs(roots: list[Path]) -> list[Path]:
    themes: list[Path] = []
    seen: set[Path] = set()
    for root in roots:
        if not root.is_dir():
            continue
        for entry in sorted(root.iterdir()):
            if not entry.is_dir() or entry.is_symlink():
                continue
            if not (entry / "index.theme").is_file():
                continue
            resolved = entry.resolve(strict=False)
            if resolved in seen:
                continue
            seen.add(resolved)
            themes.append(entry)
    return themes


def has_cache(theme: Path) -> bool:
    return any((theme / name).is_file() for name in CACHE_NAMES)


def main() -> int:
    parser = argparse.ArgumentParser(
        description="Rebuild GTK icon-theme.cache files wherever missing, across "
        "both user and root-owned icon theme roots (not just DotCortex-synced ones)."
    )
    parser.add_argument(
        "--roots",
        help="Comma-separated theme roots to scan (default: system + user icon dirs)",
    )
    parser.add_argument(
        "--force",
        action="store_true",
        help="Rebuild every theme's cache, not just themes missing one",
    )
    parser.add_argument("--dry-run", action="store_true", help="show planned actions only")
    args = parser.parse_args()

    gtk_cache = shutil.which("gtk-update-icon-cache")
    if not gtk_cache:
        print("icon-cache-rebuild: gtk-update-icon-cache not found on PATH", file=sys.stderr)
        return 1

    if args.roots:
        roots = [Path(r).expanduser() for r in args.roots.split(",") if r]
    else:
        roots = list(DEFAULT_ROOTS)

    themes = find_theme_dirs(roots)
    to_rebuild = [t for t in themes if args.force or not has_cache(t)]
    skipped = len(themes) - len(to_rebuild)

    if not to_rebuild:
        print(f"icon-cache-rebuild: scanned={len(themes)} already_cached={skipped} rebuilt=0")
        return 0

    writable = [t for t in to_rebuild if os.access(t, os.W_OK)]
    needs_sudo = [t for t in to_rebuild if t not in writable]

    for theme in writable:
        if args.dry_run:
            print(f"plan cache {theme}")
            continue
        subprocess.run([gtk_cache, "-f", "-t", str(theme)], check=False)

    if needs_sudo:
        if args.dry_run:
            for theme in needs_sudo:
                print(f"plan cache (sudo) {theme}")
        else:
            print(f"icon-cache-rebuild: {len(needs_sudo)} root-owned theme(s) need sudo — you may be prompted once")
            cmd = ["sudo", "bash", "-c", 'for t in "$@"; do "$0" -f -t "$t"; done', gtk_cache]
            cmd += [str(t) for t in needs_sudo]
            subprocess.run(cmd, check=False)

    mode = "planned" if args.dry_run else "rebuilt"
    print(
        f"icon-cache-rebuild: scanned={len(themes)} already_cached={skipped} "
        f"{mode}={len(to_rebuild)} (user={len(writable)} sudo={len(needs_sudo)})"
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

* App icon workflow notes

** Repo-managed icon theme model
- Source payloads live under =all/.local/share/icons= and =linux/.local/share/icons=.
- Live home icon roots are real directories; =icons-home-sync= links repo-backed
  theme entries into =~/.local/share/icons= and removes repo-side cache files.
- Do not commit =icon-theme.cache= or =.icon-theme.cache=.
- Some remixed themes have invalid cache metadata but still resolve icon files
  correctly. Treat cache failures as a theme-health issue, not as proof the icon
  aliases failed.

** Naming rules
- Read the application's desktop file and use its exact =Icon== value first.
- Add app-id aliases when launchers or Flatpaks may resolve those instead
  (example: =firedragon=, =org.garudalinux.firedragon=,
  =garudalinux-firedragon=).
- Put the preferred real icon name first in =--names=. Additional names become
  relative symlinks by default; use =--copy-aliases= only when a consumer cannot
  resolve aliases.
- Keep icons in the =apps= context unless the desktop file points elsewhere.

** Pixel art handling
- For pixel-art source, scale with nearest-neighbor and inspect small sizes.
- For fake-alpha PNGs, run with =--magic-pink= or
  =--transparent-color '#ff00ff'= before resizing.
- For source art with too much matte space, add =--trim-transparent= after
  color-keying. It crops transparent borders, then pads back to square so icons
  do not distort.
- For antialiased/painterly source, use =--resample lanczos=.
- Generate real size-specific PNGs; do not place a 1024px image in 16px slots.
- For Plasma/SonicDE, include high-resolution app sizes such as 64, 128, and
  256px so panels and switchers do not bilinear-upscale a 48px icon.
- Good future tools to declare in package manifests: =imagemagick= for format
  conversion/sprite sheets, =oxipng= for lossless PNG optimization,
  =pngquant= for lossy palette compression when file size matters, and
  =libresprite= or =aseprite= for manual pixel-art editing.
