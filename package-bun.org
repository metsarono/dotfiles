* Bun build manifest
Use this lane for Bun-managed source trees that need a checkout plus conditional
install/build steps rather than a simple =bun add -g=. These rows are applied
sequentially by =bun:apply= after the global Bun package pass.

- =MODE=submodule= expects =ROOT= to already exist in =.gitmodules=
- =MODE=git= clones or syncs a standalone checkout at =ROOT=
- =MODE=local= uses an already-present tree and never pulls it forward
- =INSTALL=: empty = auto (=bun install --frozen-lockfile= when a Bun lockfile exists,
  else =bun install=); =none= disables dependency install
- =BUILD=: empty or =none= skips builds
- =HEALTH=: optional command, only executed when =RUN_CHECKS=1 bun-health=
- =EXTRA=: optional semicolon-delimited build guards. =patches= accepts an ordered,
  comma-delimited patch series; =allow-untracked= names managed generated paths
  that may coexist with that series. Unknown checkout edits still fail closed.
- Quote commands that contain spaces

** DeepSeek Harness MailCortex Telegram reply seam

The browser/API =session/prompt= RPC persists its client-minted request ID on
the resulting =user/message=.  The Telegram live-feed already routes every
event for a bound session through the harness transport, but it finalizes prose
only for a Telegram-owned inbound.  Mark only the watcher-owned
=mailcortex-= request-ID namespace as such an inbound, so the ordinary feed
publishes the final answer without a watcher-side Bot API call.

Cold restoration has one additional ordering requirement: adopting a resumed
session can synchronously replay its queued =user/message=.  Install the durable
chat binding before =adopt()= so that replay cannot fall through the Bridge's
unbound-event return.  Otherwise the later =assistant/message= still reaches
the pvox side channel after the session becomes live, but the renderer sees no
pending Telegram inbound and discards the final prose at =turn/end=.

The fixture mounts the real harness feed in production registration order.  It
locks one unquoted Telegram delivery with live feed enabled and disabled, keeps
that delivery when a separate MailCortex reply tool runs, and replays the wake
inside restart adoption before the restored session is published live.

manifest still passes =--unidiff-zero= for older compatibility patches, so
line-number-only hunks are unsafe when another staged seam shifts the file.

        if len(fields) == 9:
            fields.append("")
        sys.stdout.write("\t".join(fields) + "\n")
PY
}

bun_build_safe_name() {
  printf '%s\n' "$1" | tr '/: ' '___'
}

bun_build_state_dir() {
  printf '%s/%s\n' "$BUN_BUILD_STATE_ROOT" "$(bun_build_safe_name "$1")"
}

bun_build_root_abs() {
  local root="$1"

  case "$root" in
    /*)
      printf '%s\n' "$root"
      ;;
    ~*)
      printf '%s\n' "${root/#\~/$HOME}"
      ;;
    *)
      printf '%s\n' "$DOTCORTEX_ROOT/$root"
      ;;
  esac
}

bun_build_workdir_abs() {
  local root="$1" subdir="$2" root_abs

  root_abs="$(bun_build_root_abs "$root")"
  if [ -z "$subdir" ] || [ "$subdir" = "." ]; then
    printf '%s\n' "$root_abs"
  else
    printf '%s\n' "$root_abs/$subdir"
  fi
}

bun_build_bun_version_or_missing() {
  local bun_home bin_path

  if command -v bun >/dev/null 2>&1; then
    bun --version
    return 0
  fi

  bun_home="${BUN_INSTALL:-$HOME/.bun}"
  bin_path="$bun_home/bin/bun"
  if [ -x "$bin_path" ]; then
    "$bin_path" --version
  else
    printf 'missing\n'
  fi
}

bun_build_hash_existing_files() {
  local path
  declare -A seen=()

  for path in "$@"; do
    [ -n "$path" ] || continue
    if [ -n "${seen[$path]:-}" ]; then
      continue
    fi
    seen["$path"]=1
    if [ -f "$path" ]; then
      printf '%s=%s\n' "$path" "$(sha256_file "$path")"
    fi
  done
}

bun_build_effective_install_cmd() {
  local root_abs="$1" workdir_abs="$2" install_cmd="$3"

  case "$install_cmd" in
    none)
      printf '%s\n' ""
      ;;
    "")

        printf '%s\n' 'bun install --frozen-lockfile'
      else
        printf '%s\n' 'bun install'
      fi
      ;;
    *)
      printf '%s\n' "$install_cmd"
      ;;
  esac
}

bun_build_effective_build_cmd() {
  local build_cmd="$1"

  case "$build_cmd" in
    ""|none)
      printf '%s\n' ""
      ;;
    *)
      printf '%s\n' "$build_cmd"
      ;;
  esac
}

bun_build_effective_health_cmd() {
  local health_cmd="$1"

    *)
      printf '%s\n' "$health_cmd"

bun_build_is_git_repo() {
  local root_abs="$1"
  git -C "$root_abs" rev-parse --is-inside-work-tree >/dev/null 2>&1
}

bun_build_source_fingerprint() {
  local root_abs="$1" status head worktree_diff cached_diff submodules

  if [ ! -d "$root_abs" ]; then
    printf '%s\n' 'missing'
    return 0
  fi

  if bun_build_is_git_repo "$root_abs"; then
    head="$(git -C "$root_abs" rev-parse HEAD 2>/dev/null || printf 'unknown')"
    status="$(git -C "$root_abs" status --porcelain=v1 --untracked-files=normal 2>/dev/null || true)"
    submodules="$(git -C "$root_abs" submodule status --recursive 2>/dev/null || true)"
    worktree_diff=""
    cached_diff=""

    if [ -n "$status" ]; then
      worktree_diff="$(git -C "$root_abs" diff --no-ext-diff HEAD 2>/dev/null | sha256_stdin)"
      cached_diff="$(git -C "$root_abs" diff --cached --no-ext-diff 2>/dev/null | sha256_stdin)"
    fi

    {
      printf 'head=%s\n' "$head"
      printf 'status=%s\n' "$status"
      printf 'worktree_diff=%s\n' "$worktree_diff"
      printf 'cached_diff=%s\n' "$cached_diff"
      printf 'submodules=%s\n' "$submodules"
    } | sha256_stdin
    return 0
  fi

  {
    printf 'root=%s\n' "$root_abs"
      bun_build_hash_existing_files \
      "$root_abs/package.json" \
      "$root_abs/bun.lock" \
      "$root_abs/bun.lockb" \
      "$root_abs/bunfig.toml"
  } | sha256_stdin
}

bun_build_install_fingerprint() {
  local root_abs="$1" workdir_abs="$2" install_cmd="$3"

  {
    printf 'bun=%s\n' "$(bun_build_bun_version_or_missing)"
    printf 'system=%s|%s\n' "$(uname -s)" "$(uname -m)"
    printf 'install=%s\n' "$install_cmd"
    printf 'source=%s\n' "$(bun_build_source_fingerprint "$root_abs")"
    bun_build_hash_existing_files \
      "$root_abs/package.json" \
      "$root_abs/bun.lock" \
      "$root_abs/bun.lockb" \
      "$root_abs/bunfig.toml" \
      "$workdir_abs/package.json" \
      "$workdir_abs/bun.lock" \
      "$workdir_abs/bun.lockb" \
      "$workdir_abs/bunfig.toml"
  } | sha256_stdin
}

bun_build_build_fingerprint() {
  local root_abs="$1" workdir_abs="$2" install_cmd="$3" build_cmd="$4"

  {
    printf 'build=%s\n' "$build_cmd"
    printf 'source=%s\n' "$(bun_build_source_fingerprint "$root_abs")"
    printf 'install=%s\n' "$(bun_build_install_fingerprint "$root_abs" "$workdir_abs" "$install_cmd")"
  } | sha256_stdin
}

bun_build_fingerprint_matches() {
  local stamp_path="$1" fingerprint="$2"

  [ -f "$stamp_path" ] || return 1
  [ "$(<"$stamp_path")" = "$fingerprint" ]
}

bun_build_write_fingerprint() {
  local stamp_path="$1" fingerprint="$2"

  mkdir -p "$(dirname "$stamp_path")"
  printf '%s\n' "$fingerprint" >"$stamp_path"
}

bun_build_expand_extra_path() {
  local path="$1"

  case "$path" in
    /*)
      printf '%s\n' "$path"
      ;;
    '~/'*)
      printf '%s/%s\n' "$HOME" "${path#\~/}"
      ;;
    '$HOME/'*)
      printf '%s/%s\n' "$HOME" "${path#\$HOME/}"
      ;;
    *)
      printf '%s/%s\n' "$DOTCORTEX_ROOT" "$path"
      ;;
  esac
}

bun_build_worktree_tree() {
  local root_abs="$1" base_commit="$2" index_path="$3" diff_path="$4"

  rm -f "$index_path" "$index_path.lock"
  GIT_INDEX_FILE="$index_path" git -C "$root_abs" read-tree "$base_commit"
  git -C "$root_abs" diff --binary --full-index "$base_commit" -- >"$diff_path"
  if [ -s "$diff_path" ]; then
    GIT_INDEX_FILE="$index_path" git -C "$root_abs" apply --cached --binary --whitespace=nowarn "$diff_path"
  fi
  GIT_INDEX_FILE="$index_path" git -C "$root_abs" write-tree
}

bun_build_patch_state_write() {
  local name="$1" base_commit="$2" expected_tree="$3" applied_patch="$4"
  local state_dir temporary

  state_dir="$(bun_build_state_dir "$name")/patch-series"
  mkdir -p "$state_dir"
  temporary="$(mktemp -d "$state_dir/.write.XXXXXX")"
  printf '%s\n' "$base_commit" >"$temporary/base.commit"
  printf '%s\n' "$expected_tree" >"$temporary/applied.tree"
  cp "$applied_patch" "$temporary/applied.patch"
  printf '1\n' >"$temporary/schema"
  mv "$temporary/base.commit" "$state_dir/base.commit"
  mv "$temporary/applied.tree" "$state_dir/applied.tree"
  mv "$temporary/applied.patch" "$state_dir/applied.patch"
  mv "$temporary/schema" "$state_dir/schema"
  rmdir "$temporary"
}

bun_build_patch_failure_details() {
  local name="$1" root_abs="$2" base_commit="$3"

  echo "!! [$name] checkout differs from the pinned ref in edits not recognized as the managed patch series; refusing to modify it" >&2
  git -C "$root_abs" status --short --untracked-files=all >&2 || true
  echo "!! [$name] pinned ref: $base_commit; no reset or checkout was performed" >&2
}

bun_build_apply_patch_series() {
  local name="$1" root_abs="$2" ref="$3" allow_untracked_csv="$4"
  shift 4
  local patches=("$@")
  local base_commit base_tree head patch patch_abs expected_tree current_tree
  local state_dir state_patch prior_base="" prior_tree="" prior_valid=0
  local temporary expected_index current_index prior_index expected_patch current_diff
  local entry code path candidate allowed unknown_untracked=0 transition=base
  local reverted_tree actual_tree restore_tree
  local allowed_untracked=()

  if [ "${#patches[@]}" -eq 0 ]; then
    echo "!! [$name] patch guard requires at least one patch" >&2
    return 1
  fi
  if ! bun_build_is_git_repo "$root_abs"; then
    echo "!! [$name] patch guard requires a git checkout: $root_abs" >&2
    return 1
  fi
  if [ -z "$ref" ]; then
    echo "!! [$name] patch guard requires a pinned REF" >&2
    return 1
  fi

  base_commit="$(git -C "$root_abs" rev-parse "$ref^{commit}")"
  base_tree="$(git -C "$root_abs" rev-parse "$base_commit^{tree}")"
  head="$(git -C "$root_abs" rev-parse HEAD)"
  if [ "$head" != "$base_commit" ]; then
    echo "!! [$name] patch guard expected HEAD $base_commit, got $head" >&2
    return 1
  fi
  if ! git -C "$root_abs" diff --cached --quiet --; then
    echo "!! [$name] staged checkout edits are never managed patch state; refusing to modify them" >&2
    git -C "$root_abs" status --short --untracked-files=all >&2 || true
    return 1
  fi

  if [ -n "$allow_untracked_csv" ]; then
    IFS=',' read -r -a allowed_untracked <<<"$allow_untracked_csv"
  fi
  while IFS= read -r -d '' entry; do
    code="${entry:0:2}"
    [ "$code" = "??" ] || continue
    path="${entry:3}"
    allowed=0
    for candidate in "${allowed_untracked[@]}"; do
      if [ -n "$candidate" ] && [ "$path" = "$candidate" ]; then
        allowed=1
        break
      fi
    done
    if [ "$allowed" = 0 ]; then
      echo "!! [$name] unknown untracked checkout path: $path" >&2
      unknown_untracked=1
    fi
  done < <(git -C "$root_abs" status --porcelain=v1 -z --untracked-files=all)
  if [ "$unknown_untracked" = 1 ]; then
    echo "!! [$name] refusing to modify a checkout with unknown local files" >&2
    return 1
  fi

  temporary="$(mktemp -d)"
  expected_index="$temporary/expected.index"
  current_index="$temporary/current.index"
  prior_index="$temporary/prior.index"
  expected_patch="$temporary/expected.patch"
  current_diff="$temporary/current.patch"

  rm -f "$expected_index"
  GIT_INDEX_FILE="$expected_index" git -C "$root_abs" read-tree "$base_commit"
  for patch in "${patches[@]}"; do
    patch_abs="$(bun_build_expand_extra_path "$patch")"
    if [ ! -f "$patch_abs" ]; then
      echo "!! [$name] managed patch is missing: $patch_abs" >&2
      rm -rf "$temporary"
      return 1
    fi
    if ! GIT_INDEX_FILE="$expected_index" git -C "$root_abs" apply --cached --check --unidiff-zero "$patch_abs"; then
      echo "!! [$name] patch applies neither to the ordered pristine series nor as current state: $patch_abs" >&2
      rm -rf "$temporary"
      return 1
    fi
    GIT_INDEX_FILE="$expected_index" git -C "$root_abs" apply --cached --unidiff-zero "$patch_abs"
  done
  expected_tree="$(GIT_INDEX_FILE="$expected_index" git -C "$root_abs" write-tree)"
  GIT_INDEX_FILE="$expected_index" git -C "$root_abs" diff --cached --binary --full-index "$base_commit" -- >"$expected_patch"
  if [ ! -s "$expected_patch" ]; then
    echo "!! [$name] managed patch series produced no change from $base_commit" >&2
    rm -rf "$temporary"
    return 1
  fi

  current_tree="$(bun_build_worktree_tree "$root_abs" "$base_commit" "$current_index" "$current_diff")"
  state_dir="$(bun_build_state_dir "$name")/patch-series"
  state_patch="$state_dir/applied.patch"
  if [ -f "$state_dir/schema" ] && [ "$(<"$state_dir/schema")" = 1 ] \
      && [ -f "$state_dir/base.commit" ] && [ -f "$state_dir/applied.tree" ] \
      && [ -f "$state_patch" ]; then
    prior_base="$(<"$state_dir/base.commit")"
    prior_tree="$(<"$state_dir/applied.tree")"
    if [ "$prior_base" = "$base_commit" ] && [ "$current_tree" = "$prior_tree" ]; then
      cp "$current_index" "$prior_index"
      if GIT_INDEX_FILE="$prior_index" git -C "$root_abs" apply --cached --reverse --check --binary "$state_patch" \
          && GIT_INDEX_FILE="$prior_index" git -C "$root_abs" apply --cached --reverse --binary "$state_patch"; then
        reverted_tree="$(GIT_INDEX_FILE="$prior_index" git -C "$root_abs" write-tree)"
        if [ "$reverted_tree" = "$base_tree" ]; then
          prior_valid=1
        fi
      fi
    fi
  fi

  if [ "$current_tree" = "$expected_tree" ]; then
    echo ">> [$name] patch series current"
    bun_build_patch_state_write "$name" "$base_commit" "$expected_tree" "$expected_patch"
    rm -rf "$temporary"
    return 0
  fi
  if [ "$current_tree" = "$base_tree" ]; then
    transition=base
  elif [ "$prior_valid" = 1 ]; then
    transition=prior
  else
    bun_build_patch_failure_details "$name" "$root_abs" "$base_commit"
    rm -rf "$temporary"
    return 1
  fi

  if [ "$transition" = prior ]; then
    echo ">> [$name] replace recognized prior patch series"
    if ! git -C "$root_abs" apply --reverse --check --binary "$state_patch" \
        || ! git -C "$root_abs" apply --reverse --binary "$state_patch"; then
      echo "!! [$name] recognized prior patch series could not be removed; checkout left unchanged or requires inspection" >&2
      rm -rf "$temporary"
      return 1
    fi
    actual_tree="$(bun_build_worktree_tree "$root_abs" "$base_commit" "$current_index" "$current_diff")"
    if [ "$actual_tree" != "$base_tree" ]; then
      echo "!! [$name] prior patch removal did not restore the pinned ref; refusing the new series" >&2
      rm -rf "$temporary"
      return 1
    fi
  else
    echo ">> [$name] apply patch series (${#patches[@]} patches)"
  fi

  if ! git -C "$root_abs" apply --check --binary "$expected_patch" \
      || ! git -C "$root_abs" apply --binary "$expected_patch"; then
    echo "!! [$name] preflight passed but applying the managed patch series failed" >&2
    if [ "$transition" = prior ]; then
      restore_tree="$(bun_build_worktree_tree "$root_abs" "$base_commit" "$current_index" "$current_diff")"
      if [ "$restore_tree" = "$base_tree" ] \
          && git -C "$root_abs" apply --check --binary "$state_patch" \
          && git -C "$root_abs" apply --binary "$state_patch"; then
        echo "!! [$name] restored the recognized prior patch series after failure" >&2
      else
        echo "!! [$name] could not restore the prior patch series; inspect the checkout before retrying" >&2
      fi
    fi
    rm -rf "$temporary"
    return 1
  fi

  actual_tree="$(bun_build_worktree_tree "$root_abs" "$base_commit" "$current_index" "$current_diff")"
  if [ "$actual_tree" != "$expected_tree" ]; then
    echo "!! [$name] applied patch series does not match its preflighted tree" >&2
    rm -rf "$temporary"
    return 1
  fi
  bun_build_patch_state_write "$name" "$base_commit" "$expected_tree" "$expected_patch"
  rm -rf "$temporary"
}

bun_build_apply_extra_guards() {
  local name="$1" root_abs="$2" ref="$3" extra="$4"
  local item key value patches_value="" allow_untracked=""
  local seen_patches=0 seen_allow=0
  local items=() patches=()

  [ -n "$extra" ] || return 0
  IFS=';' read -r -a items <<<"$extra"
  for item in "${items[@]}"; do
    [ -n "$item" ] || continue
    if [[ "$item" != *=* ]]; then
      echo "!! [$name] malformed EXTRA guard: $item" >&2
      return 1
    fi
    key="${item%%=*}"
    value="${item#*=}"
    case "$key" in
      patches)
        if [ "$seen_patches" = 1 ] || [ -z "$value" ]; then
          echo "!! [$name] duplicate or empty patches EXTRA guard" >&2
          return 1
        fi
        seen_patches=1
        patches_value="$value"
        ;;
      allow-untracked)
        if [ "$seen_allow" = 1 ]; then
          echo "!! [$name] duplicate allow-untracked EXTRA guard" >&2
          return 1
        fi
        seen_allow=1
        allow_untracked="$value"
        ;;
      *)
        echo "!! [$name] unsupported EXTRA guard: $key" >&2
        return 1
        ;;
    esac
  done

  if [ "$seen_patches" = 0 ]; then
    echo "!! [$name] EXTRA declares no supported build guard" >&2
    return 1
  fi
  IFS=',' read -r -a patches <<<"$patches_value"
  bun_build_apply_patch_series "$name" "$root_abs" "$ref" "$allow_untracked" "${patches[@]}"
}

bun_build_submodule_declared() {
  local root="$1" gitmodules="$DOTCORTEX_ROOT/.gitmodules"

  [ -f "$gitmodules" ] || return 1
  git -C "$DOTCORTEX_ROOT" config -f "$gitmodules" --get-regexp '^submodule\..*\.path$' 2>/dev/null \
    | awk '{print $2}' \
    | grep -Fxq "$root"
}

bun_build_ensure_checkout() {
  local update="$1" name="$2" mode="$3" root="$4" repo="$5" ref="$6"
  local root_abs parent origin branch

  root_abs="$(bun_build_root_abs "$root")"

  case "$mode" in
    submodule)
      if [[ "$root" = /* || "$root" = ~* ]]; then
        echo "!! [$name] submodule ROOT must be repo-relative: $root" >&2
        return 1
      fi
      if ! bun_build_submodule_declared "$root"; then
        echo "!! [$name] submodule path is not declared in $DOTCORTEX_ROOT/.gitmodules: $root" >&2
        return 1
      fi
      echo ">> [$name] sync submodule: $root"
      git -C "$DOTCORTEX_ROOT" submodule update --init --recursive -- "$root"
      if [ "$update" = "1" ]; then
        echo ">> [$name] update submodule from remote"
        git -C "$DOTCORTEX_ROOT" submodule update --remote --recursive -- "$root"
      fi
      if [ -n "$repo" ]; then
        origin="$(git -C "$root_abs" remote get-url origin 2>/dev/null || true)"
        if [ -n "$origin" ] && [ "$origin" != "$repo" ]; then
          echo "!! [$name] submodule origin mismatch: expected $repo, got $origin" >&2
          return 1
        fi
      fi
      ;;

    git)
      parent="$(dirname "$root_abs")"
      mkdir -p "$parent"
      if [ ! -d "$root_abs/.git" ]; then
        if [ -z "$repo" ]; then
          echo "!! [$name] MODE=git requires REPO" >&2
          return 1
        fi
        echo ">> [$name] clone git source: $repo -> $root_abs"
        git clone "$repo" "$root_abs"
      fi

      if ! bun_build_is_git_repo "$root_abs"; then
        echo "!! [$name] expected a git checkout at $root_abs" >&2
        return 1
      fi

      if [ -n "$repo" ]; then
        origin="$(git -C "$root_abs" remote get-url origin 2>/dev/null || true)"
        if [ -z "$origin" ]; then
          git -C "$root_abs" remote add origin "$repo"
        elif [ "$origin" != "$repo" ]; then
          echo "!! [$name] git origin mismatch: expected $repo, got $origin" >&2
          return 1
        fi
      fi

      if [ "$update" = "1" ]; then
        echo ">> [$name] fetch latest git refs"
        git -C "$root_abs" fetch --all --tags --prune
      fi

      if [ -n "$ref" ]; then
        if [ "$(git -C "$root_abs" rev-parse HEAD)" != "$(git -C "$root_abs" rev-parse "$ref^{commit}")" ]; then
          if ! git -C "$root_abs" diff --quiet || ! git -C "$root_abs" diff --cached --quiet; then
            echo "!! [$name] checkout differs from pinned ref and has tracked changes; refusing checkout" >&2
            return 1
          fi
          git -C "$root_abs" checkout "$ref"
        fi
        if [ "$update" = "1" ] && git -C "$root_abs" rev-parse --verify "origin/$ref" >/dev/null 2>&1; then
          git -C "$root_abs" merge --ff-only "origin/$ref"
        fi
      elif [ "$update" = "1" ]; then
        branch="$(git -C "$root_abs" symbolic-ref --quiet --short HEAD 2>/dev/null || true)"
        if [ -n "$branch" ]; then
          git -C "$root_abs" pull --ff-only origin "$branch"
        fi
      fi
      ;;

    local)
      if [ ! -d "$root_abs" ]; then
        echo "!! [$name] local source tree missing: $root_abs" >&2
        return 1
      fi
      ;;

    *)
      echo "!! [$name] unsupported MODE: $mode" >&2
      return 1
      ;;
  esac
}

bun_build_run_step() {
  local name="$1" phase="$2" workdir_abs="$3" cmd="$4"

  [ -n "$cmd" ] || return 0

  if [ ! -d "$workdir_abs" ]; then
    echo "!! [$name] workdir missing for $phase: $workdir_abs" >&2
    return 1
  fi

  echo ">> [$name] $phase: $cmd"
  "$DOTCORTEX_BUN_ENV" bash -lc 'cd "$1" && eval "$2"' bash "$workdir_abs" "$cmd"
}

set -euo pipefail

touch "$MAN"

import os
import sys

bin_dir = os.path.expanduser(sys.argv[1])
if not os.path.isdir(bin_dir):

_emit_norm() {
  awk -v FS='[ \t]+' '
    function dequote(s){
      if (s == "\"\"") return ""
      if (s ~ /^".*"$/) {
        sub(/^"/, "", s)
        sub(/"$/, "", s)
      }
      return s
    }
    $0 ~ /^[ \t]*#/ || NF==0 { next }
    {
      printf "%s|%s|%s|%s|%s|%s\n",
        dequote($1),dequote($2),dequote($3),dequote($4),dequote($5),dequote($6)

* Detach live Bun homes from DotCortex
#+BEGIN_SRC python :tangle all/.local/bin/bun-home-detach :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
from __future__ import annotations

import argparse
import shutil
import sys
from pathlib import Path


SKIP = {"manifest"}


def copy_entry(src: Path, dst: Path) -> None:
    if src.is_dir() and not src.is_symlink():
        shutil.copytree(src, dst, symlinks=True, dirs_exist_ok=True)
    else:
        dst.parent.mkdir(parents=True, exist_ok=True)
        if dst.exists() or dst.is_symlink():
            if dst.is_dir() and not dst.is_symlink():
                shutil.rmtree(dst)
            else:
                dst.unlink()
        shutil.copy2(src, dst, follow_symlinks=False)


def remove_entry(path: Path) -> None:
    if path.is_dir() and not path.is_symlink():
        shutil.rmtree(path)
    elif path.exists() or path.is_symlink():
        path.unlink()


def detach_pair(repo_home: Path, home: Path, keep_repo_copy: bool) -> int:
    if not repo_home.exists():
        return 0
    migrated = 0
    if home.is_symlink():
        home.unlink()
    home.mkdir(parents=True, exist_ok=True)
    for src in sorted(repo_home.iterdir()):
        if src.name in SKIP:
            continue
        copy_entry(src, home / src.name)
        migrated += 1
        if not keep_repo_copy:
            remove_entry(src)
    return migrated


def main() -> int:
    parser = argparse.ArgumentParser(
        description="Detach ~/.bun and ~/.bunx from DotCortex and keep live Bun state in $HOME."
    )
    parser.add_argument(
        "--repo-bun",
        default=str(Path.home() / "DotCortex/all/.bun"),
        help="repo-owned bun path to migrate away from",
    )
    parser.add_argument(
        "--repo-bunx",
        default=str(Path.home() / "DotCortex/all/.bunx"),
        help="repo-owned bunx path to migrate away from",
    )
    parser.add_argument(
        "--bun-home",
        default=str(Path.home() / ".bun"),
        help="real bun home destination",
    )
    parser.add_argument(
        "--bunx-home",
        default=str(Path.home() / ".bunx"),
        help="real bunx home destination",
    )
    parser.add_argument(
        "--keep-repo-copy",
        action="store_true",
        help="do not remove migrated live state from the repo paths",
    )
    args = parser.parse_args()

    migrated_bun = detach_pair(
        Path(args.repo_bun).expanduser().resolve(),
        Path(args.bun_home).expanduser(),
        args.keep_repo_copy,
    )
    migrated_bunx = detach_pair(
        Path(args.repo_bunx).expanduser().resolve(),
        Path(args.bunx_home).expanduser(),
        args.keep_repo_copy,
    )

    print(
        "detached bun homes: "
        f"bun={migrated_bun} bunx={migrated_bunx} "
        f"homes={Path(args.bun_home).expanduser()},{Path(args.bunx_home).expanduser()}"
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
