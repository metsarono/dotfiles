* Claude settings

The JSON below is the SHARED settings baseline. It tangles to =all/.claude/settings.shared.json= (generated, git-ignored), never to the live =settings.json=. =claude-settings-apply= merges it directly into the live regular =~/.claude/settings.json=. Keys present in the baseline are DotCortex-owned and win; absent keys remain machine-local, including =model=, =effortLevel=, and =agentPushNotifEnabled=. =enabledPlugins= deep-merges, with shared entries winning and local additions surviving. Hook commands named exactly =claude-hook-*= are DotCortex-owned: applying the baseline replaces their old registrations, including obsolete commands, while leaving private HelmCortex hook groups and other foreign registrations intact. Stow ignores the live file. The pipeline is tangle → merge; run =loom claude:settings= after changing this baseline.

* Claude settings merge (shared keys and hooks into live regular file)

=claude-settings-apply= reads the tangled shared baseline and merges it into the live regular file. It refuses a live symlink or malformed JSON. The HelmCortex userspace compiler owns its hook paths and also merges into this file; apply the two mergers serially. The live file is never a Stow target.

#+BEGIN_SRC bash :tangle all/.local/bin/claude-settings-apply :shebang "#!/usr/bin/env bash" :comments no
# ~/DotCortex/all/.local/bin/claude-settings-apply
# Merge the shared Claude settings baseline (tangled from agents-hooks.org) onto the
# machine's live regular settings.json. Keys absent from the baseline survive;
# enabledPlugins deep-merges and only bare claude-hook-* commands are replaced.
# The "merge" tier of the pipeline: tangle -> settings.shared.json, then this
# script -> settings.json. Never tangle directly to settings.json again.
set -euo pipefail

SHARED="$HOME/DotCortex/all/.claude/settings.shared.json"
LIVE="$HOME/.claude/settings.json"

if [ ! -f "$SHARED" ]; then
  echo "!! claude-settings-apply: shared baseline missing at $SHARED" >&2
  echo "   run 'tangle-one agents-hooks.org' first (or 'loom claude:settings')" >&2
  exit 1
fi
command -v jq >/dev/null 2>&1 || { echo "!! claude-settings-apply: jq not on PATH" >&2; exit 1; }
if [ -L "$LIVE" ]; then
  echo "!! claude-settings-apply: $LIVE is a symlink; detach Stow ownership before applying" >&2
  exit 1
fi
mkdir -p "$(dirname "$LIVE")"

# On a fresh machine the live file may not exist; feed jq /dev/null so $live[0] is
# null and the shared baseline is used verbatim (no model -> Claude Code default).
live_input="$LIVE"
[ -f "$LIVE" ] || live_input=/dev/null

umask 077
tmp="$(mktemp "${LIVE}.tmp.XXXXXX")"
if jq -n \
     --slurpfile shared "$SHARED" \
     --slurpfile live "$live_input" \
     'def managed: (.type == "command" and (.command | type == "string" and test("^claude-hook-[A-Za-z0-9-]+$")));
      def valid_groups: type == "array" and all(.[]; type == "object" and (.hooks | type == "array") and all(.hooks[]; type == "object"));
      ($shared[0]) as $s | ($live[0] // {}) as $l
      | if ($s | type) != "object" or ($l | type) != "object"
           or (($s.hooks // {}) | type) != "object" or (($l.hooks // {}) | type) != "object"
           or (($s.enabledPlugins // {}) | type) != "object" or (($l.enabledPlugins // {}) | type) != "object"
           or ([$s.hooks[]? | valid_groups] | all(.[]; .)) == false
           or ([$l.hooks[]? | valid_groups] | all(.[]; .)) == false
           or ([$s.hooks[]? | .[] | .hooks[] | managed] | all(.[]; .)) == false
        then error("invalid Claude settings or unowned shared hook") else . end
      | ($l.hooks // {}) as $oldhooks
      | ($s.hooks // {}) as $newhooks
      | (($oldhooks + $newhooks) | keys) as $events
      | reduce $events[] as $event ({};
          .[$event] = (([$oldhooks[$event][]? | .hooks |= map(select(managed | not)) | select(.hooks | length > 0)]
                       + ($newhooks[$event] // [])))) as $mergedhooks
      | ($l + ($s | del(.hooks, .enabledPlugins)))
      | .enabledPlugins = (($l.enabledPlugins // {}) + ($s.enabledPlugins // {}))
      | .hooks = $mergedhooks' > "$tmp" \
   && jq empty "$tmp" 2>/dev/null; then
  mv -f "$tmp" "$LIVE"
  printf '>> claude settings merged: model=%s effort=%s theme=%s\n' \
    "$(jq -r '.model // "(unset -> Claude default)"' "$LIVE")" \
    "$(jq -r '.effortLevel // "(unset)"' "$LIVE")" \
    "$(jq -r '.theme // "(unset)"' "$LIVE")"
else
  rm -f "$tmp"
  echo "!! claude-settings-apply: merge produced invalid JSON; left $LIVE untouched" >&2
  exit 1
fi

#+BEGIN_SRC python :tangle all/.local/share/dotcortex/tests/test-claude-settings-apply.py :comments no
"""The two Claude settings mergers must share one live regular file."""
import json
import os
from pathlib import Path
import shutil
import subprocess
import tempfile
import unittest


ROOT = Path(__file__).resolve().parents[5]
APPLY = ROOT / "all/.local/bin/claude-settings-apply"


class ClaudeSettingsApplyTest(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory()
        self.addCleanup(self.temp.cleanup)
        self.home = Path(self.temp.name)
        self.shared = self.home / "DotCortex/all/.claude/settings.shared.json"
        self.live = self.home / ".claude/settings.json"
        self.shared.parent.mkdir(parents=True)
        self.live.parent.mkdir(parents=True)
        self.baseline = {
            "theme": "dark",
            "enabledPlugins": {"shared@plugin": True},
            "hooks": {"Stop": [{"hooks": [{"type": "command", "command": "claude-hook-stop"}]}]},
        }
        self.private = {
            "matcher": "private",
            "hooks": [{"type": "command", "command": "VAULT=$HOME/HelmCortex; $VAULT/NEXUS/stow/userspace/.claude/hooks/private.sh"}],
        }
        self.private_start = {
            "hooks": [{"type": "command", "command": "VAULT=$HOME/HelmCortex; $VAULT/NEXUS/stow/userspace/.claude/hooks/start.sh"}],
        }
        self.foreign_prompt = {"hooks": [{"type": "prompt", "prompt": "private instruction"}]}
        self.current = {
            "theme": "light", "model": "sonnet", "effortLevel": "high",
            "agentPushNotifEnabled": True, "machineOnly": {"value": 7},
            "enabledPlugins": {"local@plugin": True, "shared@plugin": False},
            "hooks": {
                "Stop": [
                    {"hooks": [
                        {"type": "command", "command": "claude-hook-obsolete"},
                        {"type": "command", "command": "claude-hook-stop"},
                    ]},
                    self.private,
                ],
                "SessionStart": [{"hooks": [{"type": "command", "command": "foreign-command"}]},
                                 self.private_start, self.foreign_prompt],
            },
        }
        self.write(self.shared, self.baseline)
        self.write(self.live, self.current)

    @staticmethod
    def write(path, value):
        path.write_text(json.dumps(value) + "\n")

    def apply(self):
        return subprocess.run([str(APPLY)], env={**os.environ, "HOME": str(self.home)},
                              text=True, capture_output=True)

    def test_union_and_repeat_are_stable(self):
        self.assertEqual(self.apply().returncode, 0)
        first = json.loads(self.live.read_text())
        self.assertEqual(first["theme"], "dark")
        self.assertEqual(first["model"], "sonnet")
        self.assertEqual(first["effortLevel"], "high")
        self.assertTrue(first["agentPushNotifEnabled"])
        self.assertEqual(first["machineOnly"], {"value": 7})
        self.assertEqual(first["enabledPlugins"], {"shared@plugin": True, "local@plugin": True})
        self.assertIn(self.private, first["hooks"]["Stop"])
        self.assertEqual(first["hooks"]["SessionStart"], self.current["hooks"]["SessionStart"])
        self.assertEqual([entry["command"] for group in first["hooks"]["Stop"]
                          for entry in group["hooks"] if entry["command"].startswith("claude-hook-")],
                         ["claude-hook-stop"])
        self.assertEqual(self.apply().returncode, 0)
        self.assertEqual(json.loads(self.live.read_text()), first)

    def test_invalid_json_keeps_live_bytes(self):
        before = self.live.read_bytes()
        self.shared.write_text("{broken")
        self.assertNotEqual(self.apply().returncode, 0)
        self.assertEqual(self.live.read_bytes(), before)
        self.write(self.shared, self.baseline)
        self.live.write_text("{broken")
        before = self.live.read_bytes()
        self.assertNotEqual(self.apply().returncode, 0)
        self.assertEqual(self.live.read_bytes(), before)

    def test_symlink_refused_without_replacement(self):
        target = self.home / "foreign.json"
        self.write(target, self.current)
        self.live.unlink()
        self.live.symlink_to(target)
        before = target.read_bytes()
        self.assertNotEqual(self.apply().returncode, 0)
        self.assertTrue(self.live.is_symlink())
        self.assertEqual(target.read_bytes(), before)

    def test_shared_hook_must_be_dotcortex_owned(self):
        self.baseline["hooks"]["Stop"].append(self.private)
        self.write(self.shared, self.baseline)
        before = self.live.read_bytes()
        self.assertNotEqual(self.apply().returncode, 0)
        self.assertEqual(self.live.read_bytes(), before)

    def test_full_generated_baseline_keeps_private_group(self):
        self.shared.write_bytes((ROOT / "all/.claude/settings.shared.json").read_bytes())
        self.assertEqual(self.apply().returncode, 0)
        merged = json.loads(self.live.read_text())
        self.assertIn(self.private, merged["hooks"]["Stop"])
        self.assertTrue(merged["agentPushNotifEnabled"])

    def test_stow_ignores_only_live_settings_artifact(self):
        package = self.home / "packages/all/.claude"
        target = self.home / "target/.claude"
        package.mkdir(parents=True)
        target.mkdir(parents=True)
        shutil.copy2(ROOT / "all/.stow-local-ignore", package.parent)
        (package / "settings.json").write_text("{}\n")
        (package / "settings.shared.json").write_text("{}\n")
        (target / "settings.json").write_text("{\"private\": true}\n")
        result = subprocess.run(["stow", "--simulate", "-d", str(package.parent.parent),
                                 "-t", str(target.parent), "all"],
                                text=True, capture_output=True)
        self.assertEqual(result.returncode, 0, result.stderr)
        self.assertEqual((target / "settings.json").read_text(), '{"private": true}\n')


if __name__ == "__main__":
    unittest.main()

#+BEGIN_SRC bash :tangle all/.local/bin/claude-statusline :shebang "#!/usr/bin/env bash" :comments no
set -u
input=$(cat)

# User and host
user=$(whoami)
host=$(hostname -s)

# Directory — basename only
cwd=$(echo "$input" | jq -r '.workspace.current_dir // .cwd')
short_cwd=$(basename "$cwd")

# Git branch from worktree or workspace
git_branch=$(echo "$input" | jq -r '.worktree.branch // .workspace.git_worktree // empty')
if [ -z "$git_branch" ] && command -v git >/dev/null 2>&1; then
  git_branch=$(git -C "$cwd" symbolic-ref --short HEAD 2>/dev/null || git -C "$cwd" rev-parse --short HEAD 2>/dev/null)
fi

# Time
time_str=$(date +%H:%M)

# Model short name
model=$(echo "$input" | jq -r '.model.display_name // empty' | sed 's/Claude //')

# Context USED (not remaining)
used=$(echo "$input" | jq -r '.context_window.used_percentage // empty')

# Claude's documented status-line total_input_tokens is the canonical current
# input-context value.  The fallback mirrors its documented current_usage
# components and excludes output tokens.  current_usage is null immediately
# after compaction; PostCompact clears the owning governor state then.
tokens=$(echo "$input" | jq -r '
  if .context_window.total_input_tokens != null
  then .context_window.total_input_tokens
  elif .context_window.current_usage != null
  then ((.context_window.current_usage.input_tokens // 0) +
        (.context_window.current_usage.cache_creation_input_tokens // 0) +
        (.context_window.current_usage.cache_read_input_tokens // 0))
  else 0
  end')

# Persist only the small, non-secret status snapshot used by the owner.
# Emacs-owned sessions use a channel file under XDG_RUNTIME_DIR; PTY-supervised
# sessions use their private runtime directory.  Both writes are atomic.
owner="${CLAUDE_IDLE_OWNER:-}"
channel="${CLAUDE_IDLE_CHANNEL:-}"
session_id=$(echo "$input" | jq -r '.session_id // empty')
runtime_dir=""
snapshot_path=""
if [ "$owner" = "pty-supervisor" ] || [ -z "$owner" ]; then
  runtime_dir="${CLAUDE_IDLE_RUNTIME_DIR:-}"
  snapshot_path="$runtime_dir/status.json"
  # Claude 2.1.220 may launch statusLine commands without propagating the
  # supervisor's custom CLAUDE_IDLE_* variables.  Recover the owner from the
  # private per-session registry using the validated session ID.  The registry
  # entry must still name a live PID with an owned PTY; the supervisor repeats
  # channel/session validation at the compaction gate.
  if [ -z "$channel" ] || [ -z "$runtime_dir" ]; then
    registry_root="${XDG_RUNTIME_DIR:-}/claude-idle-compaction"
    for state_file in "$registry_root"/claude-*/state.json; do
      [ -r "$state_file" ] || continue
      candidate_session=$(jq -r '.session_id // empty' "$state_file" 2>/dev/null || true)
      candidate_pid=$(jq -r '.pid // empty' "$state_file" 2>/dev/null || true)
      candidate_owned=$(jq -r '.pty_master_owned // false' "$state_file" 2>/dev/null || true)
      candidate_channel=$(basename "$(dirname "$state_file")")
      if [ "$candidate_session" = "$session_id" ] &&
         [[ "$candidate_channel" =~ ^claude-[0-9a-f]{32}$ ]] &&
         [[ "$candidate_pid" =~ ^[0-9]+$ ]] &&
         [ "$candidate_owned" = "true" ] &&
         kill -0 "$candidate_pid" 2>/dev/null; then
        channel="$candidate_channel"
        runtime_dir="$(dirname "$state_file")"
        snapshot_path="$runtime_dir/status.json"
        break
      fi
    done
  fi
fi
if [ -z "$channel" ] || [ -z "$runtime_dir" ]; then
  # An explicitly Emacs-owned session keeps the original channel-file path.
  if [ "$owner" != "pty-supervisor" ] && [ -n "$channel" ]; then
    runtime_dir="${XDG_RUNTIME_DIR:-}"
    snapshot_path="$runtime_dir/claude-idle-compact-${channel}.json"
  fi
fi
if [ "$owner" = "emacs" ]; then
  runtime_dir="${XDG_RUNTIME_DIR:-}"
  snapshot_path="$runtime_dir/claude-idle-compact-${channel}.json"
fi
if [[ "$channel" =~ ^claude-[0-9a-f]+$ && -n "$session_id" &&
      "$session_id" =~ ^[A-Za-z0-9._:-]+$ && "$tokens" =~ ^[0-9]+$ &&
      -n "$snapshot_path" && -d "$runtime_dir" ]]; then
  snapshot_tmp=$(mktemp "$runtime_dir/.claude-idle-compact.XXXXXX" 2>/dev/null || true)
  if [ -n "${snapshot_tmp:-}" ]; then
    if jq -cn --arg channel "$channel" --arg session_id "$session_id" \
         --arg model "$model" \
         --argjson tokens "$tokens" --argjson timestamp "$(date +%s)" \
         '{channel:$channel, session_id:$session_id,
           model:(if $model == "" then null else $model end),
           current_input_context_tokens:$tokens, update_timestamp:$timestamp}' \
         >"$snapshot_tmp" 2>/dev/null; then
      chmod 600 "$snapshot_tmp" 2>/dev/null || true
      mv -f "$snapshot_tmp" "$snapshot_path"
    else
      rm -f "$snapshot_tmp"
    fi
  fi
fi
if [ -n "$tokens" ] && [ "$tokens" != "null" ] && [ "$tokens" -gt 0 ] 2>/dev/null; then
  tk=$(echo "$tokens" | awk '{if($1>=1000) printf "%.1fK",$1/1000; else printf "%d",$1}')
else
  tk=""
fi

# 5-hour rolling window — remaining (prefer dedicated field, fall back to 100-used)
five_hr=$(echo "$input" | jq -r '
  if .rate_limits.five_hour.remaining_percentage != null
  then .rate_limits.five_hour.remaining_percentage
  elif .rate_limits.five_hour.used_percentage != null
  then (100 - .rate_limits.five_hour.used_percentage)
  else empty
  end')

# Weekly quota — remaining % and days to reset
seven_day_used=$(echo "$input" | jq -r '.rate_limits.seven_day.used_percentage // empty')
seven_day_resets=$(echo "$input" | jq -r '.rate_limits.seven_day.resets_at // empty')
week_pct=""
week_days=""
if [ -n "$seven_day_used" ] && [ "$seven_day_used" != "null" ]; then
  week_pct=$(echo "$seven_day_used" | awk '{printf "%.0f", 100-$1}')
fi
if [ -n "$seven_day_resets" ] && [ "$seven_day_resets" != "null" ] && [ "$seven_day_resets" -gt 0 ] 2>/dev/null; then
  now=$(date +%s)
  secs=$((seven_day_resets - now))
  [ $secs -lt 0 ] && secs=0
  week_days=$(awk "BEGIN{printf \"%.1f\", $secs/86400}")
fi

printf "\033[38;5;166m %s\033[0m" "$user@$host"
printf " \033[33m %s\033[0m" "$short_cwd"
if [ -n "$git_branch" ]; then
  printf " \033[34m %s\033[0m" "$git_branch"
fi
printf "\033[38;5;59m  %s\033[0m" "$time_str"
# The Caddy creature — this session's Pokémon from the sealed assignment ledger
# (HelmCortex FORGE/bin/pokedex statusline; gazetteer sealed 2026-08-24, msg 708).
# Silent when HelmCortex or the ledger is absent — never breaks the statusline.
creature=""
pokedex_bin="$HOME/HelmCortex/FORGE/bin/pokedex"
if [ -x "$pokedex_bin" ]; then
  creature=$(printf '%s' "$input" | "$pokedex_bin" statusline 2>/dev/null)
fi
# Consolidate "partner of Fable" with model "Fable 5": row 2 renders the partner
# as the model chip and row 1 drops the model segment (his word, msg 717).
if [ -n "$creature" ] && [ -n "$model" ]; then
  partner_name="${creature##*partner of }"
  if [ "$partner_name" != "$creature" ]; then
    case "$model" in
      "$partner_name"*)
        creature="${creature%partner of *}partner of $(printf '\033[36m✦ %s\033[0m' "$model")"
        model=""
        ;;
    esac
  fi
fi
if [ -n "$model" ]; then
  printf " \033[36m✦ %s\033[0m" "$model"
fi
if [ -n "$tk" ] && [ -n "$used" ]; then
  printf " \033[38;5;59m%s (%s%%)\033[0m" "$tk" "$(printf '%.0f' "$used")"
elif [ -n "$used" ]; then
  printf " \033[38;5;59m(%s%%)\033[0m" "$(printf '%.0f' "$used")"
fi
# Row 2: creature + the quota stats that overflowed row 1 on portrait panes (msg 717).
row2=""

#+BEGIN_SRC javascript :tangle all/.config/opencode/plugins/continuity.js :comments no
import { chmod, mkdir, readFile, rename, writeFile } from "node:fs/promises";
import { execFile } from "node:child_process";
import { homedir } from "node:os";
import { dirname, join } from "node:path";
import { promisify } from "node:util";

const SESSION = /^[A-Za-z0-9._:-]{1,240}$/;
const REFERENCE = /^continuity:\/\/[A-Za-z0-9._:/-]{1,150}$/;
const execute = promisify(execFile);
const RECEIPT_MAX_AGE_MS = 1800 * 1000;
const CLOCK_SKEW_MS = 10 * 1000;
const idleTimers = new Map();

const idleDelayMs = () => {
  const value = Number(process.env.OPENCODE_WARM_IDLE_DELAY_SECONDS ?? 900);
  if (!Number.isFinite(value) || value < 0) return 900 * 1000;
  return Math.min(value, 24 * 60 * 60) * 1000;
};

const stateRoot = () => {
  if (process.env.OPENCODE_WARM_STATE_DIR) return process.env.OPENCODE_WARM_STATE_DIR;
  const stateHome = process.env.XDG_STATE_HOME || join(homedir(), ".local", "state");
  return join(stateHome, "opencode-idle-continuity");
};

const writeEvent = async (sessionID) => {
  if (!SESSION.test(sessionID)) return;
  const target = join(stateRoot(), "events", `${sessionID}.json`);
  const directory = dirname(target);
  await mkdir(directory, { recursive: true, mode: 0o700 });
  await chmod(directory, 0o700);
  const temporary = `${target}.tmp-${process.pid}-${Date.now()}`;
  const payload = JSON.stringify({
    schema: "dotcortex.opencode-continuity-event.v1",
    type: "session.compacted",
    session_id: sessionID,
    observed_at: Date.now() / 1000,
  }) + "\n";
  await writeFile(temporary, payload, { encoding: "utf8", mode: 0o600 });
  await rename(temporary, target);
};

const handoffFor = async (sessionID) => {
  try {
    const raw = await readFile(join(stateRoot(), "handoff.json"), "utf8");
    const value = JSON.parse(raw);
    if (value?.schema !== "dotcortex.opencode-continuity-receipt.v1") return null;
    if (value?.kind !== "handoff" || value?.session_id !== sessionID) return null;
    if (typeof value?.reference !== "string" || !REFERENCE.test(value.reference)) return null;
    if (typeof value?.created_at !== "number" || !Number.isFinite(value.created_at)) return null;
    const age = Date.now() - value.created_at * 1000;
    if (age < -CLOCK_SKEW_MS || age > RECEIPT_MAX_AGE_MS) return null;
    const critical = value?.critical;
    if (critical !== undefined && (critical === null || typeof critical !== "object" || Array.isArray(critical))) return null;
    const clean = {};
    for (const key of ["objective", "next", "prohibition"]) {
      const item = critical?.[key];
      if (typeof item === "string" && item.length > 0 && item.length <= 500) clean[key] = item;
    }
    return {
      reference: value.reference,
      checkpoint: typeof value.checkpoint_id === "string" ? value.checkpoint_id : "unknown",
      sourceMessage: typeof value.message_id === "string" ? value.message_id : "unknown",
      critical: clean,
    };
  } catch {
    return null;
  }
};

const dispatchController = async (sessionID, eventType) => {
  if (!SESSION.test(sessionID)) return;
  const binary = process.env.OPENCODE_WARM_EVENT_BIN || join(homedir(), ".local", "bin", "opencode-warm-event");
  try {
    await execute(binary, [
      "--session", sessionID,
      "--event", eventType,
      "--observed-at", String(Date.now() / 1000),
    ], { timeout: 30000, maxBuffer: 65536 });
  } catch {
    // The controller is fail-closed and writes its own bounded state. A refusal
    // must never destabilize OpenCode's event loop.
  }
};

const dispatchAfterQuietPeriod = async (sessionID) => {
  const previous = idleTimers.get(sessionID);
  if (previous) clearTimeout(previous);
  const delay = idleDelayMs();
  if (delay === 0) {
    await dispatchController(sessionID, "session.idle");
    return;
  }
  const timer = setTimeout(() => {
    idleTimers.delete(sessionID);
    void dispatchController(sessionID, "session.idle");
  }, delay);
  timer.unref?.();
  idleTimers.set(sessionID, timer);
};

export const OpenCodeContinuityPlugin = async () => ({
  event: async ({ event }) => {
    const sessionID = event?.properties?.sessionID;
    if (typeof sessionID !== "string") return;
    if (event?.type === "session.compacted") {
      const pending = idleTimers.get(sessionID);
      if (pending) clearTimeout(pending);
      idleTimers.delete(sessionID);
      await writeEvent(sessionID);
      await dispatchController(sessionID, "session.compacted");
    } else if (event?.type === "session.idle") {
      await dispatchAfterQuietPeriod(sessionID);
    }
  },
  "experimental.session.compacting": async ({ sessionID }, output) => {
    const handoff = await handoffFor(sessionID);
    if (!handoff) return;
    output.context.push(`HelmCortex continuity handoff: ${JSON.stringify(handoff)}`);
  },
  "experimental.compaction.autocontinue": async ({ sessionID }, output) => {
    if (await handoffFor(sessionID)) output.enabled = false;
  },
});

* OpenCode secret hygiene (2026-09-25)
Until 2026-09-25 every interactive zsh/bash exported all of =~/.env=, so an agent launched from a terminal inherited every API key in its process environment, and one =env= in its bash tool would have sent them all to its model provider. The shells no longer do that (=shell.org=, =withkeys=); these two layers still guard shells or parents that carry keys anyway:

- =opencode-clean= starts OpenCode with an allowlisted environment: no keys at all. OpenCode's own provider logins live in its auth file, and the MCP servers above load their own keys (e.g. =opencode-mcp-brave-search= reads =~/.env= itself), so nothing needs the inherited keys. Non-interactive shells the agent spawns do not re-read =~/.env= (=.bashrc= returns early; =.zshrc= is interactive-only).
- The =permission= block in the global config above denies the read tool on =~/.env*=, harness auth files, =~/.ssh=, Borg passphrases and KeePass files, and makes bash commands that name them (or dump the environment) ask first. The read-tool rule alone would not cover =cat ~/.env= through bash; the environment rule alone would not cover the files.

The root fix landed the same day: keys load only into the tool that needs them (=withkeys=, tools reading =~/.env= themselves).

* Git commit guard
Block the exact failure mode that caused the recent regression:

- refuse commits from a dirty tracked worktree
- reject backup/runtime spillover files
- reject generated output commits that do not include canonical Org source
- reject giant sweep commits unless explicitly overridden
- refuse commits while any stash entry exists (rule 12e — a stash is transport, never storage; the 2026-08-06 unpopped pre-pull stash silently reverted four days of work)

Escape hatches exist for intentional exceptional cases:

- =DOTCORTEX_GUARD_ALLOW_DIRTY=1=
- =DOTCORTEX_GUARD_ALLOW_BULK=1=
- =DOTCORTEX_GUARD_ALLOW_JUNK=1=
- =DOTCORTEX_GUARD_ALLOW_GENERATED=1=
- =DOTCORTEX_GUARD_ALLOW_STASH=1=

#+BEGIN_SRC bash :tangle all/.local/bin/dotcortex-git-guard :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/dotcortex-git-guard
set -euo pipefail

note() {
  printf 'dotcortex-guard: %s\n' "$*" >&2
}

die() {
  note "$1"
  exit 1
}

repo="$(git rev-parse --show-toplevel 2>/dev/null || true)"
[ -n "$repo" ] || exit 0
cd "$repo"

allow_dirty="${DOTCORTEX_GUARD_ALLOW_DIRTY:-0}"
allow_bulk="${DOTCORTEX_GUARD_ALLOW_BULK:-0}"
allow_junk="${DOTCORTEX_GUARD_ALLOW_JUNK:-0}"
allow_generated="${DOTCORTEX_GUARD_ALLOW_GENERATED:-0}"
allow_stash="${DOTCORTEX_GUARD_ALLOW_STASH:-0}"

staged=()
while IFS= read -r -d '' path; do
  staged+=("$path")
done < <(git diff --cached --name-only -z --diff-filter=ACMRD)

[ "${#staged[@]}" -gt 0 ] || exit 0

unstaged=()
while IFS= read -r -d '' path; do
  unstaged+=("$path")
done < <(git diff --name-only -z --diff-filter=ACMRD)

if [ "${#unstaged[@]}" -gt 0 ] && [ "$allow_dirty" != "1" ]; then
  note "refusing commit from a dirty tracked worktree"
  printf '%s\n' "${unstaged[@]:0:20}" | sed 's/^/  - /' >&2
  if [ "${#unstaged[@]}" -gt 20 ]; then
    note "... and $((${#unstaged[@]} - 20)) more"
  fi
  note "stage intentionally, or commit the in-flight work as its own wip checkpoint — never stash it away"
  note "override once with: DOTCORTEX_GUARD_ALLOW_DIRTY=1 git commit ..."
  exit 1
fi

stash_count="$(git stash list | wc -l)"
if [ "$stash_count" -gt 0 ] && [ "$allow_stash" != "1" ]; then
  note "refusing commit while $stash_count stash entries exist — a stash is transport, never storage (rule 12e)"
  git stash list --format='  - %gd  %cr  %s' | head -12 >&2
  if [ "$stash_count" -gt 12 ]; then
    note "... and $((stash_count - 12)) more"
  fi
  note "resolve each: git stash pop/apply it, or preserve it as a real ref:"
  note "  git tag rescue/stash-<name> \$(git rev-parse 'stash@{N}') && git stash drop 'stash@{N}'"
  note "override once (triage in progress) with: DOTCORTEX_GUARD_ALLOW_STASH=1 git commit ..."
  exit 1
fi

declare -A roots=()
bad_paths=()
has_org_source=0
generated_paths=0

for path in "${staged[@]}"; do
  root="${path%%/*}"
  roots["$root"]=1

  case "$path" in
    *.org)
      has_org_source=1
      ;;
  esac

* Network-address guard (rule 21)
AGENTS.md rule 21 says DotCortex holds no network addresses, because the repository is public and mirrored. Until 2026-09-24 that rule was prose only, and three tailnet addresses sat in public history for a month (introduced 2026-08-25 by two "bind to the tailnet IP, not 0.0.0.0" commits). This guard makes the rule mechanical.

- It scans only *added* lines, so existing history never blocks work but no new address can enter.
- It runs at commit time (=--staged=, from the pre-commit and pre-merge-commit hooks) and again at push time (=--pre-push=, from the pre-push hook, over every commit the push would publish).
- It refuses tailnet (CGNAT, 100.64/10) and private (RFC 1918) IPv4 literals.
- There is no override flag. A known-fake fixture address can be named in =DOTCORTEX_GUARD_ALLOWED_ADDRESSES= (space-separated); the default allows only the CGNAT base address that the herdr-web test uses as a stand-in.

Resolve a machine's own tailnet address at runtime instead, with =tailscale ip -4= filtered to the first IPv4 line (see the Recall units in =services.org=). Per-host addresses that are not the machine's own belong in machine-local, uncommitted env files.

#+BEGIN_SRC bash :tangle all/.local/bin/dotcortex-address-guard :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/dotcortex-address-guard
set -euo pipefail

usage() {
  cat <<'USAGE'
Usage: dotcortex-address-guard --staged
       dotcortex-address-guard --pre-push <remote-name> <remote-url>  (reads refs on stdin)
       dotcortex-address-guard --range <rev-range>

Refuse tailnet (100.64/10) and private (RFC 1918) IPv4 literals in lines
being added to DotCortex (AGENTS.md rule 21). Only added lines are scanned.
Known-fake fixture addresses: DOTCORTEX_GUARD_ALLOWED_ADDRESSES (space-separated).
USAGE
}

ADDRESS_RE='(100\.(6[4-9]|[7-9][0-9]|1[01][0-9]|12[0-7])|10\.[0-9]{1,3}|172\.(1[6-9]|2[0-9]|3[01])|192\.168)\.[0-9]{1,3}\.[0-9]{1,3}'
ALLOWED=" ${DOTCORTEX_GUARD_ALLOWED_ADDRESSES:-100.64.0.1} "

# Read a unified diff on stdin; print "path: address" for every added-line hit.
scan_diff() {
  awk -v re="$ADDRESS_RE" -v allowed="$ALLOWED" '
    /^\+\+\+ / { path = substr($0, 5); sub(/^b\//, "", path); next }
    /^\+/ {
      line = substr($0, 2)
      while (match(line, re)) {
        hit = substr(line, RSTART, RLENGTH)
        before = (RSTART > 1) ? substr(line, RSTART - 1, 1) : ""
        after = substr(line, RSTART + RLENGTH, 1)
        if (before !~ /[0-9.]/ && after !~ /[0-9]/ && index(allowed, " " hit " ") == 0)
          print path ": " hit
        line = substr(line, RSTART + RLENGTH)
      }
    }'
}

report() {
  local hits="$1" where="$2"
  [ -z "$hits" ] && return 0
  printf 'dotcortex-address-guard: refusing %s: network addresses in a public repo (AGENTS.md rule 21)\n' "$where" >&2
  printf '%s\n' "$hits" | sort -u | sed 's/^/  - /' >&2
  printf 'dotcortex-address-guard: resolve the address at runtime (tailscale ip -4) or keep it in a machine-local env file\n' >&2
  return 1
}

case "${1:-}" in
  -h|--help) usage; exit 0 ;;
  --staged)
    report "$(git diff --cached -U0 --no-color --no-ext-diff | scan_diff)" "commit" ;;
  --range)
    [ -n "${2:-}" ] || { usage >&2; exit 2; }
    report "$(git log -p -U0 --no-color --no-ext-diff --format= "$2" | scan_diff)" "range $2" ;;
  --pre-push)
    zero="0000000000000000000000000000000000000000"
    status=0
    while read -r local_ref local_sha remote_ref remote_sha; do
      [ "$local_sha" = "$zero" ] && continue
      if [ "$remote_sha" = "$zero" ]; then
        hits="$(git log -p -U0 --no-color --no-ext-diff --format= "$local_sha" --not --remotes | scan_diff)"
      else
        hits="$(git log -p -U0 --no-color --no-ext-diff --format= "$remote_sha..$local_sha" | scan_diff)"
      fi
      report "$hits" "push of $local_ref" || status=1
    done
    exit "$status" ;;
  *) usage >&2; exit 2 ;;
esac
#+END_SRC

* Safe pull guard
Wrapper around =git pull --ff-only= that refuses if the working tree has any
dirty tracked files. Prevents the exact failure mode where a fast-forward pull
silently succeeds even with uncommitted local changes that don't conflict with
the incoming commits, destroying the ability to audit what was in-flight.

Use =dotcortex-pull= instead of =git pull --ff-only= for any automated or
remote-triggered pull. The override =DOTCORTEX_PULL_ALLOW_DIRTY=1= exists for
intentional exceptions (e.g. pulling while stash is present and you know it).

#+BEGIN_SRC bash :tangle all/.local/bin/dotcortex-pull :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/dotcortex-pull
set -euo pipefail

die() {
  printf 'dotcortex-pull: %s\n' "$*" >&2
  exit 1
}

repo="$(git rev-parse --show-toplevel 2>/dev/null || true)"
[ -n "$repo" ] || die "not inside a git repo"
cd "$repo"

allow_dirty="${DOTCORTEX_PULL_ALLOW_DIRTY:-0}"

unstaged=()
while IFS= read -r -d '' path; do
  unstaged+=("$path")
done < <(git diff --name-only -z --diff-filter=ACMRD)

if [ "${#unstaged[@]}" -gt 0 ] && [ "$allow_dirty" != "1" ]; then
  printf 'dotcortex-pull: refusing pull from a dirty tracked worktree\n' >&2
  printf '%s\n' "${unstaged[@]:0:20}" | sed 's/^/  - /' >&2
  [ "${#unstaged[@]}" -gt 20 ] && printf 'dotcortex-pull: ... and %d more\n' "$((${#unstaged[@]} - 20))" >&2
  printf 'dotcortex-pull: stash or commit first\n' >&2
  printf 'dotcortex-pull: override once with: DOTCORTEX_PULL_ALLOW_DIRTY=1 dotcortex-pull\n' >&2
  exit 1
fi

exec git pull --ff-only "$@"

    bytes=$(printf '%s' "$extracted" | wc -c)

    if [ "$bytes" -gt 6000 ]; then

      printf '\n(… LightCone truncated at 6000 bytes — full section in the TODO file)\n'

* Telegram delivery guard
*event: Stop (wired first in the Stop array above).*

In a Telegram-bridged session the assistant's transcript text never reaches Mètsàtron — only a Telegram reply tool call does. This Stop hook lives in the SHARED baseline so it deploys to =~/.claude/settings.json= at USER scope: it therefore protects every warm bot (Fable/Auryn/Haiku/Opus) regardless of working directory, unlike the old HelmCortex-root-only wiring that most bots never loaded (Claude Code loads the cwd's own =.claude/settings.json= plus user scope, not ancestor project dirs).

At turn end, if the LAST user prompt actually arrived FROM Telegram (carries a =<channel source="...telegram...">= tag) and the turn produced spoken assistant text but made no Telegram delivery call (=reply=/=edit_message=), it blocks the stop and tells the model to deliver. A plain TUI prompt is never blocked (the terminal is what the user sees there). Gated to channel-bound sessions; fail-OPEN on any error (never wedge a session); never loops (honors =stop_hook_active=). The transcript read retries once to survive a flush race, and logs the distinct failure shapes (missing path / present-but-unreadable / no transcript_path) so a real fault is diagnosable rather than silent.

#+BEGIN_SRC bash :tangle all/.local/bin/claude-hook-telegram-delivery-guard :shebang "#!/usr/bin/env bash" :comments no
# ~/DotCortex/all/.local/bin/claude-hook-telegram-delivery-guard
# Stop hook. Fail-OPEN on any error (never wedge a session).
# User-scope: loaded for every session via ~/.claude/settings.json, so it
# protects all warm bots (Fable/Auryn/Haiku/Opus) regardless of working dir --
# unlike the old HelmCortex-root-only wiring that most bots never loaded.
INPUT=$(cat 2>/dev/null || true)
HOOK_NAME="telegram-delivery-guard"
LOG_DIR="${XDG_STATE_HOME:-$HOME/.local/state}/claude-hooks"
ERROR_LOG="$LOG_DIR/telegram-delivery-guard.log"

log() {
  local message="${1:-unknown}"
  message="${message//$'\n'/ }"
  mkdir -p "$LOG_DIR" 2>/dev/null || return 0
  printf '%s hook=%s %s\n' "$(date '+%Y-%m-%dT%H:%M:%S%z')" "$HOOK_NAME" "$message" >> "$ERROR_LOG" 2>/dev/null || true
}

# Gate 1: only channel-bound (bridged) sessions -- else the transcript IS the user's view.
[ -n "${CLAUDE_IDLE_CHANNEL:-}" ] || exit 0
command -v python3 >/dev/null 2>&1 || exit 0

# Gate 2: never loop -- if a prior Stop already blocked this turn, let it pass.
STOP_ACTIVE=$(printf '%s' "$INPUT" | python3 -c 'import json,sys
try: print("1" if json.load(sys.stdin).get("stop_hook_active") else "0")
except Exception: print("0")' 2>/dev/null)
[ "$STOP_ACTIVE" = "1" ] && exit 0

TP=$(printf '%s' "$INPUT" | python3 -c 'import json,sys
try: print(json.load(sys.stdin).get("transcript_path","") or "")
except Exception: print("")' 2>/dev/null)
if [ -z "$TP" ]; then
  log "error=no transcript_path in Stop payload"
  exit 0
fi
# The transcript can lag the Stop event by a flush; retry once before giving up.
[ -r "$TP" ] || sleep 0.15
if [ ! -r "$TP" ]; then
  if [ -e "$TP" ]; then log "error=transcript present but unreadable: $TP"
  else log "error=transcript path does not exist: $TP"; fi
  exit 0
fi

VERDICT=$(python3 -c '
import json, sys, re
tp = sys.argv[1]
recs = []
try:
    with open(tp) as f:
        for line in f:
            line = line.strip()
            if not line: continue
            try: recs.append(json.loads(line))
            except Exception: pass
except Exception:
    print("pass"); sys.exit(0)

def texts(content):
    out = []
    if isinstance(content, str):
        out.append(content)
    elif isinstance(content, list):
        for b in content:
            if isinstance(b, dict) and b.get("type") == "text":
                t = b.get("text") or ""
                if t: out.append(t)
    return out

# Last REAL user prompt: type=user, role=user, carrying text.
# (tool_result turns are role=user but carry only tool_result blocks -- not prompts.)
last_user = -1
last_text = ""
for i, d in enumerate(recs):
    if d.get("type") != "user": continue
    m = d.get("message") or {}
    if m.get("role") != "user": continue
    joined = "\n".join(texts(m.get("content"))).strip()
    if joined:
        last_user = i
        last_text = joined

if last_user < 0:
    print("pass"); sys.exit(0)

# Only guard prompts that arrived FROM Telegram; a plain TUI prompt needs no
# Telegram reply because the terminal is what the user sees there.
if not re.search(r"<channel\b[^>]*telegram", last_text, re.I):
    print("pass"); sys.exit(0)

deliver = re.compile(r"telegram.*(reply|edit_message)", re.I)
spoke = delivered = False
for d in recs[last_user+1:]:
    if d.get("type") != "assistant": continue
    c = (d.get("message") or {}).get("content")
    if not isinstance(c, list): continue
    for b in c:
        if not isinstance(b, dict): continue
        if b.get("type") == "text" and (b.get("text") or "").strip():
            spoke = True
        elif b.get("type") == "tool_use" and deliver.search(str(b.get("name",""))):
            delivered = True

print("block" if (spoke and not delivered) else "pass")
' "$TP" 2>/dev/null) || VERDICT="pass"

if [ "$VERDICT" = "block" ]; then
  log "decision=block spoke-without-telegram-delivery"
  python3 -c 'import json,sys; sys.stdout.write(json.dumps({"decision":"block","reason":(
"TELEGRAM DELIVERY GUARD: this turn produced a response but never called the Telegram reply tool. "
"Your transcript text is INVISIBLE to Mètsàtron -- only mcp__plugin_telegram_telegram__reply reaches his chat. "
"Deliver your message now via the reply tool (chat_id from the inbound <channel> tag), turn-final, then stop.")}))'
  exit 0
fi
exit 0

* Voice reply relay (outbound — pvox speaks what the agent sends)

The other half of the voice loop, per Mètsàtron's ruling (2026-08-16): every Telegram reply an agent sends is ALSO spoken aloud through pvox, automatically, with the voice chosen by the agent's identity — no extra agent action, no extra tokens. A =PostToolUse= hook fires on the Telegram plugin's =reply= tool, lifts the already-generated =text= from the tool input, resolves the speaking identity, and fire-and-forgets it to pvox on the speaker host over SSH. The hook returns immediately (playback is fully detached), so replies are never delayed by synthesis.

Identity resolution, simplest-first: =CLAUDE_WARM_HERDR_AGENT= (set by every claude-* launcher: Fable, Haiku, Auryn, Opus, or OpenCode's Ecco hook) → the basename suffix of =TELEGRAM_STATE_DIR= (=telegram-fable= → fable) → =Default=. The identity maps onto the pvox agent roster (=pvox agents=); Nano identifies as =Nanobot= and shares the existing =Haiku= voice seat until Mètsàtron seals a dedicated voice. Delivery is split-lane (ruling 2026-08-17, the night the phone earned its voice): synthesis runs on the hook's own machine via pvox (HelmCortex is NFS-shared, so pvox and the conda engines are everywhere), and playback ships as a *headered WAV* to =PVOX_REPLY_HOST= (default =s24= — the phone, always in his pocket), falling back to =PVOX_REPLY_FALLBACK_HOST= (default =t480s=) when the phone is unreachable. The old pvox-over-SSH law forbade piping *raw* PCM across SSH because the player cannot know the rate; WAV carries its header, so that law is honored, not repealed. Disable per-session with =PVOX_REPLY_OFF=1=.

Delivery is *push-then-play* (fix 2026-08-17, the "stopped halfway" incident): the original design streamed the WAV over SSH straight into =sox='s stdin, which made the entire playback depend on the SSH connection staying alive — when the phone dozed mid-play (screen off, in pocket), the stream stalled and the audio truncated halfway. Now the play host receives the whole WAV into a temp file first, then playback detaches from the SSH session (=nohup=, backgrounded) with a Termux wake-lock held for its duration, so the network matters only during transfer and a doze can no longer cut the voice off. The SSH exit code therefore certifies *delivery*, not playback — the log field says =delivered=, honestly. Every skip and failure path logs one line to =~/.cache/pvox-reply.log= (empty-text skip with stdin byte count, render failure, per-host delivery failure, successful delivery with host + sizes) — the hook's original total silence meant every incident cost a fresh archaeology dig, against the silent-skip law.

Kitten seats render *on the play host itself* (his ruling 2026-08-17: the S24 out-muscles the fleet for small models, Q8-verified ~0.4–0.9 RTF): the hook resolves the seat via =pvox voicespec=, normalises every reply through the shared =voice-text-normalize= boundary, then ships CLEANED TEXT instead of a WAV to =termux-kitten-say= on the play host, detached with a wake lock, sharing the same play queue lock as the ship lane. The fleet render+ship path remains the lane for non-kitten seats, fallback hosts, and a dead enclave — on-device failure logs =ON-DEVICE-FAIL= and falls through. Kill switch: =PVOX_REPLY_NO_ONDEVICE=1=.

** Agent voice spool (sandboxed agents speak without the operator's SSH key)

An agent that runs as its own Unix user (Agent Boundaries phase 8: =agent-deepseek= first) cannot run the relay above: the relay ships audio to the phone over the operator's SSH identity, and holding that key is exactly what the separate user exists to prevent. So the relay is split at the user boundary (ruling 2026-09-26, "provided that the latency is the same"). The agent side only *drops* its hook payload into a spool directory; an operator-owned watcher picks each file up through inotify within milliseconds and feeds it to the unchanged =claude-hook-voice-reply=, so synthesis, voice choice, phone delivery, fallback host and play queue stay one code path.

The spool deliberately does not parse or rewrite reply text: it is a byte-preserving trust-boundary transport.  Its operator-owned destination is the hook above, which extracts and normalises the text before the shared spoken-character cap.  Ecco and DeepSeek therefore inherit the same speech boundary without a second implementation.

- Spool: =/var/spool/agent-voice=, owned by the operator, group =cortex=, mode =3730=. Agents (all in =cortex=) can create files but cannot list the directory or delete each other's files; the operator owns the directory and removes each file once spoken.
- Identity comes from the file's owning uid, never from the payload: =agent-deepseek= speaks as =DeepSeek=, =agent-opencode= as =Ecco=, any other =agent-NAME= as =Name=, and files owned by anyone else are discarded. A sandboxed agent cannot make itself speak in another agent's voice.
- Files older than two minutes are discarded rather than spoken (a watcher outage must not replay a backlog of stale replies), and a payload is capped at 64 KiB before the hook's own character cap applies.
- Owner: =agent-voice-relay install= (run once as the operator; uses =sudo -n=) creates the spool and writes =/etc/cron.d/dotcortex-agent-voice-relay=: =@reboot= plus a three-minute self-heal, both calling =serve --daemon=, which is a no-op while a watcher holds the lock. Tangling does not deploy it.
- Agent side: the DSH Telegram bridge's =media.voiceReply.command= points at this script by its DotCortex path (the operator's =~/.local= is closed to agents; =~/DotCortex= is not). The bridge spawns it with no arguments, so no arguments means =drop=.
- Log: =~/.local/state/agent-voice-relay/serve.log= (one line per spoken, stale or rejected file).

#+BEGIN_SRC bash :tangle all/.local/bin/agent-voice-relay :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/agent-voice-relay
# Split voice-reply relay across the agent user boundary. See agents-hooks.org.
set -euo pipefail

SPOOL="${AGENT_VOICE_SPOOL:-/var/spool/agent-voice}"
SELF="$(readlink -f "$0")"
MAX_BYTES=65536
STALE_SECONDS=120

usage() {
    cat <<'EOF'
usage: agent-voice-relay [drop]         agent side: hook JSON on stdin -> spool
       agent-voice-relay serve [--daemon]
                                        operator side: speak spooled replies
       agent-voice-relay install        operator, once: spool dir + cron owner
       agent-voice-relay status
EOF
}

# Identity is derived from the file owner, never read from the payload.
identity_for() {
    case "$1" in
        agent-deepseek) echo DeepSeek ;;
        agent-opencode) echo Ecco ;;
        agent-?*) local n="${1#agent-}"; echo "${n^}" ;;
        *) return 1 ;;
    esac
}

cmd_drop() {
    # Best effort: a missing spool must never fail the agent's turn.
    [ -d "$SPOOL" ] && [ -w "$SPOOL" ] || { cat >/dev/null; exit 0; }
    local name tmp
    name="$(date +%s%N)-$$-$RANDOM"
    tmp="$SPOOL/.$name.tmp"
    umask 027
    head -c "$MAX_BYTES" > "$tmp"
    mv "$tmp" "$SPOOL/$name.json"
}

STATE="${XDG_STATE_HOME:-$HOME/.local/state}/agent-voice-relay"
LOG="$STATE/serve.log"
HOOK="${AGENT_VOICE_HOOK:-$HOME/.local/bin/claude-hook-voice-reply}"

log() { printf '%s %s\n' "$(date '+%F %T')" "$*" >> "$LOG"; }

speak_one() {
    local f="$1" owner id age bytes
    [ -f "$f" ] || return 0
    owner="$(stat -c %U "$f" 2>/dev/null)" || return 0
    bytes="$(stat -c %s "$f")"
    age=$(( $(date +%s) - $(stat -c %Y "$f") ))
    if ! id="$(identity_for "$owner")"; then
        log "REJECT owner=$owner bytes=$bytes"
    elif [ "$age" -gt "$STALE_SECONDS" ]; then
        log "STALE agent=$id age=${age}s bytes=$bytes"
    elif env -u PVOX_REPLY_OFF CLAUDE_WARM_HERDR_AGENT="$id" "$HOOK" < "$f" >/dev/null 2>&1; then
        log "SPOKE agent=$id bytes=$bytes"
    else
        log "HOOK-FAIL agent=$id bytes=$bytes"
    fi
    rm -f -- "$f"
}

cmd_serve() {
    mkdir -p "$STATE"
    if [ "${1:-}" = "--daemon" ]; then
        nohup setsid "$SELF" serve </dev/null >>"$LOG" 2>&1 &
        exit 0
    fi
    exec 9>"$STATE/serve.lock"
    flock -n 9 || exit 0   # a watcher is already running
    [ -d "$SPOOL" ] || { log "FATAL no spool at $SPOOL (run agent-voice-relay install)"; exit 1; }
    command -v inotifywait >/dev/null || { log "FATAL inotifywait not on PATH"; exit 1; }
    [ -x "$HOOK" ] || { log "FATAL hook not executable: $HOOK"; exit 1; }
    log "START pid=$$ spool=$SPOOL"
    # Watch first, then list the backlog into the same stream, so a drop that
    # lands between the two is seen at least once (speak_one tolerates repeats).
    {
        # 9>&-: the watcher must not inherit the lock, or an orphaned inotifywait
        # would block the cron self-heal after its parent dies.
        inotifywait -m -q -e moved_to --format '%f' "$SPOOL" 9>&- &
        sleep 1
        (cd "$SPOOL" && ls -1 -- *.json 2>/dev/null) || true
        wait
    } | while IFS= read -r n; do
        case "$n" in *.json) speak_one "$SPOOL/$n" ;; esac
    done
    log "EXIT watcher ended (cron restarts it)"
}

cmd_install() {
    local me cron tmp path
    me="$(id -un)"
    cron=/etc/cron.d/dotcortex-agent-voice-relay
    path="$HOME/.guix-extra-profiles/core/core/bin:$HOME/.local/bin:/usr/local/bin:/usr/bin:/bin"
    sudo -n install -d -o "$me" -g cortex -m 3730 "$SPOOL"
    tmp="$(mktemp)"
    cat > "$tmp" <<EOF
# DotCortex agent voice relay owner; managed by agent-voice-relay install.
SHELL=/bin/sh
@reboot $me HOME=$HOME PATH=$path $SELF serve --daemon
*/3 * * * * $me HOME=$HOME PATH=$path $SELF serve --daemon
EOF
    sudo -n install -m 644 "$tmp" "$cron"
    rm -f "$tmp"
    echo "spool: $(stat -c '%A %U:%G' "$SPOOL") $SPOOL"
    echo "cron:  $cron"
}

cmd_status() {
    echo "spool: $(stat -c '%A %U:%G' "$SPOOL" 2>/dev/null || echo missing) $SPOOL"
    if [ -e "$STATE/serve.lock" ] && ! flock -n "$STATE/serve.lock" true; then
        echo "watcher: running"
    else
        echo "watcher: not running"
    fi
    [ -f "$LOG" ] && tail -n 5 "$LOG"
    return 0
}

case "${1:-drop}" in
    drop) cmd_drop ;;
    serve) shift; cmd_serve "$@" ;;
    install) cmd_install ;;
    status) cmd_status ;;
    -h|--help) usage ;;
    *) usage >&2; exit 1 ;;
esac
#+END_SRC

* Voice transcript relay

Auto-transcribes Telegram voice/audio notes and relays the transcript both ways. The official Telegram plugin delivers voice notes as an =attachment_file_id= in the =<channel>= tag without downloading them — without this hook the agent must notice, call =download_attachment=, and transcribe by hand. This hook makes the flow structural, per Mètsàtron's ruling (2026-08-16): the same architecture must hold wherever the agent runs — kikin today, Termux on the phone later — so it lives here in the hooks source, not in any one session's habits.

Flow: =UserPromptSubmit= fires on the inbound channel message → the hook parses =attachment_kind="voice|audio"= + =attachment_file_id= from the tag attributes (plugin-authored; body text could forge them, but senders are allowlisted and a forged file_id can only reach files this bot already holds) → downloads via the Bot API using the token in =TELEGRAM_STATE_DIR/.env= → runs =whisper-transcribe --model small= (found on PATH or at the HelmCortex FORGE fallback) → injects the transcript as =additionalContext= so the agent reads it as if typed → and for provenance parity with the Hermes agent, replies the transcript into the Telegram chat itself, quoted, threaded to the voice message, so the chat log carries what the machine heard. Idempotent per =file_id= (download, transcript, and chat echo each fire once); every failure degrades to a one-line context note rather than blocking the prompt; the token never appears in any emitted text (exception class names only).

fi

python3 - <<'PYEOF'

* File edit rollback guard
Stages exact-path rollback anchors around Claude file-edit tools in DotCortex:

- before =Edit=/=MultiEdit=/=Write=, an already-dirty tracked file is staged so its pre-edit state remains recoverable from the index
- before editing an existing untracked file, that file is staged first
- after =Write= creates a new file, the exact new path is staged immediately

The hook does not auto-stage clean tracked files after editing; doing that would overwrite the index and erase the previous rollback point.

* Remote git safety guard
Intercepts Bash tool calls that attempt remote git operations without proper precautions.
Blocks =ssh -A= (agent forwarding) and raw remote =git pull= without dirty-state checks.
Also blocks raw =git pull --ff-only= on remote machines when =dotcortex-pull= should be used instead.
Blocks DotCortex stow verbs whose machine target does not match the machine they would run on — the live host for local commands, or the ssh destination for ssh-wrapped commands. Block reasons go to stderr, which is what Claude Code shows the agent on a PreToolUse exit-2.

* Cross-harness stow target guard

The machine-target check is shared by Claude Code, Codex, and =make safe-stow=.
The harness hooks are useful early warnings, but =safe-stow= is the enforcement
boundary: a direct terminal invocation must not be able to apply an X230 or
T480s overlay on the wrong host. Shared overlays (=all linux debian devuan
termux=) are valid everywhere. Exactly one machine-specific overlay may appear
in a package set, and it must match the live hostname. Unknown package names
remain outside the machine check so future non-fleet overlays are not silently
reclassified as laptop targets.

The package mode also hands the expanded stack to =dotcortex-layers validate=
(=layers.org=): at most one =user-*= layer, it must belong to the invoking
account, and it must be file-disjoint from every other layer in the stack
(Stow has no override semantics). Package names declared nowhere in
=hosts.ssv= only *warn*. A stack without a =user-*= package keeps its
pre-registry behaviour even if the helper is missing or fails (warning only);
a stack with one fails closed. The command (hook) mode is unchanged.

The command mode accepts the Claude/Codex shell command text and understands
=loom stow:<machine>=, direct =stow <machine>=, =make safe-stow= with
=STOW_PKGS=, and the legacy =make x11-apply= X230 path. The package mode is
what the Make recipe calls after expanding =STOW_PKGS=.

#!/usr/bin/env python3
"""Fail-closed host/overlay validation for every DotCortex stow path."""

import argparse
import json
import os
import re
import shlex
import socket
import subprocess
import sys


SHARED_PACKAGES = {"all", "linux", "debian", "devuan", "termux"}
HOST_ALIASES = {
    "ThinkPad-T480s": "t480s",
    "t480s": "t480s",
    "ThinkPad-X230": "x230",
    "x230": "x230",
    "ThinkPad-T480": "t480",
    "t480": "t480",
    "ThinkPad-T1700": "t1700",
    "t1700": "t1700",
    "kikin-kushi": "t1700",
    "OpenMandriva": "openmandriva",
    "openmandriva": "openmandriva",
    "s24": "s24",
    "localhost": "s24",
    "beelink": "honey",
}
EXPECTED_HOSTS = {
    "x230": {"x230", "ThinkPad-X230"},
    "t480s": {"t480s", "ThinkPad-T480s"},
    "t480": {"t480", "ThinkPad-T480"},
    "t1700": {"t1700", "ThinkPad-T1700", "kikin-kushi"},
    "openmandriva": {"openmandriva", "OpenMandriva"},
    "s24": {"s24", "localhost"},
    "honey": {"beelink"},
}
MACHINE_PACKAGES = set(EXPECTED_HOSTS)


def sanctuary_context_active():
    """Allow the X230 public layer only inside its named guest home."""
    # The named guest is the only deliberate cross-host stow exception: the
    # Cortex migration projects the X230-shaped public layer into T1700.
    context = os.environ.get("DOTCORTEX_STOW_CONTEXT", "")
    home = os.path.realpath(os.environ.get("HOME", ""))
    guest_suffix = "/.local/share/dotcortex/guests/sanctuary-cortex/home"
    return context == "sanctuary-cortex" and home.endswith(guest_suffix)


def short_host(raw_host):
    return HOST_ALIASES.get(raw_host, raw_host)


def block(reason, target, raw_host, where, codex):
    short = short_host(raw_host)
    message = (
        "DotCortex stow target does not match the machine it would run on. "
        f"Requested target: {target}. Would run on: {where} "
        f"(alias: {short}). {reason}"
    )
    if codex:
        print(json.dumps({
            "hookSpecificOutput": {
                "hookEventName": "PreToolUse",
                "permissionDecision": "deny",
                "permissionDecisionReason": message,
            }
        }))
        return 0
    print("BLOCKED: " + message, file=sys.stderr)
    print(f"Use the stow verb for '{short}', or run this on the matching machine.", file=sys.stderr)
    return 2


def validate(packages, raw_host, where, codex=False):
    targets = [pkg for pkg in packages if pkg in MACHINE_PACKAGES]
    unique_targets = list(dict.fromkeys(targets))
    if len(unique_targets) > 1:
        return block(
            "A package set may contain only one machine-specific overlay.",
            ", ".join(unique_targets),
            raw_host,
            where,
            codex,
        )
    if not unique_targets:
        return 0
    target = unique_targets[0]
    if target == "x230" and sanctuary_context_active():
        return 0
    expected = EXPECTED_HOSTS[target]
    if raw_host not in expected and short_host(raw_host) not in expected:
        return block(
            f"Expected host: {', '.join(sorted(expected))}.",
            target,
            raw_host,
            where,
            codex,
        )
    return 0


def layers_helper():
    """Locate dotcortex-layers beside this guard, then in the usual homes."""
    here = os.path.dirname(os.path.realpath(__file__))
    for candidate in (
        os.path.join(here, "dotcortex-layers"),
        os.path.expanduser("~/.local/bin/dotcortex-layers"),
        os.path.expanduser("~/DotCortex/all/.local/bin/dotcortex-layers"),
    ):
        if os.path.isfile(candidate):
            return candidate
    return None


def validate_layers(packages):
    """Package mode only: user-layer ownership, overlap and registry checks (layers.org)."""
    has_user = any(pkg.startswith("user-") for pkg in packages)
    helper = layers_helper()
    if helper is None:
        if has_user:
            print("BLOCKED: user layer requested but dotcortex-layers is missing", file=sys.stderr)
            return 2
        return 0
    result = subprocess.run([sys.executable, helper, "validate", "--packages", " ".join(packages)])
    if result.returncode == 0:
        return 0
    if has_user:
        return 2
    print("dotcortex-stow-target-guard: warning: layer validation failed; ignored because "
          "this stack has no user layer", file=sys.stderr)
    return 0


def ssh_destination(command):
    try:
        words = shlex.split(command)
    except ValueError:
        words = command.split()
    # An explicit path (/usr/bin/ssh) and leading VAR=val assignments are still
    # ssh invocations — matching only the bare word misses them (2026-08-25).
    index = 0
    while index < len(words) and "=" in words[index] and not words[index].startswith("-") and "/" not in words[index].split("=", 1)[0]:
        index += 1
    if index >= len(words) or words[index].rsplit("/", 1)[-1] != "ssh":
        return None
    consuming_flags = {"-p", "-i", "-o", "-l", "-F", "-J", "-b", "-e", "-c", "-L", "-R", "-D", "-W", "-S"}
    index += 1
    while index < len(words):
        word = words[index]
        if word in consuming_flags:
            index += 2
            continue
        if word.startswith("-"):
            index += 1
            continue
        return word.rsplit("@", 1)[-1]
    return None


def command_targets(command):
    targets = []
    loom_pattern = r"(?:^|[;&|]\s*)(?:\S*/)?loom\s+stow(?::([A-Za-z0-9_-]+))?"
    for match in re.finditer(loom_pattern, command):
        targets.append(match.group(1) or "all")

    direct_pattern = r"(?:^|[;&|]\s*)(?:\S*/)?stow\s+(x230|t480s|t480|t1700|s24|openmandriva)\b"
    for match in re.finditer(direct_pattern, command):
        targets.append(match.group(1))

    if re.search(r"(?:^|[;&|]\s*)(?:\S*/)?make(?:\s+-f\s+\S+)?\s+(?:safe-stow|stow)\b", command):
        assignment = re.search(
            r"(?:^|\s)STOW_PKGS=(?:\"([^\"]*)\"|'([^']*)'|([^\s;&|]+))",
            command,
        )
        if assignment:
            targets.extend(shlex.split(next(value for value in assignment.groups() if value is not None)))
        else:
            targets.append("all")
    if re.search(r"(?:^|[;&|]\s*)(?:\S*/)?make(?:\s+-f\s+\S+)?\s+x11-apply\b", command):
        targets.append("x230")
    return targets


def validate_command(command, codex=False):
    targets = command_targets(command)
    if not targets:
        return 0
    destination = ssh_destination(command)
    if destination:
        raw_host = destination.split(".", 1)[0]
        where = f"ssh destination '{destination}'"
    else:
        raw_host = socket.gethostname().split(".", 1)[0]
        where = f"local host '{raw_host}'"
    return validate(targets, raw_host, where, codex)


def codex_command_from_stdin():
    try:
        payload = json.load(sys.stdin)
        tool_input = payload.get("tool_input", {})
        return tool_input.get("command", "") if isinstance(tool_input, dict) else ""
    except (json.JSONDecodeError, OSError):
        return None


def main():
    parser = argparse.ArgumentParser()
    modes = parser.add_mutually_exclusive_group(required=True)
    modes.add_argument("--packages", help="space-separated STOW_PKGS value")
    modes.add_argument("--command", help="shell command text to inspect")
    modes.add_argument("--codex-json", action="store_true", help="read a Codex PreToolUse payload from stdin")
    args = parser.parse_args()

    if args.packages is not None:
        packages = shlex.split(args.packages)
        status = validate(
            packages,
            socket.gethostname().split(".", 1)[0],
            f"local host '{socket.gethostname().split('.', 1)[0]}'",
        )
        if status != 0:
            return status
        return validate_layers(packages)
    if args.command is not None:
        return validate_command(args.command)
    command = codex_command_from_stdin()
    if command is None:
        return block("The Codex hook payload could not be parsed.", "unknown", socket.gethostname(), "local host", True)
    return validate_command(command, codex=True)


if __name__ == "__main__":
    sys.exit(main())

def simple_commands(text):
    """Split into simple commands, honouring quotes, keeping redirect operators.

    shlex with punctuation_chars emits ; & | < > as their own tokens and leaves
    quoted spans intact as single tokens, so a `>` inside "a -> b" never appears
    as an operator. That is the whole reason this is tokenised rather than
    regexed: two false positives on 2026-07-26 came from matching shell syntax
    inside quoted strings.
    """
    lexer = shlex.shlex(text, posix=True, punctuation_chars=True)
    lexer.whitespace_split = True
    try:
        tokens = list(lexer)
    except ValueError:
        return None  # unbalanced quotes / heredoc: caller falls back

    groups, current = [], []
    for tok in tokens:
        if tok in (";", "&&", "||", "|", "&", "\n"):
            groups.append(current)
            current = []
        else:
            current.append(tok)
    groups.append(current)
    return groups

* Pre-compact
Memory is destructive to truncate and must only be cleared after the Obsidian
append reports success.  The old unconditional sequence lost the live copy when
an archive command failed: the T480 facts survived elsewhere, but the promised
=DotCortex.md= dump did not exist.  A failed append now returns its status and
leaves =MEMORY.md= byte-for-byte untouched.

The hook is model-aware.  A workspace whose =.claude/MEMORY.md= is a *projection*
of a HelmCortex harness =MEMORYS.md= (detected by the =LOCAL AGENT NOTES=
markers) keeps its durable memory version-tracked in the harness — agents write
there, never in the projection — so on compaction we archive and reset only the
per-machine local-notes block and leave the durable projection intact (it
regenerates from the harness on the next compile).  A legacy workspace whose
=MEMORY.md= *is* the memory (=## Session Facts= / =## User Model=) keeps the
original dump-whole-then-truncate behaviour.

WORKSPACE=$(basename "$PWD")
MEMFILE="$PWD/.claude/MEMORY.md"
[ -f "$MEMFILE" ] || { echo 'No MEMORY.md, nothing to save.'; exit 0; }

if grep -q '<!-- BEGIN LOCAL AGENT NOTES -->' "$MEMFILE"; then
  # Harness-projected workspace: durable memory is version-tracked in the
  # HelmCortex harness MEMORYS.md; only the per-machine local-notes block is
  # scratch worth archiving. Reset just that block, leave the durable projection.
  CONTENT=$(awk '/<!-- BEGIN LOCAL AGENT NOTES -->/{f=1;next} /<!-- END LOCAL AGENT NOTES -->/{f=0} f' "$MEMFILE" | grep -v '^[[:space:]]*<!--' | sed '/^[[:space:]]*$/d')
  if [ -n "$CONTENT" ]; then
    DUMP=$(printf '## Memory Dump %s (%s local scratch)\n\n%s' "$(date +%Y-%m-%d)" "$WORKSPACE" "$CONTENT")
    if flatpak run md.obsidian.Obsidian vault=HelmCortex append path="LOGS/TODO/$WORKSPACE.md" content="$DUMP" 2>/dev/null; then
      awk '/<!-- BEGIN LOCAL AGENT NOTES -->/{print; print "<!-- Per-machine scratch only. Promote durable notes into the harness MEMORYS.md and commit. -->"; f=1; next} /<!-- END LOCAL AGENT NOTES -->/{f=0} !f' "$MEMFILE" > "$MEMFILE.tmp" && mv "$MEMFILE.tmp" "$MEMFILE"

      status=$?
      echo "Memory archive failed for $WORKSPACE; local scratch preserved." >&2
      exit "$status"
    fi
  else
    echo "No local scratch; durable memory is in the harness MEMORYS.md."
  fi
else
  # Legacy workspace: MEMORY.md itself IS the memory (## Session Facts / ## User Model).
  CONTENT=$(tail -n +2 "$MEMFILE" 2>/dev/null | sed '/^$/d' | sed '/^## Session Facts$/d' | sed '/^## User Model.*$/d')
  if [ -n "$CONTENT" ]; then
    DUMP=$(printf '## Memory Dump %s\n\n%s' "$(date +%Y-%m-%d)" "$CONTENT")
    if flatpak run md.obsidian.Obsidian vault=HelmCortex append path="LOGS/TODO/$WORKSPACE.md" content="$DUMP" 2>/dev/null; then
      printf '# Session Memory (auto-updated by Opus, max 2000 chars)\n\n## Session Facts\n\n\n## User Model (max 1500 chars for this section)\n\n' > "$MEMFILE"
      echo "Memory saved to $WORKSPACE TODO via Obsidian CLI and MEMORY.md truncated."
    else
      status=$?
      echo "Memory archive failed for $WORKSPACE; MEMORY.md preserved." >&2
      exit "$status"
    fi
  else
    echo 'MEMORY.md empty, nothing to save.'
  fi
fi
