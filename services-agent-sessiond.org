* Agent Session Daemon
Phase-2 telemetry foundation for agent sessions.

=mux-session= remains the transport/attach/create launcher. =agent-session=
is the human-facing CLI after entering or creating a tmux session.
=agent-sessiond= is the authoritative daemon and writes state under
=HelmCortex/ROOTS/DotRoot/agent-session/=. LOGS are notes/TODO storage, not a session
source. The =HelmCortex/ROOTS/AgentRoot/session-index.json= (written by FORGE/bin/agent-session)
is the cross-agent session truth; agent-sessiond reads it for summary counts only.

** thread-triage
Phase-0 Thread-Ledger evidence packet extractor. This is a read-only offline
archaeology lane over normalized =.MASTER= markdown exports; it never writes
under HelmCortex.

TERMINAL_SIGNAL_RE = re.compile(r"\b(commit|handoff|exit|done|shipped)\b", re.I)
TURN_HEADER_RE = re.compile(r"^##(?: \[(\d+)\])? ([^\n]+?)\s*$", re.M)

    "AGENT", "SYSTEM",
}
FRONTMATTER_KEYS = {

def warn(msg):
    print(f"thread-triage: WARN: {msg}", file=sys.stderr)


def parse_args(argv):

    ap.add_argument("--month")
    ap.add_argument("--root", default=str(pathlib.Path.home() / "HelmCortex"))
    ap.add_argument(

        default=str(pathlib.Path.home() / "HelmCortex/ROOTS/AgentRoot/thread-ledger.json"),

def read_head_bytes(path, max_bytes=HEAD_SCAN_MAX_BYTES):
    try:
        with open(path, "rb") as fh:
            return fh.read(max_bytes)
    except Exception as exc:
        warn(f"{path}: head read failed: {exc}")
        return b""


def read_tail_bytes(path, max_bytes=TAIL_SCAN_MAX_BYTES):
    try:
        size = path.stat().st_size
        with open(path, "rb") as fh:
            fh.seek(max(0, size - max_bytes))
            data = fh.read()
        if size > max_bytes:
            cut = data.find(b"\n")
            if cut >= 0:
                data = data[cut + 1:]
        return data
    except Exception as exc:
        warn(f"{path}: tail read failed: {exc}")
        return b""


def decode(data):
    return data.decode("utf-8", errors="replace")


def iso_from_mtime(path):
    try:
        return (datetime.fromtimestamp(path.stat().st_mtime, tz=timezone.utc)
                .isoformat(timespec="seconds").replace("+00:00", "Z"))
    except Exception:
        return None


def trunc(text, n=TURN_TRUNCATE_CHARS):
    if text is None:
        return None
    text = re.sub(r"[ \t]+\n", "\n", text).strip()
    return text if len(text) <= n else text[:n - 1] + "…"


def machine_from_path(path):
    parts = path.parts
    if ".MASTER" in parts:
        idx = parts.index(".MASTER")
        if idx + 1 < len(parts):
            return parts[idx + 1]
    return None


def harness_from_path(path):
    parts = path.parts
    for raw, canonical in HARNESS_DIRS.items():
        if raw in parts:
            return canonical
    return None


def parse_frontmatter(head_text, path):
    warnings = []
    fm = {}
    if not head_text.startswith("---\n"):
        warnings.append("frontmatter missing from bounded head scan")
        return fm, warnings
    end = head_text.find("\n---", 4)
    if end < 0:
        warnings.append("frontmatter terminator missing from bounded head scan")
        return fm, warnings
    body = head_text[4:end]
    current_key = None
    for line in body.splitlines():
        if not line.strip() or line.startswith("  - "):
            continue
        if line.startswith((" ", "\t")) and current_key:
            continue
        key, sep, val = line.partition(":")
        if not sep:
            continue
        key = key.strip()
        if key not in FRONTMATTER_KEYS:
            current_key = key
            continue
        val = val.strip()
        if ((val.startswith('"') and val.endswith('"')) or
                (val.startswith("'") and val.endswith("'"))):
            val = val[1:-1]
        fm[key] = val if val != "" else None
        current_key = key
    if "session_id" not in fm:
        warnings.append("session_id missing from frontmatter")
    if not (fm.get("cwd") or fm.get("session_cwd") or fm.get("workspace_path")):
        warnings.append("cwd missing from frontmatter")
    return fm, warnings


def parse_turns(tail_text):
    raw_matches = list(TURN_HEADER_RE.finditer(tail_text))
    matches = []
    for m in raw_matches:
        probe = tail_text[m.end():].lstrip().splitlines()
        first = probe[0].strip() if probe else ""
        if first.startswith("<time ") or re.fullmatch(r"\*[^*]+\*", first):
            matches.append(m)
    turns = []
    for idx, m in enumerate(matches):
        start = m.end()
        end = matches[idx + 1].start() if idx + 1 < len(matches) else len(tail_text)
        label = m.group(2).strip()
        body = tail_text[start:end].strip()
        lines = body.splitlines()
        if lines and (re.fullmatch(r"\*[^*]+\*", lines[0].strip()) or
                      lines[0].strip().startswith("<time ")):
            body = "\n".join(lines[1:]).strip()
        role = "user" if label.upper() in ("USER", "METSATRON", "MÈTSÀTRON") else "agent"
        turns.append({
            "n": int(m.group(1) or (len(turns) + 1)),
            "label": label,
            "role": role,
            "text": body,
        })
    return turns


def abrupt_heuristics(turns, paths_by_cwd, path, date_text):
    last = turns[-1] if turns else None
    last_agent = next((t for t in reversed(turns) if t["role"] == "agent"), None)
    last_agent_text = last_agent["text"].rstrip() if last_agent else ""
    ends_mid_tool = False
    if last_agent_text:
        ends_mid_tool = (
            last_agent_text.endswith(("**Bash**", "**Tool**", "**Skill**", "```")) or
            last_agent_text.count("```") % 2 == 1
        )
    ends_mid_list = bool(last_agent_text and MID_LIST_RE.search(last_agent_text))
    successor = None
    cwd_key = str(path.parent)
    siblings = paths_by_cwd.get(cwd_key, [])
    try:
        mtime = path.stat().st_mtime
        for other in siblings:
            if other == path:
                continue
            omt = other.stat().st_mtime
            if 0 < omt - mtime <= 48 * 3600:
                successor = str(other)
                break
    except Exception:
        pass
    return {
        "last_turn_role": last["role"] if last else None,
        "last_turn_is_agent": bool(last and last["role"] == "agent"),
        "last_agent_ends_mid_tool_call": ends_mid_tool,
        "last_agent_ends_mid_list": ends_mid_list,
        "successor_same_cwd_within_48h": bool(successor),
        "successor_path": successor,
        "source_date": date_text,
    }


def bounded_hash(path, head_bytes, tail_bytes):
    h = hashlib.sha256()
    try:
        size = path.stat().st_size
    except Exception:
        size = None
    h.update(b"thread-triage.bounded.v1\0")
    h.update(str(size).encode())
    h.update(b"\0")
    h.update(head_bytes)
    h.update(b"\0")
    h.update(tail_bytes)
    return h.hexdigest()


def ledger_skip_keys(path):
    keys = set()
    if not path.exists():
        return keys
    try:
        payload = json.loads(path.read_text(encoding="utf-8"))
    except Exception as exc:
        warn(f"ledger read failed; treating as empty skip-list: {exc}")
        return keys
    entries = payload.get("sessions") if isinstance(payload, dict) else payload
    if not isinstance(entries, list):
        warn("ledger has no list/sessions array; treating as empty skip-list")
        return keys
    for item in entries:
        if not isinstance(item, dict):
            continue
        verdict = item.get("verdict") or item.get("status") or item.get("human_verdict")
        if not verdict or verdict == "unknown":
            continue
        h = item.get("hash") or item.get("source_hash") or item.get("bounded_hash")
        mt = item.get("mtime") or item.get("source_mtime")
        if h and mt:
            keys.add((str(h), str(mt)))
    return keys


def discover(root, month):
    logs = root / "LOGS"
    for raw in HARNESS_DIRS:
        base = logs / raw
        if not base.exists():
            warn(f"missing export corpus: {base}")
            continue
        for path in base.rglob("*.md"):
            s = path.as_posix()
            if "/.MASTER/" not in s:
                continue
            if "/.CHUNKED/" in s or "/.ROLLED/" in s:
                continue
            if month and f"/{month}/" not in s:
                continue
            yield path


def packet_for(path, paths_by_cwd, skip_keys):
    warnings = []
    head_bytes = read_head_bytes(path)
    tail_bytes = read_tail_bytes(path)
    head_text = decode(head_bytes)
    tail_text = decode(tail_bytes)
    fm, fm_warnings = parse_frontmatter(head_text, path)
    warnings.extend(fm_warnings)
    turns = parse_turns(tail_text)
    if not turns:
        warnings.append("no turn headers found in bounded tail scan")
    users = [t for t in turns if t["role"] == "user"]
    agents = [t for t in turns if t["role"] == "agent"]
    last_user = users[-1]["text"] if users else ""
    stat = None
    try:
        stat = path.stat()
    except Exception as exc:
        warnings.append(f"stat failed: {exc}")
    content_hash = bounded_hash(path, head_bytes, tail_bytes)
    mtime = str(stat.st_mtime) if stat else ""
    skipped = (content_hash, mtime) in skip_keys
    machine = fm.get("machine") or fm.get("host") or machine_from_path(path)
    cwd = fm.get("cwd") or fm.get("session_cwd") or fm.get("workspace_path")
    git_branch = fm.get("git_branch") or fm.get("branch")
    packet = {
        "schema": "thread-triage.packet.v1",
        "harness": harness_from_path(path),
        "model": fm.get("model"),
        "session_id": fm.get("session_id") or path.stem,
        "date": fm.get("datetime") or fm.get("date") or fm.get("source_utc") or iso_from_mtime(path),
        "cwd": cwd,
        "machine": machine,
        "git_branch": git_branch,
        "source_path": str(path),
        "source_mtime": mtime or None,
        "source_size_bytes": stat.st_size if stat else None,
        "source_hash": content_hash,
        "source_hash_scheme": "sha256(head64k+tail64k+size)",
        "skip_list": {
            "ledger_match": skipped,
            "matched_by": "source_hash+source_mtime" if skipped else None,
        },
        "last_user_turns": [
            {"n": t["n"], "label": t["label"], "text": trunc(t["text"])}
            for t in users[-3:]
        ],
        "last_agent_turn": (
            {"n": agents[-1]["n"], "label": agents[-1]["label"], "text": trunc(agents[-1]["text"])}
            if agents else None
        ),
        "terminal_signal": {
            "regex": r"\b(commit|handoff|exit|done|shipped)\b",
            "matched": bool(TERMINAL_SIGNAL_RE.search(last_user or "")),
            "matches": sorted({m.group(1).lower() for m in TERMINAL_SIGNAL_RE.finditer(last_user or "")}),
        },
        "abrupt_end": abrupt_heuristics(turns, paths_by_cwd, path, fm.get("date")),
        "warnings": warnings,
    }
    return packet


def main(argv):
    ns = parse_args(argv)
    root = pathlib.Path(ns.root).expanduser()
    ledger = pathlib.Path(ns.ledger).expanduser()
    skip_keys = ledger_skip_keys(ledger)
    paths = sorted(discover(root, ns.month))
    if ns.limit:
        paths = paths[: max(0, ns.limit)]
    paths_by_cwd = {}
    for p in paths:
        paths_by_cwd.setdefault(str(p.parent), []).append(p)
    for path in paths:
        try:
            packet = packet_for(path, paths_by_cwd, skip_keys)
            if packet["skip_list"]["ledger_match"] and not ns.include_skipped:
                continue
            json.dump(packet, sys.stdout, ensure_ascii=False, sort_keys=True)
            sys.stdout.write("\n")
        except Exception as exc:
            warn(f"{path}: packet extraction failed: {exc}")


if __name__ == "__main__":
    main(sys.argv[1:])

# ── /proc metrics (Linux only; all fields guarded) ────────────────────────────
def proc_metrics(pid_str):
    m = {"ppid": None, "pgid": None, "sid": None, "proc_status": None,
         "cwd": None, "cpu_ticks": None, "rss_kb": None,
         "io_read_bytes": None, "io_write_bytes": None}
    try:
        m["cwd"] = os.readlink(f"/proc/{pid_str}/cwd")
    except OSError:
        pass
    try:
        raw   = pathlib.Path(f"/proc/{pid_str}/stat").read_text()
        after = raw[raw.rfind(")") + 2:]
        f     = after.split()
        m["proc_status"] = f[0]
        m["ppid"]        = int(f[1])
        m["pgid"]        = int(f[2])
        m["sid"]         = int(f[3])
        m["cpu_ticks"]   = int(f[11]) + int(f[12])
        m["rss_kb"]      = int(f[21]) * PAGE_SIZE_KB
    except Exception:
        pass
    try:
        for line in pathlib.Path(f"/proc/{pid_str}/io").read_text().splitlines():
            k, _, v = line.partition(":")
            k = k.strip()
            if k == "read_bytes":
                m["io_read_bytes"] = int(v.strip())
            elif k == "write_bytes":
                m["io_write_bytes"] = int(v.strip())
    except Exception:
        pass
    return m


# ── session ID extraction from cmdline ────────────────────────────────────────
_RESUME_RE       = re.compile(r"(?:--resume|-s)[= ]([A-Za-z0-9/_.:@-]{8,})")
_CODEX_RESUME_RE = re.compile(r"\bresume\s+([A-Za-z0-9/_.:@-]{8,})")

def session_id_from_cmdline(cmdline, agent):
    m = _RESUME_RE.search(cmdline)
    if m:
        return m.group(1), "cmdline_flag"
    if agent == "codex":
        m = _CODEX_RESUME_RE.search(cmdline)
        if m:
            return m.group(1), "cmdline_subcommand"
    return None, None

def _fd_entries(pid_str):
    """Yield readlink targets for /proc/<pid>/fd, silently skipping permission errors."""
    try:
        for entry in pathlib.Path(f"/proc/{pid_str}/fd").iterdir():
            try:
                yield os.readlink(str(entry))
            except Exception:
                pass
    except Exception:
        pass

def session_id_probe(pid_str, agent):
    """Try non-cmdline session ID extraction. Returns (sid, source, fd_path, fd_kind)."""
    # claude-family: read ~/.claude/sessions/<pid>.json written by claude at startup
    if agent in ("claude", "auryn", "auryn-chan", "helmastra"):
        session_file = _CLAUDE_SESSIONS_DIR / f"{pid_str}.json"
        try:
            data = json.loads(session_file.read_text())
            sid = data.get("sessionId")
            if sid:
                return sid, "session_file", str(session_file), "claude"
        except Exception:
            pass
    # codex: scan /proc/<pid>/fd for open rollout session jsonl
    if agent == "codex":
        for target in _fd_entries(pid_str):
            m = _CODEX_SESSION_FD_RE.search(target)
            if m:
                return m.group(1), "fd_probe", target, "codex"
    # opencode: find open log file, scan tail for most-recent session request path
    if agent == "opencode":
        log_candidates = []
        for target in _fd_entries(pid_str):
            if _OPENCODE_LOG_FD_RE.search(target):
                try:
                    log_candidates.append((os.path.getsize(target), target))
                except OSError:
                    pass
        if log_candidates:
            _, log_path = max(log_candidates)
            try:
                with open(log_path, errors="replace") as fh:
                    content = fh.read()
                matches = list(_OPENCODE_SESSION_RE.finditer(content))
                if matches:
                    return matches[-1].group(1), "fd_probe", log_path, "opencode"
            except Exception:
                pass
    return None, None, None, None


# ── PPID ancestry helpers for tmux linkage ────────────────────────────────────
def ppid_read(pid_str):
    """Return parent PID string from /proc/<pid>/stat, or None."""
    try:
        raw   = pathlib.Path(f"/proc/{pid_str}/stat").read_text()
        after = raw[raw.rfind(")") + 2:]
        return after.split()[1]
    except Exception:
        return None

def ppid_chain_walk(pid_str, pane_pid_set, max_depth=32):
    """Walk parent chain; return (matched_pane_pid_str, chain) or (None, chain)."""
    chain = []
    cur   = ppid_read(pid_str)
    for _ in range(max_depth):
        if cur is None or cur in ("0", "1"):
            break
        chain.append(cur)
        if cur in pane_pid_set:
            return cur, chain
        cur = ppid_read(cur)
    return None, chain


# ── model context window table (static, source always labeled "model_table") ─
CONTEXT_WINDOW_SIZES = {
    "claude-sonnet-5-5": 1000000,  # native 1M (Claude Code 2.1.285 model catalog)
    "claude-opus-4-7":   200000, "claude-opus-4-5":   200000,
    "claude-sonnet-4-6": 200000, "claude-sonnet-4-5": 200000,
    "claude-haiku-4-5":  200000,
}

def _ctx_window(model):
    if not model:
        return None
    if model in CONTEXT_WINDOW_SIZES:
        return CONTEXT_WINDOW_SIZES[model]
    for k, v in CONTEXT_WINDOW_SIZES.items():
        if model.startswith(k):
            return v
    return None

# ── claude model/context probe (read project JSONL) ──────────────────────────
_CLAUDE_PROJECTS_DIR = home / ".claude/projects"

def _claude_model_probe(pid_str, session_id):
    """Return model/usage dict from the most recent assistant JSONL entry."""
    try:
        sf   = _CLAUDE_SESSIONS_DIR / f"{pid_str}.json"
        data = json.loads(sf.read_text())
        cwd  = data.get("cwd") or ""
        if not cwd:
            return {}
        encoded = cwd.replace("/", "-")
        jf = _CLAUDE_PROJECTS_DIR / encoded / f"{session_id}.jsonl"
        if not jf.exists():
            for d in _CLAUDE_PROJECTS_DIR.iterdir():
                c = d / f"{session_id}.jsonl"
                if c.exists():
                    jf = c
                    break
            else:
                return {}
        with open(jf, "rb") as fh:
            fh.seek(0, 2)
            size = fh.tell()
            fh.seek(max(0, size - 65536))
            tail = fh.read().decode(errors="replace")
        model = None
        usage = {}
        for line in tail.splitlines():
            line = line.strip()
            if not line:
                continue
            try:
                entry = json.loads(line)
                msg   = entry.get("message", {})
                if msg.get("role") == "assistant" and msg.get("model") and msg.get("usage"):
                    model = msg["model"]
                    usage = msg["usage"]
            except Exception:
                pass
        if not model:
            return {}
        inp = ((usage.get("input_tokens") or 0) +
               (usage.get("cache_read_input_tokens") or 0) +
               (usage.get("cache_creation_input_tokens") or 0))
        win = _ctx_window(model)
        pct = round(inp / win * 100, 1) if (win and inp) else None
        mts = jf.stat().st_mtime
        return {
            "provider":              "anthropic",
            "provider_source":       "claude_jsonl",
            "model":                 model,
            "model_source":          "claude_jsonl",
            "model_confidence":      "high",
            "context_size":          inp or None,
            "context_size_source":   "claude_jsonl",
            "context_window":        win,
            "context_window_source": "model_table" if win else "unknown",
            "context_percent":       pct,
            "metadata_ts":           mts,
        }
    except Exception:
        return {}


# ── codex model probe (read logs_2.sqlite) ───────────────────────────────────
# NOTE: logs_2.sqlite can be multi-GB / tens of millions of rows. A per-PID
# query (WHERE process_uuid LIKE 'pid:<pid>:%') forces SQLite to walk the
# ts index across the whole table looking for LIKE matches, which can take
# minutes per PID with several live codex processes. Instead we run ONE
# bounded query per collection cycle (ts >= now - horizon, LIMIT N), cache
# the most-recent match per pid in a module-level dict, and turn
# _codex_model_probe() into a plain dict lookup. The ts lower bound lets
# the descending index scan terminate at the horizon regardless of how
# many rows match feedback_log_body LIKE '%model=%'.
_CODEX_LOG_DB       = home / ".codex/logs_2.sqlite"
_CODEX_MODEL_RE     = re.compile(r'\bmodel=([A-Za-z0-9._:/-]+)')
_CODEX_PID_RE       = re.compile(r'^pid:(\d+):')
_CODEX_PROBE_HORIZON_S = 48 * 3600
_CODEX_PROBE_LIMIT     = 2000
_codex_model_by_pid    = None  # populated on first probe of this cycle

def _codex_model_probe_scan():
    """One bounded pass over logs_2.sqlite, cached per collection cycle.

    Returns {pid_str: extra_dict} built from the most-recent (ts DESC)
    matching row per pid within the horizon window."""
    result = {}
    if not _CODEX_LOG_DB.exists():
        return result
    try:
        conn = sqlite3.connect(f"file:{_CODEX_LOG_DB}?mode=ro", uri=True, timeout=5)
        try:
            conn.execute("PRAGMA busy_timeout=2000")
            horizon = time.time() - _CODEX_PROBE_HORIZON_S
            rows = conn.execute(
                "SELECT process_uuid, feedback_log_body, ts FROM logs "
                "WHERE ts >= ? AND feedback_log_body LIKE '%model=%' "
                "ORDER BY ts DESC, ts_nanos DESC LIMIT ?",
                (horizon, _CODEX_PROBE_LIMIT)
            ).fetchall()
        finally:
            conn.close()
        for (process_uuid, body, ts) in rows:
            if not body or not process_uuid:
                continue
            pm = _CODEX_PID_RE.match(process_uuid)
            if not pm:
                continue
            pid_str = pm.group(1)
            if pid_str in result:
                continue  # already have the most-recent (rows are ts DESC)
            mm = _CODEX_MODEL_RE.search(body)
            if mm:
                result[pid_str] = {
                    "provider":              "openai",
                    "provider_source":       "codex_log",
                    "model":                 mm.group(1),
                    "model_source":          "codex_log",
                    "model_confidence":      "high",
                    "context_size":          None,
                    "context_size_source":   "unknown",
                    "context_window":        None,
                    "context_window_source": "unknown",
                    "context_percent":       None,
                    "metadata_ts":           float(ts) if ts else None,
                }
    except Exception:
        return {}
    return result

def _codex_model_probe(pid_str):
    """Extract most-recent model for this PID from Codex logs_2.sqlite.

    Backed by a single bounded scan per collection cycle (see
    _codex_model_probe_scan) instead of a per-PID query."""
    global _codex_model_by_pid
    if _codex_model_by_pid is None:
        _codex_model_by_pid = _codex_model_probe_scan()
    return _codex_model_by_pid.get(str(pid_str), {})

HEAD_SCAN_MAX_BYTES = 65536      # bounded head read -- never full-file parse
TAIL_SCAN_MAX_BYTES = 65536      # bounded tail read
EXACT_COUNT_MAX_BYTES = 300_000  # below this, count lines exactly; above, estimate


def iso_from_epoch(ts):
    try:
        return (datetime.fromtimestamp(float(ts), tz=timezone.utc)
                 .isoformat(timespec="seconds").replace("+00:00", "Z"))
    except Exception:
        return None


def iso_from_mtime(path):
    try:
        return iso_from_epoch(path.stat().st_mtime)
    except Exception:
        return None


def read_head(path, max_bytes=HEAD_SCAN_MAX_BYTES):
    """Read a bounded prefix; never seeks past max_bytes."""
    try:
        with open(path, "rb") as fh:
            data = fh.read(max_bytes)
        return data.decode(errors="replace")
    except Exception as exc:
        warnings_global.append(f"{path.name}: head read failed: {exc}")
        return ""


def read_tail(path, max_bytes=TAIL_SCAN_MAX_BYTES):
    """Read a bounded suffix; drops a possibly-truncated first partial line."""
    try:
        size = path.stat().st_size
        with open(path, "rb") as fh:
            fh.seek(max(0, size - max_bytes))
            data = fh.read()
        text = data.decode(errors="replace")
        lines = text.split("\n")
        if size > max_bytes and lines:
            lines = lines[1:]
        return lines
    except Exception as exc:
        warnings_global.append(f"{path.name}: tail read failed: {exc}")
        return []


def iter_json_lines(lines):
    """Yield parsed dicts, skipping unparseable or empty lines silently."""
    for ln in lines:
        ln = ln.strip()
        if not ln:
            continue
        try:
            obj = json.loads(ln)
        except Exception:
            continue
        if isinstance(obj, dict):
            yield obj


def estimate_message_count(path, tail_lines):
    """Exact line count for small files; tail-density estimate for large ones."""
    try:
        size = path.stat().st_size
    except Exception:
        return None, "unknown"
    if size <= EXACT_COUNT_MAX_BYTES:
        try:
            with open(path, "rb") as fh:
                count = sum(1 for _ in fh)
            return count, "exact"
        except Exception:
            return None, "unknown"
    sample = [l for l in tail_lines if l.strip()]
    if not sample:
        return None, "unknown"
    tail_bytes = sum(len(l.encode(errors="replace")) + 1 for l in sample)
    if tail_bytes <= 0:
        return None, "unknown"
    avg_len = tail_bytes / len(sample)
    if avg_len <= 0:
        return None, "unknown"
    return round(size / avg_len), "estimated_from_tail_sample"


def truncate(s, n=140):
    if not s:
        return s
    s = " ".join(s.split())
    return s if len(s) <= n else s[: n - 1] + "…"


# -- claude sample -------------------------------------------------------------
def sample_claude(path):
    warnings = []
    session_id = path.stem

    cwd = None
    title = None
    git_branch = None
    for obj in iter_json_lines(read_head(path).splitlines()):
        if cwd is None and isinstance(obj.get("cwd"), str):
            cwd = obj["cwd"]
        if git_branch is None and isinstance(obj.get("gitBranch"), str):
            git_branch = obj["gitBranch"]
        if title is None and obj.get("type") == "user":
            msg = obj.get("message") or {}
            content = msg.get("content")
            if isinstance(content, str):
                title = content
            elif isinstance(content, list):
                for block in content:
                    if isinstance(block, dict) and block.get("type") == "text":
                        title = block.get("text")
                        break
        if cwd and title and git_branch:
            break
    if cwd is None:
        warnings.append("no cwd found in bounded head scan")
    if title is None:
        warnings.append("no first user message found in bounded head scan")

    tail_lines = read_tail(path)
    model = None
    updated_at = None
    for obj in reversed(list(iter_json_lines(tail_lines))):
        if updated_at is None and isinstance(obj.get("timestamp"), str):
            updated_at = obj["timestamp"]
        if git_branch is None and isinstance(obj.get("gitBranch"), str):
            git_branch = obj["gitBranch"]
        if model is None:
            msg = obj.get("message") or {}
            if msg.get("role") == "assistant" and msg.get("model") and msg.get("usage"):
                model = msg["model"]
        if model and updated_at:
            break
    if model is None:
        warnings.append("no assistant model found in bounded tail scan")
    if updated_at is None:
        updated_at = iso_from_mtime(path)
        warnings.append("no timestamp in tail scan; fell back to file mtime")

    msg_count, msg_count_source = estimate_message_count(path, tail_lines)

    return {
        "agent": "claude",
        "session_id": session_id,
        "cwd": cwd,
        "model": model,
        "git_branch": git_branch,
        "title": truncate(title),
        "updated_at": updated_at,
        "message_count": msg_count,
        "message_count_source": msg_count_source,
        "resume_cmd": f"claude --resume {session_id}",
        "source_path": str(path),
        "size_bytes": path.stat().st_size if path.exists() else None,
        "warnings": warnings,
    }

def newest_files(root, pattern, n, exclude_substr=None):
    if not root.exists():
        return []
    files = []
    for f in root.rglob(pattern):
        if exclude_substr and exclude_substr in f.as_posix():
            continue
        try:
            files.append((f.stat().st_mtime, f))
        except OSError:
            continue
    files.sort(key=lambda t: t[0], reverse=True)
    return [f for _, f in files[:n]]

DotCortex state helpers:

  exec "$hc_bin" "$@"

** Collector schedule (cron)

The daemon is snapshot-based: nothing keeps =state.<host>.json= fresh unless something runs =agent-sessiond once= on a schedule. The 2026-07-06→08-19 gap — six weeks of frozen state files read as "daemon dead" — was exactly this missing piece; the daemon was fine, it was simply never scheduled. This apply script installs two user-crontab lines per machine (marker-managed, idempotent, all other crontab lines preserved): a =once= snapshot every 10 minutes (sub-second run), and an hourly =agent-session scan= index refresh guarded on the HelmCortex FORGE index tool actually existing on that machine, at a per-host minute derived from the hostname so the fleet's scanners never meet on the shared index (2026-08-21: the :07-everywhere schedule plus a fixed temp path corrupted =session-index.json=; the index tool now locks and stages through unique temp files as well). Termux (S24) has no cron by default — the script fails loud there rather than pretending; schedule via =crond= from the cronie package or an SSH-driven collector from a hub machine.

#+BEGIN_SRC bash :tangle all/.local/bin/agent-sessiond-cron-apply :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

TAG="agent-sessiond-cron-apply"
MARK="# dotcortex:agent-sessiond"

usage() {
  cat <<'EOF'
Usage: agent-sessiond-cron-apply [--remove] [--dry-run]

Install (or refresh) the user-crontab collector schedule for agent-sessiond:
  - agent-sessiond once   every 10 minutes (state snapshot, sub-second)
  - agent-session scan    hourly, only when the HelmCortex FORGE index tool
                          exists on this machine

Idempotent: managed lines carry the marker "# dotcortex:agent-sessiond" and
are replaced wholesale on every run. All other crontab lines are preserved.

Options:
  --remove     remove the managed lines instead of installing
  --dry-run    print the resulting crontab without installing it
  -h, --help   show this help
EOF
}

MODE=install
DRY=0
for arg in "$@"; do
  case "$arg" in
    -h|--help|help) usage; exit 0 ;;
    --remove) MODE=remove ;;
    --dry-run) DRY=1 ;;
    *) echo "$TAG: unknown argument: $arg" >&2; usage >&2; exit 2 ;;
  esac
done

if ! command -v crontab >/dev/null 2>&1; then
  echo "$TAG: no crontab binary on this host (Termux needs cronie + a running crond) — nothing installed" >&2
  exit 1
fi

mkdir -p "$HOME/.local/state/agent-sessiond"

existing="$(crontab -l 2>/dev/null | grep -vF "$MARK" || true)"

# Post-cutover every machine scans the one canonical index over NFS. A fixed
# minute (:07 everywhere) made the writers collide every hour; the index tool
# now locks, but staggering per host keeps the lock uncontended. Deterministic
# from the hostname so reruns are idempotent.
scan_minute=$(( ($(printf '%s' "$(hostname)" | cksum | cut -d' ' -f1) % 50) + 5 ))

managed=""
if [ "$MODE" = install ]; then
  managed="$(printf '%s\n%s' \
    "*/10 * * * * \$HOME/.local/bin/agent-sessiond once >/dev/null 2>>\$HOME/.local/state/agent-sessiond/cron-errors.log $MARK" \
    "$scan_minute * * * * [ -x \$HOME/HelmCortex/FORGE/bin/agent-session ] && \$HOME/HelmCortex/FORGE/bin/agent-session scan >/dev/null 2>>\$HOME/.local/state/agent-sessiond/cron-errors.log $MARK")"
fi

result="$(printf '%s\n%s\n' "$existing" "$managed" | sed '/^[[:space:]]*$/d')"

if [ "$DRY" = 1 ]; then
  printf '%s\n' "$result"
  exit 0
fi

printf '%s\n' "$result" | crontab -
if [ "$MODE" = install ]; then
  echo "$TAG: collector schedule installed (sessiond once @10min; index scan hourly when FORGE tool present)"
else
  echo "$TAG: managed collector lines removed"
fi
#+END_SRC

.RECIPEPREFIX := |
.PHONY: sessions-once sessions-watch sessions-status sessions-top sessions-top-all sessions-cron

AGENT_SESSIOND ?= $(firstword $(wildcard $(HOME)/.local/bin/agent-sessiond $(HOME)/DotCortex/all/.local/bin/agent-sessiond))
AGENT_SESSION ?= $(firstword $(wildcard $(HOME)/.local/bin/agent-session $(HOME)/DotCortex/all/.local/bin/agent-session))
AGENT_TOP ?= $(firstword $(wildcard $(HOME)/.local/bin/agent-top $(HOME)/DotCortex/all/.local/bin/agent-top))
AGENT_SESSIOND_CRON ?= $(firstword $(wildcard $(HOME)/.local/bin/agent-sessiond-cron-apply $(HOME)/DotCortex/all/.local/bin/agent-sessiond-cron-apply))

sessions-once:
| @chmod +x $(AGENT_SESSIOND) $(AGENT_SESSION) 2>/dev/null || true
| $(AGENT_SESSIOND) once

sessions-watch:
| @chmod +x $(AGENT_SESSIOND) $(AGENT_SESSION) 2>/dev/null || true
| $(AGENT_SESSIOND) watch

sessions-status:
| @chmod +x $(AGENT_SESSIOND) $(AGENT_SESSION) 2>/dev/null || true
| $(AGENT_SESSIOND) status

sessions-top:
| @chmod +x $(AGENT_TOP) 2>/dev/null || true
| $(AGENT_TOP) once

| @chmod +x $(AGENT_TOP) 2>/dev/null || true
| $(AGENT_TOP) once --all

sessions-cron:
| @chmod +x $(AGENT_SESSIOND_CRON) 2>/dev/null || true
| $(AGENT_SESSIOND_CRON)
