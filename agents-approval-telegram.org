* Installation boundary

The existing paired Telegram plugin remains the sole Bot API poller. Its private
Unix socket renders a held item from the private queue. An allowlisted callback
passes the Admiral sender ID, item ID and random nonce to the exact verdict
helper. The helper rechecks identity, seat, held status, deadline and one-use
answer under the queue lock. It never executes a command. Native approvals and
all accepted messages also stamp presence via the same runtime authority.

Enable only after staging the complete series against the pinned fleet 0.0.7
baseline. =approval-telegram-stage= copies it, checks the source hash, dry-runs
and applies the one patch with zero fuzz, adds the bridge and compiles the whole
candidate. An unsupported source fails before any live change. Existing fleet
patches are already in this exact baseline; preserve them. No installer or live
restart is included. Provision the helper paths and manifest in the private bot
launcher environment and review the complete staged candidate before deploying.

The plugin environment needs SO_APPROVAL_KIKIN_MANIFEST, APPROVAL_VERDICT_HELPER,
APPROVAL_PRESENCE_HELPER and, optionally, APPROVAL_TELEGRAM_HOLD_SOCKET. The hook
launcher uses the same socket/state dir and manifest. No token is passed to the
hook or helpers. Public environment templates contain blank values.

// Runs inside the EXISTING Telegram plugin, which owns its token and poller.
// The gate speaks over a private Unix socket; never starts getUpdates itself.
import { chmodSync } from 'fs'
import { join, isAbsolute } from 'path'

export async function runApprovalHelper(value: Record<string, unknown>, presence = false) {
  const helper = process.env[presence ? 'APPROVAL_PRESENCE_HELPER' : 'APPROVAL_VERDICT_HELPER']
  if (!helper || !isAbsolute(helper) || helper.includes('YOUR_')) return null
  const proc = Bun.spawn([helper], { stdin: new Blob([JSON.stringify(value)]), stdout: 'pipe', stderr: 'ignore' })
  const timer = setTimeout(() => proc.kill(), 2000)
  try {
    const output = await new Response(proc.stdout).text()
    return (await proc.exited) === 0 ? output : null
  } finally { clearTimeout(timer) }
}

export async function approvalPresence(sender: string) {
  await runApprovalHelper({ sender }, true)
}

export async function handleApprovalCallback(ctx: any, data: string, access: {allowFrom: string[]}) {
  const match = /^gate:([ad]):([0-9a-f]{24}):([0-9a-f]{32})$/.exec(data)
  if (!match) return false
  const sender = String(ctx.from?.id ?? '')
  if (!access.allowFrom.includes(sender)) {
    await ctx.answerCallbackQuery({text:'Not authorized.'}).catch(() => {})
    return true
  }
  const [, answer, id, nonce] = match
  const result = await runApprovalHelper({id, nonce, sender, behavior:answer === 'a' ? 'allow' : 'deny'})
  const label = result === null ? 'Expired, already answered, or unauthorized.' : answer === 'a' ? 'Allowed once.' : 'Denied.'
  await ctx.answerCallbackQuery({text:label}).catch(() => {})
  if (result !== null) await ctx.editMessageReplyMarkup({reply_markup:{inline_keyboard:[]}}).catch(() => {})
  return true
}

export function startApprovalHolds(bot: any, loadAccess: () => {allowFrom:string[]}, stateDir: string) {
  // Missing helper leaves the ordinary channel untouched and the gate's send
  // fails closed immediately. A reviewed installation enables this seam.
  if (!process.env.APPROVAL_VERDICT_HELPER) return
  const path = process.env.APPROVAL_TELEGRAM_HOLD_SOCKET ?? join(stateDir, 'approval-telegram-hold.sock')
  if (!isAbsolute(path)) throw new Error('approval hold socket must be absolute')
  const server = Bun.serve({ unix:path, maxRequestBodySize:4096, async fetch(request) {
    if (request.method !== 'POST' || new URL(request.url).pathname !== '/hold') return new Response('', {status:404})
    try {
      const value = await request.json() as {id:string; nonce:string}
      if (Object.keys(value).sort().join(',') !== 'id,nonce' || !/^[0-9a-f]{24}$/.test(value.id) || !/^[0-9a-f]{32}$/.test(value.nonce)) return new Response('', {status:400})
      // Preview comes from the held queue record, never from caller-supplied prose.
      const raw = await runApprovalHelper({...value, action:'preview'})
      if (raw === null) return new Response('', {status:409})
      const details = JSON.parse(raw)
      const sender = details.admiral
      if (!loadAccess().allowFrom.includes(sender)) return new Response('', {status:403})
      const preview = JSON.stringify(details.request.tool_input, null, 2)
      const text = `Held approval ${value.id}\n${details.request.tool_name}: ${preview}\n\nTimeout: ${details.timeout_seconds}s; then deny and morning queue.`
      // Full preview is mandatory. Never truncate into an uninformed allow.
      if (text.length > 3900) return new Response('', {status:413})
      void bot.api.sendMessage(sender, text, { reply_markup:{inline_keyboard:[[
        {text:'Allow once', callback_data:`gate:a:${value.id}:${value.nonce}`},
        {text:'Deny', callback_data:`gate:d:${value.id}:${value.nonce}`},
      ]]}}).catch(() => runApprovalHelper({...value, action:'delivery_failed'}))
      // Accepted by the paired plugin; delivery completes asynchronously. A
      // failed Bot API send is recorded in the same queue and releases the hold.
      return new Response('sent')
    } catch { return new Response('', {status:503}) }
  }})
  chmodSync(path, 0o600)
  process.once('exit', () => server.stop(true))
}

"""Ask the existing credential-owning plugin to display a held queue request."""
import json
import os
from pathlib import Path
import re
import socket
import sys


def main():
    try:
        value = json.loads(sys.stdin.read(1024*1024+1))
        if value.get('schema') != 'dotcortex.approval.hold.v1' or not re.fullmatch('[0-9a-f]{24}',value.get('id','')) or not re.fullmatch('[0-9a-f]{32}',value.get('nonce','')):
            return 1
        state = Path(os.environ.get('TELEGRAM_STATE_DIR') or Path.home()/'.claude/channels/telegram')
        path = os.environ.get('APPROVAL_TELEGRAM_HOLD_SOCKET') or str(state/'approval-telegram-hold.sock')
        body = json.dumps({'id':value['id'],'nonce':value['nonce']}).encode()
        with socket.socket(socket.AF_UNIX,socket.SOCK_STREAM) as channel:
            channel.settimeout(0.65)
            channel.connect(path)
            channel.sendall(b'POST /hold HTTP/1.1\r\nHost: local\r\nContent-Type: application/json\r\nConnection: close\r\nContent-Length: '+str(len(body)).encode()+b'\r\n\r\n'+body)
            response = channel.recv(4096)
        return 0 if response.startswith(b'HTTP/1.1 200 ') else 1
    except Exception:
        return 1


if __name__ == '__main__':
    raise SystemExit(main())

"""Preflight the complete new Telegram seam against the pinned fleet baseline.
Only creates a candidate directory; never installs, restarts or edits upstream.
"""
import argparse
import hashlib
import json
from pathlib import Path
import shutil
import subprocess

PINNED_SHA = 'd902d7195a527b579b889cfdbf251fc24fd224a68e319ac6b7c3b691515c3458'


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--upstream', type=Path, required=True)
    parser.add_argument('--destination', type=Path, required=True)
    args = parser.parse_args()
    upstream = args.upstream.resolve(strict=True)
    if args.destination.exists() or args.destination.is_symlink(): parser.error('destination must be new')
    # Complete pinned runtime is staged before applying ANY patch.
    body = (upstream/'server.ts').read_bytes()
    if hashlib.sha256(body).hexdigest() != PINNED_SHA: parser.error('unsupported upstream hash; live runtime untouched')
    version = json.loads((upstream/'.claude-plugin/plugin.json').read_text()).get('version')
    if version != '0.0.7': parser.error('unsupported plugin version')
    here = Path(__file__).resolve().parents[1]/'share/dotcortex/approval-telegram'
    shutil.copytree(upstream,args.destination,symlinks=True)
    patch = here/'holds.patch'
    subprocess.run(['patch','--batch','--fuzz=0','--dry-run','-p1','-i',str(patch)],cwd=args.destination,check=True)
    subprocess.run(['patch','--batch','--fuzz=0','-p1','-i',str(patch)],cwd=args.destination,check=True)
    shutil.copy2(here/'approval-holds.ts',args.destination/'approval-holds.ts')
    final=(args.destination/'server.ts').read_text()
    for seam in ['startApprovalHolds(bot, loadAccess, STATE_DIR)', 'await handleApprovalCallback(ctx, data, loadAccess())', 'await approvalPresence(String(from.id))']:
        if final.count(seam)!=1: raise ValueError('invalid generated seam')
    # No package install. Existing lock and dependencies must already be present.
    subprocess.run(['bun','build',str(args.destination/'server.ts'),'--target=bun','--outdir',str(args.destination/'approval-preflight')],cwd=args.destination,check=True)
    print('Review-ready candidate: '+str(args.destination))
    return 0


if __name__ == '__main__':
    raise SystemExit(main())
