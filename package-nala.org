* Brave stable canary gate (T480s)
The native Brave package is deliberately kept in the Nala manifest, but a Chromium
regression must not be allowed to turn a desktop update into a broken browser.  The
T480s policy is a bounded hold, not a permanent security pin: =known_good= is the
last GUI-verified version, =rejected= records only the failed stable candidate, and
the next strictly newer stable candidate automatically re-enters the isolated GUI
canary.  The gate verifies Brave's signed =InRelease=, the Packages-file digest,
and the package SHA-256 before extracting it.  Canary profiles are always temporary
and never use the real Brave profile.

#+BEGIN_SRC text :tangle debian/.nala/manifest/brave-policy.ssv :padline no :comments none
# KEY VALUE — T480s native Brave policy; the rejected version is advanced by the gate.
known_good 1.92.143
rejected 1.95.101
#+END_SRC

#+BEGIN_SRC bash :tangle debian/.local/bin/brave-canary-gate :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755) :comments no
set -euo pipefail

POLICY="${BRAVE_POLICY:-$HOME/.nala/manifest/brave-policy.ssv}"
STATE="${BRAVE_STATE:-${XDG_STATE_HOME:-$HOME/.local/state}/dotcortex/brave-canary.state}"
TEST="${BRAVE_GATE_TEST:-0}"
mkdir -p "$(dirname "$STATE")"

read_policy() {
  local key value
  while read -r key value _; do
    case "$key" in
      known_good) KNOWN_GOOD="$value" ;;
      rejected) REJECTED="$value" ;;
    esac
  done < "$POLICY"
}
read_state() {
  [ -f "$STATE" ] || return 0
  local key value
  while read -r key value _; do
    case "$key" in
      known_good) KNOWN_GOOD="$value" ;;
      rejected) REJECTED="$value" ;;
    esac
  done < "$STATE"
}
write_state() {
  local d="$(dirname "$STATE")" tmp
  tmp="$(mktemp "$d/.brave-state.XXXXXX")"
  printf 'known_good %s\nrejected %s\n' "$KNOWN_GOOD" "$REJECTED" > "$tmp"
  chmod 600 "$tmp"
  mv -f "$tmp" "$STATE"
}
log_action() { [ "$TEST" = 1 ] && printf '%s\n' "$*" >> "${BRAVE_GATE_TEST_ACTIONS:?}" || true; }

read_policy
read_state
candidate="${BRAVE_GATE_TEST_CANDIDATE:-$(apt-cache policy brave-browser | awk '/Candidate:/{print $2; exit}')}"
installed="${BRAVE_GATE_TEST_INSTALLED:-$(dpkg-query -W -f='${Version}\n' brave-browser 2>/dev/null || true)}"
[ -n "$candidate" ] || { echo 'brave-canary-gate: no apt candidate' >&2; exit 1; }

# Test mode exercises the state machine without network, GUI, package, or profile effects.
if [ "$TEST" = 1 ]; then
  if dpkg --compare-versions "$candidate" le "$KNOWN_GOOD"; then
    echo gate=baseline; exit 0
  fi
  if [ "$candidate" = "$REJECTED" ]; then
    log_action hold; echo gate=known-bad-rejected; exit 0
  fi
  if [ "${BRAVE_GATE_TEST_CANARY:-fail}" = pass ]; then
    KNOWN_GOOD="$candidate"; REJECTED=""; log_action "approve $candidate"; write_state
    echo gate=approved; exit 0
  fi
  REJECTED="$candidate"; log_action "rollback $KNOWN_GOOD"; log_action hold; write_state
  echo gate=rollback-hold; exit 0
fi

# Existing rejected releases stay held.  A newer stable release is the only event
# that clears this bounded hold and demands a new canary.
if [ "$candidate" = "$REJECTED" ] || dpkg --compare-versions "$candidate" le "$KNOWN_GOOD"; then
  sudo apt-mark hold brave-browser >/dev/null
  exit 0
fi

repo=https://brave-browser-apt-release.s3.brave.com
td="$(mktemp -d "${TMPDIR:-/tmp}/brave-gate.XXXXXX")"
trap 'find "$td" -depth -delete 2>/dev/null || true' EXIT
curl -fsSL "$repo/dists/stable/InRelease" -o "$td/InRelease"
gpgv --keyring /usr/share/keyrings/brave-browser-release-keyring.gpg "$td/InRelease" >/dev/null
curl -fsSL "$repo/dists/stable/main/binary-amd64/Packages" -o "$td/Packages"
packages_sha="$(awk '$0 == "SHA256:" {ok=1; next} ok && $3 == "main/binary-amd64/Packages" {print $1; exit}' "$td/InRelease")"
printf '%s  %s\n' "$packages_sha" "$td/Packages" | sha256sum -c - >/dev/null
stanza="$(awk -v RS='' -v ver="$candidate" '$1 == "Package:" && $2 == "brave-browser" && $0 ~ "Version: " ver {print; exit}' "$td/Packages")"
asset="$(awk '$1 == "Filename:" {print $2}' <<<"$stanza")"
digest="$(awk '$1 == "SHA256:" {print $2}' <<<"$stanza")"
[ -n "$asset" ] && [ -n "$digest" ] || { echo "brave-canary-gate: signed candidate $candidate absent" >&2; exit 1; }
deb="$td/$(basename "$asset")"
curl -fL "$repo/$asset" -o "$deb"
printf '%s  %s\n' "$digest" "$deb" | sha256sum -c - >/dev/null
dpkg-deb -x "$deb" "$td/root"
bin="$td/root/opt/brave.com/brave/brave-browser"
test -x "$bin"

# Import only non-secret desktop variables from the live XFCE session.
session_pid="$(pgrep -u "$(id -u)" -x xfce4-session | head -1)"
while IFS= read -r kv; do
  case "$kv" in DISPLAY=*|XAUTHORITY=*|DBUS_SESSION_BUS_ADDRESS=*|XDG_RUNTIME_DIR=*|XDG_DATA_DIRS=*|XDG_CURRENT_DESKTOP=*|DESKTOP_SESSION=*|HOME=*|PATH=*|LANG=*|LC_*|UBUNTU_MENUPROXY=*) export "$kv";; esac
done < <(tr '\0' '\n' < "/proc/$session_pid/environ")
unset GTK_MODULES
log="$td/canary.log"
"$bin" --no-sandbox --user-data-dir="$td/profile" --no-first-run --no-default-browser-check >/dev/null 2>"$log" & canary_pid=$!
sleep 8
if kill -0 "$canary_pid" 2>/dev/null && DISPLAY="${DISPLAY:-:0}" xdotool search --class brave >/dev/null 2>&1; then
  kill "$canary_pid" 2>/dev/null || true
  wait "$canary_pid" 2>/dev/null || true
  sudo apt-mark unhold brave-browser >/dev/null || true
  sudo DEBIAN_FRONTEND=noninteractive apt-get install -y --allow-downgrades "$deb" >/dev/null
  KNOWN_GOOD="$candidate"; REJECTED=""; write_state
else
  kill "$canary_pid" 2>/dev/null || true
  wait "$canary_pid" 2>/dev/null || true
  fallback="/var/cache/apt/archives/brave-browser_${KNOWN_GOOD}_amd64.deb"
  [ -r "$fallback" ] || { echo "brave-canary-gate: no recoverable $KNOWN_GOOD fallback" >&2; exit 1; }
  sudo DEBIAN_FRONTEND=noninteractive apt-get install -y --allow-downgrades "$fallback" >/dev/null
  REJECTED="$candidate"; write_state
  sudo apt-mark hold brave-browser >/dev/null
  echo "brave-canary-gate: rejected $candidate; held at $KNOWN_GOOD" >&2
fi
#+END_SRC

#+BEGIN_SRC bash :tangle debian/.local/share/dotcortex/tests/brave-canary-gate.sh :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755) :comments no
set -euo pipefail

ROOT="$(cd "$(dirname "$0")/../../../../.." && pwd)"
GATE="$ROOT/debian/.local/bin/brave-canary-gate"
tmp="$(mktemp -d)"
trap 'find "$tmp" -depth -delete 2>/dev/null || true' EXIT
policy="$tmp/policy"; state="$tmp/state"; actions="$tmp/actions"; profile="$tmp/real-profile"
printf '%s\n' 'known_good 1.92.143' 'rejected 1.95.101' > "$policy"
mkdir "$profile"; printf '%s\n' untouched > "$profile/sentinel"; before="$(sha256sum "$profile/sentinel")"

run_gate() {
  : > "$actions"
  BRAVE_GATE_TEST=1 BRAVE_POLICY="$policy" BRAVE_STATE="$state" BRAVE_GATE_TEST_ACTIONS="$actions" \
    BRAVE_GATE_TEST_CANDIDATE="$1" BRAVE_GATE_TEST_CANARY="${2:-fail}" BRAVE_GATE_TEST_INSTALLED=1.92.143 \
    "$GATE" >/dev/null
}

run_gate 1.95.101 pass
grep -qx hold "$actions"
run_gate 1.96.0 fail
grep -qx 'rollback 1.92.143' "$actions"
grep -qx hold "$actions"
grep -qx 'rejected 1.96.0' <(awk '$1 == "rejected" {print $1, $2}' "$state")
run_gate 1.96.0 pass
if grep -qx 'approve 1.96.0' "$actions"; then
  echo 'known-bad candidate was re-canaried' >&2
  exit 1
fi
run_gate 1.97.0 pass
grep -qx 'approve 1.97.0' "$actions"
grep -qx 'known_good 1.97.0' <(awk '$1 == "known_good" {print $1, $2}' "$state")
[ "$(sha256sum "$profile/sentinel")" = "$before" ]
grep -q '\[ "$pkg" = brave-browser \] && continue' "$ROOT/debian/.local/bin/nala-apply"
printf '%s\n' 'brave canary gate fixtures: ok'
#+END_SRC

* GitHub release-backed .deb manifest
For Debian-family apps distributed as GitHub release artifacts instead of apt repos.
Columns: REPO, PKG, ASSET_NAME_OR_GLOB, ARCH, SCOPE, EXTRA
- PKG is the Debian package name from the .deb metadata, not the asset filename.
- ARCH: =amd64=, =arm64=, =all=, or =any=
- SCOPE: =shared=, =devuan-only=, =ubuntu-only=, =vendefoul-only=, =chatgpt-supported=

The same manifest accepts =direct://URL= for a one-off HTTPS asset, but the
release helper rejects every non-HTTPS direct URL.  OpenAI's official lane uses
=signed-deb://BASE= instead: the helper resolves the current =chatgpt= package
from BASE's signed =dists/stable/InRelease= and SHA-256-checked =Packages=
metadata, then verifies the .deb against that metadata before installation.
The pinned repository key is embedded in the helper, so key retrieval is not a
runtime trust decision.  The support predicate is deliberately narrow: Debian
13 and Ubuntu 24.04/26.04 only; Devuan is not treated as Debian for this lane.

* RustDesk unattended-access autostart (x230)
The native .deb ships a =rustdesk.service= systemd unit (=ExecStart=/usr/bin/rustdesk --service=,
=User=root=), but RustDesk's own binary self-disables that unit
(=systemctl disable rustdesk; systemctl stop rustdesk=) whenever it expects a desktop-session
instance to take over instead — this is by design, not a bug to fight. The systemd unit is a
fallback for a machine with no logged-in user; once a real session exists, RustDesk wants to run
in that session's tray instead.

it needs =rustdesk --tray= autostarted in the desktop session. Without this entry nothing was ever
listening, which is why the systemd unit kept disabling itself with no replacement — confirmed via
=journalctl -u rustdesk= showing the same self-disable pattern going back to Feb 2026.

scope_matches() {
  local scope="$1"
  case "$scope" in
    all) return 0 ;;
    devuan) [ "$IS_DEVUAN" -eq 1 ] ;;
    ubuntu) [ "$OS_ID" = "ubuntu" ] ;;
    vendefoul) [ "$OS_ID" = "vendefoul" ] ;;
    *) return 0 ;;
  esac

CHANGED=0

parse_repo_line() {
  local line="$1"
  local -a parts=()

  mapfile -t parts < <(printf '%s\n' "$line" | xargs -n1 printf '%s\n')
  while [ "${#parts[@]}" -lt 6 ]; do
    parts+=("")
  done
  printf '%s|%s|%s|%s|%s|%s\n' \
    "${parts[0]}" "${parts[1]}" "${parts[2]}" \
    "${parts[3]}" "${parts[4]}" "${parts[5]}"
}

while IFS='|' read -r name type key_url repo_spec os_match _extra; do
  [ -z "$name" ] && continue

  if ! scope_matches "$os_match"; then
    echo "  skip $name (os=$os_match, this=$OS_ID)"
    continue
  fi

  repo_spec="${repo_spec//\{\{ARCH\}\}/$ARCH}"

  case "$type" in
    key+sources)
      # key_url is GPG key, repo_spec is URL to .sources file
      keyring="/usr/share/keyrings/${name}-keyring.gpg"
      sources_file="/etc/apt/sources.list.d/${name}.sources"

      if [ -f "$sources_file" ]; then
        echo "  ok $name (already configured)"
        continue
      fi

      echo "  add $name repo..."
      sudo install -d -m 755 "$(dirname "$keyring")" "$(dirname "$sources_file")"

      # Download and install key
      if [ -n "$key_url" ]; then
        if echo "$key_url" | grep -q '\.gpg$'; then
          sudo wget -qO "$keyring" "$key_url"
        else
          wget -qO- "$key_url" | gpg --dearmor | sudo tee "$keyring" >/dev/null
        fi
        sudo chmod go+r "$keyring"
      fi

      # Download and install sources file
      sudo wget -qO "$sources_file" "$repo_spec"
      CHANGED=1
      echo "  done $name"
      ;;

    ppa)
      # Ubuntu PPA — repo_spec is the PPA string
      if [ "$OS_ID" != "ubuntu" ]; then
        echo "  skip $name (PPA, not Ubuntu)"
        continue
      fi
      if grep -rq "$repo_spec" /etc/apt/sources.list.d/ 2>/dev/null; then
        echo "  ok $name (already configured)"
        continue
      fi
      echo "  add $name PPA..."
      sudo add-apt-repository -y "ppa:$repo_spec"
      CHANGED=1
      ;;

    key+list)
      keyring="${_extra:-}"
      [ -n "$keyring" ] || keyring="/usr/share/keyrings/${name}-keyring.gpg"
      list_file="/etc/apt/sources.list.d/${name}.list"

      if [ -f "$list_file" ] && grep -Fqx "$repo_spec" "$list_file" 2>/dev/null; then
        echo "  ok $name (already configured)"
        continue
      fi

      echo "  add $name repo..."
      sudo install -d -m 755 "$(dirname "$keyring")" "$(dirname "$list_file")"

      if [ -n "$key_url" ]; then
        if echo "$key_url" | grep -q '\.gpg$'; then
          sudo wget -qO "$keyring" "$key_url"
        else
          wget -qO- "$key_url" | gpg --dearmor --yes | sudo tee "$keyring" >/dev/null
        fi
        sudo chmod go+r "$keyring"
      fi

      printf '%s\n' "$repo_spec" | sudo tee "$list_file" >/dev/null
      CHANGED=1
      ;;

    *)
      echo "  unknown repo type: $type for $name" >&2
      ;;
  esac
done < <(
  while IFS= read -r raw; do
    line="${raw#${raw%%[![:space:]]*}}"
    line="${line%${line##*[![:space:]]}}"
    [ -z "$line" ] && continue
    case "$line" in
      \#*) continue ;;
    esac
    parse_repo_line "$line"
  done < "$MAN"
)

if [ "$CHANGED" -eq 1 ]; then
  echo "Repos changed, updating package lists..."
  APT="apt-get"
  command -v nala &>/dev/null && APT="nala"
  sudo $APT update
fi

install_local_deb() {
  local apt_bin="$1" bundle="$2"
  local pkg_name pkg_version installed_status installed_version
  if ! pkg_name="$(dpkg-deb --field "$bundle" Package 2>/dev/null)" || [ -z "$pkg_name" ]; then
    echo "  ! cannot extract package name from $(basename "$bundle")" >&2
    return 1
  fi
  if ! pkg_version="$(dpkg-deb --field "$bundle" Version 2>/dev/null)" || [ -z "$pkg_version" ]; then
    echo "  ! cannot extract package version from $(basename "$bundle")" >&2
    return 1
  fi
  package_is_exactly_installed() {
    installed_status=""
    installed_version=""
    IFS=$'\t' read -r installed_status installed_version < <(dpkg-query -W -f='${Status}\t${Version}\n' "$pkg_name" 2>/dev/null) || return 1
    [ "$installed_status" = "install ok installed" ] && [ "$installed_version" = "$pkg_version" ]
  }
  echo "  -> installing $(basename "$bundle") with $apt_bin"
  if ! sudo DEBIAN_FRONTEND=noninteractive "$apt_bin" install -y "$bundle"; then
    echo "  -> $apt_bin install failed; retrying with dpkg + apt-get -f" >&2
  fi
  # Nala sometimes returns 0 but silently skips — verify the package is installed
  if package_is_exactly_installed; then
    return 0
  fi
  echo "  -> $apt_bin did not install $pkg_name, retrying with dpkg + apt-get -f"
  if ! sudo DEBIAN_FRONTEND=noninteractive dpkg -i "$bundle"; then
    echo "  -> dpkg dependency setup incomplete; running apt-get -f" >&2
  fi
  sudo DEBIAN_FRONTEND=noninteractive apt-get install -f -y
  if package_is_exactly_installed; then
    return 0
  fi
  echo "  ! package $pkg_name is not installed at bundle version $pkg_version after dependency repair" >&2
  return 1
}

  if ! bundle_pkg="$(dpkg-deb -f "$bundle" Package 2>/dev/null)" || [ -z "$bundle_pkg" ]; then
    echo "  ! cannot extract package name from $bundle" >&2
    return 1
  fi
  if ! bundle_ver="$(dpkg-deb -f "$bundle" Version 2>/dev/null)" || [ -z "$bundle_ver" ]; then
    echo "  ! cannot extract package version from $bundle" >&2
    return 1
  fi
  effective_pkg="$pkg"
  compare_ver="$bundle_ver"

  if [ -n "$bundle_pkg" ] && [ "$bundle_pkg" != "$pkg" ]; then
    echo "  ! manifest package $pkg differs from deb metadata $bundle_pkg; using deb metadata"
    effective_pkg="$bundle_pkg"
  fi

  installed="$(installed_version "$effective_pkg")"
  echo "  release: pkg=$effective_pkg arch=$want_arch latest=$compare_ver installed=${installed:-none}"
  if [ "$installed" = "$compare_ver" ]; then
    echo "  -> already current, skipping install for $effective_pkg"
    return 0
  fi

  install_local_deb "$apt_bin" "$bundle"
}

nala_hostname() {
  printf '%s\n' "${NALA_HOSTNAME:-$(hostname -s 2>/dev/null || hostname)}"

# Hostnames the provisioning registries know: column 1 of host-excludes.ssv and
# host-gates.ssv, repo copy and stowed copy (env overrides win, as in their helpers).
known_host_scope() {
  local root="${DOTCORTEX_ROOT:-$HOME/DotCortex}" f
  for f in "${HOST_EXCLUDES:-}" "$HOME/.provision/host-excludes.ssv" "$root/all/.provision/host-excludes.ssv" \
           "${PROVISION_GATES:-}" "$HOME/.provision/host-gates.ssv" "$root/all/.provision/host-gates.ssv"; do
    [ -n "$f" ] && [ -f "$f" ] || continue
    awk -v h="$1" '/^[ \t]*#/ || NF==0 { next } $1==h { found=1; exit } END { exit(found?0:1) }' "$f" && return 0
  done
  return 1
}

scope_matches() {
  local scope="$1"
  case "$scope" in
    shared) return 0 ;;
    devuan-only) [ "$IS_DEVUAN" -eq 1 ] ;;
    ubuntu-only) [ "$OS_ID" = "ubuntu" ] ;;
    vendefoul-only) [ "$OS_ID" = "vendefoul" ] ;;
    *)
      # Host scope: a SCOPE that names a host the provisioning registries already
      # know (host-excludes.ssv, host-gates.ssv; keyed by hostname -s) applies on
      # that host only: kikin-kushi, beelink. A truly unknown scope keeps the old
      # behaviour (installs everywhere) but says so on stderr.
      if known_host_scope "$scope"; then
        [ "$scope" = "$(nala_hostname)" ]
        return
      fi
      echo "nala-apply: warning: unknown SCOPE '$scope' treated as shared (see package-nala.org)" >&2
      return 0
      ;;
  esac
}

# --- Gated hosts: additive-only, dry-run by default, never removes ---
manifest_rows() {
  awk -v FS='[ \t]+' '
    function dq(s){ return (s == "\"\"" ? "" : s) }
    /^[ \t]*#/ || NF==0 { next }
    { printf "%s|%s|%s|%s|%s\n", dq($1), dq($2), dq($3), dq($4), dq($5) }
  ' "$MAN"
}

gated_apply() {
  local mode="$1" pkg _ver _cat scope _extra sim
  local satisfied=() install=() missing=() held=()
  echo "nala-apply: gated host (mode: $mode): additive only; nothing is ever removed here"
  echo "  manifest: $MAN"
  while IFS='|' read -r pkg _ver _cat scope _extra; do
    [ -z "$pkg" ] && continue
    scope_matches "$scope" || continue
    if [ -x "$HOST_EXCLUDES_BIN" ] && "$HOST_EXCLUDES_BIN" nala "$pkg"; then
      continue
    fi
    if [ "$(dpkg-query -W -f='${db:Status-Abbrev}' "$pkg" 2>/dev/null)" = "ii " ]; then
      satisfied+=("$pkg")
    elif apt-cache show "$pkg" >/dev/null 2>&1; then
      install+=("$pkg")
    else
      missing+=("$pkg")
    fi
  done < <(manifest_rows)

  if [ -x "$HOST_EXCLUDES_BIN" ]; then
    while IFS= read -r pkg; do
      [ -n "$pkg" ] && dpkg -s "$pkg" >/dev/null 2>&1 && held+=("$pkg")
    done < <("$HOST_EXCLUDES_BIN" nala)
  fi

  echo "  satisfied (${#satisfied[@]}): ${satisfied[*]:-}"
  echo "  to install (${#install[@]}): ${install[*]:-}"
  [ "${#missing[@]}" -eq 0 ] || echo "  not in apt repos (${#missing[@]}): ${missing[*]}"
  [ "${#held[@]}" -eq 0 ] || echo "  host-excluded but installed (${#held[@]}): ${held[*]} -- never removed on a gated host"
  echo "  skipped on gated host: nala bootstrap, nala upgrade, removals, Brave canary gate, release-backed .deb sync"

  if [ "${#install[@]}" -eq 0 ]; then
    echo "Nothing to install."
    return 0
  fi

  # With --no-remove apt aborts if the transaction would remove anything; the
  # Remv/Purg scan is a second, independent refusal.
  if ! sim="$(apt-get -s --no-remove -o APT::Get::AutomaticRemove=false install "${install[@]}" 2>&1)"; then
    echo "REFUSED: apt-get simulation failed; a gated host installs only transactions that remove nothing" >&2
    printf '%s\n' "$sim" | tail -20 >&2
    return 3
  fi
  if printf '%s\n' "$sim" | grep -qE '^(Remv|Purg) '; then
    echo "REFUSED: the install transaction would remove packages on a gated host:" >&2
    printf '%s\n' "$sim" | grep -E '^(Remv|Purg) ' >&2
    return 3
  fi
  echo "  simulation: $(printf '%s\n' "$sim" | grep -c '^Inst ') package(s) would be installed, 0 removed"

  if [ "$mode" != additive ]; then
    echo "dry-run (default on gated hosts): nothing installed. PROVISION_APPLY=1 installs the ${#install[@]} package(s) above; removals are never run here."
    return 0
  fi
  sudo DEBIAN_FRONTEND=noninteractive apt-get install -y --no-remove -o APT::Get::AutomaticRemove=false "${install[@]}"
  echo "Done (additive)."
}

      REMOVE_LIST+=("$excl_pkg")
    fi
  done < <("$HOST_EXCLUDES_BIN" nala)

  if [ "${#REMOVE_LIST[@]}" -gt 0 ]; then
    echo "Removing ${#REMOVE_LIST[@]} host-excluded package(s)..."
    printf "  %s\n" "${REMOVE_LIST[@]}"
    sudo $APT remove -y "${REMOVE_LIST[@]}" || {
      echo "$APT remove failed, falling back to apt-get remove..."
      sudo apt-get remove -y "${REMOVE_LIST[@]}"
    }
  fi
fi

if [ -x "$REL_SYNC_HELPER" ]; then
  "$REL_SYNC_HELPER"

* Release-backed .deb diff
#+BEGIN_SRC bash :tangle debian/.local/bin/nala-release-diff :shebang "#!/usr/bin/env bash"
set -euo pipefail

MAN="${NALA_RELEASES_SSV:-$HOME/.nala/manifest/releases.ssv}"

[ -f "$MAN" ] || exit 0

OS_ID="unknown"
if [ -f "${NALA_OS_RELEASE:-/etc/os-release}" ]; then
  # shellcheck disable=SC1091
  . "${NALA_OS_RELEASE:-/etc/os-release}"
  OS_ID="$ID"
fi
IS_DEVUAN=0
[ -f "${NALA_DEVUAN_VERSION_FILE:-/etc/devuan_version}" ] && IS_DEVUAN=1
ARCH="${NALA_ARCH:-$(dpkg --print-architecture)}"

chatgpt_supported_platform() {
  [ "$IS_DEVUAN" -eq 0 ] || return 1
  case "$OS_ID:${VERSION_ID:-}" in
    debian:13|ubuntu:24.04|ubuntu:26.04) return 0 ;;
    *) return 1 ;;
  esac
}

scope_matches() {
  local scope="$1"
  case "$scope" in
    shared|"") return 0 ;;
    unsupported-devuan) return 1 ;;
    chatgpt-supported) chatgpt_supported_platform ;;
    devuan-only) [ "$IS_DEVUAN" -eq 1 ] ;;
    ubuntu-only) [ "$OS_ID" = "ubuntu" ] ;;
    vendefoul-only) [ "$OS_ID" = "vendefoul" ] ;;
    *) return 0 ;;
  esac
}

arch_matches() {
  local want="$1"
  case "$want" in
    any|all|"") return 0 ;;
    "$ARCH") return 0 ;;
    *) return 1 ;;
  esac
}

gh_latest_json() {
  local repo="$1" url="https://api.github.com/repos/${repo}/releases/latest"
  if [ -n "${GH_TOKEN:-}" ]; then
    curl -fsSL -H "Authorization: Bearer $GH_TOKEN" "$url"
  else
    curl -fsSL "$url"
  fi
}

normalize_ver() {
  sed -E 's/^v//' <<<"$1"
}

installed_version() {
  local pkg="$1"
  dpkg-query -W -f='${Version}\n' "$pkg" 2>/dev/null || true
}

if grep -q '^signed-deb://' "$MAN"; then
  SYNC_HELPER="${NALA_RELEASE_SYNC_HELPER:-$HOME/.local/bin/nala-release-sync}"
  [ -x "$SYNC_HELPER" ] || SYNC_HELPER="$(dirname "$0")/nala-release-sync"
  NALA_RELEASE_DIFF_ONLY=1 NALA_RELEASES_SSV="$MAN" "$SYNC_HELPER"
  exit $?
fi

echo "[release-backed debs]"
while IFS='|' read -r repo pkg asset_glob want_arch scope _extra; do
  [ -n "$repo" ] || continue

  if ! scope_matches "$scope"; then
    echo "  . $pkg skipped (scope=$scope, this=$OS_ID)"
    continue
  fi

  if ! arch_matches "$want_arch"; then
    echo "  . $pkg skipped (arch=$want_arch, this=$ARCH)"
    continue
  fi

  if [[ "$repo" == direct://* ]]; then
    latest_ver="${_extra%%,*}"
    resolved_asset_name="$asset_glob"
  else
    json="$(gh_latest_json "$repo")"
    latest_tag="$(jq -r '.tag_name' <<<"$json")"
    latest_ver="$(normalize_ver "$latest_tag")"
    resolved_asset_name=""
    while IFS=$'\t' read -r asset_entry_name _asset_entry_url _asset_entry_digest; do
      case "$asset_entry_name" in
        $asset_glob)
          resolved_asset_name="$asset_entry_name"
          break
          ;;
      esac
    done < <(jq -r '.assets[] | [.name, .browser_download_url, (.digest // "")] | @tsv' <<<"$json")
  fi

  if [ -z "$resolved_asset_name" ]; then
    echo "  ? $pkg (no release asset matching $asset_glob in $repo)"
    continue
  fi

  have="$(installed_version "$pkg")"
  if [ -z "$have" ]; then
    echo "  + $pkg (missing -> $latest_ver, asset $resolved_asset_name)"
    continue
  fi

  if dpkg --compare-versions "$have" ge "$latest_ver"; then
    echo "  = $pkg ($have, asset $resolved_asset_name)"
  else
    echo "  ~ $pkg ($have -> $latest_ver, asset $resolved_asset_name)"
  fi
done < <(
  awk -v FS='[ \t]+' '
    function dq(s){ return (s == "\"\"" ? "" : s) }
    /^[ \t]*#/ || NF==0 { next }
    {
      printf "%s|%s|%s|%s|%s|%s\n",
        dq($1), dq($2), dq($3), dq($4), dq($5), dq($6)
    }
  ' "$MAN"
)
#+END_SRC
