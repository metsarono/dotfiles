"""File safety preflight hook for Hermes.

Intercepts write operations and enforces the Git Safety Covenant.
"""

import subprocess
import os
import json
import datetime
import re
from pathlib import Path

# Tools that write to files
WRITE_TOOLS = {"patch", "writefile", "write_file", "create_file"}

# Terminal commands that write to files
WRITE_PATTERNS = re.compile(
    r'(\bsed\s+-i|\btee\b|\becho\b.*>|\bcp\b|\bmv\b|\bdd\b|\btruncate\b|>\s*\S)',
    re.IGNORECASE
)

# Covenant message injected on every turn
GIT_COVENANT = """[SOVEREIGNTY COVENANT — fires every turn]

Before calling patch, writefile, write_file, create_file, or any terminal command
that writes to disk: run `git diff HEAD -- <file>` on the specific file.

If the file has uncommitted changes you did not make, STOP and report.
Do not proceed. This is not a suggestion. Violations are logged.
"""

def preflight(tool_name: str, args: dict, task_id: str, ctx, **kwargs):
    """Pre-tool-call hook: intercept write operations and check git status."""
    if tool_name not in WRITE_TOOLS and tool_name != "terminal":
        return
    
    # Extract file path from tool args
    file_path = None
    if tool_name in WRITE_TOOLS:
        file_path = args.get("path") or args.get("file_path") or args.get("target")
    elif tool_name == "terminal":
        cmd = args.get("command") or args.get("cmd", "")
        # Check if terminal command writes to files
        if not WRITE_PATTERNS.search(cmd):
            return
        # For terminal commands, we can't reliably extract all file paths from
        # arbitrary bash commands, so we just warn
        ctx.inject_message(
            f"[PREFLIGHT WARNING] terminal command appears to write to disk:\n"
            f"  {cmd[:300]}\n"
            f"Confirm this is intentional and files are in a clean git state.",
            role="user"
        )
        return
    
    if not file_path:
        return
    
    # Convert to absolute path
    try:
        abs_path = Path(file_path).expanduser().resolve()
        repo_dir = abs_path.parent
        
        # Check git status for this specific file
        result = subprocess.run(
            ["git", "diff", "HEAD", "--", str(abs_path)],
            cwd=str(repo_dir),
            capture_output=True,
            text=True,
            timeout=10
        )
        
        if result.stdout.strip():
            # File has uncommitted changes from prior session
            ctx.inject_message(
                f"[COVENANT VIOLATION INTERCEPTED]\n"
                f"Tool `{tool_name}` was called on `{abs_path}` which has uncommitted changes "
                f"from a prior session that you did not make.\n"
                f"STOP. Do not touch this file. Report the conflict to Mètsàtron.",
                role="user"
            )
    except subprocess.TimeoutExpired:
        # SSHFS timeout - warn but don't block
        ctx.inject_message(
            f"[PREFLIGHT WARNING] git diff timed out for `{abs_path}`. "
            f"Proceed only with explicit confirmation.",
            role="user"
        )
    except Exception:
        # Git not available, not a repo, etc. - silently allow
        pass


def postflight(tool_name: str, args: dict, result: str, task_id: str, **kwargs):
    """Post-tool-call hook: log all write operations."""
    if tool_name not in WRITE_TOOLS:
        return
    
    file_path = args.get("path") or args.get("file_path") or args.get("target")
    
    # Log to audit trail
    log = Path.home() / ".hermes" / "logs" / "file-writes.jsonl"
    log.parent.mkdir(parents=True, exist_ok=True)
    
    entry = {
        "ts": datetime.datetime.now().isoformat(),
        "tool": tool_name,
        "file": file_path,
        "task_id": task_id,
        "result": result[:500]  # Truncate long results
    }
    
    with open(log, "a") as f:
        f.write(json.dumps(entry) + "\n")


def register(ctx):
    """Register hooks with Hermes plugin system."""
    # Covenant injection - runs before every LLM call
    def inject_covenant(**kwargs):
        return {"context": GIT_COVENANT}
    
    ctx.register_hook("pre_llm_call", inject_covenant)
    ctx.register_hook("pre_tool_call", preflight)
    ctx.register_hook("post_tool_call", postflight)

"""Agent routing orchestration for Hermes.

Routes tasks to appropriate agents based on ai-usage --json output,
provider quotas, task complexity, and workspace rules.
"""

import json
import subprocess
from pathlib import Path
from typing import Dict, Any, Tuple, Optional

class AgentRouter:
    """Routes tasks to optimal AI agents based on provider capacity."""
    
    # Provider model mappings (heavy/complex tasks → heavy model, light/fast tasks → light model)
    PROVIDER_MODELS = {
        "codex": {
            "heavy": "gpt-5.4",
            "light": "gpt-5.4-mini"
        },
        "claude": {
            "heavy": "claude-sonnet-5-5",
            "light": "claude-haiku-4-5"
        },
        "opencode": {
            "heavy": "minimax-2.7",
            "light": "gemma-4"
        },
        "gemini": {
            "heavy": "gemini-1.5-pro",
            "light": "gemini-1.5-flash"
        }
    }
    
    # Task types requiring specific agents
    TASK_ROUTING = {
        "complex-coding": ["claude", "codex"],
        "fast-coding": ["opencode", "gemini"],
        "research": ["claude", "gemini"],
        "planning": ["claude"],
        "simple-edits": ["opencode", "gemini"]
    }
    
    def __init__(self):
        self.usage_data = None
        self._refresh_usage()
    
    def _refresh_usage(self):
        """Refresh ai-usage data."""
        try:
            result = subprocess.run(
                ["ai-usage", "--json"],
                capture_output=True,
                text=True,
                timeout=10
            )
            if result.returncode == 0:
                self.usage_data = json.loads(result.stdout)
            else:
                print(f"WARNING: ai-usage failed: {result.stderr}")
                self.usage_data = {}
        except Exception as e:
            print(f"WARNING: ai-usage unavailable: {e}")
            self.usage_data = {}
    
    def _get_provider_capacity(self, provider: str) -> float:
        """Calculate remaining capacity (0.0 to 1.0) for provider."""
        provider_data = self.usage_data.get(provider, {})
        if not provider_data:
            return 0.5  # Default if unknown
        
        caps = provider_data.get("caps", {})
        used = provider_data.get("usage", {})
        
        # Calculate remaining percentage across isotropic and output caps
        isotropic_cap = max(caps.get("isotropic_requests_1_day", 0), 1)
        isotropic_used = used.get("isotropic_requests_1_day", 0)
        
        output_cap = max(caps.get("output_tokens_per_day", 0), 1)
        output_used = used.get("output_tokens_per_day", 0)
        
        isotropic_pct = 1.0 - (isotropic_used / isotropic_cap)
        output_pct = 1.0 - (output_used / output_cap)
        
        # Use the most restrictive capacity
        return max(0.0, min(isotropic_pct, output_pct))
    
    def _is_over_quota(self, provider: str, threshold: float = 0.8) -> bool:
        """Check if provider is over quota threshold."""
        return self._get_provider_capacity(provider) < (1.0 - threshold)
    
    def route_task(self, task_description: str, workspace_context: Dict[str, Any]) -> Tuple[str, str]:
        """
        Route task to optimal agent based on rules, quotas, and availability.
        
        Returns: (agent_name, model_name)
        """
        self._refresh_usage()
        
        # Determine task type from description and context
        task_type = self._classify_task(task_description, workspace_context)
        
        # Get preferred providers for this task type
        preferred = self.TASK_ROUTING.get(task_type, ["claude", "codex"])
        
        # Filter by capacity
        available = []
        for provider in preferred:
            if self._is_over_quota(provider, threshold=0.8):
                # Provider over 80% quota - skip
                continue
            capacity = self._get_provider_capacity(provider)
            available.append((provider, capacity))
        
        # If all preferred providers over quota, fall back to free tier
        if not available:
            free_tier = ["opencode"]
            for provider in free_tier:
                available.append((provider, 1.0))
        
        # Sort by capacity (highest first)
        available.sort(key=lambda x: x[1], reverse=True)
        
        selected_provider = available[0][0]
        
        # Determine model weight (heavy vs light)
        model_weight = self._assess_complexity(task_description, workspace_context)
        model = self.PROVIDER_MODELS[selected_provider][model_weight]
        
        return selected_provider, model
    
    def _classify_task(self, description: str, context: Dict[str, Any]) -> str:
        """Classify task type from description and context."""
        desc_lower = description.lower()
        
        # Simple keyword-based classification
        if any(k in desc_lower for k in ["refactor", "rewrite", "complex", "architecture"]):
            return "complex-coding"
        elif any(k in desc_lower for k in ["fast", "quick", "simple", "edit", "patch"]):
            return "simple-edits"
        elif any(k in desc_lower for k in ["research", "investigate", "discover", "find"]):
            return "research"
        elif any(k in desc_lower for k in ["plan", "design", "strategy"]):
            return "planning"
        else:
            return "fast-coding"
    
    def _assess_complexity(self, description: str, context: Dict[str, Any]) -> str:
        """Assess if task needs heavy or light model."""
        desc_lower = description.lower()
        
        # Heavy model indicators
        heavy_indicators = [
            "complex", "refactor", "architecture", "design",
            "comprehensive", "deep", "thorough", "investigate"
        ]
        
        # Light model indicators
        light_indicators = [
            "quick", "fast", "simple", "patch", "edit",
            "small", "minor", "fix"
        ]
        
        heavy_count = sum(1 for k in heavy_indicators if k in desc_lower)
        light_count = sum(1 for k in light_indicators if k in desc_lower)
        
        if heavy_count > light_count:
            return "heavy"
        else:
            return "light"


def register_router(ctx):
    """Register agent router with Hermes."""
    router = AgentRouter()
    
    def route_pre_llm(**kwargs):
        """Pre-LLM hook: inject routing context."""
        user_msg = kwargs.get("user_message", "")
        context = kwargs.get("context", {})
        
        if user_msg:
            agent, model = router.route_task(user_msg, context)
            routing_info = f"[AGENT ROUTER] Task routed to: {agent} ({model})"
            return {"context": routing_info}
        
        return {}
    
    ctx.register_hook("pre_llm_call", route_pre_llm)

#+BEGIN_SRC python :tangle linux/.local/bin/hermes-usage-state :shebang "#!/usr/bin/env python3" :comments no
import argparse
import json
import subprocess
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

CACHE_PATH = Path.home() / ".hermes" / "runtime" / "usage-state.json"
DEFAULT_AI_USAGE = Path.home() / ".local" / "bin" / "ai-usage"


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="Normalize ai-usage --json output into Hermes usage state."
    )
    parser.add_argument(
        "--ai-usage",
        default=str(DEFAULT_AI_USAGE),
        help="Path to ai-usage executable (default: %(default)s)",
    )
    parser.add_argument(
        "--providers",
        default="codex,claude,copilot,cursor,antigravity,openrouter,deepinfra",
        help="Comma-separated providers to probe via ai-usage",
    )
    parser.add_argument(
        "--mode",
        choices=("auto", "normal", "conserve", "burn"),
        default="auto",
        help="Override recommended mode (default: auto)",
    )
    parser.add_argument(
        "--write-cache",
        action="store_true",
        help="Write normalized output to ~/.hermes/runtime/usage-state.json",
    )
    return parser.parse_args()


def iso_now() -> str:
    return datetime.now(timezone.utc).isoformat()


def parse_iso(ts: str | None) -> datetime | None:
    if not ts or not isinstance(ts, str):
        return None
    try:
        return datetime.fromisoformat(ts.replace("Z", "+00:00"))
    except ValueError:
        return None


def minutes_until(ts: str | None) -> int | None:
    dt = parse_iso(ts)
    if dt is None:
        return None
    delta = dt - datetime.now(timezone.utc)
    return max(0, int(delta.total_seconds() // 60))


def normalize_window(raw: dict[str, Any] | None) -> dict[str, Any] | None:
    if not isinstance(raw, dict):
        return None
    used = raw.get("usedPercent")
    if used is None:
        return None
    try:
        used_val = float(used)
    except (TypeError, ValueError):
        return None
    remaining = max(0.0, 100.0 - used_val)
    resets_at = raw.get("resetsAt")
    return {
        "used_percent": used_val,
        "remaining_percent": remaining,
        "window_minutes": raw.get("windowMinutes"),
        "resets_at": resets_at,
        "minutes_to_reset": minutes_until(resets_at),
        "reset_description": raw.get("resetDescription"),
    }


def canonical_budget_kind(provider: str, usage: dict[str, Any], item: dict[str, Any]) -> str:
    if usage.get("openRouterUsage") is not None:
        return "credit_balance"
    if any(isinstance(usage.get(key), dict) for key in ("primary", "secondary", "tertiary")):
        return "windowed_quota"
    if provider == "deepinfra":
        return "token_cost"
    return "unknown"


def normalize_item(item: dict[str, Any]) -> dict[str, Any]:
    provider = str(item.get("provider") or "unknown")
    raw = item.get("raw") if isinstance(item.get("raw"), dict) else {}
    usage = raw.get("usage") if isinstance(raw.get("usage"), dict) else {}
    openrouter_usage = usage.get("openRouterUsage") if isinstance(usage.get("openRouterUsage"), dict) else None
    provider_cost = usage.get("providerCost") if isinstance(usage.get("providerCost"), dict) else None
    windows = {
        "short": normalize_window(usage.get("primary")),
        "long": normalize_window(usage.get("secondary")),
        "extra": normalize_window(usage.get("tertiary")),
    }
    windows = {k: v for k, v in windows.items() if v is not None}
    status = str(item.get("state") or "unknown")
    error = raw.get("error") if isinstance(raw.get("error"), dict) else None
    login_method = None
    identity = usage.get("identity") if isinstance(usage.get("identity"), dict) else {}
    if isinstance(identity, dict):
        login_method = identity.get("loginMethod")
    if not login_method:
        login_method = usage.get("loginMethod")
    budget_kind = canonical_budget_kind(provider, usage, item)

    burnable = False
    conserve = False
    healthy = status == "ok"

    short = windows.get("short")
    longw = windows.get("long")
    if provider == "codex" and short:
        mins = short.get("minutes_to_reset")
        rem = short.get("remaining_percent")
        if mins is not None and rem is not None and mins <= 480 and rem >= 20:
            burnable = True
        if longw:
            mins2 = longw.get("minutes_to_reset")
            rem2 = longw.get("remaining_percent")
            if mins2 is not None and rem2 is not None and mins2 <= 720 and rem2 >= 5:
                burnable = True
    if provider in {"openrouter", "deepinfra"}:
        conserve = True
    if provider == "gemini" and short and float(short.get("used_percent", 0)) >= 100:
        healthy = False
    if error is not None:
        healthy = False

    priority_score = 0
    if healthy:
        priority_score += 20
    if burnable:
        priority_score += 30
    if conserve:
        priority_score -= 20
    if provider == "openrouter" and openrouter_usage:
        if float(openrouter_usage.get("balance", 0) or 0) <= 0:
            priority_score -= 15
    return {
        "status": "unavailable" if status == "error" else status,
        "label": item.get("label"),
        "source": item.get("source"),
        "login_method": login_method,
        "budget": {"kind": budget_kind},
        "windows": windows,
        "credits": raw.get("credits") if isinstance(raw.get("credits"), dict) else None,
        "provider_cost": provider_cost,
        "openrouter_usage": openrouter_usage,
        "burnable": burnable,
        "conserve": conserve,
        "healthy": healthy,
        "priority_score": priority_score,
        "raw_summary": item.get("summary"),
        "error": error,
    }


def recommend_mode(providers: dict[str, Any]) -> str:
    codex = providers.get("codex") or {}
    short = ((codex.get("windows") or {}).get("short") or {})
    longw = ((codex.get("windows") or {}).get("long") or {})
    if codex.get("healthy"):
        mins = short.get("minutes_to_reset")
        rem = short.get("remaining_percent")
        if mins is not None and rem is not None and mins <= 480 and rem >= 20:
            return "burn"
        mins2 = longw.get("minutes_to_reset")
        rem2 = longw.get("remaining_percent")
        if mins2 is not None and rem2 is not None and mins2 <= 720 and rem2 >= 5:
            return "burn"
    openrouter = providers.get("openrouter") or {}
    gemini = providers.get("gemini") or {}
    nvidia_available = True
    if openrouter.get("healthy") is False and gemini.get("healthy") is not False and nvidia_available:
        return "conserve"
    return "normal"


def main() -> int:
    args = parse_args()
    cmd = [args.ai_usage, "--json", "--providers", args.providers]
    proc = subprocess.run(cmd, capture_output=True, text=True)
    if proc.returncode != 0:
        raise SystemExit(proc.stderr.strip() or proc.stdout.strip() or f"ai-usage failed with {proc.returncode}")
    payload = json.loads(proc.stdout)
    if not isinstance(payload, list):
        raise SystemExit("ai-usage JSON was not a list")
    providers = {}
    for item in payload:
        if not isinstance(item, dict):
            continue
        provider = str(item.get("provider") or "unknown")
        providers[provider] = normalize_item(item)
    mode = recommend_mode(providers) if args.mode == "auto" else args.mode
    state = {
        "generated_at": iso_now(),
        "source": "ai-usage --json",
        "ttl_seconds": 300,
        "mode_recommendation": mode,
        "providers": providers,
    }
    text = json.dumps(state, indent=2, sort_keys=True)
    print(text)
    if args.write_cache:
        CACHE_PATH.parent.mkdir(parents=True, exist_ok=True)
        CACHE_PATH.write_text(text + "\n", encoding="utf-8")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

BURN_OVERRIDES: dict[str, list[dict[str, Any]]] = {
    "coding": [
        {"provider": "openai-codex", "model": "gpt-5.4", "why": "burn premium coding quota"},
        {"provider": "openai-codex", "model": "gpt-5.4-mini", "why": "burn small disposable quota"},

        {"provider": "openai-codex", "model": "gpt-5.4", "why": "burn premium research/coding quota"},
    ],
}

CONSERVE_DEMOTIONS = {"openai-codex", "openrouter", "deepinfra", "claude"}


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="Emit ordered Hermes routing candidates for a task class.")
    parser.add_argument("--task", choices=tuple(TASK_MATRIX.keys()), required=True)
    parser.add_argument("--mode", choices=("auto", "normal", "conserve", "burn"), default="auto")
    parser.add_argument("--usage-state", default=str(USAGE_STATE_PATH))
    parser.add_argument("--json", action="store_true", help="Emit machine-readable JSON")
    return parser.parse_args()


def load_usage_state(path: str) -> dict[str, Any]:
    p = Path(path)
    if not p.exists():
        return {"mode_recommendation": "normal", "providers": {}}
    try:
        return json.loads(p.read_text())
    except Exception:
        return {"mode_recommendation": "normal", "providers": {}}


def candidate_score(candidate: dict[str, Any], mode: str, usage_state: dict[str, Any]) -> tuple[int, dict[str, Any]]:
    providers = usage_state.get("providers") or {}
    pstate = providers.get(candidate["provider"], {}) if isinstance(providers, dict) else {}
    score = 0
    healthy = pstate.get("healthy", True)
    if healthy:
        score += 20
    else:
        score -= 40
    score += int(pstate.get("priority_score", 0) or 0)
    if mode == "burn" and candidate["provider"] == "openai-codex":
        score += 60
    if mode == "conserve" and candidate["provider"] in CONSERVE_DEMOTIONS:
        score -= 50
    if candidate["provider"] == "deepinfra":
        score -= 30
    detail = {
        "healthy": healthy,
        "provider_state": pstate,
    }
    return score, detail


def main() -> int:
    args = parse_args()
    usage_state = load_usage_state(args.usage_state)
    mode = usage_state.get("mode_recommendation", "normal") if args.mode == "auto" else args.mode
    candidates = list(TASK_MATRIX[args.task])
    if mode == "burn":
        candidates = BURN_OVERRIDES.get(args.task, []) + candidates
    scored = []
    seen = set()
    for candidate in candidates:
        key = (candidate["provider"], candidate["model"])
        if key in seen:
            continue
        seen.add(key)
        score, detail = candidate_score(candidate, mode, usage_state)
        scored.append({
            **candidate,
            "score": score,
            "healthy": detail["healthy"],
            "provider_state": detail["provider_state"],
        })
    scored.sort(key=lambda item: item["score"], reverse=True)
    payload = {
        "task": args.task,
        "mode": mode,
        "generated_from": args.usage_state,
        "candidates": scored,
    }
    if args.json:
        print(json.dumps(payload, indent=2))
    else:
        print(f"task={args.task} mode={mode}")
        for item in scored:
            state = "healthy" if item["healthy"] else "degraded"
            print(f"- {item['provider']} / {item['model']} [{item['score']}] ({state}) — {item['why']}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
