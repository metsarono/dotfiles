* Intent

** DotCortex symlink health scanner
Lists broken symlinks under =$HOME=. With =--fix-stow= it removes the ones that point into the DotCortex checkout: Stow orphans, left behind when a file leaves the repo or moves to another layer (Stow's unstow walks the package, not the target, so it never sees them).

Stow writes *relative* links (=.local/bin/x -> ../../DotCortex/linux/.local/bin/x=), which the first version of this scanner missed: it compared the raw link text against =$HOME/DotCortex=, so only absolute links matched (found 2026-09-28, the Honey launcher move left four relative orphans). A link's target is now resolved against the link's own directory before the comparison, both lexically and physically (a checkout reached through a symlink counts), and only a target *inside* the checkout (=<checkout>/...=, never a sibling such as =DotCortex-old=) is removed. The checkout itself is never scanned: the repo tracks deliberately dangling links of its own. Scan mode tags each orphan =(stow orphan)= so the report is the dry-run of =--fix-stow=.

=DOTCORTEX_ROOT= overrides the checkout (default =$HOME/DotCortex=), for tests and non-default checkouts.

#+BEGIN_SRC bash :tangle linux/.local/bin/dotcortex-stow-health :shebang "#!/usr/bin/env bash" :mkdirp yes
#!/usr/bin/env bash
set -euo pipefail

ROOT="${HOME}"
DOT="${DOTCORTEX_ROOT:-${HOME}/DotCortex}"
mode="scan"

case "${1-}" in
  --fix-stow) mode="fix-stow" ;;
  -h|--help)
    echo "Usage: dotcortex-stow-health [--fix-stow]"
    echo "  List broken symlinks under \$HOME; --fix-stow removes those pointing into $DOT."
    exit 0 ;;
  "") ;;
  *) echo "dotcortex-stow-health: unknown argument: $1" >&2; exit 2 ;;
esac

DOT_LEX="$(realpath -m -s -- "$DOT")"
DOT_REAL="$(realpath -m -- "$DOT")"

# into_checkout LINK TARGET: true when TARGET, resolved from LINK's directory, lies inside the checkout.
into_checkout() {
  local link="$1" target="$2" abs lex phys
  case "$target" in
    /*) abs="$target" ;;
    *) abs="$(dirname -- "$link")/$target" ;;
  esac
  lex="$(realpath -m -s -- "$abs")"
  phys="$(realpath -m -- "$abs")"
  case "$lex/" in "$DOT_LEX"/?*|"$DOT_REAL"/?*) return 0 ;; esac
  case "$phys/" in "$DOT_LEX"/?*|"$DOT_REAL"/?*) return 0 ;; esac
  return 1
}

printf 'DotCortex symlink health - mode: %s\n\n' "$mode"

broken=0 orphans=0
# Find broken symlinks under $HOME, but never inside the checkout, and ignore Wine and similar
while IFS= read -r -d '' link; do
  target="$(readlink "$link" 2>/dev/null || printf '?')"
  broken=$((broken + 1))
  if into_checkout "$link" "$target"; then
    orphans=$((orphans + 1))
    printf 'BROKEN: %s -> %s (stow orphan)\n' "$link" "$target"
    if [[ "$mode" == "fix-stow" ]]; then
      echo "  removing stow symlink: $link"
      rm -v -- "$link"
    fi
  else
    printf 'BROKEN: %s -> %s\n' "$link" "$target"
  fi
done < <(
  find "$ROOT" \( -path "$DOT_LEX" -o -path "$DOT_REAL" \) -prune -o -xtype l \
    ! -path "$HOME/.wine/*" \
    ! -path "$HOME/.PlayOnLinux/*" \
    -print0 2>/dev/null
)

printf '\n%d broken, %d stow orphan(s)%s\n' "$broken" "$orphans" \
  "$([[ "$mode" == "fix-stow" ]] && echo " removed" || echo "; --fix-stow removes them")"
#+END_SRC

** Symlink health scanner tests
Run with =python3 all/.local/share/dotcortex/tests/test-dotcortex-stow-health.py -v= from the repo root. Everything happens in a scratch =$HOME=; nothing real is scanned.

#+BEGIN_SRC python :tangle all/.local/share/dotcortex/tests/test-dotcortex-stow-health.py :shebang "#!/usr/bin/env python3" :comments no
"""Tests for dotcortex-stow-health (dotcortex.org): relative and absolute Stow orphans."""

import os
import shutil
import subprocess
import tempfile
import unittest
from pathlib import Path

ROOT = Path(__file__).resolve().parents[5]
SCANNER = ROOT / "linux/.local/bin/dotcortex-stow-health"


class Home:
    def __init__(self, symlinked_checkout=False):
        self.tmp = Path(tempfile.mkdtemp())
        self.home = self.tmp / "home"
        self.home.mkdir()
        if symlinked_checkout:
            real = self.home / "src/dc"
            real.mkdir(parents=True)
            os.symlink("src/dc", self.home / "DotCortex")
        else:
            (self.home / "DotCortex").mkdir()
        dot = self.home / "DotCortex"
        (dot / "all").mkdir()
        (dot / "all/kept").write_text("x\n")
        (self.home / ".local/bin").mkdir(parents=True)
        (self.home / ".local/share").mkdir(parents=True)
        (self.home / "DotCortex-old").mkdir()

    def link(self, rel, target):
        path = self.home / rel
        path.parent.mkdir(parents=True, exist_ok=True)
        os.symlink(target, path)
        return path

    def run(self, *args):
        env = dict(os.environ, HOME=str(self.home))
        env.pop("DOTCORTEX_ROOT", None)
        return subprocess.run(["bash", str(SCANNER), *args], capture_output=True, text=True, env=env)

    def close(self):
        shutil.rmtree(self.tmp)


class StowHealthTests(unittest.TestCase):
    def setUp(self):
        self.box = Home()
        b = self.box
        self.orphans = [
            b.link(".local/bin/honey-opus", "../../DotCortex/honey/.local/bin/honey-opus"),
            b.link(".local/share/honey-claude", "../../DotCortex/honey/.local/share/honey-claude"),
            b.link(".oldabs", str(b.home / "DotCortex/gone")),
            b.link(".config/deep/x", "../../DotCortex/all/../linux/gone"),
        ]
        self.kept = [
            b.link(".foreign", "/nonexistent/elsewhere"),
            b.link(".sibling", "DotCortex-old/gone"),
            b.link(".valid", "DotCortex/all/kept"),
            b.link("DotCortex/all/inrepo", "/home/nobody/x"),
        ]

    def tearDown(self):
        self.box.close()

    def test_scan_tags_orphans_and_removes_nothing(self):
        out = self.box.run()
        self.assertEqual(out.returncode, 0, out.stderr)
        for path in self.orphans:
            self.assertIn(f"BROKEN: {path} -> {os.readlink(path)} (stow orphan)", out.stdout)
            self.assertTrue(path.is_symlink())
        self.assertIn(f"BROKEN: {self.kept[0]} -> /nonexistent/elsewhere\n", out.stdout)
        self.assertNotIn("(stow orphan)", out.stdout.split(str(self.kept[1]))[1].splitlines()[0])
        self.assertNotIn(".valid", out.stdout)
        self.assertNotIn("inrepo", out.stdout, "the checkout itself is never scanned")
        self.assertIn("6 broken, 4 stow orphan(s)", out.stdout)

    def test_fix_stow_removes_relative_and_absolute_orphans_only(self):
        out = self.box.run("--fix-stow")
        self.assertEqual(out.returncode, 0, out.stderr)
        for path in self.orphans:
            self.assertFalse(path.is_symlink(), path)
        for path in self.kept:
            self.assertTrue(path.is_symlink(), path)
        self.assertIn("4 stow orphan(s) removed", out.stdout)

    def test_unknown_argument_is_refused(self):
        self.assertEqual(self.box.run("--fix").returncode, 2)


class SymlinkedCheckoutTests(unittest.TestCase):
    def test_orphans_through_either_path_are_removed(self):
        box = Home(symlinked_checkout=True)
        try:
            via_link = box.link(".a", "DotCortex/gone")
            via_real = box.link(".b", "src/dc/gone")
            inside = box.link("src/dc/all/inrepo", "/home/nobody/x")
            out = box.run("--fix-stow")
            self.assertEqual(out.returncode, 0, out.stderr)
            self.assertFalse(via_link.is_symlink())
            self.assertFalse(via_real.is_symlink())
            self.assertTrue(inside.is_symlink(), "the real checkout is never scanned either")
        finally:
            box.close()


if __name__ == "__main__":
    unittest.main()

** DotCortex backup - mirror DotCortex to a target
#+BEGIN_SRC bash :tangle linux/.local/bin/dotcortex-backup-dotfiles :shebang "#!/usr/bin/env bash" :mkdirp yes
#!/usr/bin/env bash
set -euo pipefail

SRC="$HOME/DotCortex"

if [[ ! -d "$SRC" ]]; then
  echo "Source repo not found at $SRC" >&2
  exit 1
fi

DEST_ROOT="${1-}"
if [[ -z "$DEST_ROOT" ]]; then
  echo "Usage: dotcortex-backup-dotfiles /path/to/backup-root" >&2
  exit 1
fi

DEST_ROOT="${DEST_ROOT%/}"
DEST="${DEST_ROOT}/DotCortex"

mkdir -p "$DEST_ROOT"

echo "Mirroring $SRC -> $DEST"
rsync -avh --delete "$SRC"/ "$DEST"/
#+END_SRC
